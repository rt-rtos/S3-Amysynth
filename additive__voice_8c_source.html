67 /* 3) oscs 1..N: one PARTIAL sine per harmonic. partials_note_on() sets
68 * each child's status/pitch base but NOT its wave - PARTIAL selects the

70 * Pitch offset is log2(ratio); amp is the spectrum level, scaled per
71 * block by the parent's envelope through COEF_VEL. */
72 for (uint8_t i = 0; i < n; i++) {

78 e->freq_coefs[COEF_CONST] = add_ratio_octaves(voice->ratio[i]);
79 e->amp_coefs[COEF_CONST] = voice->level[i];

87 e->eg0_times[1] = (uint32_t)voice->decay_ms[i];

89 }
