76static inline void zero(SAMPLE* a) {
77 const size_t nbytes = AMY_BLOCK_SIZE * sizeof(SAMPLE);
78#if defined(__XTENSA__) && defined(CONFIG_IDF_TARGET_ESP32S3)
79 // ESP32-S3 PIE: clear the FM-operator scratch block with a 128-bit vector
80 // store loop. This block clear/copy is the render path's one real PIE win
81 // (~10% on a 6-op FM voice); nothing else in AMY vectorizes (int32 SAMPLE has
82 // no 32-bit vector multiply, oscillators gather, filters recur). The scratch
83 // is 16-byte aligned (malloc_caps_block in algo_init) and nbytes is a compile
84 // -time multiple of 16, so the guard passes; anything unaligned falls to bzero.
85 //
86 // The aligned inner loop is the same one esp-dsp uses in dsps_memset_aes3.S /
87 // dsps_memcpy_aes3.S (Apache-2.0) - credit there for the approach. This is not
88 // a clean-room copy of those kernels, though: it carries none of their scalar
89 // head/tail machinery for unaligned buffers (which is most of their length),
90 // taking the vector path only for whole aligned blocks and deferring anything
91 // else to libc. Written compactly as original code (no lines copied), so it
92 // needs no separate .S file and keeps this source MIT with no vendored asm.
93 if ((((uintptr_t)a | nbytes) & 15u) == 0) {
94 void *pp = a; // ee.vst.128.ip post-increments the address register
95 __asm__ volatile(
96 "ee.xorq q0, q0, q0\n\t"
97 "loopnez %1, .Lamy_zero%=\n\t"
98 "ee.vst.128.ip q0, %0, 16\n"
99 ".Lamy_zero%=:"
100 : "+&r"(pp)
101 : "r"(nbytes / 16)
102 : "memory");
103 return;
104 }
105#endif
106 bzero((void *)a, nbytes);
107}

111static inline void copy(SAMPLE* a, SAMPLE* b) {
112 const size_t nbytes = AMY_BLOCK_SIZE * sizeof(SAMPLE);
113#if defined(__XTENSA__) && defined(CONFIG_IDF_TARGET_ESP32S3)
114 // ESP32-S3 PIE: copy the block with paired 128-bit vector load/store (see zero()).
115 if ((((uintptr_t)a | (uintptr_t)b | nbytes) & 15u) == 0) {
116 const void *s = a;
117 void *d = b;
118 __asm__ volatile(
119 "loopnez %2, .Lamy_copy%=\n\t"
120 "ee.vld.128.ip q0, %1, 16\n\t"
121 "ee.vst.128.ip q0, %0, 16\n"
122 ".Lamy_copy%=:"
123 : "+&r"(d), "+&r"(s)
124 : "r"(nbytes / 16)
125 : "memory");
126 return;
127 }
128#endif
129 bcopy((void *)a, (void *)b, nbytes);
130}

213SAMPLE render_algo(SAMPLE* buf, uint16_t osc, uint8_t core) {
214 struct FmAlgorithm algo = *algorithm_for(synth[osc]->algorithm); // LOCAL EDIT: custom rows
215 SAMPLE max_value = 0;
216
217 // starts at op 6
218 SAMPLE* in_buf;
219 SAMPLE* out_buf = NULL;
220
221 SAMPLE* const BUS_ONE = scratch + (SCRATCH_BLOCKS_PER_CORE * core) * AMY_BLOCK_SIZE;
222 SAMPLE* const BUS_TWO = BUS_ONE + AMY_BLOCK_SIZE;
223 SAMPLE* const SCRATCH = BUS_TWO + AMY_BLOCK_SIZE;
224
225 //for (int i = 0; i < SCRATCH_BLOCKS_PER_CORE; ++i)
226 // zero(BUS_ONE + i * AMY_BLOCK_SIZE);
227
228 SAMPLE amp = SHIFTR(F2S(msynth[osc]->amp), 2); // Arbitrarily divide FM voice output by 4 to make it more in line with other oscs.
229 for(uint8_t op=0;op<MAX_ALGO_OPS;op++) {
230 SAMPLE feedback_level = 0;
231 SAMPLE mod_amp = F2S(1.0f);
232 if(algo.ops[op] & FB_IN) {
233 feedback_level = F2S(synth[osc]->feedback);
234 } // main algo voice stores feedback, not the op
235
236 if(algo.ops[op] & IN_BUS_ONE) {
237 in_buf = BUS_ONE;
238 } else if(algo.ops[op] & IN_BUS_TWO) {
239 in_buf = BUS_TWO;
240 } else {
241 // no in_buf
242 in_buf = NULL;
243 }
244 if ( (algo.ops[op] & IN_BUS_ONE)
245 && (algo.ops[op] & OUT_BUS_ONE)
246 /* && !(algo.ops[op] & OUT_BUS_ADD) */ ) { // IN_BUS_ONE + OUT_BUS_ONE is never ADD
247 // Input is BUS_ONE and output overwrites it, use a temp buffer.
248 out_buf = SCRATCH;
249 } else if (algo.ops[op] & OUT_BUS_ONE) {
250 out_buf = BUS_ONE;
251 } else if (algo.ops[op] & OUT_BUS_TWO) {
252 out_buf = BUS_TWO;
253 } else {
254 out_buf = buf;
255 // We apply the msynth amp to every buf that goes into the final output buffer.
256 mod_amp = amp;
257 }
258 if (!(algo.ops[op] & OUT_BUS_ADD)) {
259 // Output is not being accumulated, so have to clear it first.
260 zero(out_buf);
261 }
262
263 SAMPLE value = 0;
264 // As with mod_source, an algo_source osc can have been freed since it
265 // was named, leaving synth[] NULL; render_algo is reached past
266 // amy_render's null skip.
267 if(AMY_IS_SET(synth[osc]->algo_source[op])
268 && synth[synth[osc]->algo_source[op]] != NULL
269 && synth[synth[osc]->algo_source[op]]->role == SYNTH_IS_ALGO_SOURCE) {
270 value = render_mod(in_buf, out_buf, synth[osc]->algo_source[op], feedback_level, osc, mod_amp);
271 } // If osc is not set, output has already been cleared.
272 if (out_buf == buf && value > max_value) max_value = value;
273
274 if (out_buf == SCRATCH) {
275 // We had to invoke the spare buffer, meaning we're overwriting BUS_ONE
276 copy(out_buf, BUS_ONE);
277 }
278 }
279 //fprintf(stderr, "render_algo: time %.3f osc %d last_amp %f amp %f max_val=%f\n", amy_global.time, osc, (msynth[osc]->last_amp), (msynth[osc]->amp), S2F(max_value));
280 // TODO, i need to figure out what happens on note offs for algo_sources.. they should still render..
281 return max_value;
282}

284// LOCAL EDIT (S3-Amysynth): reach zero()'s PIE block clear from amy_render()'s
285// buffer clears (per_osc_fb, fbl, chorus delay_mod) without a second copy of the
286// kernel. Upstream accelerates only the FM scratch here, which is where the
287// measured win is; the render-loop clears are the same shape and cost us one
288// wrapper rather than duplicated inline asm. Appended (rather than folded into
289// zero()) so algorithms.c stays byte-identical to upstream above this line.
290//
291// One "block" is AMY_BLOCK_SIZE samples - the fixed size zero() clears. Callers
292// pass 1, or AMY_NCHANS for the interleaved fbl buffers. Whether the vector path
293// or libc runs is zero()'s own alignment test: buffers allocated by
294// malloc_caps_block() take the vector path, anything else falls back.
295void amy_block_zero_blocks(SAMPLE *p, int nblocks) {
296 for (int i = 0; i < nblocks; ++i) zero(p + i * AMY_BLOCK_SIZE);
297}

307// LOCAL EDIT (S3-Amysynth): custom operator programs, the RAM rows behind
308// algorithm indices amy_num_algorithms .. amy_num_algorithms+AMY_NUM_CUSTOM_ALGORITHMS-1.
309// ops[] uses the FmOperatorFlags encoding of the fixed table (one byte per
310// render slot, slot s renders algo_source[s]). The write is six plain byte
311// stores with no lock: callers double-buffer across the rows and only ever
312// rewrite a row no live osc references, so the render body never observes a
313// half-written program.
314void amy_set_custom_algorithm(uint8_t slot, const uint8_t ops[MAX_ALGO_OPS]) {
315 if (slot >= AMY_NUM_CUSTOM_ALGORITHMS || ops == NULL) return;
316 for (uint8_t i = 0; i < MAX_ALGO_OPS; i++) custom_algorithms[slot].ops[i] = ops[i];
317}

319// LOCAL EDIT (S3-Amysynth): read access to any program's ops[] bytes (fixed
320// table or custom row), so an editor can draw the routing of the algorithm a
321// voice is playing. NULL when index is past both ranges.
322const uint8_t *amy_algorithm_ops(uint16_t index) {
323 if (index < amy_num_algorithms) return algorithms[index].ops;
324 index -= amy_num_algorithms;
325 if (index >= AMY_NUM_CUSTOM_ALGORITHMS) return NULL;
326 return custom_algorithms[index].ops;
327}
