107void fx_push_dist(void)
108{
109 /* One set of dist fields serves both scopes; the event decides which.
110 * Naming no osc puts them at bus scope, and with synth and bus unset too
111 * that bus is AMY_DEFAULT_BUS, where everything here renders. Naming a
112 * synth would instead fan them out over that synth's voices. */
113 amy_event *e = amy_helpers_event_begin();
114 /* bus_dist_type is a stage mask; author all three enables explicitly
115 * so a mask change turns dropped stages off. */
116 e->dist_clip = !!(s_fx.bus_dist_type & 1u);
117 e->dist_fold = !!(s_fx.bus_dist_type & 2u);
118 e->dist_crush = !!(s_fx.bus_dist_type & 4u);
119 e->dist_bits = (uint8_t)s_fx.bus_dist_bits;
120 e->dist_rate = (uint16_t)s_fx.bus_dist_rate;
121 /* A bus sum has no per-note modulation sources, so only the CONST coef
122 * of each rail reaches it: drive linear 1..16, mix linear 0..1. */
123 e->dist_drive_coefs[COEF_CONST] = (float)s_fx.bus_dist_drive;
124 e->dist_mix_coefs[COEF_CONST] = (float)s_fx.bus_dist_mix / 100.0f;
125 amy_helpers_event_send(e);
126}

130/* Re-impose the cached global FX after a patch load so a preset cannot hijack
131 * the shared EQ/chorus/echo/reverb. No-op when the user opted into letting
132 * presets drive global FX. Safe from the sequencer/arp/drone task contexts:
133 * each fx_push_* serialises through the shared amy_helpers mutex.
134 *
135 * These events are queued AFTER the patch's own FX deltas, so they win, and
136 * both drain in the same render quantum - no audible blip. */
137void synth_ui_fx_reassert_global(void)
138{
139 if (s_fx.presets_alter_global) return;
140 fx_push_eq();
141 fx_push_chorus();
142 fx_push_echo();
143 fx_push_reverb();
144}

146void amy_fx_set_master_volume(float v)
147{
148 v = SEQ_CLAMP_F32(v, 0.0f, 2.0f);
149 s_master_volume = v;
150 /* Direct write to amy_global.volume[]: an aligned float store, atomic on
151 * Xtensa. Called from synth_ui_task, never the render body. */
152 for (int b = 0; b < amy_global.config.max_buses; b++) {
153 amy_global.volume[b] = v;
154 }
155}
