102/* Re-impose the cached global FX after a patch load so a synth's preset
103 * cannot hijack the shared EQ/chorus/echo/reverb. No-op when the user has
104 * opted into letting presets drive global FX. Safe to call from the
105 * sequencer/arp/drone task contexts: each fx_push_* serialises through the
106 * shared amy_helpers mutex.
107 *
108 * Cost is four queued events per patch change (a rare, user-driven action)
109 * and zero in the render loop. The reassert events are queued AFTER the
110 * patch's own FX deltas, so they win; both drain in the same render quantum
111 * on Core 1, so there is no audible blip. */
112void synth_ui_fx_reassert_global(void)
113{
114 if (s_fx.presets_alter_global) return;
115 fx_push_eq();
116 fx_push_chorus();
117 fx_push_echo();
118 fx_push_reverb();
119}

121void amy_fx_set_master_volume(float v)
122{
123 if (v < 0.0f) v = 0.0f;
124 if (v > 2.0f) v = 2.0f;
125 s_master_volume = v;
126 /* Write directly to amy_global.volume[] — an aligned float store,
127 * atomic on Xtensa. Called from synth_ui_task (not the render body),
128 * so the AMY locking rule (no add_delta_to_queue inside the locked
129 * render loop) is satisfied. */
130 for (int b = 0; b < AMY_NUM_BUSES; b++) {
131 amy_global.volume[b] = v;
132 }
133}

135float amy_fx_get_master_volume(void)
136{
137 return s_master_volume;
