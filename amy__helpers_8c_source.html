7/*
8 * amy_event is large enough to be risky on small task stacks. First-party synth
9 * modules share one scratch event and serialize access around amy_add_event().
10 * Callers must not hold the returned pointer after send/cancel.

16/* Bounded wait so a leaked begin/send pair surfaces as a loud assert instead
17 * of hanging every AMY sender. Sized well above the worst-case legitimate hold
18 * (a patch-string load building four FX reassert events); if we ever wait this
19 * long the mutex was leaked, not merely contended. */
20#define AMY_HELPERS_EVENT_TIMEOUT_MS 250

30/* Lock hierarchy: s_event_mutex is the OUTERMOST first-party lock. It is held
31 * ACROSS amy_add_event(), so the acquisition order is
32 * s_event_mutex -> (amy_queue_lock | SEQ_LOCK).
33 * The render body (amy_render on Core 1) holds amy_queue_lock for its whole
34 * duration, so it must NEVER call an ingress helper. Register its handle here
35 * so a debug build can catch that inversion. Optional: the guard is skipped
36 * until a handle is set. */
37void amy_helpers_set_render_task(TaskHandle_t render_task)
38{
39 s_render_task = render_task;
40}

42amy_event *amy_helpers_event_begin(void)
43{
44 amy_helpers_init();
45#if !defined(NDEBUG)
46 /* Render must not re-enter the ingress seam while holding amy_queue_lock. */
47 configASSERT(s_render_task == NULL ||
48 xTaskGetCurrentTaskHandle() != s_render_task);
49#endif
50 if (xSemaphoreTake(s_event_mutex,
51 pdMS_TO_TICKS(AMY_HELPERS_EVENT_TIMEOUT_MS)) != pdTRUE) {
52 /* A prior begin/send pair leaked the mutex: fail loud, never hang. */
53 configASSERT(0 && "amy_helpers: s_event_mutex timeout, leaked begin/send");
54 }
55 s_event = amy_default_event();
56 return &s_event;
57}

72/* ── Typed ingress entry points ─────────────────────────────────────────
73 * amy_add_event() routes by shape: an event with sequence[] set is
74 * tick-scheduled whole (sequencer_add_event / SEQ_LOCK); one without is
75 * applied now (add_delta_to_queue / amy_queue_lock). The two entry points
76 * below make that route visible at the call site. The note-send signature is
77 * also the seam where per-step parameter-lock fields will attach. */

92void amy_helpers_config_send(amy_event *event)
93{
94 /* Config sends must not carry a sequence[] tuple — that shape would be

98 configASSERT(AMY_IS_UNSET(event->sequence[SEQUENCE_TAG]) &&
99 AMY_IS_UNSET(event->sequence[SEQUENCE_TICK]) &&
100 AMY_IS_UNSET(event->sequence[SEQUENCE_PERIOD]));
