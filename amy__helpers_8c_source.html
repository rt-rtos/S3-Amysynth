20/*
21 * AMY ingest pump.
22 *
23 * amy_add_event() runs AMY's full ingest in the caller's context, and for a
24 * patch-number event that includes the patch-string parse - a multi-KB stack
25 * frame with millisecond-scale runtime. Applying it inline meant two things:
26 * every emitting task had to be sized for that worst case, and the parse ran
27 * while holding the ingress mutex, so a UI gesture could stall the sequencer,
28 * arp and live-play senders for the duration of the parse.
29 *
30 * Instead, producers copy their filled event into a FIFO and one pump task -
31 * holding the single copy of the parse-frame stack rent - drains it into
32 * amy_add_event(). The mutex hold collapses to a memcpy.
33 *
34 * The pump also serves one URGENT SOURCE: a registered drain callback for
35 * deadline-sensitive jobs produced by a context that may not block on the
36 * FIFO (the render task's decorated-step trigs, seq_trig_pump.c). The loop
37 * drains the urgent source before every FIFO item, and sends made ON the
38 * pump task (i.e. from that callback) apply inline instead of re-entering
39 * the FIFO - so a decorated note never waits out a parse backlog, and the
40 * pump cannot deadlock against its own queue. One doorbell (a task
41 * notification) covers both sources; producers ring it after enqueueing.
42 *
43 * Ordering: the copy into the FIFO happens BEFORE the mutex is released, so
44 * FIFO order equals the global emission order the previous inline scheme
45 * produced. Voice-kill -> patch-load -> note-on chains cannot reorder across
46 * producers, and events carry their own time/sequence fields, so AMY-side
47 * scheduling is untouched. Urgent jobs jumping the FIFO does not weaken
48 * this: they originate on the render task, which never participates in the
49 * producer mutex, so no order between them and FIFO events ever existed.
50 *
51 * Backpressure: a full queue blocks the producer rather than dropping - losing
52 * a patch-define or a note-off is worse than a short stall, and the wait is
53 * strictly narrower than the old "block for someone else's whole parse".
54 *
55 * The pump loop must never acquire a lock beyond what amy_add_event() takes
56 * internally - in particular never s_event_mutex, which a producer can hold
57 * while blocked on a full FIFO only this task drains - and must never block
58 * on anything but its own doorbell: it inherits the failure potential the
59 * inline scheme had, where one stalled parse froze render, the tick-slaved
60 * sequencer and all controls at once.
61 */
62#define AMY_INGEST_QUEUE_DEPTH 64
63/* Shallower internal-RAM queue used only when the PSRAM storage alloc fails;
64 * degraded burst headroom is acceptable, refusing to run is not. */
65#define AMY_INGEST_QUEUE_DEPTH_FALLBACK 16
66#define AMY_INGEST_TASK_STACK 8192
67/* Same tier as the UI producers AND the TinyUSB device task: a long patch
68 * parse time-slices with USB exactly as the old inline scheme did, instead of
69 * strictly preempting it. Note the pump does NOT outrank every producer - the
70 * NimBLE host and the render task sit far above it - so drain latency is
71 * bounded by scheduling, not priority alone; the 64-deep FIFO absorbs bursts. */
72#define AMY_INGEST_TASK_PRIO 5
73#define AMY_INGEST_TASK_CORE 0 /* Core 1 is budgeted for the AMY DSP */

92/* Bounded wait so a leaked begin/send pair asserts loudly instead of hanging
93 * every AMY sender. Well above the worst-case legitimate hold (a patch-string
94 * load building four FX reassert events): waiting this long means leaked, not

98static void amy_ingest_task(void *arg)
99{
100 (void)arg;
101 amy_event ev;
102 for (;;) {
103 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
104 for (;;) {
105 /* Urgent jobs first - and again between every FIFO item, so a
106 * decorated-step note due next sequencer tick is never stuck
107 * behind a backlog of parses. */
108 if (s_urgent_drain != NULL && s_urgent_drain()) continue;
109 if (xQueueReceive(s_ingest_queue, &ev, 0) == pdTRUE) {
110 amy_add_event(&ev);
111 continue;
112 }
113 break;
114 }
115 }
116}

118void amy_helpers_init(void)
119{
120 if (s_event_mutex == NULL) {
121 s_event_mutex = xSemaphoreCreateMutex();
122 configASSERT(s_event_mutex != NULL);
123 }
124 if (s_ingest_queue == NULL) {
125 /* Queue storage is bulk cold data touched only by memcpy, so it lives
126 * in PSRAM to keep tens of KB out of scarce internal RAM. */
127 const size_t item = sizeof(amy_event);
128 uint8_t *storage = heap_caps_malloc(AMY_INGEST_QUEUE_DEPTH * item,
129 MALLOC_CAP_SPIRAM);
130 if (storage != NULL) {
131 s_ingest_queue = xQueueCreateStatic(AMY_INGEST_QUEUE_DEPTH, item,
132 storage,
133 &s_ingest_queue_struct);
134 } else {
135 ESP_LOGW(TAG_INGEST, "PSRAM queue alloc failed; using internal RAM");
136 s_ingest_queue = xQueueCreate(AMY_INGEST_QUEUE_DEPTH_FALLBACK, item);
137 }
138 /* Lazy init runs single-threaded during startup, before any producer
139 * task exists; a missing queue here means the seam would be used from
140 * a task with nowhere to put events. */
141 configASSERT(s_ingest_queue != NULL);
142
143 BaseType_t ok = xTaskCreatePinnedToCore(amy_ingest_task, "amy_ingest",
144 AMY_INGEST_TASK_STACK, NULL,
145 AMY_INGEST_TASK_PRIO,
146 &s_pump_task,
147 AMY_INGEST_TASK_CORE);
148 configASSERT(ok == pdPASS);
149 }
150}

165/* Lock hierarchy: s_event_mutex is the OUTERMOST first-party lock. It is held
166 * across the FIFO enqueue only; the AMY-side locks are taken later, on the pump
167 * task, so the acquisition order is
168 * s_event_mutex -> ingest FIFO, then amy_queue_lock on the pump.
169 * The render body (amy_render on Core 1) holds amy_queue_lock for its whole
170 * duration, so it must NEVER call an ingress helper: it would both invert that
171 * order and risk blocking on a FIFO the pump cannot drain. Register its handle
172 * here so a debug build can catch that inversion. Optional: the guard is
173 * skipped until a handle is set. */
174void amy_helpers_set_render_task(TaskHandle_t render_task)
175{
176 s_render_task = render_task;
177}

179amy_event *amy_helpers_event_begin(void)
180{
181 amy_helpers_init();
182#if !defined(NDEBUG)
183 /* Render must not re-enter the ingress seam while holding amy_queue_lock. */
184 configASSERT(s_render_task == NULL ||
185 xTaskGetCurrentTaskHandle() != s_render_task);
186#endif
187 if (s_pump_task != NULL && xTaskGetCurrentTaskHandle() == s_pump_task) {
188 /* Pump-context send (urgent-source expansion): private scratch, no
189 * mutex. A producer can legitimately hold s_event_mutex while blocked
190 * on a full FIFO that only this task drains - taking it here would
191 * close that cycle into a deadlock. */
192#if !defined(NDEBUG)
193 configASSERT(!s_pump_event_busy &&
194 "amy_helpers: nested pump-context begin");
195 s_pump_event_busy = true;
196#endif
197 s_pump_event = amy_default_event();
198 return &s_pump_event;
199 }
200 if (xSemaphoreTake(s_event_mutex,
201 pdMS_TO_TICKS(AMY_HELPERS_EVENT_TIMEOUT_MS)) != pdTRUE) {
202 /* A prior begin/send pair leaked the mutex: fail loud, never hang. */
203 configASSERT(0 && "amy_helpers: s_event_mutex timeout, leaked begin/send");
204 }
205 s_event = amy_default_event();
206 return &s_event;
207}

209/* INVARIANT: a send is not an apply. This hands the event to the pump; the
210 * engine applies it later. Reading AMY state (synth[]/msynth[], patch or voice
211 * info) right after a send to observe that send's effect is a bug class - the
212 * read will race the pump. Express dependencies as send ORDER, which the FIFO
213 * preserves exactly, or poll state that self-heals across UI frames. There is
214 * deliberately no flush/drain barrier; add one only when a caller genuinely
215 * needs synchronous apply. */
216void amy_helpers_event_send(amy_event *event)
217{
218 if (event == &s_pump_event) {
219 /* On the pump task a send IS an apply: routing through the FIFO from
220 * its own consumer would deadlock when full, and the urgent path
221 * exists precisely to skip the backlog. */
222 amy_add_event(event);
223#if !defined(NDEBUG)
224 s_pump_event_busy = false;
225#endif
226 return;
227 }
228 configASSERT(event == &s_event);
229 if (s_ingest_queue == NULL) {
230 /* Init failed in a build with assertions compiled out: degrade to the
231 * inline apply rather than dropping the event or crashing. */
232 amy_add_event(event);
233 xSemaphoreGive(s_event_mutex);
234 return;
235 }
236 /* Enqueue BEFORE releasing the mutex: that is what makes FIFO order equal
237 * global emission order across producers. */
238 if (xQueueSend(s_ingest_queue, event,
239 pdMS_TO_TICKS(AMY_HELPERS_EVENT_TIMEOUT_MS)) != pdTRUE) {
240 /* 64 events of sustained backlog means ingest has stalled system-wide
241 * (wedged or starved pump). Fail loud rather than hang the senders
242 * silently; events are never dropped while assertions are enabled. */
243 configASSERT(0 && "amy_helpers: ingest queue full, pump stalled");
244 }
245 amy_helpers_pump_wake();
246 xSemaphoreGive(s_event_mutex);
247}
