46// --- MIDI realtime clock OUT (AMY as clock master, AMY_MIDI_SYNC_SEND) ---
47
48// Counts sequencer ticks (AMY_SEQUENCER_PPQ) down to MIDI clock rate
49// (MIDI_SEQUENCER_PPQ, 24 PPQ).
50static uint8_t clock_out_phase = 0;
51
52#ifdef __EMSCRIPTEN__
53// The sequencer ticks in the AudioWorklet render thread, but midi_out()'s
54// EM_ASM can only reach the page's midiOutputDevice from the browser main
55// thread (the same constraint as amy_sequencer_js_hook). Queue realtime bytes
56// here and flush them from main_loop__em().
57#define CLOCK_OUT_QUEUE_LEN 64
58static volatile uint8_t clock_out_queue[CLOCK_OUT_QUEUE_LEN];
59static volatile uint8_t clock_out_write = 0;
60static volatile uint8_t clock_out_read = 0;
61
62void midi_clock_out_flush() {
63 while (clock_out_read != clock_out_write) {
64 uint8_t byte = clock_out_queue[clock_out_read];
65 clock_out_read = (clock_out_read + 1) % CLOCK_OUT_QUEUE_LEN;
66 midi_out(&byte, 1);
67 }
68}
69#endif
70
71static void midi_realtime_out(uint8_t byte) {
72#ifdef __EMSCRIPTEN__
73 uint8_t next = (clock_out_write + 1) % CLOCK_OUT_QUEUE_LEN;
74 if (next == clock_out_read) return; // queue full; drop rather than block audio
75 clock_out_queue[clock_out_write] = byte;
76 clock_out_write = next;
77#else
78 midi_out(&byte, 1);
79#endif
80}

86// Called once per sequencer tick while on the internal clock; emits MIDI
87// Timing Clock (0xF8) at 24 PPQ, i.e. every AMY_SEQUENCER_PPQ/MIDI_SEQUENCER_PPQ
88// ticks. The sequencer keeps calling this while the transport is stopped so
89// downstream slaves stay tempo-locked between Start messages.
90void midi_clock_out_tick() {
91 if (!midi_clock_out_enabled()) return;
92 if (clock_out_phase == 0) midi_realtime_out(0xF8);
93 clock_out_phase = (clock_out_phase + 1) % (AMY_SEQUENCER_PPQ / MIDI_SEQUENCER_PPQ);
94}
95
96void midi_clock_out_start() {
97 if (!midi_clock_out_enabled()) return;
98 // The first clock after Start marks the downbeat for slaves.
99 clock_out_phase = 0;
100 midi_realtime_out(0xFA);
101}
102
103void midi_clock_out_stop() {
104 if (!midi_clock_out_enabled()) return;
105 midi_realtime_out(0xFC);
106}

120 // don't forward on a note coming in through MIDI IN
121 //fprintf(stderr, "amy_send_midi_note_on: osc %d source %d note %.1f vel %.3f\n",

152void amy_received_program_change(uint8_t channel, uint8_t program) {
153 amy_event e = amy_default_event();
154 e.synth = channel;
155 e.note_source_channel = channel;
156 // MIDI patches are in blocks of 128, potentially set by an earlier CC0.
157 int bank_number = instrument_bank_number(channel);
158 if (bank_number < 0) {
159 // If the bank hasn't been set, stay within the block of 128 of the current patch
160 // (so e.g. DX7 voices remain DX7).
161 int synth_patch = instrument_get_patch_number(e.synth);
162 // ...but only if there IS a current patch. A synth built from a
163 // patch_string carries no patch number (the auto-assigned one is an
164 // allocator artifact, released as soon as it is loaded), and a synth
165 // that doesn't exist reports -1. Both used to be masked and shifted
166 // into a nonsense bank -- 511 for the unset sentinel -- which put the
167 // program change hundreds of patches past anything defined, so a bare
168 // PC on such a channel did nothing at all. Default to bank 0 (Juno).
169 uint16_t synth_patch_u16 = (uint16_t)synth_patch;
170 if (synth_patch < 0 || AMY_IS_UNSET(synth_patch_u16)) synth_patch = 0;
171 bank_number = (synth_patch & 0xFF80) >> 7;
172 // Banks 0 (Juno, patches 0-127) and 1 (DX7, 128-255) are full 128-patch
173 // banks, and bank 3 (384+) is the Gamma9001 drum kit bank -- a synth
174 // sitting on a drum kit patch should stay in the kit bank so a bare PC
175 // switches kits. Patches 256-383 (additive piano, AMYboard Web Editor
176 // base, reserved space) infer bank 2, whose PC targets are almost all
177 // undefined and silence the board (issue #758), so fall back to bank 0
178 // (Juno) only for those.
179 if (bank_number == 2) bank_number = 0;
180 }
181 e.patch_number = program + 128 * bank_number;
182 //if (channel != AMY_MIDI_CHANNEL_DRUMS) { // What would that even mean?
183 amy_add_event(&e);
184 //}
185}

205void amy_received_pitch_bend(uint8_t channel, uint8_t low_byte, uint8_t high_byte) {
206 amy_event e = amy_default_event();
207 // Currently, pitch bend is global and not applied per-channel, but preserve the info.
208 e.synth = channel;
209 e.note_source_channel = channel;
210 // The integer range is -8192 to 8191, the float range is -1/6th to +1/6th,
211 // units are octaves, so +/- 2 semitones.
212 e.pitch_bend = ((float)(((int)((high_byte << 7) | low_byte)) - 8192)) / (6.0f * 8192.0f);
213 amy_add_event(&e);
214}

243// I'm called when we get a fully formed MIDI message from any interface -- usb, gadget, uart, mac, and either sysex or normal
244void amy_event_midi_message_received(uint8_t * data, uint32_t len, uint8_t sysex) {
245 if(!sysex) {
246 uint8_t status_byte = data[0];
247 uint8_t channel = 1 + (status_byte & 0x0F);
248 if (_midi_channel_active[channel]) {
249 uint8_t status = status_byte & 0xF0;
250 // Do the AMY instrument things here
251 if(status == 0xB0 && data[1] == 0x40) amy_received_pedal(channel, data[2]);
252 else if(status == 0xB0 && data[1] == 0x7B) amy_received_all_notes_off(channel);
253 else if(status == 0XB0) amy_received_control_change(channel, data[1], data[2]);
254 else if(status == 0xC0) amy_received_program_change(channel, data[1]);
255 else if(status == 0xE0) amy_received_pitch_bend(channel, data[1], data[2]);
256 // MIDI transport (Start/Stop) only drives the sequencer when the user
257 // has opted into following external sync; otherwise a connected DAW's
258 // transport would hijack the AMYboard's own internal sequence.
259 else if(status_byte == 0xFA) { if(external_midi_sync_mode == AMY_MIDI_SYNC_FOLLOW) sequencer_midi_start(); }
260 else if(status_byte == 0xFC) { if(external_midi_sync_mode == AMY_MIDI_SYNC_FOLLOW) sequencer_midi_stop(); }
261 // midi_message_handler_to_queue keeps its time parameter -- patches.c drives
262 // it with a real e->time for the wave=AMY_MIDI osc. Live MIDI has no time
263 // of its own: it plays when it arrives.
264 midi_message_handler_to_queue(status, channel, data + 1, (uint16_t)(len - 1), AMY_UNSET_VALUE((uint32_t)0), NULL, NULL);
265 }
266 }
267
268 // Also send the external hooks if set
269 if(amy_global.config.amy_external_midi_input_hook != NULL) {
270 amy_global.config.amy_external_midi_input_hook(data, len, sysex);
271 }
272
273 // Update web MIDI out hook if set
274 #ifdef __EMSCRIPTEN__
275 EM_ASM(
276 if(typeof amy_external_midi_input_js_hook === 'function') {
277 amy_external_midi_input_js_hook(HEAPU8.subarray($0, $0+$1), $1, $2);
278 }, data, len, sysex);
279 #endif
280}

316uint16_t sysex_len = 0;
317#if defined(TULIP) || defined(AMYBOARD)
318extern const mp_obj_fun_builtin_var_t tulip_amy_send_sysex_obj;
319#endif
320uint8_t * sysex_buffer = NULL;
321// Ring buffer of sysex payload snapshots for deferred MicroPython processing.
322// parse_sysex() copies each payload into a separate slot so that a new sysex
323// arriving before the scheduled callback fires doesn't overwrite the previous
324// message. This matters when the sketch's loop() is CPU-heavy and the
325// mp_sched callback is delayed.
326char * sysex_message_copies[SYSEX_COPY_SLOTS_DIM]; // dim floored at 1 for MSVC; loops use SYSEX_COPY_SLOTS (0 off-board)
327uint8_t sysex_copy_write_idx = 0; // MIDI task writes here
328uint8_t sysex_copy_read_idx = 0; // MP callback reads here
329void parse_sysex() {
330 if(sysex_len>3) {
331 // let's use 0x00 0x03 0x45 for SPSS
332 if(sysex_buffer[0] == 0x00 && sysex_buffer[1] == 0x03 && sysex_buffer[2] == 0x45) {
333 sysex_buffer[sysex_len] = 0;
334 // zB[mode]: Reboot. Handled in pure C — no mp_sched_schedule
335 // needed, works even when loop() is hogging the scheduler.
336 // zBZ / zB0Z — bootloader mode (skip sketch.py)
337 // zB1Z — normal reboot (run sketch.py)
338 // zB2Z — ROM download/flash mode
339 if (sysex_len > 4 && sysex_buffer[3] == 'z' && sysex_buffer[4] == 'B') {
340 uint8_t mode = 0;
341 if (sysex_len > 5 && sysex_buffer[5] >= '0' && sysex_buffer[5] <= '9') {
342 mode = sysex_buffer[5] - '0';
343 }
344 if (amy_global.config.amy_external_reboot_hook) {
345 amy_global.config.amy_external_reboot_hook(mode);
346 }
347 sysex_len = 0;
348 return;
349 }
350 // zI: Ping/identity — reply with a short sysex so the web side
351 // knows the board is alive and ready. Pure C, no scheduler needed.
352 if (sysex_len > 4 && sysex_buffer[3] == 'z' && sysex_buffer[4] == 'I') {
353 uint8_t frame[] = { 0xF0, 0x00, 0x03, 0x45, 'O', 'K', 0xF7 };
354 midi_out(frame, sizeof(frame));
355 sysex_len = 0;
356 return;
357 }
358 // For Micropython hosted systems, we run MIDI on a separate "thread" (task)
359 // than MP, so just calling amy_send_message here can fail if it needs to access
360 // underlying MP resources. So we schedule it to run in the MP main loop instead.
361 // Each message gets its own ring-buffer slot so a fast-arriving next sysex
362 // doesn't overwrite an unprocessed message.
363 #if defined(TULIP) || defined(AMYBOARD)
364 {
365 // NOTE: ACK is sent from the callback (tulip_amy_send_sysex)
366 // AFTER the message is processed, not here in parse_sysex.
367 // This ensures the sender only proceeds once the ring buffer
368 // slot has been drained — receiving the ACK here would just
369 // confirm receipt, allowing the ring buffer to overflow if
370 // callbacks are slow.
371 //
372 // Do NOT stop the sequencer here. We used to do that to
373 // prevent loop() from stealing MP scheduler slots during
374 // large file transfers, but it caused sequencer_midi_start
375 // to reset next_amy_tick_us on every sysex, effectively
376 // speeding up the sequencer when knob updates arrive.
377 char *slot = sysex_message_copies[sysex_copy_write_idx];
378 if(slot) {
379 uint16_t payload_len = sysex_len - 3;
380 memcpy(slot, (char*)sysex_buffer + 3, payload_len);
381 slot[payload_len] = '\0';
382 sysex_copy_write_idx = (sysex_copy_write_idx + 1) % SYSEX_COPY_SLOTS;
383 }
384 mp_sched_schedule(MP_OBJ_FROM_PTR(&tulip_amy_send_sysex_obj), mp_const_none);
385 }
386 #else
387 amy_add_message((char*)sysex_buffer+3);
388 #endif
389 sysex_len = 0; // handled
390 } else {
391 amy_event_midi_message_received(sysex_buffer, sysex_len, 1);
392 }
393 }
394}

396void convert_midi_bytes_to_messages(uint8_t * data, size_t len, uint8_t usb) {
397 // i take any amount of bytes and add messages
398 // remember this can start in the middle of a midi message, so act accordingly
399 // running status is handled by keeping the status byte around after getting a message.
400 // remember that USB midi always comes in groups of 3 here, even if it's just a one byte message
401 // so we have USB (and mac IAC) set a usb flag so we know to end the loop once a message is parsed
402
403 for(size_t i=0;i<len;i++) {
404
405 uint8_t byte = data[i];
406
407 // Skip sysex in this parser until we get an F7. We do not pass sysex over to python (yet)
408 if(sysex_flag) {
409 if(byte == 0xF7) {
410 sysex_flag = 0;
411 parse_sysex();
412 } else if(sysex_len < MAX_SYSEX_BYTES - 1) {
413 sysex_buffer[sysex_len++] = byte;
414 }
415 // else: sysex payload has overflowed sysex_buffer (MAX_SYSEX_BYTES) --
416 // drop the extra byte rather than writing past the allocation. The
417 // "- 1" keeps room for the NUL terminator parse_sysex() writes at
418 // sysex_buffer[sysex_len].
419 } else {
420 if(byte & 0x80) { // new status byte
421 // System Real-Time messages (0xF8-0xFF) may be interleaved
422 // anywhere in the stream -- even between the data bytes of
423 // another message -- and must NOT disturb running status. So
424 // handle them with a scratch buffer, leaving current_midi_message[]
425 // and midi_message_slot untouched.
426 if(byte >= 0xF8) {
427 if(byte == 0xF8) { // clock. don't forward this on to Tulip userspace
428 midi_clock_received();
429 } else { // start/continue/stop/active-sensing/reset/etc
430 uint8_t rt[1] = { byte };
431 amy_event_midi_message_received(rt, 1, 0);
432 }
433 if(usb) i = len+1; // exit the loop if usb
434 } else {
435 // Channel Voice (0x80-0xE0) or System Common (0xF0-0xF7):
436 // these begin a fresh message and cancel running status.
437 sysex_flag = 0; sysex_len = 0;
438 current_midi_message[0] = byte;
439 midi_message_slot = 0; // drop any half-collected data bytes
440 if(byte == 0xF4 || byte == 0xF5 || byte == 0xF6) {
441 // 1-byte System Common (undefined / tune request)
442 amy_event_midi_message_received(current_midi_message, 1, 0);
443 if(usb) i = len+1; // exit the loop if usb
444 } else if(byte == 0xF0) { // sysex start
445 // everything is an AMY message until 0xF7
446 sysex_flag = 1;
447 }
448 // else: channel voice or F1/F2/F3 -- status stored, await data bytes
449 }
450 } else { // data byte of some kind
451 uint8_t status = current_midi_message[0] & 0xF0;
452
453 // a 2 bytes of data message
454 if(status == 0x80 || status == 0x90 || status == 0xA0 || status == 0xB0 || status == 0xE0 || current_midi_message[0] == 0xF2) {
455 if(midi_message_slot == 0) {
456 current_midi_message[1] = byte;
457 midi_message_slot = 1;
458 } else {
459 current_midi_message[2] = byte;
460 midi_message_slot = 0;
461 amy_event_midi_message_received(current_midi_message, 3, 0);
462 }
463 // a 1 byte data message
464 } else if (status == 0xC0 || status == 0xD0 || current_midi_message[0] == 0xF3 || current_midi_message[0] == 0xF1) {
465 current_midi_message[1] = byte;
466 amy_event_midi_message_received(current_midi_message, 2, 0);
467 if(usb) i = len+1; // exit the loop if usb
468 }
469 }
470 }
471 }

601void esp_poll_midi(void) {
602 const int uart_num = esp_get_uart(amy_global.config.midi_uart);
603 uint8_t data[MAX_MIDI_BYTES_TO_PARSE];
604 // Block for at least one tick. The old 1/portTICK_PERIOD_MS is
605 // integer-zero whenever the tick is slower than 1ms (e.g. ESP-IDF's
606 // and MicroPython's default CONFIG_FREERTOS_HZ=100), which made this
607 // a zero-timeout read and run_midi_task() a busy-spin pinning its
608 // core at 100% from a priority-23 task.
609 TickType_t timeout = pdMS_TO_TICKS(1);
610 if (timeout == 0) timeout = 1;
611 int length = uart_read_bytes(uart_num, data, MAX_MIDI_BYTES_TO_PARSE /*MAX_MIDI_BYTES_PER_MESSAGE*MIDI_QUEUE_DEPTH*/, timeout);
612 if(length > 0) {
613 convert_midi_bytes_to_messages(data,length,0);
614 }
615}

783void midi_out(uint8_t * bytes, uint16_t len) {
784 midi_out_external_hook(bytes, len);
785
786// Is there USB gadget midi? Send it
787#if defined TUD_USB_GADGET
788 // tud_ready(): only try USB when a host has enumerated us (and the bus
789 // isn't suspended). tud_midi_stream_write has no mount check -- with no
790 // host attached (e.g. AMYboard on eurorack power alone) it fills its tx
791 // FIFO, then returns 0 forever, and the stall loop below burns ~1s per
792 // byte. MIDI clock out (24 PPQ) hits that from the render path and
793 // wedges the sequencer while starving the UART/TRS out.
794 if((amy_global.config.midi & AMY_MIDI_IS_USB_GADGET) && tud_ready()) {
795 // tud_midi_stream_write uses a small FIFO (e.g. 64 bytes). For long
796 // messages (e.g. zD sysex dumps) we must loop and yield until the
797 // USB task flushes the FIFO, otherwise bytes are silently dropped.
798 if (len > 64) fprintf(stderr, "midi_out: USB gadget, want to send %d bytes\n", (int)len);
799 uint32_t sent = 0;
800 int stall_ticks = 0;
801 while (sent < len) {
802 uint32_t n = tud_midi_stream_write(0, bytes + sent, len - sent);
803 if (n == 0) {
804#if defined(TULIP) || defined(AMYBOARD)
805 // We're running on the MP main thread; the USB task also runs
806 // here, so vTaskDelay alone won't drain the FIFO. Pump USB
807 // events directly.
808 mp_usbd_task();
809#endif
810#if defined ESP_PLATFORM
811 vTaskDelay(pdMS_TO_TICKS(1));
812#endif
813 if (++stall_ticks > 1000) {
814 fprintf(stderr, "midi_out: STALLED after %u of %u bytes\n",
815 (unsigned)sent, (unsigned)len);
816 break;
817 }
818 } else {
819 stall_ticks = 0;
820 }
821 sent += n;
822 }
823 if (len > 64) fprintf(stderr, "midi_out: USB gadget sent %u/%u bytes\n",
824 (unsigned)sent, (unsigned)len);
825 }
