26// Every platform allocates the single sysex_buffer above. The sysex copy-slot
27// ring buffer below holds backup snapshots so a fast-arriving message isn't lost
28// while a previous one waits for the deferred MicroPython mp_sched callback.
29// Only AMYBOARD creates and reads these slots (see parse_sysex()): the ring
30// exists for the SPSS z* sketch-transfer bursts over USB-gadget MIDI, which
31// only AMYboard receives. Size the ring to 0 everywhere else so we don't
32// malloc SYSEX_COPY_SLOTS x MAX_SYSEX_BYTES (32 x 16KB = 512KB): that alone
33// would exhaust e.g. the rp2350's 520KB of SRAM, and on Tulip ESP32-S3 it ate
34// a third of the free SPIRAM. With 0 slots, parse_sysex() finds a NULL slot
35// and the scheduled callback ACKs SPSS messages without processing them;
36// non-SPSS sysex is unaffected.
37#if defined(AMYBOARD)
38#define SYSEX_COPY_SLOTS 32
39#else
40#define SYSEX_COPY_SLOTS 0
41#endif
42// The alloc/free loops use SYSEX_COPY_SLOTS directly (0 => no slots off-board),
43// but the array dimension must be >= 1: MSVC (the Godot Windows build includes
44// this header transitively via amy.h) rejects a zero-length array with C2466,
45// unlike the GCC/Clang extension. Floor the dimension at 1; that lone slot is
46// never touched off-board because the loops iterate 0 times.
47#define SYSEX_COPY_SLOTS_DIM ((SYSEX_COPY_SLOTS) > 0 ? (SYSEX_COPY_SLOTS) : 1)
