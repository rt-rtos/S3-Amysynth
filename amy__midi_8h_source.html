65// Every platform allocates the single sysex_buffer above. The sysex copy-slot
66// ring buffer below holds backup snapshots so a fast-arriving message isn't lost
67// while a previous one waits for the deferred MicroPython mp_sched callback.
68// Only AMYBOARD creates and reads these slots (see parse_sysex()): the ring
69// exists for the SPSS z* sketch-transfer bursts over USB-gadget MIDI, which
70// only AMYboard receives. Size the ring to 0 everywhere else so we don't
71// malloc SYSEX_COPY_SLOTS x MAX_SYSEX_BYTES: that alone would exhaust e.g.
72// the rp2350's 520KB of SRAM, and on Tulip ESP32-S3 it ate a third of the
73// free SPIRAM. With 0 slots, parse_sysex() finds a NULL slot and the
74// scheduled callback ACKs SPSS messages without processing them; non-SPSS
75// sysex is unaffected.
76// 8 slots (128KB SPIRAM) is enough on AMYboard: the sender is flow-controlled
77// (the ACK is sent only after a slot is drained, see parse_sysex()), so it
78// can never run more than the ring depth ahead. 32 slots cost 512KB of
79// SPIRAM better spent on memorypcm samples.
80#if defined(AMYBOARD)
81#define SYSEX_COPY_SLOTS 8
82#else
83#define SYSEX_COPY_SLOTS 0
84#endif
85// The alloc/free loops use SYSEX_COPY_SLOTS directly (0 => no slots off-board),
86// but the array dimension must be >= 1: MSVC (the Godot Windows build includes
87// this header transitively via amy.h) rejects a zero-length array with C2466,
88// unlike the GCC/Clang extension. Floor the dimension at 1; that lone slot is
89// never touched off-board because the loops iterate 0 times.
90#define SYSEX_COPY_SLOTS_DIM ((SYSEX_COPY_SLOTS) > 0 ? (SYSEX_COPY_SLOTS) : 1)

130// ---- Which MIDI port -------------------------------------------------
131//
132// amy has always listened to EVERY CoreMIDI source and sent to EVERY
133// destination, which is the right default -- a plugged-in keyboard just
134// works -- and is what -1 still means here. A host with somewhere to put
135// the choice (a menu, a settings pane) needs to name the ports and pick
136// one, which is what these four are for.
137//
138// Selection is by INDEX into the same order the count/name pair reports,
139// matching amy's other device selectors (config.playback_device_id).
140// Indices SHIFT when something is plugged or unplugged, so the selection
141// is pinned by NAME as well: if the name at the stored index no longer
142// matches, the name is looked for elsewhere in the list, and a port that
143// has really gone falls back to -1 rather than silently becoming
144// whichever device inherited its number.
145//
146// Implemented for macOS (macos_midi.m); every other platform answers 0
147// ports and -1, so a host may call these unconditionally.
148#define AMY_MIDI_PORT_IN 0
149#define AMY_MIDI_PORT_OUT 1
