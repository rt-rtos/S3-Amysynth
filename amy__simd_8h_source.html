1// LOCAL EDIT (S3-Amysynth): ESP32-S3 PIE (128-bit SIMD) hooks for AMY's render path.
2//
3// Thin shim over the local `pie_dsp` component. The reasoning for why so little of AMY
4// is PIE-eligible lives in components/pie_dsp/include/pie_dsp.h - short version: SAMPLE
5// is int32 (s8.23) and PIE has no 32-bit vector multiply, the LUT oscillators are
6// gather-indexed, and the filters/delays are all recurrence-bound. What is left, once
7// the block scans turned out to be too short to amortise the vector setup, is the bulk
8// block clears and copies - which is what this covers, and all it covers.
9//
10// Kept as a separate header (rather than edits to amy.h) so an AMY re-vendor only has to
11// re-apply the call sites, not untangle a modified upstream header.
12//
13// IMPORTANT: amy.h (which defines AMY_USE_FIXEDPOINT) must be included before this.
14
15#ifndef AMY_SIMD_H
16#define AMY_SIMD_H
17
18#include "pie_dsp.h"
19
20// The int32 kernels are only correct when SAMPLE is the s8.23 fixed-point type. In a
21// float build SAMPLE is `float`, so the vector paths must compile out entirely.
22#if PIE_DSP_S3_ENABLED && defined(AMY_USE_FIXEDPOINT)
23#define AMY_PIE_SIMD 1
24#else
25#define AMY_PIE_SIMD 0
26#endif
27
28// PIE-accelerated on ESP32-S3, libc memset/memcpy everywhere else. Safe in float builds
29// too: a byte-wise clear/copy does not care what the samples mean.
30#define AMY_BLOCK_BZERO(p, nbytes) PIE_BZERO((p), (nbytes))
31#define AMY_BLOCK_BCOPY(src, dst, nbytes) PIE_BCOPY((src), (dst), (nbytes))
32
33#endif // AMY_SIMD_H
