254// on all platforms, sysclock is based on total samples played, using audio out
255// (i2s or etc) as system clock 64-bit milliseconds since start. total_blocks is
256// u32 and increments once per AMY_BLOCK_SIZE samples, so this does not wrap for
257// ~219 years at 44.1 kHz. Anything that stores an absolute deadline and
258// compares it later must use this rather than amy_sysclock(); see
259// sequencer_check_and_fill().
260uint64_t amy_sysclock64() {
261 // Integer math: computing this through float quantizes the clock once
262 // total samples exceed the 24-bit mantissa (~6 min at 48 kHz), and the
263 // u32 samples-domain multiply wrapped after 2^32 samples (~25 h).
264 return ((uint64_t)amy_global.total_blocks * (AMY_BLOCK_SIZE * 1000u)) / AMY_SAMPLE_RATE;
265}
266
267uint32_t amy_sysclock() {
268 // Time is returned in integer milliseconds; wraps at 2^32 ms = 49.7 days.
269 // This is the wire/event-facing clock and stays 32-bit for compatibility.
270 // Consumers compare event times wrap-relative (AMY_TIME_GEQ), so the
271 // rollover is handled rather than avoided.
272 return (uint32_t)amy_sysclock64();
273}

276// Parse and play a stored (fired sequencer) wire message now.
277void amy_play_message(char *message) {
278 peek_stack("play_message");
279 amy_event e;
280 size_t pos = 0;
281 do {
282 amy_clear_event(&e);
283 pos = yield_event_from_message(message, &e, pos);
284 if (pos > 0) amy_add_event(&e);
285 } while(pos > 0);
286}

288// given a wire message string play / schedule the event directly (WIRE API)
289void amy_add_message_with_sysex_flag(char *message, bool sysex) {
290 peek_stack("add_message");
291 if (sysex
292 && (amy_global.transfer_flag == AMY_TRANSFER_TYPE_FILE
293 || amy_global.transfer_flag == AMY_TRANSFER_TYPE_AUDIO)) {
294 // Transfer status can't change mid-message, so the whole string is
295 // one chunk of transfer payload.
296 parse_transfer_message(message, (uint16_t)strlen(message));
297 return;
298 }
299 // Fast pre-check of this message for a leading 'H' (ticks) scheduling
300 // command, only recognized as the very first character of the message.
301 if (message[0] == 'H') {
302 handle_ticks_message(message);
303 } else {
304 // Not scheduled: parse and play every command in the message now.
305 amy_play_message(message);
306 }
307}

321// given an event play / schedule the event directly (C API)
322// diverts sequeuncer events, and handles special-case reset.
323void amy_add_event(amy_event *e) {
324 peek_stack("add_event");
325 // was amy_process_event
326 if(AMY_IS_SET(e->ticks[TICKS_TICK]) || AMY_IS_SET(e->ticks[TICKS_PERIOD]) || AMY_IS_SET(e->ticks[TICKS_TAG])) {
327 // C-API ticks event: serialize it to a wire message and hand it to
328 // the sequencer, so scheduled events have a single storage format.
329 char *buf = (char *)malloc_caps(MAX_MESSAGE_LEN, amy_global.config.ram_caps_sequencer);
330 // LOCAL EDIT (S3-Amysynth): wire strings are control-path data with
331 // their own caps (ram_caps_sequencer), so a big patch exhausting
332 // ram_caps_events (which also backs per-osc synth state) can't make
333 // scheduled events and CANCELS silently drop. Fall back to the events
334 // pool before giving up.
335 if (buf == NULL)
336 buf = (char *)malloc_caps(MAX_MESSAGE_LEN, amy_global.config.ram_caps_events);
337 if (buf == NULL) {
338 amy_oom("add_event ticks");
339 return;
340 }
341 sprint_event(e, buf, MAX_MESSAGE_LEN, /* wirecode */ true);
342 amy_add_message(buf);
343 free(buf);
344 } else if (AMY_IS_SET(e->reset_osc) && (e->reset_osc & RESET_PATCH) && AMY_IS_SET(e->patch_number)) {
345 // We're resetting just one patch, do it now. But RESET_PATCH with no patch_number should propagate to deltas.
346 patches_reset_patch(e->patch_number);
347 AMY_UNSET(e->reset_osc);
348 } else {
349 // e->time (if set) plus latency gives this delta's playback time --
350 // for near-term ordering (e.g. noteon_delay) and output-latency
351 // compensation, not for long-horizon scheduling; use ticks= for that.
352 uint32_t playback_time = amy_sysclock();
353 if(AMY_IS_SET(e->time)) playback_time = e->time;
354 playback_time += amy_global.latency_ms;
355 // UINT32_MAX is the "unset" sentinel for a u32 field, and the clock does
356 // land on it for one millisecond every 49.7 days. Nudge by 1ms so the
357 // event doesn't read back as having no time at all.
358 if(AMY_IS_UNSET(playback_time)) playback_time++;
359 e->time = playback_time;
360 amy_event_to_deltas_queue(e, 0, /* oscs_per_voice= */ 0, &amy_global.delta_queue);
361 }
362}

467// CPU overload failsafe: silence and reset the synth so the host stays responsive,
468// then play a descending bleep so the user knows AMY stopped on purpose.
469void amy_overload_failsafe() {
470 fprintf(stderr, "AMY: CPU overload (render %" PRIu32 " us > %" PRIu32 " threshold), resetting synth\n", amy_global.render_us, amy_global.overload_threshold_us);
471 // Drop everything scheduled first -- an overloaded delta queue can hold
472 // hundreds of pending events that would re-wedge us as they play out.
473 amy_grab_lock();
474 amy_deltas_reset();
475 amy_release_lock();
476 amy_event e = amy_default_event();
477 e.reset_osc = RESET_ALL_NOTES | RESET_ALL_OSCS | RESET_SEQUENCER;
478 amy_add_event(&e);
479 // "doot doot doot doot" -- a descending minor arpeggio, distinct from the
480 // startup bleep, so the user knows AMY hit its limit and stopped on
481 // purpose. Scheduled after the reset plays.
482 uint32_t start = amy_sysclock() + 50;
483 e = amy_default_event();
484 e.osc = AMY_OSCS - 1;
485 e.wave = SINE;
486 float doots[] = {880.00f, 659.26f, 523.25f, 440.00f}; // A5 E5 C5 A4
487 for (int i = 0; i < 4; i++) {
488 e.time = start + i * 160;
489 e.freq_coefs[COEF_CONST] = doots[i];
490 e.velocity = 1.0f;
491 amy_add_event(&e);
492 e.time += 110;
493 e.velocity = 0;
494 amy_add_event(&e);
495 }
496 if (amy_global.config.amy_external_overload_hook != NULL)
497 amy_global.config.amy_external_overload_hook(amy_get_render_load());
498 // Start the load measure over so we don't re-trip while recovering.
499 amy_global.render_us = 0;
500 amy_global.overload_count = 0;
501}
502
503// Called by platform render loops once per block: busy_us is the time spent
504// rendering the block. Keeps a smoothed load estimate in
505// amy_global.render_us and trips the failsafe when it stays pinned at/above
506// amy_global.overload_threshold_us for amy_global.overload_blocks blocks.
507void amy_overload_check(uint32_t render_us) {
508 amy_global.render_us = (uint32_t)(((int32_t)amy_global.render_us)
509 + ((((int32_t)render_us)
510 - ((int32_t)amy_global.render_us)) >> 5)); // 0.03125 * delta
511 if (amy_global.overload_threshold_us <= 0) return;
512 if (amy_global.render_us >= amy_global.overload_threshold_us) {
513 if (++amy_global.overload_count > amy_global.overload_blocks) {
514 amy_overload_failsafe();
515 }
516 } else {
517 amy_global.overload_count = 0;
518 }
519}

521void amy_set_render_load_threshold(float threshold) {
522 // Dynamically adjust the overload threshold.
523 amy_global.config.overload_threshold = threshold;
524 amy_global.overload_threshold_us = (uint32_t)(threshold * ((float)AMY_BLOCK_US));
525}

553// Tracks whether the allocations made by amy_start() are still live, so that
554// amy_stop() is idempotent: calling it twice in a row (e.g. a rejected live()
555// followed by another live(), or amy.stop() twice from Python) would otherwise
556// double-free the bus, filter, and osc arrays, which the deinit paths free
557// without NULLing.
558static int amy_started = 0; // Must be int; uint8_t or bool causes bootloop (?).
559
560void amy_start(amy_config_t c) {
561 amy_started = 1;
562 global_init(c);
563 amy_profiles_init();
564 transfer_init();
565 oscs_init();
566 amy_platform_init();
567 run_midi(); // Must be after platform_init in case F_CPU is modified on RP2040 Arduino.
568 if(AMY_HAS_DEFAULT_SYNTHS) amy_default_synths();
569 if(AMY_HAS_STARTUP_BLEEP) {
570 if(AMY_HAS_DEFAULT_SYNTHS)
571 amy_bleep_synth(0); // bleep using the default sinewave synth voice.
572 else
573 amy_bleep(0); // bleep using raw oscs.
574 }
575#if !defined(ESP_PLATFORM) && !defined(PICO_ON_DEVICE) && !defined(ARDUINO) && !defined(__EMSCRIPTEN__) && !defined(AMY_NO_MINIAUDIO)
576 if (amy_global.config.audio == AMY_AUDIO_IS_MINIAUDIO)
577 miniaudio_start();
578#endif
579}
580
581void amy_stop() {
582 if (!amy_started) return;
583 amy_started = 0;
584#if !defined(ESP_PLATFORM) && !defined(PICO_ON_DEVICE) && !defined(ARDUINO) && !defined(AMY_NO_MINIAUDIO)
585 if (amy_global.config.audio == AMY_AUDIO_IS_MINIAUDIO)
586 miniaudio_stop();
587#endif
588 stop_midi();
589 amy_platform_deinit();
590 oscs_deinit();
591 global_deinit();
592}
