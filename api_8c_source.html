213// on all platforms, sysclock is based on total samples played, using audio out (i2s or etc) as system clock
214uint32_t amy_sysclock() {
215 // LOCAL EDIT (S3-Amysynth): integer math. The float version had two bugs:
216 // the u32 samples-domain multiply (total_blocks * AMY_BLOCK_SIZE) wrapped
217 // at 2^32 samples (~24.9 h at 48 kHz, not the intended 49.7 days), and the
218 // 24-bit float mantissa quantized the clock as uptime grew (~2.7 ms steps
219 // after 12 h), making everything slaved to it - the sequencer above all -
220 // audibly lumpy. Integer math is exact and wraps like a plain u32
221 // millisecond counter (2^32 ms = 49.7 days). Upstream-PR candidate.
222 // Time is returned in integer milliseconds.
223 return (uint32_t)(((uint64_t)amy_global.total_blocks * (AMY_BLOCK_SIZE * 1000u)) / AMY_SAMPLE_RATE);
224}

227// Flag indicating whether the current amy_add_message call is from an
228// external sysex source (in which case transfer_flag should route data
229// to parse_transfer_message) or from an internal amy.send() call (in
230// which case it should be processed as a normal wire command).
231// Without this, a sketch's amy.send(note=36) during a file transfer
232// would get base64-decoded and written to the file as garbage.
233bool amy_parsing_from_sysex = false;
234
235// given a wire message string play / schedule the event directly (WIRE API)
236void amy_add_message(char *message) {
237 peek_stack("add_message");
238 amy_event e;
239 size_t pos = 0;
240 do {
241 amy_clear_event(&e);
242 pos = yield_event_from_message(message, &e, pos);
243 if (pos > 0) amy_add_event(&e);
244 } while(pos > 0);
245}
246
247// Like amy_add_message but marks the message as coming from an external
248// sysex source so the file transfer routing in amy_parse_message applies.
249void amy_add_message_from_sysex(char *message) {
250 amy_parsing_from_sysex = true;
251 amy_add_message(message);
252 amy_parsing_from_sysex = false;
253}

255// given an event play / schedule the event directly (C API)
256// diverts sequeuncer events, and handles special-case reset.
257void amy_add_event(amy_event *e) {
258 peek_stack("add_event");
259 // was amy_process_event
260 if(AMY_IS_SET(e->sequence[SEQUENCE_TICK]) || AMY_IS_SET(e->sequence[SEQUENCE_PERIOD]) || AMY_IS_SET(e->sequence[SEQUENCE_TAG])) {
261 uint8_t added = sequencer_add_event(e);
262 (void)added; // we don't need to do anything with this info at this time
263 e->status = EVENT_SEQUENCE;
264 } else if (AMY_IS_SET(e->reset_osc) && (e->reset_osc & RESET_PATCH) && AMY_IS_SET(e->patch_number)) {
265 // We're resetting just one patch, do it now. But RESET_PATCH with no patch_number should propagate to deltas.
266 patches_reset_patch(e->patch_number);
267 AMY_UNSET(e->reset_osc);
268 } else {
269 // if time is set, play then
270 // if time and latency is set, play in time + latency
271 // if time is not set, play now
272 // if time is not set + latency is set, play in latency
273 uint32_t playback_time = amy_sysclock();
274 if(AMY_IS_SET(e->time)) playback_time = e->time;
275 playback_time += amy_global.latency_ms;
276 e->time = playback_time;
277 e->status = EVENT_SCHEDULED;
278 amy_event_to_deltas_queue(e, 0, &amy_global.delta_queue);
279 }

317void amy_default_synths() {
318 // Configure several default synthesizers for "out of box" playability.
319
320 // sine wave "bleeper" on ch 0 (not a MIDI channel)
321 // synth 0 sinewave.
322 amy_event e = amy_default_event();
323 e.synth = 0;
324 e.num_voices = 1;
325 e.oscs_per_voice = 1;
326 e.wave = SINE;
327 amy_add_event(&e);
328
329 // GM drum synth on channel 10
330 e = amy_default_event();
331 e.synth = AMY_MIDI_CHANNEL_DRUMS; // 10
332 e.num_voices = 6;
333#ifdef GAMMA9001
334 e.patch_number = 384; // Gamma9001 drum kit 0 (baked TR-808 bank); kits 1+ at 385+ via PC bank MSB 3
335#else
336 e.patch_number = 258; // Set up in headers.py to use midi_note_cmd to match some midi note events to PCM samples
337#endif
338 e.synth_flags = SYNTH_FLAGS_NOTES_VIA_MIDI | SYNTH_FLAGS_IGNORE_NOTE_OFFS; // Ensure note events go via midi_note_cmd
339 amy_add_event(&e);
340
341 // DX7 6 note poly on channel 2
342 e = amy_default_event();
343 e.synth = 2;
344 e.num_voices = 6;
345 e.patch_number = 128;
346 amy_add_event(&e);
347
348 // Juno 6 poly on channel 1
349 // Define this last so if we release it, the oscs aren't fragmented.
350 e = amy_default_event();
351 e.synth = 1;
352 e.num_voices = 6;
353 e.patch_number = 0;
354 amy_add_event(&e);
355 // Add some MIDI CCs for the Juno (defined in midi_mappings.c).
356 amy_global.config.amy_external_midi_input_hook = juno_filter_midi_handler;
357}
