106/* Shared ascending bubble sort for collect helpers. */
107static void arp_sort_asc(uint8_t *out, uint8_t n)
108{
109 for (uint8_t i = 0; i + 1 < n; i++)
110 for (uint8_t j = 0; j + 1 < n - i; j++)
111 if (out[j] > out[j + 1]) { uint8_t t = out[j]; out[j] = out[j+1]; out[j+1] = t; }
112}

114/* UP mode: REST is a transparent skip — all non-empty notes are collected.
115 * Returns sorted ascending; caller uses sorted[note_idx] directly. */
116static uint8_t arp_collect_up(uint8_t out[ARP_MAX_SLOTS])
117{
118 uint8_t n = 0;
119 for (uint8_t i = 0; i < ARP_MAX_SLOTS; i++) {
120 if (s_arp.slots[i] == -1) break; /* empty sentinel: stop */
121 if (s_arp.slots[i] == ARP_REST) continue; /* REST in UP: skip */
122 out[n++] = (uint8_t)s_arp.slots[i];
123 }
124 arp_sort_asc(out, n);
125 return n;
126}
127
128/* DOWN mode: identical skip-REST logic to UP; caller reverses with
129 * sorted[count-1-note_idx]. */
130static uint8_t arp_collect_down(uint8_t out[ARP_MAX_SLOTS])
131{
132 return arp_collect_up(out);
133}

135/* Snap a chromatic note. Default: the arp's own scale/root. With follow_quant
136 * ON, use the global quantizer with the melodic-layer precedence: an active
137 * chord progression still wins (it owns the arp's root/scale), and a disabled
138 * global quantizer means no snapping at all. */
139static uint8_t arp_snap(uint8_t chromatic)
140{
141 uint8_t root = s_arp.root_note;
142 uint8_t scale_idx = s_arp.scale_index;
143 if (s_arp.follow_quant && !sequencer_core_progression_arp_owned()) {
144 if (!sequencer_core_get_quantizer_enabled()) {
145 return sequencer_core_clamp_melodic_note((int32_t)chromatic);
146 }
147 root = sequencer_core_get_quantizer_root_note();
148 scale_idx = sequencer_core_get_quantizer_scale();
149 }
150 const musical_scale_t *scale = quantizer_get_scale(scale_idx);
151 uint8_t snapped = quantizer_snap_midi_note(chromatic, root, scale);
152 return sequencer_core_clamp_melodic_note((int32_t)snapped);
153}

176/* Push the arp filter including the bipolar EG1->cutoff sweep depth
177 * (filter_env_amount, -8..+8 octaves, as on melodic rows). Mirrors
178 * melodic_filter_apply(): wire the coef only when the filter is enabled with a
179 * nonzero amount, and push EG1 breakpoints alongside it - AMY treats a
180 * never-configured breakpoint set as an always-open unity gate. */
181static void arp_apply_filter(const seq_filter_t *f)
182{
183 if (!f) return;
184 amy_event *e = amy_helpers_event_begin();
185 e->synth = sequencer_core_arp_synth();
186 if (f->enabled) {
187 e->filter_type = f->filter_type;
188 e->filter_freq_coefs[COEF_CONST] = f->cutoff_hz;
189 e->resonance = f->resonance;
190 if (f->filter_env_amount != 0.0f) {
191 e->filter_freq_coefs[COEF_EG1] = f->filter_env_amount;
192 }
193 } else {
194 e->filter_type = FILTER_NONE;
195 }
196 /* KS string decay from the authored feedback field; 0 = never authored,
197 * keep AMY's build-time 0.9 default. */
198 if (s_arp.patch == SEQ_PATCH_KS && f->feedback > 0.0f) {
199 e->feedback = SEQ_CLAMP_F32(f->feedback, 0.0f, 1.0f);
200 }
201 amy_helpers_event_send(e);
202
203 if (f->enabled && f->filter_env_amount != 0.0f) {
204 sequencer_core_push_envelope_eg1(sequencer_core_arp_synth(), 0,
205 &s_arp.vp.env1);
206 }
207}

209/* Recompute and push the LFO carrier frequency at the current BPM. Called by
210 * sequencer_core_set_bpm() after s_bpm updates. Must NOT be called from the
211 * render body - amy_queue_lock is held there; set_bpm() runs on the UI task. */
212void arp_core_refresh_lfo_freq(void)
213{
214 if (!s_arp.vp.lfo_authored || !s_arp.vp.lfo.enabled)
215 return;
216 /* Only patches with a reserved carrier pair (wave numbers / wavetables /
217 * bass presets) run the native LFO; the software stepper tracks BPM
218 * per-step on its own. */
219 uint8_t carrier;
220 if (!sequencer_core_lfo_native_layout(s_arp.patch, &carrier, NULL))
221 return;
222
223 amy_event *e = amy_helpers_event_begin();
224 e->synth = sequencer_core_arp_synth();
225 e->osc = carrier;
226 e->freq_coefs[COEF_CONST] = lfo_rate_to_hz(s_arp.vp.lfo.rate,
227 sequencer_core_get_bpm());
228 amy_helpers_event_send(e);
229
230 /* Keep the wobble modulator (carrier+1) BPM-synced as well. */
231 e = amy_helpers_event_begin();
232 e->synth = sequencer_core_arp_synth();
233 e->osc = (uint8_t)(carrier + 1u);
234 e->freq_coefs[COEF_CONST] = lfo_rate_to_hz((lfo_rate_t)s_arp.vp.lfo.wob_rate,
235 sequencer_core_get_bpm());
236 amy_helpers_event_send(e);
237}

250/* (Re)build the arp synth slot for the current patch and params, then
251 * re-impose any authored ADSR / filter. Mirrors drone_rebuild().
252 *
253 * Scheduled events are cleared FIRST and re-emitted afterwards (arp_mark_dirty
254 * at the end): AMY's 500 µs sequencer poll runs on its own task and fires arp
255 * note-ons autonomously, so a live schedule during a patch rebuild lets a
256 * note-on resolve against half-updated voice/osc tables. That strands an osc
257 * AUDIBLE outside any voice, unreachable by later kills (which resolve through
258 * the CURRENT voice map) - permanent ringing when scrolling arp patches.
259 * Cost: the arp is quiet for at most one UI frame around a rebuild. */
260static void arp_rebuild(void)
261{
262 arp_core_clear_all();
263 arp_kill_voices();
264
265 sequencer_core_arp_configure(s_arp.patch, sequencer_core_arp_voices(),
266 s_arp.vp.filter_authored, s_arp.vp.filter.feedback);
267 /* Raw wave/wavetable patches have no built-in EG0; always push the
268 * envelope so notes decay. Juno/DX7 strings and bass/FM presets carry
269 * their own envelope as part of their character: only override when
270 * the user authored one. */
271 if (sequencer_core_is_wave_patch(s_arp.patch) || s_arp.vp.env_authored) {
272 sequencer_core_push_envelope(sequencer_core_arp_synth(), &s_arp.vp.env);
273 }
274 /* Patches with a reserved carrier pair (wave numbers, bass presets)
275 * take the native LFO here; the software stepper excludes them. */
276 uint8_t carrier, coupled;
277 if (sequencer_core_lfo_native_layout(s_arp.patch, &carrier, &coupled)) {
278 bool lfo_on = s_arp.vp.lfo_authored && s_arp.vp.lfo.enabled;
279 voice_apply_native_lfo_topo(sequencer_core_arp_synth(),
280 lfo_on ? &s_arp.vp.lfo : NULL,
281 sequencer_core_get_bpm(),
282 carrier, coupled);
283 }
284 /* arp_apply_filter also pushes the EG1 breakpoints whenever it wires the
285 * sweep coef. */
286 if (s_arp.vp.filter_authored) {
287 arp_apply_filter(&s_arp.vp.filter);
288 }
289 /* EG1: push only when authored. Otherwise an instrument that routes its
290 * own bp1 keeps its patch-string values. */
291 if (s_arp.vp.env1_authored) {
292 sequencer_core_push_envelope_eg1(sequencer_core_arp_synth(), 0, &s_arp.vp.env1);
293 }
294 /* Distortion is per-osc state the reconfigure above cleared - reassert. */
295 arp_reapply_dist();
296 /* Any reconfigure above resets AMY's per-osc portamento_alpha to 0 -
297 * reassert. */
298 arp_push_portamento();
299
300 /* Re-emit the schedule cleared at the top (coalesced onto the next
301 * arp_core_service() frame; no-op while disabled). */
302 arp_mark_dirty();
303}

307void arp_core_init(void)
308{
309 memset(&s_arp, 0, sizeof(s_arp));
310 voice_params_init_defaults(&s_arp.vp); /* unauthored, amp trim at unity */
311 /* Seed the bipolar EG1->cutoff sweep so an enabled arp filter has its
312 * plucky-sweep character; the graph editor's EG1 page edits it
313 * (-8..+8 oct, 0 = no sweep). */
314 s_arp.vp.filter.filter_env_amount = ARP_FILTER_EG1_DEPTH_OCT;
315 s_arp.enabled = CONFIG_SEQ_ARP_DEFAULT_ENABLED;
316 s_arp.dir = ARP_UP;
317 s_arp.octaves = CONFIG_SEQ_ARP_DEFAULT_OCTAVES;
318 s_arp.rate = ARP_RATE_1_16;
319 s_arp.gate_pct = CONFIG_SEQ_ARP_DEFAULT_GATE_PCT;
320 s_arp.scale_index = CONFIG_SEQ_ARP_DEFAULT_SCALE;
321 s_arp.root_note = CONFIG_SEQ_ARP_DEFAULT_ROOT_NOTE;
322 s_arp.patch = CONFIG_SEQ_ARP_DEFAULT_PATCH;
323 /* Default ADSR mirrors the melodic compile-time defaults; unauthored until
324 * the user commits in the graph editor (the patch's own env wins). */
325 s_arp.vp.env.attack_ms = 4; /* tiny curve, prevents a digital click */
326 s_arp.vp.env.decay_ms = 250;
327 s_arp.vp.env.sustain_pct = 30; /* low sustain keeps the sequence energetic */
328 s_arp.vp.env.release_ms = 200;
329 s_arp.vp.env.eg_type = 0; /* ENVELOPE_NORMAL */
330 /* EG1 is deliberately slower than the amp env: the classic "plucky amp
331 * decay, slower filter tail" shape. */
332 s_arp.vp.env1.attack_ms = 15;
333 s_arp.vp.env1.decay_ms = 400;
334 s_arp.vp.env1.sustain_pct = 20;
335 s_arp.vp.env1.release_ms = 300;
336 s_arp.vp.env1.eg_type = 0; /* ENVELOPE_NORMAL */
337 /* Default filter: bypass (unauthored; patch's filter wins until commit). */
338 s_arp.vp.filter.filter_type = 0; /* SEQ_FILTER_NONE */
339 s_arp.vp.filter.cutoff_hz = 800.0f;
340 s_arp.vp.filter.resonance = 1.0f;
341 /* Default LFO: disabled (bypass until the user commits). */
342 s_arp.vp.lfo.wave = LFO_WAVE_SINE;
343 s_arp.vp.lfo.rate = LFO_RATE_1BAR;
344 s_arp.vp.lfo.depth = 50;
345 s_arp.vp.lfo.targets = LFO_TGT_BIT(LFO_TARGET_FILTER);
346 s_arp.octaves = SEQ_CLAMP_U8(s_arp.octaves, 1, ARP_OCT_MAX);
347 if (s_arp.scale_index >= quantizer_scale_count()) s_arp.scale_index = 0;
348 for (uint8_t i = 0; i < ARP_MAX_SLOTS; i++) s_arp.slots[i] = -1;
349
350 DIAG_HEAP_CHECK("arp_init: before arp_configure");
351 arp_rebuild();
352 DIAG_HEAP_CHECK("arp_init: after arp_configure");
353 /* If the arp boots enabled, let the first service tick emit rather than
354 * emitting inline during init. */
355 arp_mark_dirty();
356 ESP_LOGI(TAG, "arp_core initialized (synth %u)", sequencer_core_arp_synth());
357}

359void arp_core_refresh(void)
360{
361 arp_core_clear_all();
362 /* The clear above deleted the scheduled note-offs of anything still
363 * sounding - silence them now or they hang forever. On disable this is the
364 * only note-off they will ever get. */
365 arp_kill_voices();
366
367 if (!s_arp.enabled || s_arp.solo_muted) {
368 return;
369 }
370
371 /* ── SLOT mode: play slots in stored order, rests preserved ── */
372 if (s_arp.dir == ARP_SLOT) {
373 uint8_t active = 0;
374 for (uint8_t i = 0; i < ARP_MAX_SLOTS; i++)
375 if (s_arp.slots[i] != -1) active++;
376 if (active == 0) return;
377
378 uint32_t rate = s_rate_ticks[s_arp.rate];
379 uint8_t steps = (uint8_t)(active * s_arp.octaves);
380 uint32_t period = (uint32_t)steps * rate;
381 uint32_t gate = (rate * s_arp.gate_pct) / 100u;
382 if (gate < 1) gate = 1;
383
384 uint32_t tag_base = sequencer_core_arp_tag_base();
385 uint8_t step_i = 0;
386
387 /* Per-target amp trim; clamp so velocity stays <=1 (AMY cap). */
388 float arp_vel = 0.9f * s_arp.vp.amp_trim;
389 if (arp_vel > 1.0f) arp_vel = 1.0f;
390
391 for (uint8_t oct = 0; oct < s_arp.octaves; oct++) {
392 for (uint8_t si = 0; si < ARP_MAX_SLOTS; si++) {
393 int16_t v = s_arp.slots[si];
394 if (v == -1) continue; /* unused: not part of the cycle */
395
396 uint32_t tick_on = 1u + (uint32_t)step_i * rate;
397
398 if (v != ARP_REST) {
399 int32_t chromatic = (int32_t)v + (int32_t)oct * 12;
400 uint8_t play_note = arp_snap(sequencer_core_clamp_melodic_note(chromatic));
401 sequencer_core_arp_emit_note(tag_base + (uint32_t)step_i * 2u,
402 play_note, arp_vel, tick_on, gate, period);
403 }
404 /* REST: tag stays cleared (arp_clear_all already ran); step_i still advances. */
405 step_i++;
406 }
407 }
408 return;
409 }
410
411 /* ── UP / DOWN: sorted pitch order ── */
412 uint8_t sorted[ARP_MAX_SLOTS];
413 uint8_t count = (s_arp.dir == ARP_DOWN)
414 ? arp_collect_down(sorted)
415 : arp_collect_up(sorted);
416 if (count == 0) {
417 return; /* nothing to play */
418 }
419
420 uint32_t rate = s_rate_ticks[s_arp.rate];
421 uint8_t steps = (uint8_t)(count * s_arp.octaves);
422 uint32_t period = (uint32_t)steps * rate;
423 uint32_t gate = (rate * s_arp.gate_pct) / 100u;
424 if (gate < 1) gate = 1;
425
426 uint32_t tag_base = sequencer_core_arp_tag_base();
427
428 /* Apply per-target amp trim for UP/DOWN modes. */
429 float arp_vel = 0.9f * s_arp.vp.amp_trim;
430 if (arp_vel > 1.0f) arp_vel = 1.0f;
431
432 for (uint8_t i = 0; i < steps; i++) {
433 /* index within the sorted set, advancing by octave every `count`. */
434 uint8_t note_idx = i % count;
435 uint8_t octave = i / count;
436 /* Direction: UP walks sorted ascending; DOWN walks it in reverse. */
437 uint8_t pick = (s_arp.dir == ARP_DOWN)
438 ? sorted[count - 1 - note_idx]
439 : sorted[note_idx];
440 int32_t chromatic = (int32_t)pick + (int32_t)octave * 12;
441 uint8_t play_note = arp_snap(sequencer_core_clamp_melodic_note(chromatic));
442
443 uint32_t tick_on = 1u + (uint32_t)i * rate;
444 sequencer_core_arp_emit_note(tag_base + (uint32_t)i * 2u,
445 play_note, arp_vel, tick_on, gate, period);
446 }
447}

711/* ── Software LFO fallback ───────────────────────────────────────────────
712 * Patches with a reserved carrier pair get the AMY-native voice-local LFO
713 * (arp_rebuild). Every other patch owns its whole osc layout, so it runs the
714 * same 20 Hz software stepper as non-wave melodic tracks (canonical impl:
715 * sequencer_core_lfo_service), modulating each checked target's COEF_CONST
716 * rail. WOBBLE has no software analog and is ignored, as on melodic patch
717 * tracks. */
718static float s_swlfo_phase = 0.0f;
719static float s_swlfo_rnd = 0.0f;
720static bool s_swlfo_active = false;
721static uint8_t s_swlfo_targets = 0; /* targets driven while active - the set
722 * to restore, even if edited since */
723
724/* seq_core_editors.c internals shared with this stepper. Mirrored prototypes:
725 * seq_core_internal.h cannot be included here (it defines a TU-local TAG). */
726float lfo_next_rand(void);
727void lfo_push_target_neutral(uint8_t synth_id, lfo_target_t target);
728
729/* lfo_rate_to_hz capped to the stepper's usable band - mirrors seq_lfo_sw_hz
730 * (seq_core_internal.h); needs >= 4 stepper samples per LFO cycle. */
731static inline float arp_swlfo_hz(lfo_rate_t rate, uint16_t bpm)
732{
733 float hz = lfo_rate_to_hz(rate, bpm);
734 return (hz > SEQ_LFO_SW_MAX_HZ) ? SEQ_LFO_SW_MAX_HZ : hz;
735}
736
737static float arp_swlfo_eval(lfo_wave_t wave, float ph)
738{
739 switch (wave) {
740 case LFO_WAVE_SINE: return sinf(2.0f * 3.14159265f * ph);
741 case LFO_WAVE_TRIANGLE: return (ph < 0.5f) ? (4.0f * ph - 1.0f)
742 : (3.0f - 4.0f * ph);
743 case LFO_WAVE_SAW_UP: return 2.0f * ph - 1.0f;
744 case LFO_WAVE_SAW_DOWN: return 1.0f - 2.0f * ph;
745 case LFO_WAVE_SQUARE: return (ph < 0.5f) ? 1.0f : -1.0f;
746 case LFO_WAVE_RANDOM: return s_swlfo_rnd;
747 default: return 0.0f;
748 }
749}
750
751static void arp_swlfo_service(void)
752{
753 const seq_lfo_t *lfo = &s_arp.vp.lfo;
754 /* Only patches with NO reserved carrier pair - wave numbers and bass
755 * presets are served natively by arp_rebuild, so stepping them here would
756 * double-modulate. Distortion now has COEF_MOD rails too (dist_*_coefs),
757 * so it rides the carrier on native patches like every other target and
758 * the stepper is fully off there (mirrors sequencer_core_lfo_service). */
759 bool native = sequencer_core_lfo_native_layout(s_arp.patch, NULL, NULL);
760 bool want = s_arp.enabled && lfo->enabled && lfo->targets != 0 && !native;
761
762 if (!want) {
763 if (s_swlfo_active) {
764 /* Restore every rail the stepper was driving (mirrors
765 * lfo_restore_target_neutrals in seq_core_editors.c). */
766 s_swlfo_active = false;
767 uint8_t syn = sequencer_core_arp_synth();
768 for (int t = 0; t < LFO_TARGET_COUNT; t++) {
769 if (!(s_swlfo_targets & LFO_TGT_BIT(t))) continue;
770 if (t == LFO_TARGET_FILTER)
771 sequencer_core_push_filter(syn, &s_arp.vp.filter,
772 s_arp.patch == SEQ_PATCH_KS);
773 else if (t == LFO_TARGET_DIST_DRIVE || t == LFO_TARGET_DIST_MIX) {
774 /* One restore event covers both dist bits. */
775 if (t == LFO_TARGET_DIST_DRIVE ||
776 !(s_swlfo_targets & LFO_TGT_BIT(LFO_TARGET_DIST_DRIVE)))
777 voice_apply_dist(syn, &s_arp.vp.dist);
778 }
779 else
780 lfo_push_target_neutral(syn, (lfo_target_t)t);
781 }
782 }
783 return;
784 }
785
786 if (!s_swlfo_active) {
787 s_swlfo_active = true;
788 s_swlfo_phase = 0.0f;
789 }
790 /* want requires !native, so the stepper only ever runs on PATCH-mode
791 * tracks - record the full target set for the deactivation restore. */
792 s_swlfo_targets = lfo->targets;
793
794 float ph = s_swlfo_phase +
795 arp_swlfo_hz(lfo->rate, sequencer_core_get_bpm()) * 0.05f;
796 if (ph >= 1.0f) {
797 ph -= 1.0f;
798 if (lfo->wave == LFO_WAVE_RANDOM) s_swlfo_rnd = lfo_next_rand();
799 }
800 s_swlfo_phase = ph;
801
802 float val = arp_swlfo_eval(lfo->wave, ph);
803 float d = (float)lfo->depth / 100.0f;
804
805 if (!native) {
806 amy_event *e = amy_helpers_event_begin();
807 e->synth = sequencer_core_arp_synth();
808 if (LFO_HAS_TGT(lfo, LFO_TARGET_FILTER)) {
809 float base = (s_arp.vp.filter.enabled && s_arp.vp.filter.cutoff_hz > 0.0f)
810 ? s_arp.vp.filter.cutoff_hz : 1000.0f;
811 e->filter_freq_coefs[COEF_CONST] =
812 base * powf(2.0f, voice_lfo_filter_octaves(lfo) * val);
813 }
814 if (LFO_HAS_TGT(lfo, LFO_TARGET_AMP))
815 e->amp_coefs[COEF_CONST] = 1.0f - d * (0.5f - 0.5f * val);
816 if (LFO_HAS_TGT(lfo, LFO_TARGET_PAN))
817 e->pan_coefs[COEF_CONST] = 0.5f + d * 0.5f * val;
818 /* SCAN needs a wavetable voice - WAVE mode / native only. */
819 amy_helpers_event_send(e);
820 } /* !native */
821
822 /* DIST targets on a PATCH-mode arp (no carrier): step around the committed
823 * dist block, same law as the native rail (voice_push_dist_lfo). Native
824 * patches drive it via COEF_MOD, so !native excludes them like the rest. */
825 if (!native && (lfo->targets & LFO_TGT_DIST_MASK))
826 voice_push_dist_lfo(sequencer_core_arp_synth(), &s_arp.vp.dist, lfo, val);
827
828 if (!native && LFO_HAS_TGT(lfo, LFO_TARGET_PITCH)) {
829 /* Absolute Hz: see SEQ_LFO_PITCH_BASE_HZ - a bare ratio here lands
830 * ~8.8 octaves down. Osc 0 only: a synth-wide push would rewrite
831 * patch-internal modulator oscs' rates (stopgap - revisit, see the
832 * sequencer stepper). */
833 amy_event *pe = amy_helpers_event_begin();
834 pe->synth = sequencer_core_arp_synth();
835 pe->osc = 0;
836 pe->freq_coefs[COEF_CONST] = SEQ_LFO_PITCH_BASE_HZ * powf(2.0f, d * VOICE_LFO_DEPTH_PITCH * val);
837 amy_helpers_event_send(pe);
838 }
839}
