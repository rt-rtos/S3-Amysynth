110/* Refresh coalescing: setters mark the arp dirty instead of re-emitting the
111 * whole sequence inline. arp_core_service() (called once per UI frame) performs
112 * a single re-emit when dirty, so a fast encoder spin through octaves/rate/gate
113 * collapses many setter calls into at most one full re-emit per frame instead of
114 * one per detent. Each re-emit clears + re-schedules up to ARP_MAX_STEPS events,
115 * every one through the shared AMY event mutex, so this removes a real burst of
116 * mutex traffic from Core 0 during edits. */
117static volatile bool s_arp_dirty = false;

127/* Shared ascending bubble sort for collect helpers. */
128static void arp_sort_asc(uint8_t *out, uint8_t n)
129{
130 for (uint8_t i = 0; i + 1 < n; i++)
131 for (uint8_t j = 0; j + 1 < n - i; j++)
132 if (out[j] > out[j + 1]) { uint8_t t = out[j]; out[j] = out[j+1]; out[j+1] = t; }
133}
134
135/* UP mode: REST is a transparent skip — all non-empty notes are collected.
136 * Returns sorted ascending; caller uses sorted[note_idx] directly. */
137static uint8_t arp_collect_up(uint8_t out[ARP_MAX_SLOTS])
138{
139 uint8_t n = 0;
140 for (uint8_t i = 0; i < ARP_MAX_SLOTS; i++) {
141 if (s_arp.slots[i] == -1) break; /* empty sentinel: stop */
142 if (s_arp.slots[i] == ARP_REST) continue; /* REST in UP: skip */
143 out[n++] = (uint8_t)s_arp.slots[i];
144 }
145 arp_sort_asc(out, n);
146 return n;
147}
148
149/* DOWN mode: identical skip-REST logic to UP; caller reverses with
150 * sorted[count-1-note_idx]. */
151static uint8_t arp_collect_down(uint8_t out[ARP_MAX_SLOTS])
152{
153 return arp_collect_up(out);
154}

175/* Configure the arp's AMY synth slot as a bare oscillator (WAVE mode).
176 *
177 * Shared 2-osc skeleton (voice_build_wave): osc0 pitch follows the MIDI note
178 * delivered by sequencer_core_arp_emit_note() (COEF_NOTE=1), amplitude is
179 * velocity-scaled + EG0-gated so the shared ADSR editor and custom envelopes
180 * work exactly as they do in PATCH mode; osc1 is the native LFO carrier.
181 *
182 * The caller is responsible for pushing an EG0 envelope afterwards (arp_rebuild
183 * always does this in WAVE mode so even the default env is applied). */
184static void arp_configure_wave_synth(void)
185{
186 uint8_t synth = sequencer_core_arp_synth();
187 uint8_t voices = sequencer_core_arp_voices();
188
189 /* Shared skeleton: pool (2 oscs/voice — osc 1 is the AMY-native LFO
190 * carrier, dormant when no LFO is authored; always allocating it avoids a
191 * pool reset when the LFO is toggled later) + osc0 note-following carrier
192 * with velocity + EG0 amplitude. */
193 voice_wave_cfg_t cfg = {
194 .synth = synth,
195 .num_voices = voices,
196 .oscs_per_voice = 2,
197 .wave = s_arp.wave,
198 .osc0_amp_const = 1.0f,
199 .osc0_amp_vel = 1.0f,
200 .ks_feedback_authored = s_arp.vp.filter_authored,
201 .ks_feedback_q = s_arp.vp.filter.resonance,
202 .wt_preset = -1,
203 };
204 voice_build_wave(&cfg);
205
206 /* Plucky amp (EG0 in the skeleton) / slower independent filter sweep
207 * (EG1): only wired once the user has authored+enabled a filter, so a
208 * stock arp voice with no filter is unaffected. arp_rebuild() pushes valid
209 * EG1 breakpoints alongside this so the coef never reads a stuck 1.0 (AMY
210 * treats a never-configured breakpoint set as a permanent unity gate). */
211 if (s_arp.vp.filter_authored && s_arp.vp.filter.enabled) {
212 amy_event *e = amy_helpers_event_begin();
213 e->synth = synth;
214 e->osc = 0;
215 e->filter_freq_coefs[COEF_EG1] = ARP_FILTER_EG1_DEPTH_OCT;
216 amy_helpers_event_send(e);
217 }
218
219 /* Native LFO routing (shared applier): active => osc0 mod coupling +
220 * osc1 carrier; inactive => coupling cleared, carrier dormant. */
221 bool lfo_on = s_arp.vp.lfo_authored && s_arp.vp.lfo.enabled;
222 voice_apply_native_lfo(synth, lfo_on ? &s_arp.vp.lfo : NULL,
223 sequencer_core_get_bpm());
224}
225
226/* Recompute and push the LFO carrier frequency at the current BPM.
227 * Called by sequencer_core_set_bpm() after s_bpm is updated so the carrier
228 * stays in sync when the user changes tempo. Must NOT be called from the
229 * render body (amy_queue_lock is held there); sequencer_core_set_bpm() runs
230 * from the UI task, which is safe. */
231void arp_core_refresh_lfo_freq(void)
232{
233 if (!s_arp.vp.lfo_authored || !s_arp.vp.lfo.enabled || s_arp.source != ARP_SRC_WAVE)
234 return;
235
236 amy_event *e = amy_helpers_event_begin();
237 e->synth = sequencer_core_arp_synth();
238 e->osc = 1;
239 e->freq_coefs[COEF_CONST] = lfo_rate_to_hz(s_arp.vp.lfo.rate,
240 sequencer_core_get_bpm());
241 amy_helpers_event_send(e);
242}
243
244/* Push the current glide time straight to the arp synth. e->osc is left unset
245 * and e->velocity is unset (not a note on/off), so patches_event_has_voices()
246 * fans this out to every voice's base osc — see amy.c/patches.c dispatch. */
247static void arp_push_portamento(void)
248{
249 amy_event *e = amy_helpers_event_begin();
250 e->synth = sequencer_core_arp_synth();
251 e->portamento_ms = s_arp.portamento_ms;
252 amy_helpers_event_send(e);
253}

255/* (Re)build the arp synth slot for the current source and params, then
256 * re-impose any authored ADSR / filter. Mirrors drone_rebuild(). */
257static void arp_rebuild(void)
258{
259 if (s_arp.source == ARP_SRC_WAVE) {
260 arp_configure_wave_synth();
261 /* WAVE mode has no patch envelope; always push the arp's env (authored
262 * or default) so EG0 breakpoints are valid and notes decay correctly. */
263 seq_env_t env_to_push = s_arp.vp.env;
264 voice_env_apply_ks_noise_floor(&env_to_push,
265 s_arp.wave == KS, s_arp.wave == NOISE);
266 sequencer_core_push_envelope(sequencer_core_arp_synth(), &env_to_push);
267 } else {
268 sequencer_core_arp_configure(s_arp.patch, sequencer_core_arp_voices(),
269 s_arp.vp.filter_authored, s_arp.vp.filter.resonance);
270 /* Raw wave/wavetable patches have no built-in EG0; always push the
271 * envelope so notes decay. Juno/DX7 patch strings AND the bass/FM
272 * presets carry their own envelopes (they're part of the preset's
273 * character): only push over those when the user has authored one. */
274 if (sequencer_core_is_wave_patch(s_arp.patch) || s_arp.vp.env_authored) {
275 sequencer_core_push_envelope(sequencer_core_arp_synth(), &s_arp.vp.env);
276 }
277 }
278 /* Filter re-apply is source-agnostic: both WAVE and PATCH respect it. */
279 if (s_arp.vp.filter_authored) {
280 sequencer_core_push_filter(sequencer_core_arp_synth(), &s_arp.vp.filter,
281 s_arp.wave == KS);
282 }
283 /* EG1 (independent second envelope): push whenever the user has authored
284 * one directly, OR whenever WAVE mode just wired filter_freq_coefs[COEF_EG1]
285 * above (that coef must never read a stuck permanent 1.0). PATCH mode with
286 * no authored EG1 is left alone — if the loaded patch already routes its
287 * own bp1, its own patch-string values keep driving it. */
288 bool wave_filter_eg1 = (s_arp.source == ARP_SRC_WAVE) &&
289 s_arp.vp.filter_authored && s_arp.vp.filter.enabled;
290 if (s_arp.vp.env1_authored || wave_filter_eg1) {
291 sequencer_core_push_envelope_eg1(sequencer_core_arp_synth(), 0, &s_arp.vp.env1);
292 }
293 /* Any reconfigure above (patch load or WAVE pool re-init) resets AMY's
294 * per-osc portamento_alpha to 0 — reassert regardless of source. */
295 arp_push_portamento();
296}

300void arp_core_init(void)
301{
302 memset(&s_arp, 0, sizeof(s_arp));
303 voice_params_init_defaults(&s_arp.vp); /* unauthored, amp trim at unity */
304 s_arp.enabled = CONFIG_SEQ_ARP_DEFAULT_ENABLED;
305 s_arp.dir = ARP_UP;
306 s_arp.octaves = CONFIG_SEQ_ARP_DEFAULT_OCTAVES;
307 s_arp.rate = ARP_RATE_1_16;
308 s_arp.gate_pct = CONFIG_SEQ_ARP_DEFAULT_GATE_PCT;
309 s_arp.scale_index = CONFIG_SEQ_ARP_DEFAULT_SCALE;
310 s_arp.root_note = CONFIG_SEQ_ARP_DEFAULT_ROOT_NOTE;
311 s_arp.patch = CONFIG_SEQ_ARP_DEFAULT_PATCH;
312 s_arp.source = ARP_SRC_PATCH; /* preserve existing PATCH behaviour */
313 s_arp.wave = ARP_DEFAULT_WAVE; /* SAW_DOWN — sensible WAVE default */
314 /* Default ADSR mirrors the melodic compile-time defaults; not authored until
315 * the user commits in the graph editor (patch's own env wins until then). */
316 s_arp.vp.env.attack_ms = 4; /* tiny curve to prevent a digital click */
317 s_arp.vp.env.decay_ms = 250; /* medium-short decay lets the body breathe */
318 s_arp.vp.env.sustain_pct = 30; /* low sustain keeps the sequence energetic */
319 s_arp.vp.env.release_ms = 200; /* controlled tail, no muddy low-end */
320 s_arp.vp.env.eg_type = 0; /* ENVELOPE_NORMAL */
321 /* Second envelope (EG1): slower than the amp env above — the classic
322 * "plucky amp decay, slower filter tail" shape once the filter is
323 * authored+enabled in WAVE mode (see arp_configure_wave_synth()). */
324 s_arp.vp.env1.attack_ms = 15;
325 s_arp.vp.env1.decay_ms = 400;
326 s_arp.vp.env1.sustain_pct = 20;
327 s_arp.vp.env1.release_ms = 300;
328 s_arp.vp.env1.eg_type = 0; /* ENVELOPE_NORMAL */
329 /* Default filter: bypass (unauthored; patch's filter wins until commit). */
330 s_arp.vp.filter.filter_type = 0; /* SEQ_FILTER_NONE */
331 s_arp.vp.filter.cutoff_hz = 800.0f;
332 s_arp.vp.filter.resonance = 1.0f;
333 /* Default LFO: disabled (bypass until the user commits). */
334 s_arp.vp.lfo.wave = LFO_WAVE_SINE;
335 s_arp.vp.lfo.rate = LFO_RATE_1BAR;
336 s_arp.vp.lfo.depth = 50;
337 s_arp.vp.lfo.targets = LFO_TGT_BIT(LFO_TARGET_FILTER);
338 if (s_arp.octaves < 1) s_arp.octaves = 1;
339 if (s_arp.octaves > ARP_OCT_MAX) s_arp.octaves = ARP_OCT_MAX;
340 if (s_arp.scale_index >= quantizer_scale_count()) s_arp.scale_index = 0;
341 for (uint8_t i = 0; i < ARP_MAX_SLOTS; i++) s_arp.slots[i] = -1;
342
343 ARP_HEAP_CHECK("arp_init: before arp_configure");
344 arp_rebuild();
345 ARP_HEAP_CHECK("arp_init: after arp_configure");
346 /* If the arp boots enabled (with seeded slots), schedule an initial emit on
347 * the first service tick rather than emitting inline during init. */
348 arp_mark_dirty();
349 ESP_LOGI(TAG, "arp_core initialized (synth %u)", sequencer_core_arp_synth());
350}

352void arp_core_refresh(void)
353{
354 arp_core_clear_all();
355
356 if (!s_arp.enabled) {
357 return;
358 }
359
360 /* ── SLOT mode: play slots in stored order, rests preserved ── */
361 if (s_arp.dir == ARP_SLOT) {
362 uint8_t active = 0;
363 for (uint8_t i = 0; i < ARP_MAX_SLOTS; i++)
364 if (s_arp.slots[i] != -1) active++;
365 if (active == 0) return;
366
367 uint32_t rate = s_rate_ticks[s_arp.rate];
368 uint8_t steps = (uint8_t)(active * s_arp.octaves);
369 uint32_t period = (uint32_t)steps * rate;
370 uint32_t gate = (rate * s_arp.gate_pct) / 100u;
371 if (gate < 1) gate = 1;
372
373 uint32_t tag_base = sequencer_core_arp_tag_base();
374 uint8_t step_i = 0;
375
376 /* Apply per-target amp trim; clamp so velocity stays ≤1 (AMY cap). */
377 float arp_vel = 0.9f * s_arp.vp.amp_trim;
378 if (arp_vel > 1.0f) arp_vel = 1.0f;
379
380 for (uint8_t oct = 0; oct < s_arp.octaves; oct++) {
381 for (uint8_t si = 0; si < ARP_MAX_SLOTS; si++) {
382 int16_t v = s_arp.slots[si];
383 if (v == -1) continue; /* unused: not part of the cycle */
384
385 uint32_t tick_on = 1u + (uint32_t)step_i * rate;
386
387 if (v != ARP_REST) {
388 int32_t chromatic = (int32_t)v + (int32_t)oct * 12;
389 uint8_t play_note = arp_snap(sequencer_core_clamp_melodic_note(chromatic));
390 sequencer_core_arp_emit_note(tag_base + (uint32_t)step_i * 2u,
391 play_note, arp_vel, tick_on, gate, period);
392 }
393 /* REST: tag stays cleared (arp_clear_all already ran); step_i still advances. */
394 step_i++;
395 }
396 }
397 return;
398 }
399
400 /* ── UP / DOWN: sorted pitch order ── */
401 uint8_t sorted[ARP_MAX_SLOTS];
402 uint8_t count = (s_arp.dir == ARP_DOWN)
403 ? arp_collect_down(sorted)
404 : arp_collect_up(sorted);
405 if (count == 0) {
406 return; /* nothing to play */
407 }
408
409 uint32_t rate = s_rate_ticks[s_arp.rate];
410 uint8_t steps = (uint8_t)(count * s_arp.octaves);
411 uint32_t period = (uint32_t)steps * rate;
412 uint32_t gate = (rate * s_arp.gate_pct) / 100u;
413 if (gate < 1) gate = 1;
414
415 uint32_t tag_base = sequencer_core_arp_tag_base();
416
417 /* Apply per-target amp trim for UP/DOWN modes. */
418 float arp_vel = 0.9f * s_arp.vp.amp_trim;
419 if (arp_vel > 1.0f) arp_vel = 1.0f;
420
421 for (uint8_t i = 0; i < steps; i++) {
422 /* index within the sorted set, advancing by octave every `count`. */
423 uint8_t note_idx = i % count;
424 uint8_t octave = i / count;
425 /* Direction: UP walks sorted ascending; DOWN walks it in reverse. */
426 uint8_t pick = (s_arp.dir == ARP_DOWN)
427 ? sorted[count - 1 - note_idx]
428 : sorted[note_idx];
429 int32_t chromatic = (int32_t)pick + (int32_t)octave * 12;
430 uint8_t play_note = arp_snap(sequencer_core_clamp_melodic_note(chromatic));
431
432 uint32_t tick_on = 1u + (uint32_t)i * rate;
433 sequencer_core_arp_emit_note(tag_base + (uint32_t)i * 2u,
434 play_note, arp_vel, tick_on, gate, period);
435 }
436}

499void arp_set_patch(uint16_t patch_number)
500{
501 /* Full shared catalog (Juno/DX7/piano/waves/bass/wavetable/FM) — the same
502 * ceiling as melodic; kind dispatch happens in sequencer_core_arp_configure(). */
503 patch_number = SEQ_CLAMP_U16(patch_number, 0, SEQ_PATCH_FULL_MAX);
504 if (s_arp.patch == patch_number) return;
505 s_arp.patch = patch_number;
506 /* In WAVE mode: store the new patch number but leave the synth slot alone.
507 * It will be applied when the source is switched back to ARP_SRC_PATCH. */
508 if (s_arp.source == ARP_SRC_PATCH) {
