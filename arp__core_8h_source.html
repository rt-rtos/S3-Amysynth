11/* ── Standalone arpeggiator ──────────────────────────────────────────────
12 * Independent of the sequencer's layers: it owns up to ARP_MAX_SLOTS note
13 * inputs, its own scale/root, its own AMY synth, and runs in sync with the
14 * global tempo via repeating AMY SEQUENCE events.
15 *
16 * Slots store the RAW chromatic MIDI value. Special sentinels:
17 * -1 (empty) : slot unused; not part of the playback cycle.
18 * ARP_REST (-2): deliberate silent step (SLOT mode only); occupies time but
19 * plays nothing. Meaningless in UP/DOWN — store only -1 there.
20 * The arp snaps to its own scale only at playback (and for display), so the
21 * stored value never absorbs its own quantized output — no compounding drift. */
22
23#define ARP_MAX_SLOTS 8
24#define ARP_REST ((int16_t)-2) /* deliberate silent step in ARP_SLOT mode */
25
26typedef enum {
27 ARP_UP = 0,
28 ARP_DOWN = 1,
29 ARP_SLOT = 2, /* slot-order: position preserved, ARP_REST honoured */
30 ARP_DIR_COUNT
31} arp_dir_t;

33/* Sound source toggle — mirrors drone_source_t in drone_core.h.
34 * ARP_SRC_PATCH: load a DX7/Juno patch (existing behaviour, default).
35 * ARP_SRC_WAVE : configure a bare AMY oscillator with the user-chosen waveform;
36 * the arp sequence still drives pitch via normal note events. */
37typedef enum {
38 ARP_SRC_WAVE = 0, /* raw AMY waveform oscillator — no patch */
39 ARP_SRC_PATCH = 1, /* DX7/Juno patch (existing behaviour) */
40} arp_source_t;

171/* ── Portamento / glide (AMY-native, PORTAMENTO_MS delta) ──

174 * render-path change needed here. 0 = off (default, matches AMY's own reset
175 * value). Unlike the scheduling setters above, this does not mark the arp

177 * note re-emits, but IS wiped by a patch/synth reconfigure (AMY resets

179 * every source/patch/wave change. */
