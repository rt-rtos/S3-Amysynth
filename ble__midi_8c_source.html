47/* Strip BLE-MIDI packet framing and feed the plain MIDI bytes to midi_core.
48 * Packet = header byte, then interleaved timestamp bytes and MIDI messages.
49 * A high-bit byte after a data byte (or at message boundary) is a timestamp;
50 * two consecutive high-bit bytes are timestamp-then-status. Running-status
51 * data bytes arrive with no preceding timestamp and pass straight through. */
52static void ble_midi_ingest_packet(const uint8_t *data, uint16_t len)
53{
54 if (len < 2) return; /* header only: nothing to do */
55 bool last_was_ts = false;
56 for (uint16_t i = 1; i < len; i++) {
57 uint8_t b = data[i];
58 if (b & 0x80u) {
59 if (!last_was_ts) { /* timestamp byte: swallow */
60 last_was_ts = true;
61 continue;
62 }
63 last_was_ts = false; /* status byte */
64 midi_core_feed(&b, 1);
65 } else {
66 last_was_ts = false; /* data byte */
67 midi_core_feed(&b, 1);
68 }
69 }
70}

72static int midi_chr_access(uint16_t conn_handle, uint16_t attr_handle,
73 struct ble_gatt_access_ctxt *ctxt, void *arg)
74{
75 (void)conn_handle; (void)attr_handle; (void)arg;
76
77 switch (ctxt->op) {
78 case BLE_GATT_ACCESS_OP_READ_CHR:
79 /* BLE-MIDI spec: reads return no payload. */
80 return 0;
81 case BLE_GATT_ACCESS_OP_WRITE_CHR: {
82 uint16_t out_len = 0;
83 int rc = ble_hs_mbuf_to_flat(ctxt->om, s_rx_buf,
84 sizeof(s_rx_buf), &out_len);
85 if (rc != 0) return BLE_ATT_ERR_INSUFFICIENT_RES;
86#if CONFIG_SYNTH_BLE_MIDI_LOG
87 ESP_LOGI(TAG, "rx %u bytes", (unsigned)out_len);
88#endif
89 ble_midi_ingest_packet(s_rx_buf, out_len);
90 return 0;
91 }
92 default:
93 return BLE_ATT_ERR_UNLIKELY;
94 }
95}

119static int ble_midi_gap_event(struct ble_gap_event *event, void *arg)
120{
121 (void)arg;
122 switch (event->type) {
123 case BLE_GAP_EVENT_CONNECT:
124 if (event->connect.status == 0) {
125 s_connected = true;
126 midi_core_reset();
127 ESP_LOGI(TAG, "central connected");
128 } else {
129 ESP_LOGW(TAG, "connect failed (%d), re-advertising",
130 event->connect.status);
131 ble_midi_advertise();
132 }
133 return 0;
134
135 case BLE_GAP_EVENT_DISCONNECT:
136 s_connected = false;
137 ESP_LOGI(TAG, "central disconnected (reason %d)",
138 event->disconnect.reason);
139 if (s_disconnect_cb) s_disconnect_cb();
140 ble_midi_advertise();
141 return 0;
142
143 case BLE_GAP_EVENT_ADV_COMPLETE:
144 ble_midi_advertise();
145 return 0;
146
147 case BLE_GAP_EVENT_REPEAT_PAIRING: {
148 /* Peer lost its bond: delete ours so it can pair again. Without
149 * this, re-pairing after the host "forgets" us fails. */
150 struct ble_gap_conn_desc desc;
151 if (ble_gap_conn_find(event->repeat_pairing.conn_handle, &desc) == 0) {
152 ble_store_util_delete_peer(&desc.peer_id_addr);
153 }
154 return BLE_GAP_REPEAT_PAIRING_RETRY;
155 }
156
157 default:
158 return 0;
159 }
160}

162static void ble_midi_advertise(void)
163{
164 /* GAP calls against a host that is stopping or not yet synced return
165 * BLE_HS_EDISABLED; the disconnect/adv-complete events that fire during
166 * teardown would otherwise spam retries here. */
167 if (s_stopping || !ble_hs_synced()) return;
168
169 /* 31-byte legacy ADV payload: flags (3) + 128-bit service UUID (18).
170 * The device name does not fit alongside them, so it goes in the scan
171 * response - standard practice for BLE-MIDI peripherals. */
172 struct ble_hs_adv_fields fields = { 0 };
173 fields.flags = BLE_HS_ADV_F_DISC_GEN | BLE_HS_ADV_F_BREDR_UNSUP;
174 fields.uuids128 = (ble_uuid128_t *)&midi_svc_uuid;
175 fields.num_uuids128 = 1;
176 fields.uuids128_is_complete = 1;
177 int rc = ble_gap_adv_set_fields(&fields);
178 if (rc != 0) {
179 ESP_LOGE(TAG, "adv_set_fields rc=%d", rc);
180 return;
181 }
182
183 struct ble_hs_adv_fields rsp = { 0 };
184 rsp.name = (const uint8_t *)BLE_MIDI_DEVICE_NAME;
185 rsp.name_len = sizeof(BLE_MIDI_DEVICE_NAME) - 1;
186 rsp.name_is_complete = 1;
187 rc = ble_gap_adv_rsp_set_fields(&rsp);
188 if (rc != 0) {
189 ESP_LOGE(TAG, "adv_rsp_set_fields rc=%d", rc);
190 return;
191 }
192
193 struct ble_gap_adv_params adv_params = { 0 };
194 adv_params.conn_mode = BLE_GAP_CONN_MODE_UND;
195 adv_params.disc_mode = BLE_GAP_DISC_MODE_GEN;
196 rc = ble_gap_adv_start(s_own_addr_type, NULL, BLE_HS_FOREVER,
197 &adv_params, ble_midi_gap_event, NULL);
198 if (rc != 0 && rc != BLE_HS_EALREADY) {
199 ESP_LOGE(TAG, "adv_start rc=%d", rc);
200 }
201}
