95void cv_trigger_generate_events(float *cv_inputs) {
96 // Check to see if triggers are run based on new vector of trigger values.
97 cv_trigger_t *cv_trig = cv_trigger_root;
98 while(cv_trig) {
99 float cv_val = cv_inputs[cv_trig->trigger_cv];
100 bool rising = (cv_trig->thresh_trigger > cv_trig->thresh_reset);
101 int polarity = rising ? 1 : -1;
102 if ((polarity * cv_val) >= (polarity * cv_trig->thresh_trigger)) {
103 if (cv_trig->state <= CV_TRIG_TRIGGERED) {
104 ++cv_trig->state;
105 if (cv_trig->state == CV_TRIG_TRIGGERED) {
106 // execute the event
107 float note = 0;
108 if (AMY_IS_SET(cv_trig->pitch_cv)) {
109 // resample the pitch CV
110 float cv_input = amy_global.config.amy_external_coef_hook(cv_trig->pitch_cv);
111 note = midi_note_for_logfreq(
112 cv_trig->pitch_offset
113 + cv_trig->pitch_scale * cv_input
114 );
115 }
116 char message[AMY_WIRE_COMMAND_LEN];
117 substitute_midi_special_values(message, cv_trig->message_template, 0, 0, note);
118 //fprintf(stderr, "update_external_cv_in: message %s\n", message);
119 amy_add_message(message);
120 }
121 }
122 } else if ((polarity * cv_val) < (polarity * cv_trig->thresh_reset)) {
123 if (cv_trig->state != CV_TRIG_READY) {
124 // Reset
125 cv_trig->state = CV_TRIG_READY;
126 //fprintf(stderr, "update_external_cv_in: RESET message %s\n", cv_trig->message_template);
127 }
128 }
129 cv_trig = cv_trig->next;
130 }
131}
