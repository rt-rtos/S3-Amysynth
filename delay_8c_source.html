6int is_power_of_two(int val) {
7 // Returns log_2(val) if val == 2**n, else -1.
8 int log_2_val = 0;
9 while(val) {
10 if (val == 1) return log_2_val;
11 if (val & 1) return -1;
12 val >>= 1;
13 ++log_2_val;
14 }
15 return -1; // zero is not a power of 2.
16}

34 qspi_used += sizeof(uint32_t);
35 void *result = (void *)(qspi_heap + qspi_used);
36 qspi_used += num_bytes;
37 return result;

41 uint32_t last_alloc = *(uint32_t *)((uint8_t*)ptr - sizeof(uint32_t));
42 if (ptr == (void *)(qspi_heap + (qspi_used - last_alloc))) {
43 // We're just freeing the last alloc, yay.

47 fprintf(stderr, "qspi_free: punt (ptr = qspi_heap + %d)\n", (uint8_t*)ptr - (uint8_t *)qspi_heap);

57delay_line_t *new_delay_line(int len, int fixed_delay, int ram_type) {
58 // Check that len is a power of 2.
59 //printf("new_delay_line: len %d fixed_del %d\n", len, fixed_delay);
60 int log_2_len = is_power_of_two(len);
61 if (log_2_len < 0) {
62 fprintf(stderr, "delay line len must be power of 2, not %d\n", len);
63 abort();
64 }
65 delay_line_t *delay_line = (delay_line_t*)malloc_caps(sizeof(delay_line_t) + len * sizeof(SAMPLE), ram_type);
66 if (delay_line == NULL) {

68 return NULL;
69 }
70 delay_line->samples = (SAMPLE*)(((uint8_t*)delay_line) + sizeof(delay_line_t));
71 delay_line->len = len;
72 delay_line->log_2_len = log_2_len;
73 delay_line->fixed_delay = fixed_delay;

77 }
78 //fprintf(stderr, "new_delay_line: len %d fixed_del %d ->0x%x\n", len, fixed_delay, (uint32_t)delay_line);

87static SAMPLE FRACTIONAL_SAMPLE(PHASOR phase, const SAMPLE *delay, int index_mask, int index_bits) {
88 // Interpolated sample copied from oscillators.c:render_lut
89 uint32_t base_index = INT_OF_P(phase, index_bits);
90 SAMPLE frac = S_FRAC_OF_P(phase, index_bits);
91 SAMPLE b = delay[base_index];
92 SAMPLE c = delay[(base_index + 1) & index_mask];
93 // linear interpolation.
94 SAMPLE sample = b + MUL8_SS((c - b), frac);
95 return sample;
96}

98void delay_line_in_out(SAMPLE *in, SAMPLE *out, int n_samples, SAMPLE* mod_in, SAMPLE mod_scale, delay_line_t *delay_line, SAMPLE mix_level, SAMPLE feedback_level) {
99 // Read and write the next n_samples from/to the delay line.
100 // mod_in is a per-sample modulation of the maximum delay, where 1 gives
101 // the max delay, -1 gives no delay, and 0 gives max_delay/2.
102 // mod_scale is a constant scale factor applied to each value in mod_in,
103 // used e.g. to flip the sign of the delay.
104 // Also supports input feedback from a non-modulated feedback delay output.
105 int delay_len = delay_line->len;
106 int index_mask = delay_len - 1; // will be all 1s because len is guaranteed 2**n.
107 int index_bits = delay_line->log_2_len;
108
109 int index_in = delay_line->next_in;
110 int index_feedback = (index_in - delay_line->fixed_delay) & index_mask;
111
112 SAMPLE *delay = delay_line->samples;
113 SAMPLE half_mod_scale = SHIFTR(mod_scale, 1);
114 while(n_samples-- > 0) {
115 SAMPLE next_in = *in++ + MUL8_SS(feedback_level,
116 delay[index_feedback++]);
117 index_feedback &= index_mask;
118
119 PHASOR phase_out = I2P(index_in, index_bits)
120 - S2P(F2S(0.5) + MUL8_SS(half_mod_scale, *mod_in++));
121 //if(index_out >= delay_len) index_out -= delay_len;
122 //if(index_out < 0) index_out += delay_len;
123 SAMPLE sample = FRACTIONAL_SAMPLE(phase_out, delay, index_mask, index_bits);
124 *out++ += MUL8_SS(mix_level, sample); // mix delayed + original.
125 delay[index_in++] = next_in;
126 index_in &= index_mask;
127 }
128 delay_line->next_in = index_in;
129}

142static inline SAMPLE LPF(SAMPLE samp, SAMPLE *state, SAMPLE lpcoef, SAMPLE lpgain, SAMPLE gain) {
143 // 1-pole lowpass filter (exponential smoothing).
144 // Smoothing. lpcoef=1 => no smoothing; lpcoef=0.001 => much smoothing.
145 *state += SMULR6(lpcoef, samp - *state);
146 // Cross-fade between smoothed and original. lpgain=0 => all smoothed, 1 => all dry.
147 return SMULR6(SHIFTR(gain, 1), *state + SMULR6(lpgain, samp - *state));
148}

150void delay_line_in_out_fixed_delay(SAMPLE *in, SAMPLE *out, int n_samples, int delay_samples, delay_line_t *delay_line, SAMPLE mix_level, SAMPLE feedback_level, SAMPLE filter_coef) {
151 // Read and write the next n_samples from/to the delay line.
152 // Simplified version of delay_line_in_out() that uses a fixed integer delay
153 // for the whole block.
154 if (filter_coef == 0) {
155 while(n_samples-- > 0) {
156 SAMPLE delay_out = DEL_OUT(delay_line, 0);
157 SAMPLE next_in = *in++ + SMULR6(feedback_level, delay_out);
158 DEL_IN(delay_line, next_in);
159 *out++ += MUL8_SS(mix_level, delay_out);
160 }
161 } else if (filter_coef > 0) {
162 // Positive filter coef is a pole on the positive real line, to get low-pass effect.
163 // We apply the filter on the way *in* to the delay line.
164 while(n_samples-- > 0) {
165 SAMPLE delay_out = DEL_OUT(delay_line, 0);
166 SAMPLE next_in = *in++ + SMULR6(feedback_level, delay_out);
167 // Peek at the last value we wrote to the delay line to get the most recent filter output.
168 SAMPLE last_filter_result = delay_line->samples[(delay_line->next_in - 1) & (delay_line->len - 1)];
169 SAMPLE filter_result = next_in + SMULR6(filter_coef, last_filter_result - next_in);
170 DEL_IN(delay_line, filter_result);
171 *out++ += MUL8_SS(mix_level, delay_out);
172 }
173 } else {
174 // Negative filter coef is a zero on the positive real axis to get high-pass.
175 // We apply the FIR zero on the way *out* of the delay line.
176 while(n_samples-- > 0) {
177 SAMPLE delay_out = DEL_OUT(delay_line, 0);
178 SAMPLE prev_delay_out = DEL_OUT(delay_line, 1);
179 SAMPLE output = delay_out + SMULR6(filter_coef, prev_delay_out);
180 SAMPLE next_in = *in++ + SMULR6(feedback_level, output);
181 DEL_IN(delay_line, next_in);
182 *out++ += MUL8_SS(mix_level, output);
183 }
184 }
185}

286// Cache one delay line's state in locals for the reverb loop, the same way
287// delay_line_in_out() hoists its line's state. DEL_IN/DEL_OUT re-read
288// samples/next_in/fixed_delay/len through the struct pointer on every call;
289// with ten delay lines interleaved per sample that redundant traffic is a
290// measurable fraction of the loop, and on platforms that place delay lines
291// in external RAM the struct headers are slow to touch.
292//
293// `##` is the standard (C89) preprocessor token-pasting operator: it joins
294// the macro argument P with the suffix into one identifier at compile time.
295// So DL_LOAD(e1, rev->ref_1) declares four ordinary locals named e1_s
296// (samples pointer), e1_m (index mask = len-1), e1_n (next_in write index),
297// e1_f (fixed_delay read offset), and DL_WRITE(e1, x)/DL_READ(e1) refer to
298// those same locals. This gives each of the ten lines its own distinct
299// scalars (which the compiler can keep in registers) without hand-writing
300// forty declarations. The names are generated at compile time, so a wrong
301// prefix is an "undeclared identifier" error, never a silent wrong-line bug.
302// Only next_in changes inside the loop; it is written back after the loop.
303#define DL_LOAD(P, LINE) \
304 SAMPLE *P##_s = (LINE)->samples; \
305 int P##_m = (LINE)->len - 1; \
306 int P##_n = (LINE)->next_in; \
307 int P##_f = (LINE)->fixed_delay
308#define DL_WRITE(P, val) do { P##_s[P##_n] = (val); P##_n = (P##_n + 1) & P##_m; } while (0)
309#define DL_READ(P) (P##_s[(P##_n - P##_f) & P##_m])

311void stereo_reverb(reverb_params_t *rev, SAMPLE *r_in, SAMPLE *l_in, SAMPLE *r_out, SAMPLE *l_out, int n_samples, SAMPLE level) {
312 // Stereo reverb. *{r,l}_in each point to n_samples input samples.
313 // n_samples are written to {r,l}_out.
314 // Recreate
315 // https://github.com/duvtedudug/Pure-Data/blob/master/extra/rev2%7E.pd
316 // an instance of the Stautner-Puckette multichannel reverberator from
317 // https://www.ee.columbia.edu/~dpwe/e4896/papers/StautP82-reverb.pdf
318
319
320 // Early-reflection lines e1..e6, reverb delay lines dl1..dl4.
321 DL_LOAD(e1, rev->ref_1);
322 DL_LOAD(e2, rev->ref_2);
323 DL_LOAD(e3, rev->ref_3);
324 DL_LOAD(e4, rev->ref_4);
325 DL_LOAD(e5, rev->ref_5);
326 DL_LOAD(e6, rev->ref_6);
327 DL_LOAD(dl1, rev->delay_1);
328 DL_LOAD(dl2, rev->delay_2);
329 DL_LOAD(dl3, rev->delay_3);
330 DL_LOAD(dl4, rev->delay_4);
331
332 SAMPLE lpfcoef = rev->lpfcoef, lpfgain = rev->lpfgain, liveness = rev->liveness;
333 SAMPLE f1state = rev->f1state, f2state = rev->f2state;
334 SAMPLE f3state = rev->f3state, f4state = rev->f4state;
335
336 while(n_samples--) {
337 // Early echo reflections.
338 SAMPLE in_r = *r_in++;
339 SAMPLE in_l;
340 if (l_in) in_l = *l_in++;
341 else in_l = in_r;
342 SAMPLE r_acc, l_acc;
343 r_acc = MUL0_SS(F2S(0.0625f), in_r);
344 l_acc = MUL0_SS(F2S(0.0625f), in_l);
345
346 DL_WRITE(e1, l_acc);
347 SAMPLE d_out = DL_READ(e1);
348 l_acc = r_acc - d_out;
349 r_acc += d_out;
350
351 DL_WRITE(e2, l_acc);
352 d_out = DL_READ(e2);
353 l_acc = r_acc - d_out;
354 r_acc += d_out;
355
356 DL_WRITE(e3, l_acc);
357 d_out = DL_READ(e3);
358 l_acc = r_acc - d_out;
359 r_acc += d_out;
360
361 DL_WRITE(e4, l_acc);
362 d_out = DL_READ(e4);
363 l_acc = r_acc - d_out;
364 r_acc += d_out;
365
366 DL_WRITE(e5, l_acc);
367 d_out = DL_READ(e5);
368 l_acc = r_acc - d_out;
369 r_acc += d_out;
370
371 DL_WRITE(e6, l_acc);
372 l_acc = DL_READ(e6);
373
374
375 // Reverb delays & matrix.
376 SAMPLE d1 = DL_READ(dl1);
377 d1 = LPF(d1, &f1state, lpfcoef, lpfgain, liveness);
378 d1 += r_acc;
379 *r_out++ = in_r + MUL8_SS(level, d1);
380
381 SAMPLE d2 = DL_READ(dl2);
382 d2 = LPF(d2, &f2state, lpfcoef, lpfgain, liveness);
383 d2 += l_acc;
384 if (l_out != NULL) *l_out++ = in_l + MUL8_SS(level, d2);
385
386 SAMPLE d3 = DL_READ(dl3);
387 d3 = LPF(d3, &f3state, lpfcoef, lpfgain, liveness);
388
389 SAMPLE d4 = DL_READ(dl4);
390 d4 = LPF(d4, &f4state, lpfcoef, lpfgain, liveness);
391
392 // Mixing and feedback.
393 DL_WRITE(dl1, d1 + d2 + d3 + d4);
394 DL_WRITE(dl2, d1 - d2 + d3 - d4);
395 DL_WRITE(dl3, d1 + d2 - d3 - d4);
396 DL_WRITE(dl4, d1 - d2 - d3 + d4);
397 }
398
399 // Write back the advanced write indices and the filter state.
400 rev->ref_1->next_in = e1_n;
401 rev->ref_2->next_in = e2_n;
402 rev->ref_3->next_in = e3_n;
403 rev->ref_4->next_in = e4_n;
404 rev->ref_5->next_in = e5_n;
405 rev->ref_6->next_in = e6_n;
406 rev->delay_1->next_in = dl1_n;
407 rev->delay_2->next_in = dl2_n;
408 rev->delay_3->next_in = dl3_n;
409 rev->delay_4->next_in = dl4_n;
410 rev->f1state = f1state;
411 rev->f2state = f2state;
412 rev->f3state = f3state;
413 rev->f4state = f4state;
414}
