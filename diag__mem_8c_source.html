21const char *diag_mem_region(const void *p)
22{
23 if (p == NULL) return "(null)";
24 // Order matters: the data-RAM tests are the cheap common case, and
25 // esp_ptr_executable() must come last because it is the broadest - it
26 // accepts IRAM and RTC IRAM as well as the flash text window.
27 if (esp_ptr_in_dram(p)) return "DRAM";
28 if (esp_ptr_external_ram(p)) return "PSRAM";
29 if (esp_ptr_in_iram(p)) return "IRAM";
30 if (esp_ptr_in_drom(p)) return "DROM/flash";
31 if (esp_ptr_in_rtc_dram_fast(p)) return "RTC-DRAM";
32 if (esp_ptr_in_rtc_iram_fast(p)) return "RTC-IRAM";
33 if (esp_ptr_in_rtc_slow(p)) return "RTC-slow";
34 if (esp_ptr_executable(p)) return "text/flash-XIP";
35 return "?";
36}
