1// Reporter registry. Deliberately allocation-free: a fixed table sized well
2// above the number of diagnostics the firmware has, so registration cannot
3// fail in practice and cannot fragment the internal heap it is meant to be

23bool diag_register_reporter(const char *name, diag_reporter_fn fn)
24{
25 if (fn == NULL) return false;
26 if (s_num_reporters >= DIAG_MAX_REPORTERS) {
27 ESP_LOGW(TAG, "reporter table full (%d), dropping '%s'",
28 DIAG_MAX_REPORTERS, name ? name : "?");
29 return false;
30 }
31 s_reporters[s_num_reporters].name = name ? name : "?";
32 s_reporters[s_num_reporters].fn = fn;
33 s_num_reporters++;
34 return true;
35}

37void diag_init(void)
38{
39 // Each of these compiles to a no-op stub when its Kconfig gate is off, so
40 // registering unconditionally costs one table slot and nothing else. The
41 // order here is the order the report lines appear in the log.
42#if CONFIG_AMYSYNTH_RENDER_STATS
43 diag_register_reporter("render", render_stats_report);
44#endif
45#if CONFIG_AMYSYNTH_RTOS_STATS
46 diag_register_reporter("rtos", rtos_stats_report);
47#endif
48#if defined(CONFIG_AMY_PROFILE_COARSE) || defined(CONFIG_AMY_PROFILE_FULL)
49 diag_register_reporter("amy-profile", amy_profile_report);
50#endif
51
52 if (s_num_reporters == 0) {
53 ESP_LOGD(TAG, "no diagnostics enabled");
54 return;
55 }
56 for (uint8_t i = 0; i < s_num_reporters; i++) {
57 ESP_LOGI(TAG, "reporter %u: %s", i, s_reporters[i].name);
58 }
59}
60
61void diag_report_all(void)
62{
63 for (uint8_t i = 0; i < s_num_reporters; i++) {
64 s_reporters[i].fn();
65 }
66}
