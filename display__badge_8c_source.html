26/* Is every pixel of the badge footprint at x (plus its blank gap) still dark?
27 * The badge spans rows 0..7, which is exactly tile row 0, so a column is clear
28 * precisely when its first buffer byte is zero. */
29static bool slot_is_clear(u8g2_t *u8g2, int x)
30{
31 const int width = (int)u8g2_GetDisplayWidth(u8g2);
32 if (x < 0 || x + BADGE_W > width) {
33 return false;
34 }
35
36 const uint8_t *row0 = u8g2_GetBufferPtr(u8g2);
37 int lo = x - BADGE_GAP;
38 int hi = x + BADGE_W + BADGE_GAP;
39 if (lo < 0) {
40 lo = 0;
41 }
42 if (hi > width) {
43 hi = width;
44 }
45
46 for (int col = lo; col < hi; col++) {
47 if (row0[col] != 0) {
48 return false;
49 }
50 }
51 return true;
52}

54/* Nearest clear slot to the view's preferred x, or -1 when the top row is full.
55 * Searching outward keeps the badge in its hand-tuned home when free, and moves
56 * it the shortest distance - not to a fixed fallback corner - when not. */
57static int find_slot(u8g2_t *u8g2, uint8_t preferred_x)
58{
59 const int width = (int)u8g2_GetDisplayWidth(u8g2);
60 const int anchor = preferred_x;
61
62 for (int delta = 0; delta < width; delta++) {
63 if (slot_is_clear(u8g2, anchor - delta)) {
64 return anchor - delta;
65 }
66 if (delta != 0 && slot_is_clear(u8g2, anchor + delta)) {
67 return anchor + delta;
68 }
69 }
70 return -1;
71}
