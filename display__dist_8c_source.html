5/* Two-panel layout (128x64), geometry deliberately matched to display_lfo.c so
6 * the editors read as siblings: left panel a live transfer curve, right column
7 * the five parameter rows. All three distortion types are static waveshapers,
8 * so the curve IS the effect - the soft knee of CLIP, the fold count of FOLD,
9 * the quantization staircase of CRUSH.
10 *
11 * The curve shows the SHAPER ALONE, at full wet, against a dotted unity
12 * diagonal. Mix is deliberately excluded even though folding it in would be
13 * mathematically honest (blending dry and wet is memoryless, so the composite
14 * is still a valid transfer function): at low Mix every type collapses to a
15 * near-diagonal, which is exactly where the shapes differ most and the picture
16 * is needed most. Keeping Mix out isolates Drive's effect, holds the picture
17 * still while Mix is swept by ear, and loses nothing - Mix is a scalar that
18 * reads fine as a number. CRUSH's sample-hold rate has memory and therefore no
19 * transfer-curve form at all; it is not drawn either. */
20#define DIST_DIV_X 60 /* vertical divider between the two panels */
21#define DIST_RCOL_X 64 /* right-column text x */
22
23/* Curve box: a square-ish plot inside the left panel, centred on its own
24 * origin so the identity diagonal runs corner to corner. */
25#define DIST_CV_X0 4 /* left edge of the plot */
26#define DIST_CV_HW 25 /* half width in pixels (x = -1 .. +1) */
27#define DIST_CV_CY 35 /* vertical centre = output 0 */
28#define DIST_CV_HH 19 /* half height in pixels (y = -1 .. +1) */

31/* Float mirror of AMY's dist_process() shaping (components/amy/src/filters.c),
32 * minus the wet/dry blend - see the file header on why Mix stays out. Kept in
33 * float for the display only: the point is the shape, not bit equivalence with
34 * the fixed-point render path. */
35static float dist_transfer(const seq_dist_t *d, float x)
36{
37 if (d->type == 0) return x; /* OFF: every stage bypassed */
38
39 /* Enabled stages compose in the engine's fixed clip -> fold -> crush
40 * order, each re-applying drive as its own pre-gain - exactly what the
41 * stacked passes in dist_block do at mix = 1. */
42 float drive = (float)d->drive;
43 float y = x;
44 if (d->type & 1u) { /* CLIP: cubic soft knee */
45 float v = y * drive;
46 if (v > 1.0f) v = 1.0f;
47 if (v < -1.0f) v = -1.0f;
48 y = v * (1.0f - v * v * 0.33333334f);
49 }
50 if (d->type & 2u) { /* FOLD: triangle wavefolder */
51 float v = y * drive;
52 float w = v + 1.0f;
53 w -= 4.0f * floorf(w * 0.25f);
54 y = 1.0f - fabsf(w - 2.0f);
55 }
56 if (d->type & 4u) { /* CRUSH: saturate then quantize */
57 float v = y * drive;
58 if (v > 1.0f) v = 1.0f;
59 if (v < -1.0f) v = -1.0f;
60 /* bits == 0 would make the shift below undefined. Every store path
61 * clamps to >= 1, but the renderer must not depend on that. */
62 if (d->bits >= 1u && d->bits <= 23u) {
63 float qscale = (float)(1u << (d->bits - 1u));
64 v = floorf(v * qscale + 0.5f) / qscale;
65 }
66 y = v;
67 }
68 return y;
69}

71static void draw_curve(u8g2_t *u8g2, const seq_dist_t *d)
72{
73 const uint8_t cx = DIST_CV_X0 + DIST_CV_HW;
74 const uint8_t y_top = DIST_CV_CY - DIST_CV_HH;
75 const uint8_t y_bot = DIST_CV_CY + DIST_CV_HH;
76
77 /* Unity diagonal (y = x), dotted so the curve stays the dominant mark. It
78 * is the reference a transfer plot actually wants: the shaper's departure
79 * from it IS the effect, and it passes through the origin, so it carries
80 * the zero point too - which is why it replaces a plain horizontal axis
81 * rather than joining one. */
82 for (int i = 0; i <= 2 * DIST_CV_HW; i += 3) {
83 int px = DIST_CV_X0 + i;
84 int py = DIST_CV_CY + DIST_CV_HH - (i * DIST_CV_HH) / DIST_CV_HW;
85 u8g2_DrawPixel(u8g2, px, py);
86 }
87
88 int prev_x = 0, prev_y = 0;
89 bool have_prev = false;
90 for (int i = 0; i <= 2 * DIST_CV_HW; i++) {
91 float xn = (float)(i - DIST_CV_HW) / (float)DIST_CV_HW;
92 float yn = dist_transfer(d, xn);
93
94 /* Clamp into the box rather than letting a folded/boosted value run
95 * off the panel and collide with the divider. */
96 int py = DIST_CV_CY - (int)lrintf(yn * (float)DIST_CV_HH);
97 if (py < (int)y_top) py = y_top;
98 if (py > (int)y_bot) py = y_bot;
99 int px = DIST_CV_X0 + i;
100
101 if (have_prev) u8g2_DrawLine(u8g2, prev_x, prev_y, px, py);
102 else u8g2_DrawPixel(u8g2, px, py);
103 prev_x = px; prev_y = py; have_prev = true;
104 }
105}
