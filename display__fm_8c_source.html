29static uint8_t layout_width(const fm_layout_t *l, uint8_t node, uint8_t depth)
30{
31 if (depth >= FM_GRAPH_OPS) return 1;
32 uint8_t w = 0;
33 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
34 if (l->parent[i] == node) w = (uint8_t)(w + layout_width(l, i, (uint8_t)(depth + 1)));
35 }
36 return w ? w : 1;
37}

39static void layout_place(fm_layout_t *l, uint8_t node, uint8_t col, uint8_t row)
40{
41 if (row >= FM_GRAPH_OPS) return;
42 l->col[node] = col;
43 l->row[node] = row;
44 if (row + 1 > l->rows) l->rows = (uint8_t)(row + 1);
45 uint8_t c = col;
46 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
47 if (l->parent[i] != node) continue;
48 layout_place(l, i, c, (uint8_t)(row + 1));
49 c = (uint8_t)(c + layout_width(l, i, 0));
50 }
51}

53static void layout_build(const fm_graph_view_t *g, fm_layout_t *l)
54{
55 memset(l, 0, sizeof(*l));
56 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
57 l->parent[i] = FM_NO_OP;
58 for (uint8_t t = 0; t < FM_GRAPH_OPS; t++) {
59 if (g->out_mask[i] & (1u << t)) { l->parent[i] = t; break; }
60 }
61 }
62 /* Placement is a tree walk from each carrier; an op that neither routes
63 * to the output nor reaches a carrier (never true for a compiled graph)
64 * is parked on the carrier row so it still shows. */
65 bool placed[FM_GRAPH_OPS] = { false };
66 uint8_t c = 0;
67 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
68 if (l->parent[i] != FM_NO_OP) continue;
69 layout_place(l, i, c, 0);
70 c = (uint8_t)(c + layout_width(l, i, 0));
71 }
72 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
73 uint8_t n = i, guard = 0;
74 while (l->parent[n] != FM_NO_OP && guard++ < FM_GRAPH_OPS) n = l->parent[n];
75 placed[i] = (l->parent[n] == FM_NO_OP);
76 }
77 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
78 if (!placed[i]) { l->col[i] = c++; l->row[i] = 0; }
79 }
80 if (l->rows == 0) l->rows = 1;
81}

88static uint8_t row_pitch(const fm_layout_t *l)
89{
90 if (l->rows <= 1) return FM_ROW_PITCH;
91 uint8_t span = (uint8_t)(FM_BOTTOM_Y - FM_BOX_H + 1 - FM_TOP_Y); /* 38 */
92 uint8_t p = (uint8_t)(span / (l->rows - 1));
93 if (p > FM_ROW_PITCH) p = FM_ROW_PITCH;
94 if (p < FM_BOX_H) p = FM_BOX_H;
95 return p;
96}

103/* Connector from op `from` (above) down into op `to`: straight when aligned,
104 * otherwise an L through the middle of the gap. */
105static void draw_connector(u8g2_t *u8g2, const fm_layout_t *l, uint8_t from, uint8_t to)
106{
107 uint8_t cx0 = (uint8_t)(box_x(l, from) + FM_BOX_W / 2);
108 uint8_t cx1 = (uint8_t)(box_x(l, to) + FM_BOX_W / 2);
109 uint8_t y0 = (uint8_t)(box_y(l, from) + FM_BOX_H); /* first row below `from` */
110 uint8_t y1 = box_y(l, to); /* top row of `to` */
111 if (y1 <= y0) {
112 /* Touching or overlapping rows (deep chain / fan-out across rows):
113 * only a horizontal stub can show the link. */
114 if (cx0 != cx1) u8g2_DrawHLine(u8g2, cx0 < cx1 ? cx0 : cx1, y1, (uint8_t)(cx0 > cx1 ? cx0 - cx1 : cx1 - cx0));
115 return;
116 }
117 if (cx0 == cx1) {
118 u8g2_DrawVLine(u8g2, cx0, y0, (uint8_t)(y1 - y0));
119 return;
120 }
121 uint8_t ym = (uint8_t)((y0 + y1) / 2);
122 u8g2_DrawVLine(u8g2, cx0, y0, (uint8_t)(ym - y0 + 1));
123 u8g2_DrawHLine(u8g2, cx0 < cx1 ? cx0 : cx1, ym, (uint8_t)((cx0 > cx1 ? cx0 - cx1 : cx1 - cx0) + 1));
124 u8g2_DrawVLine(u8g2, cx1, ym, (uint8_t)(y1 - ym));
125}
