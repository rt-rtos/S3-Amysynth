6/* Two-panel layout (128x64): a left checklist of the 5 modulation targets
7 * (each an independent checkbox — the one LFO carrier drives every checked
8 * one) and a right column with the shared WAVE/RATE/DEPTH/EN parameters.
9 * Both panels use the 5x7 font at a tight pitch so every row (incl. the
10 * last target and the En row) ends above the hint strip at y=57. */
11#define LFO_DIV_X 60 /* vertical divider between the two panels */
12#define LFO_ROW0_Y 21 /* first checklist row baseline */
13#define LFO_ROW_DY 8 /* checklist row pitch */
14#define LFO_RCOL_X 64 /* right-column text x */

58static void draw_wave_icon(u8g2_t *u8g2, uint8_t x0, uint8_t y_center,
59 lfo_wave_t wave, bool invert)
60{
61 if ((uint8_t)wave >= LFO_WAVE_COUNT) return;
62 u8g2_SetDrawColor(u8g2, invert ? 0 : 1);
63 for (uint8_t i = 0; i < 16; i++) {
64 int16_t y = (int16_t)y_center + s_wave_icon[wave][i];
65 if (y >= 0 && y < 64)
66 u8g2_DrawPixel(u8g2, x0 + i, (uint8_t)y);
67 }
68 /* For saw waves, draw the reset edge. */
69 if (wave == LFO_WAVE_SAW_UP || wave == LFO_WAVE_SAW_DOWN) {
70 u8g2_DrawVLine(u8g2, x0 + 15, y_center - 7, 14);
71 }
72 u8g2_SetDrawColor(u8g2, 1);
73}

93void lfo_view_draw(u8g2_t *u8g2, const lfo_view_t *v)
94{
95 const seq_lfo_t *l = &v->lfo;
96
97 /* ── Header bar ── */
98 u8g2_SetFont(u8g2, u8g2_font_6x10_tf);
99 char hdr[20];
100 if (v->target_label && v->target_label[0]) {
101 snprintf(hdr, sizeof(hdr), "LFO %s", v->target_label);
102 } else {
103 snprintf(hdr, sizeof(hdr), "LFO L%u T%u%s",
104 v->layer_idx + 1u, v->track_idx + 1u, v->apply_all ? ">L" : ">T");
105 }
106 u8g2_DrawStr(u8g2, 1, 10, hdr);
107 u8g2_DrawHLine(u8g2, 0, 13, 128);
108 u8g2_DrawVLine(u8g2, LFO_DIV_X, 15, 42);
109
110 /* Panels use the compact font so all rows clear the hint strip. */
111 u8g2_SetFont(u8g2, u8g2_font_5x7_tr);
112
113 /* ── Left panel: target checklist ── */
114 for (int t = 0; t < LFO_TARGET_COUNT; t++) {
115 uint8_t base = LFO_ROW0_Y + (uint8_t)t * LFO_ROW_DY;
116 bool selected = (v->cursor == (uint8_t)t);
117 bool on = (l->targets & LFO_TGT_BIT(t)) != 0;
118
119 if (selected) {
120 u8g2_SetDrawColor(u8g2, 1);
121 u8g2_DrawBox(u8g2, 0, base - 7, LFO_DIV_X, 9);
122 u8g2_SetDrawColor(u8g2, 0);
123 }
124 /* checkbox: filled when the target is active, framed when not */
125 if (on) u8g2_DrawBox(u8g2, 2, base - 6, 6, 6);
126 else u8g2_DrawFrame(u8g2, 2, base - 6, 6, 6);
127 u8g2_DrawStr(u8g2, 11, base, target_name(t));
128 if (selected) u8g2_SetDrawColor(u8g2, 1);
129 }
130
131 /* ── Right panel: shared LFO parameters ── */
132 char buf[12];
133
134 /* WAVE row: label + live waveform icon */
135 bool wav_sel = (v->cursor == LFO_FLD_WAVE);
136 if (wav_sel) {
137 u8g2_SetDrawColor(u8g2, 1);
138 u8g2_DrawBox(u8g2, LFO_DIV_X + 1, 15, 127 - LFO_DIV_X, 12);
139 u8g2_SetDrawColor(u8g2, 0);
140 }
141 u8g2_DrawStr(u8g2, LFO_RCOL_X, 24, "Wav");
142 if (wav_sel && v->editing) u8g2_DrawBox(u8g2, 123, 17, 3, 6); /* color 0 while selected */
143 draw_wave_icon(u8g2, 92, 20, l->wave, wav_sel); /* restores color 1 */
144 if (wav_sel) u8g2_SetDrawColor(u8g2, 1);
145
146 snprintf(buf, sizeof(buf), "Rte %s", rate_label(l->rate));
147 draw_right_row(u8g2, 36, buf, v->cursor == LFO_FLD_RATE, v->editing);
148
149 snprintf(buf, sizeof(buf), "Dep %u%%", (unsigned)l->depth);
150 draw_right_row(u8g2, 45, buf, v->cursor == LFO_FLD_DEPTH, v->editing);
151
152 snprintf(buf, sizeof(buf), "En %s", l->enabled ? "ON" : "--");
153 draw_right_row(u8g2, 54, buf, v->cursor == LFO_FLD_EN, false);
154}
