7/* Two-panel layout (128x64): left checklist of the modulation targets (one LFO
8 * carrier drives every checked one), right column of shared WAVE/RATE/DEPTH/EN
9 * parameters. Both use the 5x7 font at a tight pitch so every row ends above
10 * the hint strip at y=57. The panel holds five checklist rows, so the targets
11 * are split across two tabs - the five carrier targets, then the distortion
12 * pair - and the shoulder button flips between them. */
13#define LFO_DIV_X 60 /* vertical divider between the two panels */

15#define LFO_ROW_DY 8 /* checklist row pitch */
16#define LFO_RCOL_X 64 /* right-column text x */

73static void draw_wave_icon(u8g2_t *u8g2, uint8_t x0, uint8_t y_center,
74 lfo_wave_t wave, bool invert)
75{
76 if ((uint8_t)wave >= LFO_WAVE_COUNT) return;
77 u8g2_SetDrawColor(u8g2, invert ? 0 : 1);
78 for (uint8_t i = 0; i < 16; i++) {
79 int16_t y = (int16_t)y_center + s_wave_icon[wave][i];
80 if (y >= 0 && y < 64)
81 u8g2_DrawPixel(u8g2, x0 + i, (uint8_t)y);
82 }
83 /* For saw waves, draw the reset edge. */
84 if (wave == LFO_WAVE_SAW_UP || wave == LFO_WAVE_SAW_DOWN) {
85 u8g2_DrawVLine(u8g2, x0 + 15, y_center - 7, 14);
86 }
87 u8g2_SetDrawColor(u8g2, 1);
88}

90/* Right-column parameter row ("Label value"), inverted when selected, with a
91 * caret at the far right while adjusting. `scroll` is '^'/'v' on the window's
92 * edge rows when more parameters lie that way, or 0; drawn inside the row's
93 * colour context so it stays legible on the inverted bar. */
94static void draw_right_row(u8g2_t *u8g2, uint8_t baseline, const char *text,
95 bool selected, bool editing, char scroll)
96{
97 if (selected) {
98 u8g2_SetDrawColor(u8g2, 1);
99 u8g2_DrawBox(u8g2, LFO_DIV_X + 1, baseline - 7, 127 - LFO_DIV_X, 9);
100 u8g2_SetDrawColor(u8g2, 0);
101 }
102 u8g2_DrawStr(u8g2, LFO_RCOL_X, baseline, text);
103 if (scroll) {
104 char s[2] = { scroll, '\0' };
105 u8g2_DrawStr(u8g2, LFO_SCROLL_X, baseline, s);
106 }
107 if (selected && editing) {
108 /* filled caret at the row's right edge = "turn to adjust" */
109 u8g2_DrawBox(u8g2, 123, baseline - 6, 3, 6);
110 }
111 if (selected) u8g2_SetDrawColor(u8g2, 1);
112}

114void lfo_view_draw(u8g2_t *u8g2, const lfo_view_t *v)
115{
116 const seq_lfo_t *l = &v->lfo;
117
118 /* ── Header bar ── */
119 u8g2_SetFont(u8g2, u8g2_font_6x10_tf);
120 char hdr[20];
121 if (v->target_label && v->target_label[0]) {
122 snprintf(hdr, sizeof(hdr), "LFO %s", v->target_label);
123 } else {
124 snprintf(hdr, sizeof(hdr), "LFO L%u T%u%s",
125 v->layer_idx + 1u, v->track_idx + 1u, v->src_badge);
126 }
127 u8g2_DrawStr(u8g2, 1, 10, hdr);
128 /* En chip lives in the header: the right panel is full with the five
129 * parameter rows. Cursor on LFO_FLD_EN highlights it. */
130 {
131 bool en_sel = (v->cursor == LFO_FLD_EN);
132 const char *en_txt = l->enabled ? "ON" : "--";
133 uint8_t tw = (uint8_t)u8g2_GetStrWidth(u8g2, en_txt);
134 if (en_sel) {
135 u8g2_DrawBox(u8g2, (uint8_t)(127 - tw - 4), 1, (uint8_t)(tw + 4), 11);
136 u8g2_SetDrawColor(u8g2, 0);
137 } else {
138 u8g2_DrawFrame(u8g2, (uint8_t)(127 - tw - 4), 1, (uint8_t)(tw + 4), 11);
139 }
140 u8g2_DrawStr(u8g2, (uint8_t)(127 - tw - 2), 10, en_txt);
141 if (en_sel) u8g2_SetDrawColor(u8g2, 1);
142 }
143 u8g2_DrawHLine(u8g2, 0, 13, 128);
144 u8g2_DrawVLine(u8g2, LFO_DIV_X, 15, 42);
145
146 /* Panels use the compact font so all rows clear the hint strip. */
147 u8g2_SetFont(u8g2, u8g2_font_5x7_tr);
148
149 /* ── Left panel: target checklist, one tab at a time ── */
150 bool dist_tab = (v->tgt_tab != 0);
151 uint8_t tgt_lo = dist_tab ? LFO_TARGET_DIST_DRIVE : 0;
152 uint8_t tgt_hi = dist_tab ? LFO_TARGET_COUNT : LFO_TARGET_DIST_DRIVE;
153 for (uint8_t t = tgt_lo; t < tgt_hi; t++) {
154 uint8_t base = LFO_ROW0_Y + (uint8_t)(t - tgt_lo) * LFO_ROW_DY;
155 bool selected = (v->cursor == t);
156 bool on = (l->targets & LFO_TGT_BIT(t)) != 0;
157
158 if (selected) {
159 u8g2_SetDrawColor(u8g2, 1);
160 u8g2_DrawBox(u8g2, 0, base - 7, LFO_DIV_X, 9);
161 u8g2_SetDrawColor(u8g2, 0);
162 }
163 /* checkbox: filled when the target is active, framed when not */
164 if (on) u8g2_DrawBox(u8g2, 2, base - 6, 6, 6);
165 else u8g2_DrawFrame(u8g2, 2, base - 6, 6, 6);
166 u8g2_DrawStr(u8g2, 11, base, target_name(t));
167 /* Distortion targets without a serving stepper (drone): inert, struck
168 * through like the WOBBLE rows on software tracks. Current colour
169 * already matches the row context (0 on the inverted selection bar). */
170 if (dist_tab && v->dist_inert)
171 u8g2_DrawHLine(u8g2, 11, (uint8_t)(base - 3),
172 u8g2_GetStrWidth(u8g2, target_name(t)));
173 if (selected) u8g2_SetDrawColor(u8g2, 1);
174 }
175
176 /* Tab marker: the only cue that half the targets live behind the shoulder
177 * button. It shares the bottom checklist row, which on tab 1 is empty and
178 * on tab 0 may carry the selection bar - so follow that row's colour. */
179 {
180 bool last_row_sel = (!dist_tab && v->cursor == LFO_TARGET_SCAN);
181 u8g2_SetDrawColor(u8g2, last_row_sel ? 0 : 1);
182 u8g2_DrawStr(u8g2, LFO_TAB_X, LFO_TAB_Y, dist_tab ? "2/2" : "1/2");
183 u8g2_SetDrawColor(u8g2, 1);
184 }

186 /* ── Right panel: shared LFO parameters, LFO_PANEL_SLOTS rows at 8 px
187 * pitch. More parameters than slots, so the window scrolls by one when the
188 * cursor reaches the last row. ── */
189 static const uint8_t slot_y[LFO_PANEL_SLOTS] = { 22, 31, 39, 47, 55 };
190 char buf[14];
191
192 uint8_t first = 0; /* first parameter row shown in the window */
193 if (v->cursor >= LFO_FLD_WAVE && v->cursor < LFO_FLD_WAVE + LFO_PANEL_ROWS) {
194 uint8_t row = (uint8_t)(v->cursor - LFO_FLD_WAVE);
195 if (row >= LFO_PANEL_SLOTS)
196 first = (uint8_t)(row - LFO_PANEL_SLOTS + 1u);
197 }
198
199 for (uint8_t s = 0; s < LFO_PANEL_SLOTS; s++) {
200 uint8_t row = (uint8_t)(first + s);
201 uint8_t base = slot_y[s];
202 uint8_t fld = (uint8_t)(LFO_FLD_WAVE + row);
203 bool sel = (v->cursor == fld);
204
205 /* Edge carets are the only cue that the panel continues off-screen. */
206 char scroll = 0;
207 if (s == 0 && first > 0) scroll = '^';
208 if (s == LFO_PANEL_SLOTS - 1 && row + 1u < LFO_PANEL_ROWS) scroll = 'v';
209
210 if (fld == LFO_FLD_WAVE) {
211 /* WAVE row: label + live waveform icon instead of a value string. */
212 if (sel) {
213 u8g2_SetDrawColor(u8g2, 1);
214 u8g2_DrawBox(u8g2, LFO_DIV_X + 1, base - 7, 127 - LFO_DIV_X, 9);
215 u8g2_SetDrawColor(u8g2, 0);
216 }
217 u8g2_DrawStr(u8g2, LFO_RCOL_X, base, "Wav");
218 if (sel && v->editing) u8g2_DrawBox(u8g2, 123, base - 6, 3, 6); /* color 0 while selected */
219 draw_wave_icon(u8g2, 92, base - 6, l->wave, sel); /* restores color 1 */
220 if (sel) u8g2_SetDrawColor(u8g2, 1);
221 continue;
222 }
223
224 switch (fld) {
225 case LFO_FLD_RATE:
226 snprintf(buf, sizeof(buf), "Rte %s", rate_label(l->rate));
227 break;
228 case LFO_FLD_DEPTH:
229 snprintf(buf, sizeof(buf), "Dep %u%%", (unsigned)l->depth);
230 break;
231 case LFO_FLD_FLT_OCT:
232 /* Octaves around the authored cutoff - the unit the log2 filter
233 * rail works in (voice_config.h). Resolves the legacy
234 * depth-derived sentinel, so this is the effective swing. */
235 snprintf(buf, sizeof(buf), "Flt %.2foc",
236 (double)voice_lfo_filter_octaves(l));
237 break;
238 case LFO_FLD_WOB_RATE:
239 snprintf(buf, sizeof(buf), "WRt %s", rate_label((lfo_rate_t)l->wob_rate));
240 break;
241 case LFO_FLD_WOB_DEPTH: {
242 /* Depth reaches speak dB of dip below the authored depth
243 * (voice_config.h); rate-only has no depth breathing to
244 * measure, so it speaks rate-swing octaves instead. */
245 uint8_t wob_db = voice_wob_depth_to_db(l->wob_depth);
246 if (wob_db == 0) snprintf(buf, sizeof(buf), "Wob OFF");
247 else if (l->wob_reach == WOB_REACH_RATE)
248 snprintf(buf, sizeof(buf), "Wob %.2foc",
249 (double)((float)l->wob_depth / 100.0f
250 * VOICE_WOB_DEPTH_RATE));
251 else snprintf(buf, sizeof(buf), "Wob %udB", (unsigned)wob_db);
252 break;
253 }
254 case LFO_FLD_WOB_MODE: {
255 /* Which of the carrier's two rails the wobble reaches. */
256 static const char *reach_lbl[WOB_REACH_COUNT] =
257 { "Dep+Rt", "Dep", "Rate" };
258 uint8_t wr = (l->wob_reach < WOB_REACH_COUNT) ? l->wob_reach : 0;
259 snprintf(buf, sizeof(buf), "WTo %s", reach_lbl[wr]);
260 break;
261 }
262 default:
263 buf[0] = '\0';
264 break;
265 }
266 draw_right_row(u8g2, base, buf, sel, v->editing, scroll);

268 /* WOBBLE rows are inert when the target runs the software stepper
269 * (patch strings without a native carrier): strike them through
270 * rather than hide them - the values persist and re-arm the moment
271 * the patch turns native. Match the row's colour context so the
272 * strike stays legible on the inverted selection bar. */
273 if (!v->wob_native &&
274 (fld == LFO_FLD_WOB_RATE || fld == LFO_FLD_WOB_DEPTH ||
275 fld == LFO_FLD_WOB_MODE)) {
276 u8g2_SetDrawColor(u8g2, sel ? 0 : 1);
277 u8g2_DrawHLine(u8g2, LFO_RCOL_X, (uint8_t)(base - 3),
278 u8g2_GetStrWidth(u8g2, buf));
279 u8g2_SetDrawColor(u8g2, 1);
280 }
281 }
