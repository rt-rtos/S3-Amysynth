176/* ── Peak/Duck dB amp helpers ──────────────────────────────────────────────
177 * These are the SINGLE source of truth for the engine math. Both
178 * drone_configure_wave_synth() and drone_get_amp_levels_norm() call these
179 * helpers — never inline the formula elsewhere.
180 *
181 * Knob→dB/linear mapping:
182 * peak_lin = amp_peak (linear 0..1, identity mapping)
183 * duck_db = amp_duck * 40.0f (0 dB at knob=0, -40 dB floor at knob=1)
184 *
185 * AMY engine coefs (from the two-equation solve):
186 * m = duck_db / 120
187 * const_sent = peak_lin * 10^(-duck_db / 40)
188 *
189 * Result:
190 * ON-beat (LFO=+1): const_sent * 10^(3*m) = peak_lin
191 * OFF-beat (LFO=-1): const_sent * 10^(-3*m) = peak_lin * 10^(-duck_db/20)
192 */
193static inline float s_amp_peak_lin(void)
194{
195 /* amp_trim is a per-target volume trim (0..1, default 1.0) set by the graph
196 * editor amp mode. Fold it into amp_peak here so all callers automatically
197 * pick up the trimmed level (single source of truth rule applies). */
198 float v = s_d.amp_peak * s_d.vp.amp_trim;
199 if (v < 0.0f) v = 0.0f;
200 if (v > 1.0f) v = 1.0f;
201 return v;
202}
203
204static inline float s_amp_duck_db(void)
205{
206 return s_d.amp_duck * 40.0f;
207}
208
209static inline float s_amp_m(void)
210{
211 return s_amp_duck_db() / 120.0f;
212}
213
214static inline float s_amp_const_sent(void)
215{
216 return s_amp_peak_lin() * powf(10.0f, -s_amp_duck_db() / 40.0f);
217}

237/* Count the notes in a chord formula (up to the first -1 sentinel, max DRONE_CHORD_MAX_NOTES).
238 * Uses the shared quantizer_chord_intervals() table; returns 0 if chord is out of range. */
239static uint8_t drone_chord_note_count(chord_type_t chord)
240{
241 const int8_t *row = quantizer_chord_intervals(chord);
242 if (!row) return 0;
243 uint8_t n = 0;
244 for (uint8_t i = 0; i < DRONE_CHORD_MAX_NOTES; i++) {
245 if (row[i] < 0) break;
246 n++;
247 }
248 return n;
249}

251/* ── Synth configuration (WAVE mode) ──
252 * Build a `voices`-voice synth, each voice = osc0 carrier (NOTE-following, so
253 * its pitch comes from the voice's note-on) amplitude-gated by osc1 PULSE LFO.
254 * A multi-voice main synth therefore sounds a chord when fed multiple notes. */
255/* `wave` is the AMY waveform to use for the carrier (osc0). Callers pass
256 * s_d.wave for WAVE-mode, or the patch-derived wave for PATCH-mode wave patches.
257 * `wt_preset` selects which built-in wavetable bank to play when wave==WAVETABLE
258 * (index into pcm_wavetable_base..+pcm_wavetable_samples-1); pass -1 for any
259 * non-wavetable wave, where it is a no-op. */
260static void drone_configure_wave_synth(uint8_t synth, uint8_t voices, uint16_t wave,
261 int16_t wt_preset)
262{
263 float lfo_hz = drone_lfo_hz();
264 /* AMY 1.2.12 (#720) dB/exp amp model: amp = const_sent * 10^(3*m*LFO).
265 * Peak/Duck mapping (see s_amp_* helpers for full derivation):
266 * ON-beat (LFO=+1) = peak_lin = amp_peak
267 * OFF-beat(LFO=-1) = peak_lin * 10^(-duck_db/20), duck_db = amp_duck*40. */
268 float m = s_amp_m();
269 float const_sent = s_amp_const_sent();
270
271 /* Shared skeleton: pool (build-your-own synth: N voices, 2 oscs/voice, no
272 * patch) + osc0 NOTE-following carrier (COEF_NOTE=1) so each voice plays
273 * its own chord note. Velocity does not scale amp (osc0_amp_vel=0); EG0
274 * multiplies the whole amp, so the ADSR envelope shapes the drone
275 * swell/fade *around* the LFO stutter. */
276 voice_wave_cfg_t cfg = {
277 .synth = synth,
278 .num_voices = voices,
279 .oscs_per_voice = DRONE_OSCS_PER_VC,
280 .wave = wave,
281 .osc0_amp_const = const_sent,
282 .osc0_amp_vel = 0.0f,
283 .ks_feedback_authored = false, /* fixed 0.9 KS default */
284 .ks_feedback_q = 0.0f,
285 .wt_preset = wt_preset,
286 };
287 voice_build_wave(&cfg);
288
289 /* osc1 = PULSE LFO. Absolute Hz (note-follow off), full const amp.
290 * The PULSE duty IS the gate length: 0.5 = 50/50 square (legacy), lower =
291 * a shorter "on" fraction per subdivision = a tighter percussive chop. */
292 amy_event *e = amy_helpers_event_begin();
293 e->synth = synth;
294 e->osc = 1;
295 e->wave = PULSE;
296 e->duty_coefs[COEF_CONST]= s_d.gate_len;
297 e->freq_coefs[COEF_CONST]= lfo_hz;
298 e->freq_coefs[COEF_NOTE] = 0.0f; /* absolute Hz, ignore note */
299 e->amp_coefs[COEF_CONST] = 1.0f;
300 e->amp_coefs[COEF_VEL] = 0.0f; /* turn off the default vel/eg amp */
301 e->amp_coefs[COEF_EG0] = 0.0f;
302 amy_helpers_event_send(e);
303
304 /* osc0 drone specializations: amp = const + mod(=osc1) stutter gate, and
305 * the LPF24 sweep filter. Sent after osc1 so mod_source always points at
306 * a configured gate oscillator. */
307 e = amy_helpers_event_begin();
308 e->synth = synth;
309 e->osc = 0;
310 e->amp_coefs[COEF_MOD] = m;
311 e->mod_source = 1; /* osc1 of this voice (base-osc rel) */
312 e->filter_type = FILTER_LPF24;
313 e->filter_freq_coefs[COEF_CONST] = s_d.sweep_hi;
314 e->resonance = s_d.resonance;
315 amy_helpers_event_send(e);
316}

341/* (Re)build both carrier synths for the current source/params. Main is sized to
342 * the chord's note count; the sub is always a single voice. */
343static void drone_rebuild(void)
344{
345 uint8_t chord_n = drone_chord_note_count(s_d.chord);
346 if (chord_n < 1) chord_n = 1;
347
348 if (s_d.source == DRONE_SRC_WAVE) {
349 drone_configure_wave_synth(DRONE_SYNTH_MAIN, chord_n, s_d.wave, -1);
350 if (s_d.sub_enabled) {
351 drone_configure_wave_synth(DRONE_SYNTH_SUB, DRONE_SUB_VOICES, s_d.wave, -1);
352 }
353 } else if (s_d.source == DRONE_SRC_PATCH && s_d.patch >= SEQ_PATCH_WAVE_BASE) {
354 /* PATCH-mode wave virtual patches (257-261, and — AMY_WAVETABLE —
355 * 267-271): configure as a wave synth using the full drone
356 * stutter/filter/mod setup, waveform derived from the patch number.
357 * Sends these ranges as AMY patch numbers would be silent (they are
358 * not real AMY patches), so we intercept here like the melodic
359 * sequencer does in sequencer_configure_synth(). */
360 static const uint16_t s_wave_for_patch[] = {
361 SINE, SAW_DOWN, SAW_UP, PULSE, TRIANGLE,
362 };
363 uint16_t wave = SAW_DOWN;
364 int16_t wt_preset = -1;
365#if CONFIG_AMY_WAVETABLE
366 if (s_d.patch >= SEQ_PATCH_WAVETABLE_BASE && s_d.patch <= SEQ_PATCH_WAVETABLE_MAX) {
367 wave = WAVETABLE;
368 wt_preset = (int16_t)(pcm_wavetable_base + (s_d.patch - SEQ_PATCH_WAVETABLE_BASE));
369 } else
370#endif
371 {
372 uint16_t widx = s_d.patch - SEQ_PATCH_WAVE_BASE;
373 wave = (widx < 5) ? s_wave_for_patch[widx] : SAW_DOWN;
374 }
375 drone_configure_wave_synth(DRONE_SYNTH_MAIN, chord_n, wave, wt_preset);
376 if (s_d.sub_enabled) {
377 drone_configure_wave_synth(DRONE_SYNTH_SUB, DRONE_SUB_VOICES, wave, wt_preset);
378 }
379 } else {
380 drone_configure_patch_synth(DRONE_SYNTH_MAIN, chord_n);
381 if (s_d.sub_enabled) {
382 drone_configure_patch_synth(DRONE_SYNTH_SUB, DRONE_SUB_VOICES);
383 }
384 }
385 /* Re-impose the user's custom ADSR if authored (rebuild/patch resets the
386 * synth oscs). Deferred authority, matching the melodic + arp behaviour. */
387 if (s_d.vp.env_authored) {
388 seq_env_t env_to_push = s_d.vp.env;
389 /* Attack-floor only (is_ks=false): the drone never zeroes KS sustain,
390 * and KS/NOISE are excluded from its wave cycle anyway — this guards
391 * stale persisted values without changing the envelope shape. */
392 voice_env_apply_ks_noise_floor(&env_to_push, false,
393 s_d.wave == KS || s_d.wave == NOISE);
394 sequencer_core_push_envelope(DRONE_SYNTH_MAIN, &env_to_push);
395 if (s_d.sub_enabled) {
396 sequencer_core_push_envelope(DRONE_SYNTH_SUB, &env_to_push);
397 }
398 }
399 /* Second envelope (EG1): only pushed when authored — nothing in the
400 * drone's own WAVE/PATCH setup above routes a coef to COEF_EG1, so this
401 * is a no-op unless the loaded PATCH-mode instrument already does. */
402 if (s_d.vp.env1_authored) {
403 sequencer_core_push_envelope_eg1(DRONE_SYNTH_MAIN, 0, &s_d.vp.env1);
404 if (s_d.sub_enabled) {
405 sequencer_core_push_envelope_eg1(DRONE_SYNTH_SUB, 0, &s_d.vp.env1);
406 }
407 }
408 s_d.last_lfo_hz = drone_lfo_hz();
409}

421/* Start/stop the sustained drone voices. The main synth gets a note-on per
422 * chord note (one voice each); the sub gets a single note at the chord ROOT
423 * shifted by sub_interval. On disable we release the same notes so the ADSR
424 * release fades them out. */
425static void drone_apply_enabled(void)
426{
427 const int8_t *formula = quantizer_chord_intervals(s_d.chord);
428 uint8_t chord_n = drone_chord_note_count(s_d.chord);
429 if (chord_n < 1) {
430 /* NULL or empty chord: play root note only. */
431 drone_note(DRONE_SYNTH_MAIN, s_d.enabled, (float)s_d.root_note);
432 } else {
433 for (uint8_t i = 0; i < chord_n; i++) {
434 if (!formula || formula[i] < 0) break;
435 int midi = SEQ_CLAMP_INT((int)s_d.root_note + (int)formula[i], 0, 127);
436 drone_note(DRONE_SYNTH_MAIN, s_d.enabled, (float)midi);
437 }
438 }
439
440 if (s_d.sub_enabled) {
441 int sub = SEQ_CLAMP_INT((int)s_d.root_note + (int)s_d.sub_interval, 12, 108);
442 drone_note(DRONE_SYNTH_SUB, s_d.enabled, (float)sub);
443 }
444}

458void drone_core_init(void)
459{
460 amy_helpers_init();
461
462 memset(&s_d, 0, sizeof(s_d));
463 voice_params_init_defaults(&s_d.vp); /* unauthored, amp trim at unity */
464 s_d.enabled = false;
465 s_d.source = DRONE_SRC_WAVE;
466 s_d.wave = SAW_DOWN;
467 s_d.chord = CHORD_MIN7;
468 s_d.root_note = DRONE_ROOT_DEFAULT;
469 s_d.resonance = 1.5f;
470 s_d.amp_peak = 0.5f; /* on-beat level (linear; 0.5 = -6 dB) */
471 s_d.amp_duck = 0.5f; /* duck depth knob (0.5 -> 20 dB duck) */
472 s_d.rate = DRONE_RATE_1_16;
473 s_d.patch = 25;
474 s_d.sub_enabled = true;
475 s_d.sub_interval = -12; /* one octave below */
476 s_d.sweep_lo = 600.0f;
477 s_d.sweep_hi = 2000.0f;
478 s_d.sweep_bars = 4;
479 s_d.gate_len = 0.5f; /* 50/50 square = legacy gate */
480 s_d.swing_pct = 0; /* straight grid */
481 s_d.blip_depth = 0.0f; /* filter-zap off by default */
482 s_d.pattern = DRONE_PAT_FULL;
483 s_d.last_blip_cutoff = -1.0f;
484 /* Default ADSR: slow swell suited to a drone. Not authored until the user
485 * commits in the graph editor (the raw wave plays at full sustain otherwise,
486 * since an unauthored env is not pushed and EG0 holds at 1.0 on a held note). */
487 s_d.vp.env.attack_ms = 200;
488 s_d.vp.env.decay_ms = 300;
489 s_d.vp.env.sustain_pct = 100;
490 s_d.vp.env.release_ms = 600;
491 s_d.vp.env.eg_type = 0; /* ENVELOPE_NORMAL */

509void drone_core_service(void)
510{
511 /* Drain the coalesced rebuild BEFORE the enabled gate: a param changed
512 * while the drone is off must still rebuild (cheap synth reconfig only;
513 * drone_apply_enabled() is what actually sounds notes) so the change is
514 * live on the next enable. Drained before the LFO-hz tracking below so a
515 * rebuild and the tempo push land in the right order within the frame. */
516 if (s_d_dirty) {
517 s_d_dirty = false;
518 drone_rebuild();
519 if (s_d.enabled) drone_apply_enabled();
520 }
521
522 if (!s_d.enabled) return;
523
524 /* Keep the LFO locked to tempo: re-send osc1 freq if the BPM changed the
525 * stutter rate by more than a small epsilon. */
526 float lfo_hz = drone_lfo_hz();
527 if (fabsf(lfo_hz - s_d.last_lfo_hz) > 0.01f) {
528 amy_event *e = amy_helpers_event_begin();
529 e->synth = DRONE_SYNTH_MAIN;
530 e->osc = 1;
531 e->freq_coefs[COEF_CONST] = lfo_hz;
532 e->freq_coefs[COEF_NOTE] = 0.0f;
533 amy_helpers_event_send(e);
534 if (s_d.sub_enabled) {
535 e = amy_helpers_event_begin();
536 e->synth = DRONE_SYNTH_SUB;
537 e->osc = 1;
538 e->freq_coefs[COEF_CONST] = lfo_hz;
539 e->freq_coefs[COEF_NOTE] = 0.0f;
540 amy_helpers_event_send(e);
541 }
542 s_d.last_lfo_hz = lfo_hz;
543 }
544
545 /* Everything below is a pure function of the global musical clock, NOT of how
546 * often this service runs. At 48 PPQ one bar (4 beats) = 192 ticks. This keeps
547 * the sweep, the stutter subdivision grid, swing, the step pattern and the
548 * per-step filter blip all frame-rate-independent and beat-locked: they
549 * advance with tempo automatically and stay coherent with the sequencer/arp
550 * bar grid across BPM changes. */
551 const uint32_t bar_ticks = (uint32_t)(AMY_SEQUENCER_PPQ * 4); /* 192 */
552 uint32_t now = sequencer_ticks();
553
554 /* (a) Slow bar-length sweep = the BASE cutoff. */
555 uint32_t period_ticks = (uint32_t)s_d.sweep_bars * bar_ticks;
556 if (period_ticks < 1) period_ticks = 1;
557 uint32_t pos = now % period_ticks;
558 float phase = (2.0f * (float)M_PI) * ((float)pos / (float)period_ticks);
559 float mid = 0.5f * (s_d.sweep_lo + s_d.sweep_hi);
560 float half = 0.5f * (s_d.sweep_hi - s_d.sweep_lo);
561 float base = mid + half * sinf(phase);
562
563 /* (b) Stutter subdivision grid. ticks_per_sub = 192 / (4 * rate_mult), e.g.
564 * 1/16 -> rate_mult 4 -> 12 ticks/sub. Guard against 0. The pattern mask is
565 * an 8-step per-bar grid, so the step index wraps mod DRONE_PAT_STEPS. */
566 float subs_per_bar = 4.0f * s_rate_mult[s_d.rate];
567 uint32_t ticks_per_sub = (uint32_t)((float)bar_ticks / subs_per_bar + 0.5f);
568 if (ticks_per_sub < 1) ticks_per_sub = 1;
569
570 /* (c) Swing: push ODD subdivisions later by swing_pct% of one subdivision.
571 * Applied by offsetting the clock used for the sub-phase calc. */
572 uint32_t pos_in_bar = now % bar_ticks;
573 uint32_t sub_index_raw = pos_in_bar / ticks_per_sub;
574 uint32_t swing_off = 0;
575 if ((sub_index_raw & 1u) && s_d.swing_pct > 0) {
576 swing_off = (uint32_t)((float)ticks_per_sub * (float)s_d.swing_pct / 100.0f);
577 }
578 /* effective position within the current subdivision (0..1), swing-shifted */
579 uint32_t swung = (now + bar_ticks - swing_off) % bar_ticks; /* avoid underflow */
580 uint32_t sub_index = (swung / ticks_per_sub) % DRONE_PAT_STEPS;
581 float frac_in_sub = (float)(swung % ticks_per_sub) / (float)ticks_per_sub;
582
583 /* (d) Pattern mask: a 0 bit closes the filter for that step (skips it). */
584 bool step_on = (s_pattern_mask[s_d.pattern] >> sub_index) & 1u;
585
586 /* (e) Per-step filter blip: a downward zap at the start of each open step,
587 * decaying as we move through the subdivision. cutoff = base * (1 - depth *
588 * env), env = exp(-k * frac). AMY's logfreq downward slew-limit smooths the
589 * attack edge into something musical. */
590 float cutoff;
591 if (!step_on) {
592 cutoff = DRONE_SWEEP_MIN; /* closed: skip this subdivision */
593 } else if (s_d.blip_depth > 0.0001f) {
594 float env = expf(-4.0f * frac_in_sub); /* 1 -> ~0 across the step */
595 cutoff = base * (1.0f - s_d.blip_depth * env);
596 if (cutoff < DRONE_SWEEP_MIN) cutoff = DRONE_SWEEP_MIN;
597 } else {
598 cutoff = base; /* blip off = plain sweep (legacy) */
599 }
600
601 /* Skip redundant pushes (cutoff barely changed) to keep the event queue light
602 * when the drone is idling on a sustained step. */
603 if (fabsf(cutoff - s_d.last_blip_cutoff) > 1.0f) {
604 drone_push_cutoff(DRONE_SYNTH_MAIN, cutoff);
605 if (s_d.sub_enabled) {
606 drone_push_cutoff(DRONE_SYNTH_SUB, cutoff * 0.5f); /* sub sits lower */
607 }
608 s_d.last_blip_cutoff = cutoff;
609 }
610}
