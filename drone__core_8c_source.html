173/* ── Peak/Duck dB amp helpers ──────────────────────────────────────────────
174 * SINGLE source of truth for the engine math: drone_configure_wave_synth() and
175 * drone_get_amp_levels_norm() both call these - never inline the formula.
176 *
177 * Knobs: peak_lin = amp_peak; duck_db = amp_duck * 40 (0..-40 dB floor).
178 * Coefs: m = duck_db/120; const_sent = peak_lin * 10^(-duck_db/40), solved so
179 * AMY's amp = const_sent * 10^(3*m*LFO) gives peak_lin on-beat (LFO=+1)
180 * and peak_lin * 10^(-duck_db/20) off-beat (LFO=-1).
181 */
182static inline float s_amp_peak_lin(void)
183{
184 /* Fold the graph editor's per-target trim (0..1) into amp_peak here so
185 * every caller picks up the trimmed level. */
186 float v = s_d.amp_peak * s_d.vp.amp_trim;
187 v = SEQ_CLAMP_F32(v, 0.0f, 1.0f);
188 return v;
189}
190
191static inline float s_amp_duck_db(void)
192{
193 return s_d.amp_duck * 40.0f;
194}
195
196static inline float s_amp_m(void)
197{
198 return s_amp_duck_db() / 120.0f;
199}
200
201static inline float s_amp_const_sent(void)
202{
203 return s_amp_peak_lin() * powf(10.0f, -s_amp_duck_db() / 40.0f);
204}

226 if (!row) return 0;

235/* ── Synth configuration (WAVE mode) ──
236 * Build a `voices`-voice synth, each voice = osc0 carrier (NOTE-following)
237 * amplitude-gated by an osc1 PULSE LFO, so a multi-voice main synth sounds a
238 * chord when fed multiple notes.
239 * `wave`: carrier waveform (s_d.wave, or patch-derived in PATCH mode).
240 * `wt_preset`: wavetable bank index (pcm_wavetable_base..+samples-1) when
241 * wave==WAVETABLE; -1 elsewhere (no-op). */
242static void drone_configure_wave_synth(uint8_t synth, uint8_t voices, uint16_t wave,
243 int16_t wt_preset)
244{
245 float lfo_hz = drone_lfo_hz();
246 /* AMY 1.2.12 (#720) dB/exp amp model: amp = const_sent * 10^(3*m*LFO).
247 * Peak/Duck mapping: see the s_amp_* helpers. */
248 float m = s_amp_m();
249 float const_sent = s_amp_const_sent();
250
251 /* Shared skeleton: N voices, 2 oscs each, no patch; osc0 NOTE-following so
252 * each voice plays its own chord note. Velocity does not scale amp
253 * (osc0_amp_vel=0); EG0 multiplies the whole amp, so the ADSR shapes the
254 * drone swell/fade *around* the LFO stutter. */
255 voice_wave_cfg_t cfg = {
256 .synth = synth,
257 .num_voices = voices,
258 .oscs_per_voice = DRONE_OSCS_PER_VC,
259 .wave = wave,
260 .osc0_amp_const = const_sent,
261 .osc0_amp_vel = 0.0f,
262 .ks_feedback_authored = false, /* fixed 0.9 KS default */
263 .ks_feedback = 0.0f,
264 .wt_preset = wt_preset,
265 };
266 voice_build_wave(&cfg);
267
268 /* osc1 = PULSE LFO, absolute Hz, full const amp. The PULSE duty IS the
269 * gate length: 0.5 = square, lower = tighter percussive chop. */
270 amy_event *e = amy_helpers_event_begin();
271 e->synth = synth;
272 e->osc = 1;
273 e->wave = PULSE;
274 e->duty_coefs[COEF_CONST]= s_d.gate_len;
275 e->freq_coefs[COEF_CONST]= lfo_hz;
276 e->freq_coefs[COEF_NOTE] = 0.0f; /* absolute Hz, ignore note */
277 e->amp_coefs[COEF_CONST] = 1.0f;
278 e->amp_coefs[COEF_VEL] = 0.0f; /* turn off the default vel/eg amp */
279 e->amp_coefs[COEF_EG0] = 0.0f;
280 amy_helpers_event_send(e);
281
282 /* osc0: amp = const + mod(=osc1) stutter gate, plus the LPF24 sweep filter.
283 * Sent after osc1 so mod_source always points at a configured oscillator. */
284 e = amy_helpers_event_begin();
285 e->synth = synth;
286 e->osc = 0;
287 e->amp_coefs[COEF_MOD] = m;
288 e->mod_source[0] = 1; /* osc1 of this voice (base-osc rel) */
289 e->filter_type = FILTER_LPF24;
290 e->filter_freq_coefs[COEF_CONST] = s_d.sweep_hi;
291 e->resonance = s_d.resonance;
292 amy_helpers_event_send(e);
293}

318/* (Re)build both carrier synths for the current source/params. Main is sized to
319 * the chord's note count; the sub is always a single voice. */
320static void drone_rebuild(void)
321{
322 uint8_t chord_n = drone_chord_note_count(s_d.chord);
323 if (chord_n < 1) chord_n = 1;
324
325 if (s_d.source == DRONE_SRC_WAVE) {
326 drone_configure_wave_synth(DRONE_SYNTH_MAIN, chord_n, s_d.wave, -1);
327 if (s_d.sub_enabled) {
328 drone_configure_wave_synth(DRONE_SYNTH_SUB, DRONE_SUB_VOICES, s_d.wave, -1);
329 }
330 } else if (s_d.source == DRONE_SRC_PATCH && s_d.patch >= SEQ_PATCH_WAVE_BASE) {
331 /* Virtual wave patches (257-261, plus 267-271 with AMY_WAVETABLE) are
332 * not real AMY patches - sending them as patch numbers is silent. Map
333 * to a wave synth here, as sequencer_configure_synth() does. */
334 static const uint16_t s_wave_for_patch[] = {
335 SINE, SAW_DOWN, SAW_UP, PULSE, TRIANGLE,
336 };
337 uint16_t wave = SAW_DOWN;
338 int16_t wt_preset = -1;
339#if CONFIG_AMY_WAVETABLE
340 if (s_d.patch >= SEQ_PATCH_WAVETABLE_BASE && s_d.patch <= SEQ_PATCH_WAVETABLE_MAX) {
341 wave = WAVETABLE;
342 wt_preset = (int16_t)(pcm_wavetable_base + (s_d.patch - SEQ_PATCH_WAVETABLE_BASE));
343 } else
344#endif
345 {
346 uint16_t widx = s_d.patch - SEQ_PATCH_WAVE_BASE;
347 wave = (widx < 5) ? s_wave_for_patch[widx] : SAW_DOWN;
348 }
349 drone_configure_wave_synth(DRONE_SYNTH_MAIN, chord_n, wave, wt_preset);
350 if (s_d.sub_enabled) {
351 drone_configure_wave_synth(DRONE_SYNTH_SUB, DRONE_SUB_VOICES, wave, wt_preset);
352 }
353 } else {
354 drone_configure_patch_synth(DRONE_SYNTH_MAIN, chord_n);
355 if (s_d.sub_enabled) {
356 drone_configure_patch_synth(DRONE_SYNTH_SUB, DRONE_SUB_VOICES);
357 }
358 }
359 /* Re-impose the authored ADSR: rebuild/patch resets the synth oscs.
360 * Deferred authority, matching melodic + arp. */
361 if (s_d.vp.env_authored) {
362 sequencer_core_push_envelope(DRONE_SYNTH_MAIN, &s_d.vp.env);
363 if (s_d.sub_enabled) {
364 sequencer_core_push_envelope(DRONE_SYNTH_SUB, &s_d.vp.env);
365 }
366 }
367 /* EG1: nothing in the drone's own setup routes a coef to COEF_EG1, so this
368 * is a no-op unless a loaded PATCH-mode instrument already does. */
369 if (s_d.vp.env1_authored) {
370 sequencer_core_push_envelope_eg1(DRONE_SYNTH_MAIN, 0, &s_d.vp.env1);
371 if (s_d.sub_enabled) {
372 sequencer_core_push_envelope_eg1(DRONE_SYNTH_SUB, 0, &s_d.vp.env1);
373 }
374 }
375 s_d.last_lfo_hz = drone_lfo_hz();
376}

395/* Start/stop the sustained drone voices: one note-on per chord note on main,
396 * one at root+sub_interval on the sub. Disable releases the same notes so the
397 * ADSR release fades them out. */
398static void drone_apply_enabled(void)
399{
400 const bool on = drone_sounding();
401 const int8_t *formula = quantizer_chord_intervals(s_d.chord);
402 uint8_t chord_n = drone_chord_note_count(s_d.chord);
403 if (chord_n < 1) {
404 /* NULL or empty chord: play root note only. */
405 drone_note(DRONE_SYNTH_MAIN, on, (float)s_d.root_note);
406 } else {
407 for (uint8_t i = 0; i < chord_n; i++) {
408 if (!formula || formula[i] < 0) break;
409 int midi = SEQ_CLAMP_INT((int)s_d.root_note + (int)formula[i], 0, 127);
410 drone_note(DRONE_SYNTH_MAIN, on, (float)midi);
411 }
412 }
413
414 if (s_d.sub_enabled) {
415 int sub = SEQ_CLAMP_INT((int)s_d.root_note + (int)s_d.sub_interval, 12, 108);
416 drone_note(DRONE_SYNTH_SUB, on, (float)sub);
417 }
418}

433{
434 amy_helpers_init();
435
436 memset(&s_d, 0, sizeof(s_d));
437 voice_params_init_defaults(&s_d.vp); /* unauthored, amp trim at unity */
438 s_d.enabled = false;
439 s_d.source = DRONE_SRC_WAVE;
440 s_d.wave = SAW_DOWN;
441 s_d.chord = CHORD_OFF; /* root note only until a chord is chosen */
442 s_d.root_note = DRONE_ROOT_DEFAULT;
443 s_d.resonance = 1.5f;
444 s_d.amp_peak = 0.5f; /* on-beat level (linear; 0.5 = -6 dB) */
445 s_d.amp_duck = 0.5f; /* duck depth knob (0.5 -> 20 dB duck) */
446 s_d.rate = DRONE_RATE_1_16;
447 s_d.patch = 25;
448 s_d.sub_enabled = true;
449 s_d.sub_interval = -12; /* one octave below */
450 s_d.sweep_lo = 600.0f;
451 s_d.sweep_hi = 2000.0f;
452 s_d.sweep_bars = 4;
453 s_d.gate_len = 0.5f; /* 50/50 square = legacy gate */
454 s_d.swing_pct = 0; /* straight grid */
455 s_d.blip_depth = 0.0f; /* filter-zap off by default */
456 s_d.pattern = DRONE_PAT_FULL;
457 s_d.last_blip_cutoff = -1.0f;
458 /* Default ADSR: slow drone swell. Stays unauthored until the user commits
459 * in the graph editor - an unauthored env is never pushed, so EG0 holds at
460 * 1.0 and the raw wave plays at full sustain. */
461 s_d.vp.env.attack_ms = 200;
462 s_d.vp.env.decay_ms = 300;
463 s_d.vp.env.sustain_pct = 100;
464 s_d.vp.env.release_ms = 600;
465 s_d.vp.env.eg_type = 0; /* ENVELOPE_NORMAL */
466 /* EG1 is unrouted by the drone's own WAVE patches - filter movement comes
467 * from the sweep + BPM-synced service tick, and a second envelope-driven
468 * cutoff shift would double-modulate it. Kept as timing storage for a
469 * PATCH-mode drone whose AMY patch routes its own bp1. */
470 s_d.vp.env1.attack_ms = 15;
471 s_d.vp.env1.decay_ms = 400;
472 s_d.vp.env1.sustain_pct = 25;
473 s_d.vp.env1.release_ms = 400;
474 s_d.vp.env1.eg_type = 0; /* ENVELOPE_NORMAL */
475
476 drone_rebuild();
477 ESP_LOGI(TAG, "drone_core initialized (synths %u/%u)",
478 DRONE_SYNTH_MAIN, DRONE_SYNTH_SUB);
479}

481void drone_core_service(void)
482{
483 /* Drain the coalesced rebuild BEFORE the enabled gate: a param changed
484 * while the drone is off must still rebuild so it is live on the next
485 * enable (rebuild only reconfigures; drone_apply_enabled() sounds notes).
486 * Before the LFO-hz tracking so rebuild and tempo push order correctly. */
487 if (s_d_dirty) {
488 s_d_dirty = false;
489 drone_rebuild();
490 if (s_d.enabled) drone_apply_enabled();
491 }
492
493 if (!s_d.enabled) return;
494
495 /* Keep the LFO locked to tempo: re-send osc1 freq if the BPM changed the
496 * stutter rate by more than a small epsilon. */
497 float lfo_hz = drone_lfo_hz();
498 if (fabsf(lfo_hz - s_d.last_lfo_hz) > 0.01f) {
499 amy_event *e = amy_helpers_event_begin();
500 e->synth = DRONE_SYNTH_MAIN;
501 e->osc = 1;
502 e->freq_coefs[COEF_CONST] = lfo_hz;
503 e->freq_coefs[COEF_NOTE] = 0.0f;
504 amy_helpers_event_send(e);
505 if (s_d.sub_enabled) {
506 e = amy_helpers_event_begin();
507 e->synth = DRONE_SYNTH_SUB;
508 e->osc = 1;
509 e->freq_coefs[COEF_CONST] = lfo_hz;
510 e->freq_coefs[COEF_NOTE] = 0.0f;
511 amy_helpers_event_send(e);
512 }
513 s_d.last_lfo_hz = lfo_hz;

515
516 /* Everything below is a pure function of the global musical clock, NOT of
517 * how often this service runs - sweep, stutter grid, swing, pattern and
518 * blip stay frame-rate-independent and beat-locked across BPM changes. */
519 const uint32_t bar_ticks = (uint32_t)(AMY_SEQUENCER_PPQ * 4); /* 192 */
520 /* AMY's monotonic 48 PPQ tick counter, advanced by the audio-rate
521 * sequencer: the same clock the sequencer and arp ride. */
522 uint32_t now = sequencer_ticks();
523
524 /* (a) Slow bar-length sweep = the BASE cutoff. */
525 uint32_t period_ticks = (uint32_t)s_d.sweep_bars * bar_ticks;
526 if (period_ticks < 1) period_ticks = 1;
527 uint32_t pos = now % period_ticks;
528 float phase = (2.0f * (float)M_PI) * ((float)pos / (float)period_ticks);
529 float mid = 0.5f * (s_d.sweep_lo + s_d.sweep_hi);
530 float half = 0.5f * (s_d.sweep_hi - s_d.sweep_lo);
531 float base = mid + half * sinf(phase);
532
533 /* (b) Stutter subdivision grid; the pattern mask is an 8-step per-bar grid,
534 * so the step index wraps mod DRONE_PAT_STEPS. */
535 float subs_per_bar = 4.0f * s_rate_mult[s_d.rate];
536 uint32_t ticks_per_sub = (uint32_t)((float)bar_ticks / subs_per_bar + 0.5f);
537 if (ticks_per_sub < 1) ticks_per_sub = 1;
538
539 /* (c) Swing: push ODD subdivisions later by swing_pct% of one subdivision,
540 * by offsetting the clock used for the sub-phase calc. */
541 uint32_t pos_in_bar = now % bar_ticks;
542 uint32_t sub_index_raw = pos_in_bar / ticks_per_sub;
543 uint32_t swing_off = 0;
544 if ((sub_index_raw & 1u) && s_d.swing_pct > 0) {
545 swing_off = (uint32_t)((float)ticks_per_sub * (float)s_d.swing_pct / 100.0f);
546 }
547 /* effective position within the current subdivision (0..1), swing-shifted */
548 uint32_t swung = (now + bar_ticks - swing_off) % bar_ticks; /* avoid underflow */
549 uint32_t sub_index = (swung / ticks_per_sub) % DRONE_PAT_STEPS;
550 float frac_in_sub = (float)(swung % ticks_per_sub) / (float)ticks_per_sub;
551
552 /* (d) Pattern mask: a 0 bit closes the filter for that step (skips it). */
553 bool step_on = (s_pattern_mask[s_d.pattern] >> sub_index) & 1u;
554
555 /* (e) Per-step filter blip: a downward zap at the start of each open step,
556 * decaying across the subdivision. AMY's logfreq downward slew-limit
557 * smooths the attack edge into something musical. */
558 float cutoff;
559 if (!step_on) {
560 cutoff = DRONE_SWEEP_MIN; /* closed: skip this subdivision */
561 } else if (s_d.blip_depth > 0.0001f) {
562 float env = expf(-4.0f * frac_in_sub); /* 1 -> ~0 across the step */
563 cutoff = base * (1.0f - s_d.blip_depth * env);
564 if (cutoff < DRONE_SWEEP_MIN) cutoff = DRONE_SWEEP_MIN;
565 } else {
566 cutoff = base; /* blip off = plain sweep (legacy) */
567 }
568
569 /* Skip near-identical pushes to keep the event queue light while idling on
570 * a sustained step. */
571 if (fabsf(cutoff - s_d.last_blip_cutoff) > 1.0f) {
572 drone_push_cutoff(DRONE_SYNTH_MAIN, cutoff);
573 if (s_d.sub_enabled) {
574 drone_push_cutoff(DRONE_SYNTH_SUB, cutoff * 0.5f); /* sub sits lower */
575 }
576 s_d.last_blip_cutoff = cutoff;
577 }
578}
