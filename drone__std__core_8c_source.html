1/* drone_std_core.c — normal (free-running) drone synth.
2 *
3 * The stutter drone's sibling (see drone_std_core.h for the design split and
4 * drone_core.c for the concurrency rationale shared by both): chord + mono
5 * sub voicing on its own AMY slots, sustained output, and the standard
6 * per-voice toolset (free filter, native LFO, graph-editor envelopes) instead
7 * of the stutter machinery.
8 *
9 * All AMY interaction goes through amy_helpers deltas — never amy_queue_lock,
10 * never direct synth[] access. Callers are FreeRTOS tasks, never ISRs. */

74static uint8_t drone_std_chord_note_count(chord_type_t chord)
75{
76 const int8_t *row = quantizer_chord_intervals(chord);
77 if (!row) return 0;
78 uint8_t n = 0;
79 for (uint8_t i = 0; i < DRONE_CHORD_MAX_NOTES; i++) {
80 if (row[i] < 0) break;
81 n++;
82 }
83 return n;
84}

92/* ── Synth configuration (WAVE mode) ──
93 * Shared skeleton + free filter + native LFO. No amp MOD coupling: under the
94 * 1.2.12 dB combine model a plain CONST level is exactly level_lin (EG0 =1). */
95static void drone_std_configure_wave_synth(uint8_t synth, uint8_t voices,
96 uint16_t wave, int16_t wt_preset)
97{
98 voice_wave_cfg_t cfg = {
99 .synth = synth,
100 .num_voices = voices,
101 .oscs_per_voice = DRONE_STD_OSCS_PER_VC,
102 .wave = wave,
103 .osc0_amp_const = drone_std_level_lin(),
104 .osc0_amp_vel = 0.0f,
105 .ks_feedback_authored = false,
106 .ks_feedback_q = 0.0f,
107 .wt_preset = wt_preset,
108 };
109 voice_build_wave(&cfg);
110
111 drone_std_push_filter(synth, &s_ds.vp.filter);
112
113 /* ALWAYS applied (NULL clears): re-sending the same voice count does not
114 * reset the osc pool, so a stale COEF_MOD from a prior LFO target would
115 * keep modulating (the stale-AMP case rails — see voice_config.h). */
116 voice_apply_native_lfo(synth, drone_std_lfo_on() ? &s_ds.vp.lfo : NULL,
117 seq_get_bpm());
118}

120/* ── Synth configuration (PATCH mode) ──
121 * Real AMY patch: it owns its osc topology, so no native LFO is layered on. */
122static void drone_std_configure_patch_synth(uint8_t synth, uint8_t voices)
123{
124 amy_event *e = amy_helpers_event_begin();
125 e->synth = synth;
126 e->num_voices = voices;
127 e->patch_number = s_ds.patch;
128 amy_helpers_event_send(e);
129
130 /* Patch strings carry global EQ/chorus commands; keep them per-synth. */
131 synth_ui_fx_reassert_global();
132
133 drone_std_push_filter(synth, &s_ds.vp.filter);
134}

136/* (Re)build both synths for the current source/params. Main is sized to the
137 * chord's note count; the sub is always a single voice. */
138static void drone_std_rebuild(void)
139{
140 uint8_t chord_n = drone_std_chord_note_count(s_ds.chord);
141 if (chord_n < 1) chord_n = 1;
142
143 if (s_ds.source == DRONE_SRC_WAVE) {
144 drone_std_configure_wave_synth(DRONE_STD_SYNTH_MAIN, chord_n, s_ds.wave, -1);
145 if (s_ds.sub_enabled) {
146 drone_std_configure_wave_synth(DRONE_STD_SYNTH_SUB, DRONE_STD_SUB_VOICES,
147 s_ds.wave, -1);
148 }
149 } else if (s_ds.source == DRONE_SRC_PATCH && s_ds.patch >= SEQ_PATCH_WAVE_BASE) {
150 /* Wave/wavetable virtual patches: not real AMY patches — intercept and
151 * build as a wave synth, same mapping as the stutter drone/melodic. */
152 static const uint16_t s_wave_for_patch[] = {
153 SINE, SAW_DOWN, SAW_UP, PULSE, TRIANGLE,
154 };
155 uint16_t wave = SAW_DOWN;
156 int16_t wt_preset = -1;
157#if CONFIG_AMY_WAVETABLE
158 if (s_ds.patch >= SEQ_PATCH_WAVETABLE_BASE && s_ds.patch <= SEQ_PATCH_WAVETABLE_MAX) {
159 wave = WAVETABLE;
160 wt_preset = (int16_t)(pcm_wavetable_base + (s_ds.patch - SEQ_PATCH_WAVETABLE_BASE));
161 } else
162#endif
163 {
164 uint16_t widx = s_ds.patch - SEQ_PATCH_WAVE_BASE;
165 wave = (widx < 5) ? s_wave_for_patch[widx] : SAW_DOWN;
166 }
167 drone_std_configure_wave_synth(DRONE_STD_SYNTH_MAIN, chord_n, wave, wt_preset);
168 if (s_ds.sub_enabled) {
169 drone_std_configure_wave_synth(DRONE_STD_SYNTH_SUB, DRONE_STD_SUB_VOICES,
170 wave, wt_preset);
171 }
172 } else {
173 drone_std_configure_patch_synth(DRONE_STD_SYNTH_MAIN, chord_n);
174 if (s_ds.sub_enabled) {
175 drone_std_configure_patch_synth(DRONE_STD_SYNTH_SUB, DRONE_STD_SUB_VOICES);
176 }
177 }
178
179 /* Deferred authority: re-impose the user's envelopes after any rebuild. */
180 if (s_ds.vp.env_authored) {
181 seq_env_t env_to_push = s_ds.vp.env;
182 voice_env_apply_ks_noise_floor(&env_to_push, false,
183 s_ds.wave == KS || s_ds.wave == NOISE);
184 sequencer_core_push_envelope(DRONE_STD_SYNTH_MAIN, &env_to_push);
185 if (s_ds.sub_enabled) {
186 sequencer_core_push_envelope(DRONE_STD_SYNTH_SUB, &env_to_push);
187 }
188 }
189 if (s_ds.vp.env1_authored) {
190 sequencer_core_push_envelope_eg1(DRONE_STD_SYNTH_MAIN, 0, &s_ds.vp.env1);
191 if (s_ds.sub_enabled) {
192 sequencer_core_push_envelope_eg1(DRONE_STD_SYNTH_SUB, 0, &s_ds.vp.env1);
193 }
194 }
195 s_ds.last_lfo_bpm = (float)seq_get_bpm();
196}

208/* Start/stop the sustained voices: one note-on per chord note on the main,
209 * a single root+interval note on the sub. Disable releases the same notes. */
210static void drone_std_apply_enabled(void)
211{
212 const int8_t *formula = quantizer_chord_intervals(s_ds.chord);
213 uint8_t chord_n = drone_std_chord_note_count(s_ds.chord);
214 if (chord_n < 1) {
215 drone_std_note(DRONE_STD_SYNTH_MAIN, s_ds.enabled, (float)s_ds.root_note);
216 } else {
217 for (uint8_t i = 0; i < chord_n; i++) {
218 if (!formula || formula[i] < 0) break;
219 int midi = SEQ_CLAMP_INT((int)s_ds.root_note + (int)formula[i], 0, 127);
220 drone_std_note(DRONE_STD_SYNTH_MAIN, s_ds.enabled, (float)midi);
221 }
222 }
223
224 if (s_ds.sub_enabled) {
225 int sub = SEQ_CLAMP_INT((int)s_ds.root_note + (int)s_ds.sub_interval, 12, 108);
226 drone_std_note(DRONE_STD_SYNTH_SUB, s_ds.enabled, (float)sub);
227 }
228}

232void drone_std_core_init(void)
233{
234 amy_helpers_init();
235
236 memset(&s_ds, 0, sizeof(s_ds));
237 voice_params_init_defaults(&s_ds.vp);
238 s_ds.enabled = false;
239 s_ds.source = DRONE_SRC_WAVE;
240 s_ds.wave = SAW_DOWN;
241 s_ds.chord = CHORD_MIN7;
242 s_ds.root_note = 45; /* A2, same default as the stutter drone */
243 s_ds.level = 0.5f;
244 s_ds.patch = 25;
245 s_ds.sub_enabled = true;
246 s_ds.sub_interval = -12;
247 /* Default ADSR: slow swell (unauthored until the graph editor commits). */
248 s_ds.vp.env.attack_ms = 200;
249 s_ds.vp.env.decay_ms = 300;
250 s_ds.vp.env.sustain_pct = 100;
251 s_ds.vp.env.release_ms = 600;
252 s_ds.vp.env.eg_type = 0;
253 s_ds.vp.env1.attack_ms = 15;
254 s_ds.vp.env1.decay_ms = 400;
255 s_ds.vp.env1.sustain_pct = 25;
256 s_ds.vp.env1.release_ms = 400;
257 s_ds.vp.env1.eg_type = 0;
258 /* Filter defaults: bypass, but a sensible LPF starting point for the
259 * editor to seed from when the user first enables one. */
260 s_ds.vp.filter.filter_type = SEQ_FILTER_LPF;
261 s_ds.vp.filter.cutoff_hz = 1200.0f;
262 s_ds.vp.filter.resonance = 1.0f;
263 s_ds.vp.filter.enabled = false;
264 /* LFO defaults for the editor seed; inert until authored + enabled. */
265 s_ds.vp.lfo.mode = LFO_MODE_FREE;
266 s_ds.vp.lfo.wave = LFO_WAVE_SINE;
267 s_ds.vp.lfo.rate = LFO_RATE_1BAR;
268 s_ds.vp.lfo.depth = 50;
269 s_ds.vp.lfo.targets = LFO_TGT_BIT(LFO_TARGET_FILTER);
270
271 drone_std_rebuild();
272 ESP_LOGI(TAG, "drone_std_core initialized (synths %u/%u)",
273 DRONE_STD_SYNTH_MAIN, DRONE_STD_SYNTH_SUB);
274}

276void drone_std_core_service(void)
277{
278 /* Drain the coalesced rebuild BEFORE the enabled gate so a param changed
279 * while off is live on the next enable. */
280 if (s_ds_dirty) {
281 s_ds_dirty = false;
282 drone_std_rebuild();
283 if (s_ds.enabled) drone_std_apply_enabled();
284 }
285}

287void drone_std_core_refresh_lfo_freq(void)
288{
289 if (!drone_std_lfo_on() || s_ds.source != DRONE_SRC_WAVE) return;
290 float bpm = (float)seq_get_bpm();
291 if (fabsf(bpm - s_ds.last_lfo_bpm) < 0.5f) return;
292 s_ds.last_lfo_bpm = bpm;
293 voice_apply_native_lfo(DRONE_STD_SYNTH_MAIN, &s_ds.vp.lfo, (uint16_t)bpm);
294 if (s_ds.sub_enabled) {
295 voice_apply_native_lfo(DRONE_STD_SYNTH_SUB, &s_ds.vp.lfo, (uint16_t)bpm);
296 }

324void drone_std_set_chord(chord_type_t chord)
325{
326 if (chord >= CHORD_TYPE_COUNT) return;
327 if (s_ds.chord == chord) return;
328 /* Release the old chord before the resize or extra voices stick on. */
329 bool was_enabled = s_ds.enabled;
330 if (was_enabled) {
331 s_ds.enabled = false;
332 drone_std_apply_enabled();
333 }
334 s_ds.chord = chord;
335 drone_std_rebuild();
336 if (was_enabled) {
337 s_ds.enabled = true;
338 drone_std_apply_enabled();

389void drone_std_set_sub_interval(int8_t st)
390{
391 int v = SEQ_CLAMP_INT((int)st, -36, 0);
392 if (s_ds.sub_interval == (int8_t)v) return;
393 bool was_on = s_ds.enabled && s_ds.sub_enabled;
394 if (was_on) {
395 /* Release at the OLD interval before it changes, or the note sticks. */
396 int old_midi = SEQ_CLAMP_INT((int)s_ds.root_note + (int)s_ds.sub_interval,
397 12, 108);
398 drone_std_note(DRONE_STD_SYNTH_SUB, false, (float)old_midi);
399 }
400 s_ds.sub_interval = (int8_t)v;
401 if (was_on) {
402 int new_midi = SEQ_CLAMP_INT((int)s_ds.root_note + (int)s_ds.sub_interval,

404 drone_std_note(DRONE_STD_SYNTH_SUB, true, (float)new_midi);
