1/* drone_std_core.c - normal (free-running) drone synth.
2 *
3 * The stutter drone's sibling: chord + mono sub voicing on its own AMY slots,
4 * sustained output, and the standard per-voice toolset (free filter, native
5 * LFO, graph-editor envelopes) instead of the stutter machinery. Design split:
6 * drone_std_core.h; shared concurrency rationale: drone_core.c.
7 *
8 * All AMY interaction goes through amy_helpers deltas - never amy_queue_lock,
9 * never direct synth[] access. Callers are FreeRTOS tasks, never ISRs. */

76static uint8_t drone_std_chord_note_count(chord_type_t chord)
77{
78 const int8_t *row = quantizer_chord_intervals(chord);
79 if (!row) return 0;
80 uint8_t n = 0;
81 for (uint8_t i = 0; i < DRONE_CHORD_MAX_NOTES; i++) {
82 if (row[i] < 0) break;
83 n++;
84 }
85 return n;
86}

94/* ── Synth configuration (WAVE mode) ──
95 * Shared skeleton + free filter + native LFO. No amp MOD coupling: under the
96 * dB combine model a plain CONST level is exactly level_lin (EG0 = 1). */
97static void drone_std_configure_wave_synth(uint8_t synth, uint8_t voices,
98 uint16_t wave, int16_t wt_preset)
99{
100 voice_wave_cfg_t cfg = {
101 .synth = synth,
102 .num_voices = voices,
103 .oscs_per_voice = DRONE_STD_OSCS_PER_VC,
104 .wave = wave,
105 .osc0_amp_const = drone_std_level_lin(),
106 .osc0_amp_vel = 0.0f,
107 .ks_feedback_authored = false,
108 .ks_feedback = 0.0f,
109 .wt_preset = wt_preset,
110 };
111 voice_build_wave(&cfg);
112
113 drone_std_push_filter(synth, &s_ds.vp.filter);
114
115 /* ALWAYS applied (NULL clears): re-sending the same voice count does not
116 * reset the osc pool, so a stale COEF_MOD from a prior LFO target would
117 * keep modulating - the stale-AMP case rails (voice_config.h). */
118 voice_apply_native_lfo(synth, drone_std_lfo_on() ? &s_ds.vp.lfo : NULL,
119 seq_get_bpm());
120}

122/* ── Synth configuration (PATCH mode) ──
123 * Real AMY patch: it owns its osc topology, so no native LFO is layered on. */
124static void drone_std_configure_patch_synth(uint8_t synth, uint8_t voices)
125{
126 /* Foreign topology on a wave-built slot: void the lazy-LFO shape proof
127 * (see voice_config.h) before the patch reshapes the pool. */
128 voice_lfo_mark_foreign(synth);
129
130 amy_event *e = amy_helpers_event_begin();
131 e->synth = synth;
132 e->num_voices = voices;
133 e->patch_number = s_ds.patch;
134 amy_helpers_event_send(e);
135
136 /* Patch strings carry global EQ/chorus commands; keep them per-synth. */
137 synth_ui_fx_reassert_global();
138
139 drone_std_push_filter(synth, &s_ds.vp.filter);
140}

142/* (Re)build both synths for the current source/params. Main is sized to the
143 * chord's note count; the sub is always a single voice. */
144static void drone_std_rebuild(void)
145{
146 uint8_t chord_n = drone_std_chord_note_count(s_ds.chord);
147 if (chord_n < 1) chord_n = 1;
148
149 if (s_ds.source == DRONE_SRC_WAVE) {
150 drone_std_configure_wave_synth(DRONE_STD_SYNTH_MAIN, chord_n, s_ds.wave, -1);
151 if (s_ds.sub_enabled) {
152 drone_std_configure_wave_synth(DRONE_STD_SYNTH_SUB, DRONE_STD_SUB_VOICES,
153 s_ds.wave, -1);
154 }
155 } else if (s_ds.source == DRONE_SRC_PATCH && s_ds.patch >= SEQ_PATCH_WAVE_BASE) {
156 /* Virtual wave/wavetable patches are not real AMY patches: intercept
157 * and build as a wave synth, same mapping as stutter drone/melodic. */
158 static const uint16_t s_wave_for_patch[] = {
159 SINE, SAW_DOWN, SAW_UP, PULSE, TRIANGLE,
160 };
161 uint16_t wave = SAW_DOWN;
162 int16_t wt_preset = -1;
163#if CONFIG_AMY_WAVETABLE
164 if (s_ds.patch >= SEQ_PATCH_WAVETABLE_BASE && s_ds.patch <= SEQ_PATCH_WAVETABLE_MAX) {
165 wave = WAVETABLE;
166 wt_preset = (int16_t)(pcm_wavetable_base + (s_ds.patch - SEQ_PATCH_WAVETABLE_BASE));
167 } else
168#endif
169 {
170 uint16_t widx = s_ds.patch - SEQ_PATCH_WAVE_BASE;
171 wave = (widx < 5) ? s_wave_for_patch[widx] : SAW_DOWN;
172 }
173 drone_std_configure_wave_synth(DRONE_STD_SYNTH_MAIN, chord_n, wave, wt_preset);
174 if (s_ds.sub_enabled) {
175 drone_std_configure_wave_synth(DRONE_STD_SYNTH_SUB, DRONE_STD_SUB_VOICES,
176 wave, wt_preset);
177 }
178 } else {
179 drone_std_configure_patch_synth(DRONE_STD_SYNTH_MAIN, chord_n);
180 if (s_ds.sub_enabled) {
181 drone_std_configure_patch_synth(DRONE_STD_SYNTH_SUB, DRONE_STD_SUB_VOICES);
182 }
183 }
184
185 /* Per-osc distortion does not survive a rebuild either. */
186 drone_std_reapply_dist();
187
188 /* Deferred authority: re-impose the user's envelopes after any rebuild. */
189 if (s_ds.vp.env_authored) {
190 sequencer_core_push_envelope(DRONE_STD_SYNTH_MAIN, &s_ds.vp.env);
191 if (s_ds.sub_enabled) {
192 sequencer_core_push_envelope(DRONE_STD_SYNTH_SUB, &s_ds.vp.env);
193 }
194 }
195 if (s_ds.vp.env1_authored) {
196 sequencer_core_push_envelope_eg1(DRONE_STD_SYNTH_MAIN, 0, &s_ds.vp.env1);
197 if (s_ds.sub_enabled) {
198 sequencer_core_push_envelope_eg1(DRONE_STD_SYNTH_SUB, 0, &s_ds.vp.env1);
199 }
200 }
201 s_ds.last_lfo_bpm = (float)seq_get_bpm();
202}

214/* Whether the drone should be making sound right now: the user's switch AND not
215 * silenced by a solo elsewhere. */
216static inline bool drone_std_sounding(void)
217{
218 return s_ds.enabled && !s_ds.solo_muted;
219}

221/* Start/stop the sustained voices: one note-on per chord note on the main,
222 * a single root+interval note on the sub. Disable releases the same notes. */
223static void drone_std_apply_enabled(void)
224{
225 const bool on = drone_std_sounding();
226 const int8_t *formula = quantizer_chord_intervals(s_ds.chord);
227 uint8_t chord_n = drone_std_chord_note_count(s_ds.chord);
228 if (chord_n < 1) {
229 drone_std_note(DRONE_STD_SYNTH_MAIN, on, (float)s_ds.root_note);
230 } else {
231 for (uint8_t i = 0; i < chord_n; i++) {
232 if (!formula || formula[i] < 0) break;
233 int midi = SEQ_CLAMP_INT((int)s_ds.root_note + (int)formula[i], 0, 127);
234 drone_std_note(DRONE_STD_SYNTH_MAIN, on, (float)midi);
235 }
236 }
237
238 if (s_ds.sub_enabled) {
239 int sub = SEQ_CLAMP_INT((int)s_ds.root_note + (int)s_ds.sub_interval, 12, 108);
240 drone_std_note(DRONE_STD_SYNTH_SUB, on, (float)sub);
241 }
242}

246void drone_std_core_init(void)
247{
248 amy_helpers_init();
249
250 memset(&s_ds, 0, sizeof(s_ds));
251 voice_params_init_defaults(&s_ds.vp);
252 s_ds.enabled = false;
253 s_ds.source = DRONE_SRC_WAVE;
254 s_ds.wave = SAW_DOWN;
255 s_ds.chord = CHORD_OFF; /* root note only until a chord is chosen */
256 s_ds.root_note = 45; /* A2, same default as the stutter drone */
257 s_ds.level = 0.5f;
258 s_ds.patch = 25;
259 s_ds.sub_enabled = true;
260 s_ds.sub_interval = -12;
261 /* Default ADSR: slow swell (unauthored until the graph editor commits). */
262 s_ds.vp.env.attack_ms = 200;
263 s_ds.vp.env.decay_ms = 300;
264 s_ds.vp.env.sustain_pct = 100;
265 s_ds.vp.env.release_ms = 600;
266 s_ds.vp.env.eg_type = 0;
267 s_ds.vp.env1.attack_ms = 15;
268 s_ds.vp.env1.decay_ms = 400;
269 s_ds.vp.env1.sustain_pct = 25;
270 s_ds.vp.env1.release_ms = 400;
271 s_ds.vp.env1.eg_type = 0;
272 /* Filter defaults: bypass, but a sensible LPF starting point for the
273 * editor to seed from when the user first enables one. */
274 s_ds.vp.filter.filter_type = SEQ_FILTER_LPF;
275 s_ds.vp.filter.cutoff_hz = 1200.0f;
276 s_ds.vp.filter.resonance = 1.0f;
277 s_ds.vp.filter.enabled = false;
278 /* LFO defaults for the editor seed; inert until authored + enabled. */
279 s_ds.vp.lfo.mode = LFO_MODE_FREE;
280 s_ds.vp.lfo.wave = LFO_WAVE_SINE;
281 s_ds.vp.lfo.rate = LFO_RATE_1BAR;
282 s_ds.vp.lfo.depth = 50;
283 s_ds.vp.lfo.targets = LFO_TGT_BIT(LFO_TARGET_FILTER);
284
285 drone_std_rebuild();
286 ESP_LOGI(TAG, "drone_std_core initialized (synths %u/%u)",
287 DRONE_STD_SYNTH_MAIN, DRONE_STD_SYNTH_SUB);
288}

290void drone_std_core_service(void)
291{
292 /* Drain the coalesced rebuild BEFORE the enabled gate so a param changed
293 * while off is live on the next enable. */
294 if (s_ds_dirty) {
295 s_ds_dirty = false;
296 drone_std_rebuild();
297 if (s_ds.enabled) drone_std_apply_enabled();
298 }
299}

301void drone_std_core_refresh_lfo_freq(void)
302{
303 if (!drone_std_lfo_on() || s_ds.source != DRONE_SRC_WAVE) return;
304 float bpm = (float)seq_get_bpm();
305 if (fabsf(bpm - s_ds.last_lfo_bpm) < 0.5f) return;
306 s_ds.last_lfo_bpm = bpm;
307 voice_apply_native_lfo(DRONE_STD_SYNTH_MAIN, &s_ds.vp.lfo, (uint16_t)bpm);
308 if (s_ds.sub_enabled) {
309 voice_apply_native_lfo(DRONE_STD_SYNTH_SUB, &s_ds.vp.lfo, (uint16_t)bpm);
310 }
311}

347void drone_std_set_chord(chord_type_t chord)
348{
349 if (chord >= CHORD_TYPE_COUNT) return;
350 if (s_ds.chord == chord) return;
351 /* Release the old chord before the resize or extra voices stick on. */
352 bool was_enabled = s_ds.enabled;
353 if (was_enabled) {
354 s_ds.enabled = false;
355 drone_std_apply_enabled();
356 }
357 s_ds.chord = chord;
358 drone_std_rebuild();
359 if (was_enabled) {
360 s_ds.enabled = true;
361 drone_std_apply_enabled();
362 }
363}

412void drone_std_set_sub_interval(int8_t st)
413{
414 int v = SEQ_CLAMP_INT((int)st, -36, 0);
415 if (s_ds.sub_interval == (int8_t)v) return;
416 bool was_on = s_ds.enabled && s_ds.sub_enabled;
417 if (was_on) {
418 /* Release at the OLD interval before it changes, or the note sticks. */
419 int old_midi = SEQ_CLAMP_INT((int)s_ds.root_note + (int)s_ds.sub_interval,
420 12, 108);
421 drone_std_note(DRONE_STD_SYNTH_SUB, false, (float)old_midi);
422 }
423 s_ds.sub_interval = (int8_t)v;
424 if (was_on) {
425 int new_midi = SEQ_CLAMP_INT((int)s_ds.root_note + (int)s_ds.sub_interval,
426 12, 108);
427 drone_std_note(DRONE_STD_SYNTH_SUB, true, (float)new_midi);
428 }
