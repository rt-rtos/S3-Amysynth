9AMY_IRAM_ATTR SAMPLE compute_mod_value(uint16_t mod_osc) {
10 // Return the modulation-rate value for the specified oscillator.
11 // i.e., this oscillator is acting as modulation for something, so
12 // just calculate that modulation rate (without knowing what it
13 // modulates).
14 // Has this mod value already been calculated this frame? Can't
15 // recalculate, because compute_mod advance phase internally.
16 if (synth[mod_osc]->mod_value_clock == amy_global.total_blocks*AMY_BLOCK_SIZE)
17 return synth[mod_osc]->mod_value;
18 synth[mod_osc]->mod_value_clock = amy_global.total_blocks*AMY_BLOCK_SIZE;
19 SAMPLE value = 0;
20 if(synth[mod_osc]->wave == NOISE) value = compute_mod_noise(mod_osc);
21 if(synth[mod_osc]->wave == SAW_DOWN) value = compute_mod_saw_down(mod_osc);
22 if(synth[mod_osc]->wave == SAW_UP) value = compute_mod_saw_up(mod_osc);
23 if(synth[mod_osc]->wave == PULSE) value = compute_mod_pulse(mod_osc);
24 if(synth[mod_osc]->wave == TRIANGLE) value = compute_mod_triangle(mod_osc);
25 if(synth[mod_osc]->wave == SINE) value = compute_mod_sine(mod_osc);
26 if(pcm_samples)
27 if(synth[mod_osc]->wave == PCM) value = compute_mod_pcm(mod_osc);
28 if(AMY_HAS_CUSTOM) {
29 if(synth[mod_osc]->wave == CUSTOM) value = compute_mod_custom(mod_osc);
30 }
31 synth[mod_osc]->mod_value = value;

35AMY_IRAM_ATTR SAMPLE compute_mod_scale(uint16_t osc, uint16_t which_source) {
36 // hold_and_modify(source) evaluates the modulator's own COEF_MOD* inputs,
37 // so modulators can themselves be modulated (chained modulators). Cycles
38 // are rejected at assignment time (mod_osc_would_cause_loop in amy.c), so
39 // the recursion is bounded; the per-block memo in compute_mod_value keeps
40 // shared modulators cheap -- including one shared between an osc's own two
41 // mod slots.
42 uint16_t source = synth[osc]->mod_source[which_source];
43 if(AMY_IS_SET(source)) {
44 // synth[source] can be NULL: FREE_OSC returns a released voice's osc
45 // storage to the heap, while every osc that named it as a modulator
46 // still holds its number. amy_render's own null skip doesn't cover
47 // this descent, so recheck here -- the same reason render_osc_wave
48 // rechecks before following chained_osc.
49 if(source != osc && synth[source] != NULL) { // belt-and-braces; assignment already rejects self as source
50 hold_and_modify(source);
51 return compute_mod_value(source);
52 }
53 }
54 return 0; // 0 is no change, unlike bp scale
55}

58AMY_IRAM_ATTR SAMPLE compute_breakpoint_scale(uint16_t osc, uint8_t bp_set, uint16_t sample_offset) {
59 AMY_PROFILE_START(COMPUTE_BREAKPOINT_SCALE)
60 // given a breakpoint list, compute the scale
61 // we first see how many BPs are defined, and where we are in them?
62 int8_t found = -1;
63 int8_t release = 0;
64 uint32_t t1 = 0, t0 = 0;
65 SAMPLE v1 = 0, v0 = 0;
66 int8_t bp_r = 0;
67 t0 = 0; v0 = 0;
68 // exp2(4.328085) = exp(3.0)
69 #define EXP_RATE_VAL -4.328085f
70 const SAMPLE exponential_rate = F2S(EXP_RATE_VAL);
71 // We have to aim to overshoot to the desired gap so that we hit the target by exponential_rate time.
72 const SAMPLE exponential_rate_overshoot_factor = F2S(1.0f / (1.0f - exp2f(EXP_RATE_VAL)));
73 uint32_t elapsed = 0;
74 SAMPLE scale = F2S(1.0f);
75 int eg_type = synth[osc]->eg_type[bp_set];
76 uint32_t bp_end_times[MAX_BREAKPOINTS];
77 uint32_t cumulated_time = 0;
78 int sign = 1;
79
80 // Scan breakpoints to find which one is release (the last one)
81 bp_r = -1;
82 for(int i = 0; i < synth[osc]->max_num_breakpoints[bp_set]; ++i) {
83 uint32_t this_seg_time = synth[osc]->breakpoint_times[bp_set][i];
84 if (!AMY_IS_SET(this_seg_time))
85 break;
86 bp_r = i; // Last good segment.
87 cumulated_time += this_seg_time;
88 bp_end_times[i] = cumulated_time;
89 }
90 if(bp_r < 0) {
91 // no breakpoints, return key gate.
92 // Change: Now an empty env reads as 1.0 *all the time*.
93 // If you want a key gate, define bpX='0,1,0,1,0,0' (or maybe just '0,1,0,0').
94 //if(AMY_IS_SET(synth[osc]->note_off_clock)) scale = 0;
95 synth[osc]->last_scale[bp_set] = scale;
96 //return scale;
97 goto return_label;
98 }
99 // Fix up bp_end_times for release segment to be relative to note-off time.
100 bp_end_times[bp_r] = synth[osc]->breakpoint_times[bp_set][bp_r];
101
102 // Find out which BP we're in
103 if(AMY_IS_SET(synth[osc]->note_on_clock)) {
104 elapsed = (amy_global.total_blocks*AMY_BLOCK_SIZE - synth[osc]->note_on_clock + sample_offset) + 1;
105 for(uint8_t i = 0; i < bp_r; i++) {
106 if(elapsed < bp_end_times[i]) {
107 // We found a segment.
108 found = i;
109 break;
110 }
111 }
112 if(found < 0) {
113 // We didn't find anything, so we are in sustain.
114 found = bp_r - 1; // segment before release defines sustain
115 scale = F2S(synth[osc]->breakpoint_values[bp_set][found]);
116 synth[osc]->last_scale[bp_set] = scale;
117 //printf("env: time %lld bpset %d seg %d SUSTAIN %f\n", amy_global.total_blocks*AMY_BLOCK_SIZE, bp_set, found, S2F(scale));
118 //return scale;
119 goto return_label;
120 }
121 } else if(AMY_IS_SET(synth[osc]->note_off_clock)) {
122 release = 1;
123 elapsed = (amy_global.total_blocks*AMY_BLOCK_SIZE - synth[osc]->note_off_clock + sample_offset);
124 // Get the last t/v pair , for release
125 found = bp_r;
126 t0 = 0; // start the elapsed clock again
127 // Release starts from wherever we got to
128 v0 = synth[osc]->last_scale[bp_set];
129 if(elapsed > synth[osc]->breakpoint_times[bp_set][bp_r]) {
130 //printf("cbp: time %f osc %d amp %f OFF\n", amy_global.total_blocks*AMY_BLOCK_SIZE / (float)AMY_SAMPLE_RATE, osc, msynth[osc]->amp);
131 // Synth is now turned off in hold_and_modify, which tracks when the amplitude goes to zero (and waits a bit).
132 //AMY_UNSET(synth[osc]->note_off_clock);
133 scale = F2S(synth[osc]->breakpoint_values[bp_set][bp_r]);
134 synth[osc]->last_scale[bp_set] = scale;
135 //return scale;
136 goto return_label;
137 }
138 }
139
140 if(found<0) return scale;
141
142 t1 = bp_end_times[found];
143 v1 = F2S(synth[osc]->breakpoint_values[bp_set][found]);
144 if(found>0 && bp_r != found && !release) {
145 t0 = bp_end_times[found-1];
146 v0 = F2S(synth[osc]->breakpoint_values[bp_set][found-1]);
147 }
148 scale = v0;
149 if (v0 < 0 || v1 < 0) {
150 sign = -1;
151 v0 = -v0; v1 = -v1;
152 }
153 if(t1==t0 || elapsed==t1 || v1 == v0) {
154 // This way we return exact zero for v1 at the end of the segment, rather than BREAKPOINT_EPS
155 scale = v1;
156 } else {
157 // OK, we are transition from v0 to v1 , and we're at elapsed time between t0 and t1
158 float time_ratio = ((float)(elapsed - t0) / (float)(t1 - t0));
159 // Compute scale based on which type we have
160 if(eg_type == ENVELOPE_LINEAR) {
161 scale = v0 + MUL4_SS(v1 - v0, F2S(time_ratio));
162 } else if(eg_type == ENVELOPE_TRUE_EXPONENTIAL || eg_type == ENVELOPE_DX7) {
163#define BREAKPOINT_EPS 0.0002f
164 v0 = MAX(v0, F2S(BREAKPOINT_EPS));
165 v1 = MAX(v1, F2S(BREAKPOINT_EPS));
166 if (eg_type == ENVELOPE_DX7) {
167 // DX7 is only defined for amps up to 1.0, clip to avoid negative values (which caused a math panic when using float math).
168 v0 = MIN(F2S(1.0f), v0);
169 v1 = MIN(F2S(1.0f), v1);
170 }
171 if (eg_type == ENVELOPE_DX7 && (v1 > v0)) {
172 // Somewhat complicated relationship, see https://colab.research.google.com/drive/1qZmOw4r24IDijUFlel_eSoWEf3L5VSok#scrollTo=F5zkeACrOlum
173 // in SAMPLE version, DX7 levels are div 8 i.e. 0 to 12.375 instead of 0 to 99.
174#define LINEAR_SAMP_TO_DX7_LEVEL(samp) (S2F(log2_lut(MAX(F2S(BREAKPOINT_EPS), samp))) + 12.375f)
175#define DX7_LEVEL_TO_LINEAR_SAMP(level) (exp2_lut(F2S(level - 12.375f)))
176#define MIN_LEVEL_S 4.25f
177#define ATTACK_RANGE_S 9.375f
178#define MAP_ATTACK_LEVEL_S(level) (1 - MAX(level - MIN_LEVEL_S, 0) / ATTACK_RANGE_S)
179 // MAP_ATTACK_LEVEL_S reaches zero (and goes negative) once the
180 // DX7 level passes 13.625, which a breakpoint value above ~2.37
181 // does. log2_lut needs a positive argument, so floor these the
182 // same way v0/v1 are floored in the decay branch below.
183 SAMPLE mapped_current_level = MAX(F2S(MAP_ATTACK_LEVEL_S(LINEAR_SAMP_TO_DX7_LEVEL(v0))), F2S(BREAKPOINT_EPS));
184 SAMPLE mapped_target_level = MAX(F2S(MAP_ATTACK_LEVEL_S(LINEAR_SAMP_TO_DX7_LEVEL(v1))), F2S(BREAKPOINT_EPS));
185 float t_const = (t1 - t0) / S2F(log2_lut(mapped_current_level) - log2_lut(mapped_target_level));
186 float my_t0 = -t_const * S2F(log2_lut(mapped_current_level));
187 // This is the magic equation that shapes the DX7 attack envelopes.
188 scale = DX7_LEVEL_TO_LINEAR_SAMP(MIN_LEVEL_S + ATTACK_RANGE_S * S2F(F2S(1.0f) - exp2_lut(-F2S((my_t0 + elapsed)/t_const))));
189 } else {
190 // TRUE_EXPONENTIAL or DX7 Decay (which is also regular true_exponential)
191 v0 = MAX(v0, F2S(BREAKPOINT_EPS));
192 v1 = MAX(v1, F2S(BREAKPOINT_EPS));
193 //float dx7_exponential_rate = -logf(S2F(v1)/S2F(v0)) / (t1 - t0);
194 //scale = MUL4_SS(v0, F2S(expf(-dx7_exponential_rate * (elapsed - t0))));
195 // Claude's solution avoids SAMPLE overflow for times > 5.8 sec.
196 float log2_v0 = S2F(log2_lut(v0));
197 float log2_v1 = S2F(log2_lut(v1));
198 scale = exp2_lut(F2S(log2_v0 + (log2_v1 - log2_v0) * time_ratio));
199 }
200 } else { // ENVELOPE_NORMAL - "false exponential"? or ENVELOPE_DB
201 // After the full amount of time, the exponential decay will reach (1 - expf(-3)) = 0.95
202 // so make the target gap a little bit bigger, to ensure we meet v1
203 //scale = v0 + MUL4_SS(v1 - v0, F2S(exponential_rate_overshoot_factor * (1.0f - exp2f(-exponential_rate * time_ratio))));
204 scale = v0 + MUL4_SS(v1 - v0,
205 MUL4_SS(exponential_rate_overshoot_factor,
206 F2S(1.0f)
207 - exp2_lut(MUL4_SS(exponential_rate,
208 F2S(time_ratio)))));
209 if (scale < 0) scale = 0; // Overshoot ramp-to-zero from limited resolution?
210 //printf("false_exponential time %lld bpset %d seg %d time_ratio %f scale %f\n", amy_global.total_blocks*AMY_BLOCK_SIZE, bp_set, found, time_ratio, S2F(scale));
211 }
212 }
213 return_label:
214 if (!release) synth[osc]->last_scale[bp_set] = scale;
215 // If sign is negative, flip it back again.
216 if (sign < 0) {
217 scale = -scale; // does not mix well with no_amp_001
218 }
219 // Keep track of the most-recently returned non-release scale.
220 //static int last_seg = -1;
221 //if (osc < AMY_OSCS && found != -1 && last_seg != found) {
222 // fprintf(stderr, "\renv: time %f osc %d bpset %d seg %d type %d t0 %d t1 %d elapsed %d v0 %f v1 %f scale %f\n", amy_global.total_blocks*AMY_BLOCK_SIZE / (float)AMY_SAMPLE_RATE, osc, bp_set, found, eg_type, t0, t1, elapsed, S2F(v0), S2F(v1), S2F(scale));
223 // last_seg = found;
224 //}
225 AMY_PROFILE_STOP(COMPUTE_BREAKPOINT_SCALE)
226 return scale;
