29static float lpf2_mag(float r, float Q)
30{
31 float d1 = 1.0f - r * r;
32 float d2 = r / Q;
33 return 1.0f / sqrtf(d1 * d1 + d2 * d2 + 1e-12f);

37{

59/* 6-stage allpass phaser (AMY FILTER_PHASER): six identical first-order
60 * allpasses H1 = (a + z^-1)/(1 + a z^-1), feedback fb around the chain with a
61 * one-sample loop delay, then a 50/50 dry/wet sum. |H1| = 1, so only the chain
62 * phase moves — the dry+wet sum nulls at the -180/-540/-900 degree crossings,
63 * giving 3 notches with the middle one at the cutoff. Coefficient mapping
64 * mirrors the engine (filters.c): a from tan(pi*fc/fs), fb linear in Q up to
65 * 0.85. Unlike the biquads this depends on absolute frequency, hence f and fc
66 * instead of their ratio; fs must match AMY_SAMPLE_RATE. */
67static float phaser_mag(float f, float fc, float Q)
68{
69 const float fs = 48000.0f;
70 float t = tanf((float)M_PI * fc / fs);
71 float a = (t - 1.0f) / (t + 1.0f);
72 float fb = 0.85f * (Q - FGRAPH_RES_MIN) / (FGRAPH_RES_MAX - FGRAPH_RES_MIN);
73 fb = SEQ_CLAMP_F32(fb, 0.0f, 0.85f);
74
75 float w = 2.0f * (float)M_PI * f / fs;
76 float zr = cosf(w); /* z^-1 = e^{-jw} */
77 float zi = -sinf(w);
78
79 /* H1 = (a + z^-1) / (1 + a z^-1) */
80 float nr = a + zr, ni = zi;
81 float dr = 1.0f + a * zr, di = a * zi;
82 float dm = dr * dr + di * di + 1e-12f;
83 float h1r = (nr * dr + ni * di) / dm;
84 float h1i = (ni * dr - nr * di) / dm;
85
86 /* Hc = H1^6 */
87 float hr = h1r, hi = h1i;
88 for (int k = 1; k < 6; k++) {
89 float tr = hr * h1r - hi * h1i;
90 hi = hr * h1i + hi * h1r;
91 hr = tr;
92 }
93
94 /* Hw = Hc / (1 - fb * Hc * z^-1) */
95 float lr = hr * zr - hi * zi;
96 float li = hr * zi + hi * zr;
97 dr = 1.0f - fb * lr;
98 di = -fb * li;
99 dm = dr * dr + di * di + 1e-12f;
100 float wr = (hr * dr + hi * di) / dm;
101 float wi = (hi * dr - hr * di) / dm;
102
103 /* out = 0.5 * (dry + wet) */
104 float sr = 0.5f * (1.0f + wr);
105 float si = 0.5f * wi;
106 return sqrtf(sr * sr + si * si);
107}

109/* Display magnitude (0..1) at frequency f for cutoff fc, Q and filter type.
110 * Passband = FG_PASSBAND_NORM; a resonance spike may exceed it, capped at 1. */
111static float filter_display_mag(uint8_t type, float f, float fc, float Q)
112{
113 if (fc < 1.0f) fc = 1.0f;
114 float r = f / fc;
115 if (r < 1e-4f) r = 1e-4f;
116 float mag;
117 switch (type) {
118 case FGRAPH_FILTER_LPF: {
119 float m = lpf2_mag(r, Q);
120 mag = m * FG_PASSBAND_NORM;
121 break;
122 }
123 case FGRAPH_FILTER_LPF24: {
124 float m = lpf2_mag(r, Q);
125 mag = m * m * FG_PASSBAND_NORM; /* two cascaded biquads */
126 break;
127 }
128 case FGRAPH_FILTER_HPF: {
129 float m = hpf2_mag(r, Q);
130 mag = m * FG_PASSBAND_NORM;
131 break;
132 }
133 case FGRAPH_FILTER_BPF: {
134 float m = bpf2_mag(r, Q);
135 /* BPF unity gain at center = 1.0; scale to passband level. */
136 mag = m * FG_PASSBAND_NORM;
137 break;
138 }
139 case FGRAPH_FILTER_NOTCH: {
140 float m = notch2_mag(r, Q);
141 mag = m * FG_PASSBAND_NORM;
142 break;
143 }
144 case FGRAPH_FILTER_PHASER: {
145 float m = phaser_mag(f, fc, Q);
146 mag = m * FG_PASSBAND_NORM;
147 break;
148 }
149 default: /* FGRAPH_FILTER_NONE */
150 mag = FG_PASSBAND_NORM;
151 break;
152 }
153 return SEQ_CLAMP_F32(mag, 0.0f, 1.0f);
154}
