170// Stick to the faster mult for biquad, hpf etc, since the parameters aren't so sensitive, and parametric_eq was chewing major CPU.
171/* LOCAL EDIT (upstream PR candidate): SMULR6 keeps only 12 fractional bits of
172 * each operand. For low-cutoff biquads the split-feedback corrections e,f are
173 * ~2^-10, so they retain just 2-3 significant bits and the pole lands almost
174 * randomly around its target: sub-150 Hz HPF/BPF at Q>=2 rings up ~+12 dB
175 * into sustained clipping (or, depending on where the pole falls, loses the
176 * resonance), and the sub-100 Hz LPF numerator rounds to zero (silent
177 * output). A full 32x32->64 rounding multiply restores exact s8.23 products
178 * for a couple of extra instructions per multiply; host-side A/B against a
179 * double-precision reference matches to 3 decimals across fc=10..1000 Hz,
180 * Q=0.7..8 (see docs/filter-fixedpoint-instability-2026-07-18.md). */
181#ifdef AMY_USE_FIXEDPOINT
182static inline SAMPLE SMUL64R(SAMPLE a, SAMPLE b) {
183 return (SAMPLE)((((int64_t)a * (int64_t)b) + (1 << (S_FRAC_BITS - 1))) >> S_FRAC_BITS);
184}
185#define FILT_MUL_SS(a, b) SMUL64R(a, b)
186#else
187#define FILT_MUL_SS(a, b) SMULR6(a, b)
188#endif

194AMY_IRAM_ATTR int8_t dsps_biquad_f32_ansi(const SAMPLE *input, SAMPLE *output, int len, SAMPLE *coef, SAMPLE *w) {
195 AMY_PROFILE_START(DSPS_BIQUAD_F32_ANSI)
196
197 // Zeros then poles - Direct Form I
198 // We need 2 memories for input, and 2 for output.
199 SAMPLE x1 = w[0];
200 SAMPLE x2 = w[1];
201 SAMPLE y1 = w[2];
202 SAMPLE y2 = w[3];
203 for (int i = 0 ; i < len ; i++) {
204 SAMPLE x0 = SHIFTL(input[i], FILTER_BIQUAD_SCALEUP_BITS);
205 // SAMPLE x0 = SHIFTR(input[i], FILTER_BIQUAD_SCALEDOWN_BITS);
206 SAMPLE w0 = FILT_MUL_SS(coef[0], x0) + FILT_MUL_SS(coef[1], x1) + FILT_MUL_SS(coef[2], x2);
207 SAMPLE y0 = w0 - FILT_MUL_SS(coef[3], y1) - FILT_MUL_SS(coef[4], y2);
208 x2 = x1;
209 x1 = x0;
210 y2 = y1;
211 y1 = y0;
212 output[i] = SHIFTR(y0, FILTER_BIQUAD_SCALEUP_BITS);
213 //output[i] = SHIFTL(y0, FILTER_BIQUAD_SCALEDOWN_BITS);
214 }
215 w[0] = x1;
216 w[1] = x2;
217 w[2] = y1;
218 w[3] = y2;
219 AMY_PROFILE_STOP(DSPS_BIQUAD_F32_ANSI)
220
221 return 0;
222}

225AMY_IRAM_ATTR int8_t dsps_biquad_f32_ansi_split_fb(const SAMPLE *input, SAMPLE *output, int len, SAMPLE *coef, SAMPLE *w) {
226 AMY_PROFILE_START(DSPS_BIQUAD_F32_ANSI_SPLIT_FB)
227 // Rewrite the feeedback coefficients as a1 = -2 + e and a2 = 1 - f
228 SAMPLE x1 = w[0];
229 SAMPLE x2 = w[1];
230 SAMPLE y1 = w[2];
231 SAMPLE y2 = w[3];
232 SAMPLE e = F2S(2.0f) + coef[3]; // So coef[3] = -2 + e
233 SAMPLE f = F2S(1.0f) - coef[4]; // So coef[4] = 1 - f
234 //fprintf(stderr, "e=%f (%d) f=%f\n", S2F(e), (e < F2S(0.0625)), S2F(f));
235 for (int i = 0 ; i < len ; i++) {
236 SAMPLE x0 = SHIFTL(input[i], FILTER_SCALEUP_BITS);
237 SAMPLE w0 = FILT_MUL_SS(coef[0], x0) + FILT_MUL_SS(coef[1], x1) + FILT_MUL_SS(coef[2], x2);
238 SAMPLE y0 = w0 + SHIFTL(y1, 1) - y2;
239 y0 = y0 - FILT_MUL_SS(e, y1) + FILT_MUL_SS(f, y2);
240 x2 = x1;
241 x1 = x0;
242 y2 = y1;
243 y1 = y0;
244 output[i] = SHIFTR(y0, FILTER_SCALEUP_BITS);
245 }
246 w[0] = x1;
247 w[1] = x2;
248 w[2] = y1;
249 w[3] = y2;
250 AMY_PROFILE_STOP(DSPS_BIQUAD_F32_ANSI_SPLIT_FB)
251
252 return 0;
253}

265static inline int headroom(SAMPLE a) {
266 // How many bits bigger can this value get before overflow?
267#ifdef AMY_USE_FIXEDPOINT
268 return __builtin_clz(ABS(a) | 1) - 1; // -1 for sign bit.
269#else // !AMY_USE_FIXEDPOINT
270 // How many bits can you shift before this max overflows?
271 int bits = 0;
272 while(a < F2S(128.0) && bits < 32) {
273 a = SHIFTL(a, 1);
274 ++bits;
275 }
276 return bits;
277#endif // AMY_USE_FIXEDPOINT
278}

280static inline int nheadroom16(SAMPLE a) {
281 // Headroom with MAX(0, 16 - headroom(a)) precomputed.
282#ifdef AMY_USE_FIXEDPOINT
283 return MAX(0, 16 - (__builtin_clz(ABS(a) | 1) - 1)); // -1 for sign bit.
284#else // !AMY_USE_FIXEDPOINT
285 // How many bits can you shift before this max overflows?
286 int bits = 0;
287 while(a < F2S(128.0) && bits < 32) {
288 a = SHIFTL(a, 1);
289 ++bits;
290 }
291 return bits;
292#endif // AMY_USE_FIXEDPOINT
293}

297SAMPLE top16SMUL(SAMPLE a, SAMPLE b) {
298 // Multiply the top 15 bits of a and b.
299 //int adropped = MAX(0, 16 - headroom(a));
300 int adropped = nheadroom16(a); //MAX(0, 16 - headroom(a));
301 if (adropped) {
302 a = SHIFTR(SHIFTR(a, adropped - 1) + 1, 1);
303 }
304 int resultdrop = 23 - adropped;
305 int bdropped = MIN(resultdrop, nheadroom16(b)); // MAX(0, 16 - headroom(b)));
306 if (bdropped) {
307 //b = SHIFTR(b + (1 << (bdropped - 1)), bdropped);
308 b = SHIFTR(SHIFTR(b, bdropped - 1) + 1, 1);
309 }
310 resultdrop -= bdropped;
311 SAMPLE result = a * b;
312 if (resultdrop) {
313 //return SHIFTR(result + (1 << (resultdrop - 1)), resultdrop);
314 return SHIFTR(SHIFTR(result, resultdrop - 1) + 1, 1);
315 }
316 return result;
317}

319SAMPLE top16SMUL_a_part(SAMPLE a, int *p_resultdrop) {
320 // Just the processing of a, so we can split it out
321 int adropped = nheadroom16(a);
322 if (adropped) {
323 //a = SHIFTR(a + (1 << (adropped - 1)), adropped);
324 a = SHIFTR(SHIFTR(a, adropped - 1) + 1, 1);
325 }
326 *p_resultdrop = 23 - adropped;
327 return a;
328}

330SAMPLE top16SMUL_after_a(SAMPLE a_processed, SAMPLE b, int resultdrop, int bnheadroom16) {
331 // The rest of top16SMUL after a has been preprocessed by top16SML_a_part.
332 //int bdropped = MIN(resultdrop, MAX(0, 16 - bheadroom));
333 int bdrop = MIN(resultdrop, bnheadroom16);
334 if (bdrop) {
335 //b = SHIFTR(b + (1 << (bdrop - 1)), bdrop);
336 b = SHIFTR(SHIFTR(b, bdrop - 1) + 1, 1);
337 resultdrop -= bdrop;
338 }
339 SAMPLE result = a_processed * b;
340 if (resultdrop) {
341 //return SHIFTR(result + (1 << (resultdrop - 1)), resultdrop);
342 return SHIFTR(SHIFTR(result, resultdrop - 1) + 1, 1);
343 }
344 return result;
345}

347static inline SAMPLE top16SMUL_after_a_live(SAMPLE a_processed, SAMPLE b, int resultdrop) {
348 // Like top16SMUL_after_a, but measures b's actual headroom here instead of
349 // trusting a bound precomputed at block start. Needed for the filter state
350 // terms: the state can grow far beyond the block-start allowance within one
351 // block (e.g. resonance ringing up as the cutoff sweeps toward the signal),
352 // and an under-shifted b overflows the 32-bit product.
353 return top16SMUL_after_a(a_processed, b, resultdrop, nheadroom16(b));
354}

379AMY_IRAM_ATTR SAMPLE scan_max(SAMPLE* block, int len) {
380 AMY_PROFILE_START(SCAN_MAX)
381
382 // Find the max abs sample value in a block.
383 SAMPLE max = 0;
384 while (len--) {
385 SAMPLE val = *block++;
386 if (val > max) max = val;
387 else if ((-val) > max) max = -val;
388 }
389 AMY_PROFILE_STOP(SCAN_MAX)
390 return max;
391}

568AMY_IRAM_ATTR void parametric_eq_process(uint8_t bus, SAMPLE *block) {
569 // was void parametric_eq_process_top16block
570 // Optimized to run all 3 filters interleaved, to avoid extra buffers/buf accesses.
571 AMY_PROFILE_START(PARAMETRIC_EQ_PROCESS)
572 SAMPLE **eq_coeffs = amy_global.bus[bus]->eq.eq_coeffs;
573 SAMPLE ***eq_delay = amy_global.bus[bus]->eq.eq_delay;
574
575 for(int c = 0; c < AMY_NCHANS; ++c) {
576 SAMPLE *cblock = block + c * AMY_BLOCK_SIZE;
577 // Zeros then poles - Direct Form I
578 // We need 2 memories for input, and 2 for output.
579 SAMPLE x1 = eq_delay[c][0][0];
580 SAMPLE x2 = eq_delay[c][0][1];
581 SAMPLE y01 = eq_delay[c][0][2];
582 SAMPLE y02 = eq_delay[c][0][3];
583 SAMPLE y11 = eq_delay[c][1][2];
584 SAMPLE y12 = eq_delay[c][1][3];
585 SAMPLE y21 = eq_delay[c][2][2];
586 SAMPLE y22 = eq_delay[c][2][3];
587 int c00bits, c03bits, c04bits;
588 int c10bits, c13bits, c14bits;
589 int c20bits, c23bits, c24bits;
590 // int c21bits, c22bits, c11bits, c12bits, c01bits, c02bits;
591 // Fold the global EQ parameters into the forward-gains of each stage.
592 SAMPLE c00 = top16SMUL_a_part(top16SMUL(amy_global.bus[bus]->eq.eq[0], eq_coeffs[0][0]), &c00bits);
593 SAMPLE c03 = top16SMUL_a_part(eq_coeffs[0][3], &c03bits);
594 SAMPLE c04 = top16SMUL_a_part(eq_coeffs[0][4], &c04bits);
595 SAMPLE c10 = top16SMUL_a_part(top16SMUL(amy_global.bus[bus]->eq.eq[1], eq_coeffs[1][0]), &c10bits);
596 SAMPLE c13 = top16SMUL_a_part(eq_coeffs[1][3], &c13bits);
597 SAMPLE c14 = top16SMUL_a_part(eq_coeffs[1][4], &c14bits);
598 SAMPLE c20 = top16SMUL_a_part(top16SMUL(amy_global.bus[bus]->eq.eq[2], eq_coeffs[2][0]), &c20bits);
599 SAMPLE c23 = top16SMUL_a_part(eq_coeffs[2][3], &c23bits);
600 SAMPLE c24 = top16SMUL_a_part(eq_coeffs[2][4], &c24bits);
601 for (int i = 0 ; i < AMY_BLOCK_SIZE ; i++) {
602 SAMPLE x0 = cblock[i];
603 SAMPLE x1times2 = SHIFTL(x1, 1);
604 int xbits = nheadroom16(MAXABS2(x0, MAXABS2(x1times2, x2)));
605 // Optimize the FIR multiplies for the known structure of the zeros in LPF/BPF/HPF.
606 SAMPLE w00 = top16SMUL_after_a(c00, x0 + x1times2 + x2, c00bits, xbits);
607 int y0bits = nheadroom16(MAXABS2(y01, y02));
608 SAMPLE y00 = w00 - top16SMUL_after_a(c03, y01, c03bits, y0bits) - top16SMUL_after_a(c04, y02, c04bits, y0bits);
609 SAMPLE w10 = top16SMUL_after_a(c10, x0 - x2, c10bits, xbits);
610 int y1bits = nheadroom16(MAXABS2(y11, y12));
611 SAMPLE y10 = w10 - top16SMUL_after_a(c13, y11, c13bits, y1bits) - top16SMUL_after_a(c14, y12, c14bits, y1bits);
612 SAMPLE w20 = top16SMUL_after_a(c20, x0 - x1times2 + x2, c20bits, xbits);
613 int y2bits = nheadroom16(MAXABS2(y21, y22));
614 SAMPLE y20 = w20 - top16SMUL_after_a(c23, y21, c23bits, y2bits) - top16SMUL_after_a(c24, y22, c24bits, y2bits);
615 x2 = x1;
616 x1 = x0;
617 y02 = y01;
618 y01 = y00;
619 y12 = y11;
620 y11 = y10;
621 y22 = y21;
622 y21 = y20;
623 cblock[i] = y00 - y10 + y20;
624 }
625 eq_delay[c][0][0] = x1;
626 eq_delay[c][0][1] = x2;
627 eq_delay[c][0][2] = y01;
628 eq_delay[c][0][3] = y02;
629 eq_delay[c][1][2] = y11;
630 eq_delay[c][1][3] = y12;
631 eq_delay[c][2][2] = y21;
632 eq_delay[c][2][3] = y22;
633 }
634 AMY_PROFILE_STOP(PARAMETRIC_EQ_PROCESS)
635}

637void hpf_buf(SAMPLE *buf, SAMPLE *state) {
638 AMY_PROFILE_START(HPF_BUF)
639
640 // Implement a dc-blocking HPF with a pole at (decay + 0j) and a zero at (1 + 0j).
641 SAMPLE pole = F2S(0.99);
642 SAMPLE xn1 = state[0];
643 SAMPLE yn1 = state[1];
644 for (uint16_t i = 0; i < AMY_BLOCK_SIZE; ++i) {
645 SAMPLE w = buf[i] - xn1;
646 xn1 = buf[i];
647 buf[i] = w + FILT_MUL_SS(pole, yn1);
648 yn1 = buf[i];
649 }
650 state[0] = xn1;
651 state[1] = yn1;
652 AMY_PROFILE_STOP(HPF_BUF)

656void check_overflow(SAMPLE* block, int osc, char *msg) {
657 // Search for overflow in a sample buffer as an unusually large sample-to-sample delta.
658#ifdef AMY_DEBUG
659 SAMPLE last = block[0];
660 SAMPLE max = 0;
661 SAMPLE maxdiff = 0;
662 int len = AMY_BLOCK_SIZE;
663 while (len--) {
664 SAMPLE val = *block++;
665 SAMPLE diff = val - last;
666 if (diff < 0) diff = -diff;
667 if (diff > maxdiff) maxdiff = diff;
668 last = val;
669 if (val < 0) val = -val;
670 if (val > max) max = val;
671 }
672 if (maxdiff > F2S(0.2f))
673 fprintf(stderr, "Overflow at timeframe %.3f max=%.3f maxdiff=%.3f osc=%d msg=%s\n",
674 (float)(amy_global.total_blocks*AMY_BLOCK_SIZE) / AMY_SAMPLE_RATE,
675 S2F(max), S2F(maxdiff), osc, msg);
676
677#endif // AMY_DEBUG
678}

680SAMPLE block_norm(SAMPLE* block, int len, int bits) {
681 AMY_PROFILE_START(BLOCK_NORM)
682
683 SAMPLE max_val = 0;
684 if (bits >= 0) {
685 // do this even if bits == 0 to ensure max_val is set.
686 while(len--) {
687 *block = SHIFTL(*block, bits);
688 if (*block > 0) {
689 if (*block > max_val)
690 max_val = *block;
691 } else {
692 if (-*block > max_val)
693 max_val = -*block;
694 }
695 ++block;
696 }
697 } else {
698 // bits is negative - right-shift.
699 bits = -bits;
700 while(len--) {
701 *block = SHIFTR(*block, bits);
702 if (*block > 0) {
703 if (*block > max_val)
704 max_val = *block;
705 } else {
706 if (-*block > max_val)
707 max_val = -*block;
708 }
709 ++block;
710 }
711 }
712 AMY_PROFILE_STOP(BLOCK_NORM)
713 return max_val;
714}
