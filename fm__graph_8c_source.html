16void fm_graph_decode(const uint8_t ops[FM_GRAPH_OPS], fm_graph_view_t *out)
17{
18 uint8_t writers[3] = { 0, 0, 0 }; /* bitmask of ops currently on bus 1/2 */
19 memset(out, 0, sizeof(*out));
20 out->fb_op = FM_OP_NONE;
21 for (uint8_t s = 0; s < FM_GRAPH_OPS; s++) {
22 uint8_t b = ops[s];
23 uint8_t in = (b & IN_BUS_ONE) ? writers[BUS_ONE]
24 : (b & IN_BUS_TWO) ? writers[BUS_TWO] : 0;
25 for (uint8_t w = 0; w < FM_GRAPH_OPS; w++) {
26 if (in & (1u << w)) out->out_mask[w] |= (uint8_t)(1u << s);
27 }
28 if (b & FB_IN) out->fb_op = s;
29 uint8_t bus = (b & OUT_BUS_ONE) ? BUS_ONE : (b & OUT_BUS_TWO) ? BUS_TWO : BUS_NONE;
30 if (bus == BUS_NONE) {
31 out->out_mask[s] |= FM_OUT_BIT;
32 } else if (b & OUT_BUS_ADD) {
33 writers[bus] |= (uint8_t)(1u << s);
34 } else {
35 writers[bus] = (uint8_t)(1u << s);
36 }
37 }
38}

62bool fm_graph_edge_allowed(const uint8_t op_to[FM_GRAPH_OPS], uint8_t op, uint8_t target)
63{
64 if (op >= FM_GRAPH_OPS) return false;
65 if (target == FM_TO_OUT) return true;
66 if (target >= FM_GRAPH_OPS || target == op) return false;
67 /* A cycle would exist if op already sits on target's path to the output. */
68 return !forest_reaches(op_to, target, op);

71void fm_graph_to_forest(const fm_graph_view_t *g, uint8_t op_to[FM_GRAPH_OPS])
72{
73 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
74 uint8_t m = g->out_mask[i];
75 op_to[i] = FM_TO_OUT;
76 for (uint8_t t = 0; t < FM_GRAPH_OPS; t++) {
77 if (m & (1u << t)) { op_to[i] = t; break; }
78 }
79 }
80 /* Break cycles / orphan chains: anything that never reaches the output
81 * becomes a carrier. */
82 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
83 uint8_t op = i, guard = 0;
84 while (op_to[op] != FM_TO_OUT && guard++ < FM_GRAPH_OPS) op = op_to[op];
85 if (op_to[op] != FM_TO_OUT) op_to[i] = FM_TO_OUT;
86 }

109/* Emit `node` so that its output lands on `out` (BUS_ONE/BUS_TWO/BUS_OUT),
110 * accumulating when `add`. `live` = buses holding data that must survive. */
111static bool emit(compile_ctx_t *c, uint8_t node, uint8_t out, bool add, uint8_t live)
112{
113 if (c->emitted & (1u << node)) return false; /* cycle */
114 c->emitted |= (uint8_t)(1u << node);
115
116 /* Children, largest subtree first: only the first child may still use a
117 * bus that later children would find live. */
118 uint8_t kids[FM_GRAPH_OPS], nk = 0;
119 for (uint8_t i = 0; i < FM_GRAPH_OPS; i++) {
120 if (c->op_to[i] == node) kids[nk++] = i;
121 }
122 for (uint8_t a = 1; a < nk; a++) { /* insertion sort, desc size */
123 uint8_t k = kids[a], sk = subtree_size(c, k, 0);
124 int b = a - 1;
125 while (b >= 0 && subtree_size(c, kids[b], 0) < sk) { kids[b + 1] = kids[b]; b--; }
126 kids[b + 1] = k;
127 }
128
129 uint8_t in = BUS_NONE;
130 if (nk > 0) {
131 uint8_t busy = live;
132 if (add && out != BUS_OUT) busy |= (uint8_t)(1u << out);
133 for (uint8_t cand = BUS_ONE; cand <= BUS_TWO; cand++) {
134 if (busy & (1u << cand)) continue;
135 /* Reading and overwriting the same bus is the 0x11 scratch case
136 * AMY only implements for bus one, and never with ADD. */
137 if (cand == out && (add || out != BUS_ONE)) continue;
138 in = cand;
139 break;
140 }
141 if (in == BUS_NONE) return false;
142 for (uint8_t k = 0; k < nk; k++) {
143 uint8_t child_live = busy;
144 if (k > 0) child_live |= (uint8_t)(1u << in);
145 if (!emit(c, kids[k], in, k > 0, child_live)) return false;
146 }
147 }
148
149 if (c->n >= FM_GRAPH_OPS) return false;
150 uint8_t byte = 0;
151 if (in == BUS_ONE) byte |= IN_BUS_ONE;
152 if (in == BUS_TWO) byte |= IN_BUS_TWO;
153 if (out == BUS_OUT) {
154 byte |= OUT_BUS_ADD; /* carriers always sum into buf */
155 } else {
156 byte |= (out == BUS_ONE) ? OUT_BUS_ONE : OUT_BUS_TWO;
157 if (add) byte |= OUT_BUS_ADD;
158 }
159 if (node == c->fb_op) byte |= FB_IN | FB_OUT;
160 c->prog->ops[c->n] = byte;
161 c->prog->slot_op[c->n] = node;
162 c->n++;
163 return true;
164}

166bool fm_graph_compile(const uint8_t op_to[FM_GRAPH_OPS], uint8_t fb_op, fm_program_t *out)
167{
168 compile_ctx_t c = { .op_to = op_to, .fb_op = fb_op, .prog = out, .n = 0, .emitted = 0 };
