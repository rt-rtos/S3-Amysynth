5/* Operator-array index -> role, for algorithm 0 (see fm_voice.h / algorithms.c):
6 * 0,1,2 : deep modulator stack (feeds index 3)
7 * 3 : carrier A (chain-A output)
8 * 4 : modulator (feeds index 5)
9 * 5 : carrier B (chain-B output)
10 * All four presets keep index 0 (op6, the only slot with FB_IN in this
11 * algorithm) silenced, so `feedback` is not exercised by these fixed presets
12 * — it remains available for exploration via the FM_CUSTOM voice. */

14static void fm_preset_bass(fm_voice_t *v)

26static void fm_preset_epiano(fm_voice_t *v)
