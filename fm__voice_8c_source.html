24/* ── Custom program double-buffer ─────────────────────────────────────────
25 * The compiled program lives in one of AMY's AMY_NUM_CUSTOM_ALGORITHMS RAM
26 * rows. A changed program is written to the row NOT currently referenced,
27 * then the algorithm/algo_source event switches every synth to it in one
28 * apply under amy_queue_lock, so the render body never reads a half-written
29 * row and the FIFO ordering of edits is preserved. */
30static fm_program_t s_prog; /* what the referenced row holds */
31static uint8_t s_prog_slot = 0;
32static bool s_prog_valid = false;

34static void fm_program_identity(fm_program_t *p, uint8_t algorithm)
35{
36 const uint8_t *ops = amy_algorithm_ops(algorithm);
37 for (uint8_t i = 0; i < FM_NUM_OPS; i++) {
38 p->ops[i] = ops ? ops[i] : 0;
39 p->slot_op[i] = i;
40 }
41}

43/* Resolve the voice's routing to an AMY algorithm index + slot order,
44 * publishing a custom program to a RAM row when it changed. */
45static uint8_t fm_voice_resolve_program(const fm_voice_t *v, fm_program_t *p)
46{
47 if (v->algorithm != FM_ALGO_CUSTOM) {
48 uint8_t a = (v->algorithm < amy_num_algorithms) ? v->algorithm : 1;
49 fm_program_identity(p, a);
50 return a;
51 }
52 if (!fm_graph_compile(v->op_to, v->fb_op, p)) {
53 /* Setters never store an uncompilable graph; a corrupt voice falls
54 * back to row 1 rather than pushing garbage routing. */
55 fm_program_identity(p, 1);
56 return 1;
57 }
58 if (!s_prog_valid || memcmp(&s_prog, p, sizeof(*p)) != 0) {
59 s_prog_slot = (uint8_t)((s_prog_slot + 1u) % AMY_NUM_CUSTOM_ALGORITHMS);
60 s_prog = *p;
61 s_prog_valid = true;
62 amy_set_custom_algorithm(s_prog_slot, p->ops);
63 }
64 return (uint8_t)(amy_num_algorithms + s_prog_slot);
65}

67void fm_voice_default(fm_voice_t *v)
68{
69 if (!v) return;
70 memset(v, 0, sizeof(*v));
71 v->algorithm = 1;
72 v->feedback = 0.0f;
73 for (uint8_t i = 0; i < FM_NUM_OPS; i++) {
74 v->op_ratio[i] = 1.0f;
75 v->op_level[i] = 0.0f;
76 v->op_env[i] = (seq_env_t) {
77 .attack_ms = FM_OP_ATTACK_MS,
78 .decay_ms = FM_OP_DECAY_MS,
79 .sustain_pct = FM_OP_SUSTAIN,
80 .release_ms = FM_OP_RELEASE_MS,
81 .eg_type = ENVELOPE_DX7, /* the DX7 attack curve: what makes modulator envelopes sound right */
82 };
83 }
84 /* Algorithm 1's second chain: index 4 (OP2) modulates index 5 (OP1).
85 * Enabling only this pair makes a fresh voice audible as a 2-op tone. */
86 v->op_ratio[4] = 1.0f; v->op_level[4] = 0.5f;
87 v->op_ratio[5] = 1.0f; v->op_level[5] = 1.0f;
88 /* Seed the custom fields from the same row so entering custom mode (or a
89 * topology edit) starts from what is heard. */
90 fm_graph_view_t g;
91 fm_voice_graph(v, &g);
92 fm_graph_to_forest(&g, v->op_to);
93 v->fb_op = g.fb_op;
94}

97static void fm_voice_send_routing(uint8_t synth_id, const fm_voice_t *voice)
98{
99 fm_program_t p;
100 uint8_t algo = fm_voice_resolve_program(voice, &p);
101
102 amy_event *e = amy_helpers_event_begin();
103 e->synth = synth_id;
104 e->osc = 0;
105 e->algorithm = algo;
106 e->feedback = voice->feedback;
107 /* algo_source[s] is voice-relative; AMY offsets it by the voice's base
108 * osc internally. Slot s renders operator slot_op[s] = osc slot_op[s]+1. */
109 for (uint8_t s = 0; s < FM_NUM_OPS; s++) {
110 e->algo_source[s] = (int16_t)(p.slot_op[s] + 1);
111 }
112 /* The ALGORITHM delta force-switches the osc's eg_type[0] to the DX7
113 * curve; this voice authors ENVELOPE_NORMAL, so re-assert it in the same
114 * event or the first live edit silently changes the envelope shape. */
115 e->eg_type[0] = ENVELOPE_NORMAL;
116 amy_helpers_event_send(e);
117}

119/* One operator osc: ratio, level, own EG0. Sent after the routing event so
120 * the eg_type re-assert lands after AMY's ALGO_SOURCE naming (which forces
121 * ENVELOPE_DX7 on the operator). */
122static void fm_voice_send_op(uint8_t synth_id, const fm_voice_t *voice, uint8_t op)
123{
124 const seq_env_t *env = &voice->op_env[op];
125 amy_event *e = amy_helpers_event_begin();
126 e->synth = synth_id;
127 e->osc = (uint16_t)(op + 1);
128 e->wave = SINE;
129 e->ratio = voice->op_ratio[op];
130 e->freq_coefs[COEF_NOTE] = 1.0f;
131 e->amp_coefs[COEF_CONST] = voice->op_level[op];
132 /* VEL must be 0 on operators: AMY never delivers velocity to
133 * SYNTH_IS_ALGO_SOURCE oscs, so their velocity input stays 0. Under the
134 * dB amp-combine a nonzero VEL coef with 0 input contributes -60 dB and
135 * amp_combine_controls floors the result to exactly 0 - every operator
136 * renders silence. Velocity comes from the ALGO control osc (VEL=1),
137 * whose amp scales the carriers in render_algo. Same convention as the
138 * built-in DX7 patch strings. */
139 e->amp_coefs[COEF_VEL] = 0.0f;
140 e->amp_coefs[COEF_EG0] = 1.0f;
141 e->bp_is_set[0] = 1;
142 e->eg_type[0] = env->eg_type;
143 e->eg0_times[0] = SEQ_CLAMP_U32(env->attack_ms, VOICE_ENV_ATTACK_MIN_MS,
144 VOICE_ENV_TIME_MAX_MS);
145 e->eg0_values[0] = 1.0f;
146 e->eg0_times[1] = SEQ_CLAMP_U32(env->decay_ms, 0u, VOICE_ENV_TIME_MAX_MS);
147 e->eg0_values[1] = (float)SEQ_CLAMP_U8(env->sustain_pct, 0u, 100u) / 100.0f;
148 e->eg0_times[2] = SEQ_CLAMP_U32(env->release_ms, VOICE_ENV_RELEASE_MIN_MS,
149 VOICE_ENV_TIME_MAX_MS);
150 e->eg0_values[2] = 0.0f;
151 amy_helpers_event_send(e);
152}

249uint8_t fm_voice_step_algorithm(fm_voice_t *v, int dir)
250{
251 int n = (int)amy_num_algorithms; /* rows 1..n-1 are the DX7 set */
252 int step = (dir > 0) ? 1 : -1;
253 if (v->algorithm == FM_ALGO_CUSTOM) {
254 v->algorithm = (uint8_t)((step > 0) ? 1 : n - 1);
255 return v->algorithm;
256 }
257 int a = (int)v->algorithm + step;
258 if (a <= 0 || a >= n) {
259 fm_voice_make_custom(v); /* wrap through the custom slot */
260 return v->algorithm;
261 }
262 v->algorithm = (uint8_t)a;
263 return v->algorithm;
264}
