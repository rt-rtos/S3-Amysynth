12/* ── Generic 6-operator DX7-style FM/ALGO voice engine ───────────────────────
13 * Wraps AMY's ALGO wave type as one melodic voice: an ALGO control osc
14 * (osc 0) plus six SINE operators (osc 1..6), so oscs_per_voice is always 7.
15 *
16 * Operator index i (0..5) is osc i+1 of the voice, always. In table mode it is
17 * also AMY algo_source slot i; in custom mode the compiled program reorders
18 * the slots (fm_graph.h) and algo_source[] carries the mapping. AMY's table
19 * rows are authored DX7 op6 first, so the UI labels index i as OP(6-i): index
20 * 5 is OP1, the leftmost carrier of every DX7 chart.
21 *
22 * Routing comes from one of two sources:
23 * algorithm < FM_ALGO_CUSTOM : AMY algorithms[] row (DX7 numbering, row 0
24 * aliases row 1); op_to/fb_op are ignored.
25 * algorithm == FM_ALGO_CUSTOM: the authored forest op_to[] + fb_op,
26 * compiled onto AMY's two buses at push time.
27 * `feedback` is the amount on whichever operator the routing marks FB.
28 *
29 * Per-operator "level" is that operator's amp_coefs[COEF_CONST]: for a
30 * modulator it IS the modulation index/brightness, for a carrier it scales
31 * that carrier. Level 0 silences the operator. Each operator has its own EG0
32 * (op_env); the ALGO control osc keeps the row's per-track ADSR as a VCA over
33 * the carriers, so both shape the note. */
34
35#define FM_NUM_OPS 6
36#define FM_ALGO_CUSTOM 0xFF /* algorithm: use op_to[]/fb_op */
37
38typedef struct {
39 uint8_t algorithm;
40 uint8_t fb_op; /* custom mode: FM_OP_NONE or 0..5 */
41 uint8_t op_to[FM_NUM_OPS]; /* custom mode: FM_TO_OUT or 0..5 */
42 float op_ratio[FM_NUM_OPS]; /* per-operator frequency ratio */
43 float op_level[FM_NUM_OPS]; /* per-operator output level, 0..1 */
44 seq_env_t op_env[FM_NUM_OPS]; /* per-operator EG0 */
45 float feedback; /* 0..~1.2 */
46} fm_voice_t;
