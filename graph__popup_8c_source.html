13/* Normalised-coordinate clamp. Via seq_clamp_f32 so NaN degrades to 0.0f
14 * instead of propagating: these results feed (int) casts on the draw path,
15 * and NaN->int is undefined behaviour. */
16static inline float clampf01(float v) { return SEQ_CLAMP_F32(v, 0.0f, 1.0f); }
17
18/* Inset (in px) of the plot region inside the box border. Leaves room for the
19 * frame and an optional title strip at the top. */
20#define GPOPUP_PAD_X 4
21#define GPOPUP_PAD_TOP 4
22#define GPOPUP_PAD_BOT 4
23#define GPOPUP_TITLE_H 8 /* extra top inset when a title is shown */
24
25/* Fallback minimum normalised X gap between points until a host installs its
26 * own via graph_popup_set_min_x_gap(). Sized for marker legibility on a 128 px
27 * plot; a host that knows the real floor in its own units should install that,
28 * so the plot is never the stricter constraint. */
29#define GPOPUP_DEFAULT_MIN_X_GAP 0.004f
30
31/* Accepted range for an installed gap. Zero is legal (no spacing rule); the
32 * ceiling stops a bad conversion from silently distorting real edits. */
33#define GPOPUP_MIN_X_GAP_LO 0.0f
34#define GPOPUP_MIN_X_GAP_HI 0.05f
35
36/* Encoder step size for value adjustment (normalised units per detent). */
37#define GPOPUP_ADJUST_STEP 0.02f
38
39/* Compute the inner plot rectangle (in absolute screen pixels). */
40static void plot_rect(const gpopup_t *p,
41 uint8_t *px, uint8_t *py, uint8_t *pw, uint8_t *ph)
42{
43 /* ADSR style has no internal title strip (host draws the top bar), uses
44 * tighter insets, and reserves 3px at the bottom for tick marks. */
45 bool adsr = (p->style == GPOPUP_STYLE_ADSR);
46 uint8_t top_pad = adsr ? 2 : (GPOPUP_PAD_TOP + (p->title ? GPOPUP_TITLE_H : 0));
47 uint8_t bot_pad = adsr ? (GPOPUP_PAD_BOT + 3) : GPOPUP_PAD_BOT;
48 uint8_t side_pad = adsr ? 2 : GPOPUP_PAD_X;
49 *px = (uint8_t)(p->x + side_pad);
50 *py = (uint8_t)(p->y + top_pad);
51 *pw = (uint8_t)(p->w - 2 * side_pad);
52 *ph = (uint8_t)(p->h - top_pad - bot_pad);
53}

55/* Map a normalised point to absolute screen pixels within the plot rect.
56 * Y is inverted (screen y grows downward). */
57static void point_to_px(const gpopup_t *p, const gpopup_point_t *pt,
58 uint8_t plot_x, uint8_t plot_y,
59 uint8_t plot_w, uint8_t plot_h,
60 int *out_x, int *out_y)
61{
62 float xn = clampf01(pt->x);
63 float yn = clampf01(pt->y);
64 *out_x = (int)(plot_x + xn * (plot_w - 1));
65 *out_y = (int)(plot_y + (1.0f - yn) * (plot_h - 1));
66}

158/* ── ADSR role-aware constraints ─────────────────────────────────────────────
159 * Points are [origin, A(attack peak), D(decay end / sustain level), R(release
160 * end)] - index roles assume the host's standard 4-point ADSR seed:
161 * - X monotonic with a minimum gap so points can't cross or collapse.
162 * - origin pinned to (0,0); A.y pinned to 1.0; R.y pinned to 0.0.
163 * - only D's Y (the sustain level) is freely movable. */
164static void adsr_apply_constraints(gpopup_t *p)
165{
166 if (p->style != GPOPUP_STYLE_ADSR || p->num_points < 4) return;
167
168 gpopup_point_t *pt = p->points;
169
170 /* Role-fixed Y levels. */
171 pt[0].x = 0.0f; pt[0].y = 0.0f; /* origin */
172 pt[1].y = 1.0f; /* attack peak */
173 pt[3].y = 0.0f; /* release end */
174 /* pt[2].y is the sustain level — left as edited, just bounded 0..1. */
175 pt[2].y = clampf01(pt[2].y);
176
177 /* Monotonic X with a minimum gap: forward pass bounds each point below by
178 * its predecessor + gap, backward pass keeps an earlier point from being
179 * pushed past a later one that was dragged left. */
180 for (uint8_t i = 1; i < p->num_points; ++i) {
181 float lo = pt[i - 1].x + p->min_x_gap;
182 if (pt[i].x < lo) pt[i].x = lo;
183 if (pt[i].x > 1.0f) pt[i].x = 1.0f;
184 }
185 for (int i = (int)p->num_points - 2; i >= 1; --i) {
186 float hi = pt[i + 1].x - p->min_x_gap;
187 if (pt[i].x > hi) pt[i].x = hi;
188 if (pt[i].x < 0.0f) pt[i].x = 0.0f;
189 }
190}

192/* Whether the selected point's Y is editable in ADSR style. Only the sustain
193 * point (index 2) has a free level; A/R levels are role-fixed. */
194static bool adsr_y_editable(const gpopup_t *p, uint8_t idx)
195{
196 if (p->style != GPOPUP_STYLE_ADSR) return true;
197 return (idx == 2);
198}
199
200/* Whether the selected point's X (time) is editable in ADSR style. With
201 * adsr_lock_sx the sustain point (2) is Y-only - its X is the host's
202 * auto-derived decay time. A and R stay X-editable; the origin never is. */
203static bool adsr_x_editable(const gpopup_t *p, uint8_t idx)
204{
205 if (p->style != GPOPUP_STYLE_ADSR) return true;
206 if (idx == 0) return false; /* origin is pinned */
207 if (p->adsr_lock_sx && idx == 2) return false; /* S.x host-owned */
208 return true;
209}
210
211/* True when a point moves on BOTH axes: the explicit-decay sustain point
212 * (lock_sx off) carries decay time (X) and sustain level (Y). With lock_sx on
213 * no point is dual-free and the two-axis handling below stays dormant. */
214static bool adsr_point_dual_free(const gpopup_t *p, uint8_t idx)
215{
216 return p->style == GPOPUP_STYLE_ADSR
217 && adsr_x_editable(p, idx) && adsr_y_editable(p, idx);
218}
219
220/* Returns true if the encoder should adjust Y, false for X. Single-axis ADSR
221 * points (A/R = time/X, S = level/Y when its X is host-owned) auto-select, so
222 * there is nothing to toggle; the dual-free sustain point and the generic
223 * (non-ADSR) editor honour the manual adjust_axis_y flag. */
224static bool adsr_effective_axis_is_y(const gpopup_t *p, uint8_t idx)
225{
226 if (p->style != GPOPUP_STYLE_ADSR) return p->adjust_axis_y;
227 if (adsr_y_editable(p, idx) && !adsr_x_editable(p, idx)) return true;
228 if (adsr_x_editable(p, idx) && !adsr_y_editable(p, idx)) return false;
229 /* Both free (explicit-decay sustain): honour the axis flag. Neither free
230 * (origin): value is irrelevant, default to Y. */
231 return p->adjust_axis_y;
232}

259/* Interpolate the curve's screen-Y at a given screen column, walking the
260 * point segments. Returns the baseline if the column is outside the curve. */
261static int curve_y_at_col(const gpopup_t *p, int col,
262 uint8_t plot_x, uint8_t plot_y,
263 uint8_t plot_w, uint8_t plot_h)
264{
265 int baseline = (int)(plot_y + plot_h - 1);
266 int px0, py0, px1, py1;
267 point_to_px(p, &p->points[0], plot_x, plot_y, plot_w, plot_h, &px0, &py0);
268 if (col <= px0) return py0;
269 /* Holds the first point's level when there is no segment to walk. */
270 py1 = py0;
271 for (uint8_t i = 1; i < p->num_points; ++i) {
272 point_to_px(p, &p->points[i], plot_x, plot_y, plot_w, plot_h, &px1, &py1);
273 if (col <= px1) {
274 int span = px1 - px0;
275 if (span <= 0) return py1;
276 if (p->shape) {
277 /* Evaluate in normalised level space so the callback sees the
278 * same v0/v1/t domain the synth engine uses. */
279 float t = (float)(col - px0) / (float)span;
280 float yn = p->shape(clampf01(p->points[i - 1].y),
281 clampf01(p->points[i].y), t);
282 return (int)(plot_y + (1.0f - clampf01(yn)) * (plot_h - 1));
283 }
284 return py0 + ((py1 - py0) * (col - px0)) / span;
285 }
286 px0 = px1; py0 = py1;
287 }
288 (void)baseline;
289 return py1; /* past the last point: hold its level */
290}

292void graph_popup_draw(u8g2_t *u8g2, const gpopup_t *p)
293{
294 if (!u8g2 || !p || !p->active) return;
295
296 const bool adsr = (p->style == GPOPUP_STYLE_ADSR);
297
298 /* ADSR style is full-screen and framed by the host's top bar, so it skips
299 * the rounded border for max plot area. */
300 u8g2_SetDrawColor(u8g2, 0);
301 u8g2_DrawBox(u8g2, p->x, p->y, p->w, p->h);
302 u8g2_SetDrawColor(u8g2, 1);
303 if (!adsr) {
304 u8g2_DrawRFrame(u8g2, p->x, p->y, p->w, p->h, 2);
305 }
306
307 /* Optional title (PLAIN style only; ADSR uses the host top bar). */
308 if (p->title && !adsr) {
309 u8g2_SetFont(u8g2, u8g2_font_5x7_tr);
310 u8g2_DrawStr(u8g2, (uint8_t)(p->x + 3), (uint8_t)(p->y + 7), p->title);
311 }
312
313 uint8_t plot_x, plot_y, plot_w, plot_h;
314 plot_rect(p, &plot_x, &plot_y, &plot_w, &plot_h);
315
316 int baseline = (int)(plot_y + plot_h - 1);
317
318 /* Axes: bottom + left of the plot region. */
319 u8g2_DrawHLine(u8g2, plot_x, (uint8_t)baseline, plot_w);
320 u8g2_DrawVLine(u8g2, plot_x, plot_y, plot_h);
321
322 if (p->num_points == 0) return;
323
324 /* ADSR: fill under the curve first (one VLine per column down to the
325 * baseline) so the points and curve edge sit on top. */
326 if (adsr) {
327 int last_x;
328 {
329 int lx, ly;
330 point_to_px(p, &p->points[p->num_points - 1], plot_x, plot_y,
331 plot_w, plot_h, &lx, &ly);
332 last_x = lx;
333 }
334 for (int cx = (int)plot_x; cx <= last_x; ++cx) {
335 int cy = curve_y_at_col(p, cx, plot_x, plot_y, plot_w, plot_h);
336 cy = SEQ_CLAMP_INT(cy, (int)plot_y, baseline);
337 if (baseline - cy > 0) {
338 u8g2_DrawVLine(u8g2, (uint8_t)cx, (uint8_t)cy,
339 (uint8_t)(baseline - cy));
340 }
341 }
342 }
343
344 /* Curve, on top of any fill. With a shape callback the path is evaluated
345 * per column so non-linear segments render their true profile, joining
346 * adjacent columns with lines to keep steep runs solid; otherwise straight
347 * segments between points. */
348 if (p->shape && p->num_points > 1) {
349 int fx, fy, lx, ly;
350 point_to_px(p, &p->points[0], plot_x, plot_y, plot_w, plot_h, &fx, &fy);
351 point_to_px(p, &p->points[p->num_points - 1], plot_x, plot_y,
352 plot_w, plot_h, &lx, &ly);
353 int prev_cx = fx, prev_cy = fy;
354 for (int cx = fx; cx <= lx; ++cx) {
355 int cy = curve_y_at_col(p, cx, plot_x, plot_y, plot_w, plot_h);
356 cy = SEQ_CLAMP_INT(cy, (int)plot_y, baseline);
357 u8g2_DrawLine(u8g2, prev_cx, prev_cy, cx, cy);
358 prev_cx = cx;
359 prev_cy = cy;
360 }
361 } else {
362 int prev_x = 0, prev_y = 0;
363 for (uint8_t i = 0; i < p->num_points; ++i) {
364 int cx, cy;
365 point_to_px(p, &p->points[i], plot_x, plot_y, plot_w, plot_h, &cx, &cy);
366 if (i > 0) {
367 u8g2_DrawLine(u8g2, prev_x, prev_y, cx, cy);
368 }
369 prev_x = cx;
370 prev_y = cy;
371 }
372 }
373
374 /* ADSR: minimal time tick marks in the bottom margin (just under the axis).
375 * Spacing reflects the host's (possibly non-linear) time->X mapping. */
376 if (adsr && p->num_ticks > 0) {
377 int ty = baseline + 1;
378 for (uint8_t i = 0; i < p->num_ticks; ++i) {
379 int tx = (int)(plot_x + clampf01(p->ticks[i]) * (plot_w - 1));
380 u8g2_DrawVLine(u8g2, (uint8_t)tx, (uint8_t)ty, 2);
381 }
382 }
383
384 /* ADSR: amplitude ticks at the quarter divisions in the 2px left margin,
385 * so both edges carry a scale (0%/100% are the axis ends). */
386 if (adsr && plot_x >= 2) {
387 for (uint8_t i = 1; i < 4; ++i) { /* 25%, 50%, 75% */
388 int tyy = (int)plot_y + (int)((4u - i) * (plot_h - 1)) / 4;
389 u8g2_DrawHLine(u8g2, (uint8_t)(plot_x - 2), (uint8_t)tyy, 2);
390 }
391 }
392
393 /* Edit-mode markers + cursor highlight. */
394 if (p->mode == GPOPUP_MODE_EDIT) {
395 /* ADSR letter labels for the editable points (A, D/S, R). */
396 static const char adsr_letters[4] = { 0, 'A', 'S', 'R' };
397
398 for (uint8_t i = 0; i < p->num_points; ++i) {
399 int cx, cy;
400 point_to_px(p, &p->points[i], plot_x, plot_y, plot_w, plot_h, &cx, &cy);
401
402 /* Label above-left of the point (ADSR style, skip the origin). The
403 * sustain marker reads 'D' while its decay-time (X) axis is being
404 * adjusted, 'S' otherwise, so the dual-axis point shows which
405 * value the encoder moves. */
406 char letter = adsr_letters[i];
407 if (i == 2 && i == p->cursor && p->editing_value
408 && adsr_point_dual_free(p, i) && !adsr_effective_axis_is_y(p, i)) {
409 letter = 'D';
410 }
411 if (adsr && i > 0 && i < 4 && letter) {
412 char lb[2] = { letter, 0 };
413 u8g2_SetFont(u8g2, u8g2_font_5x7_tr);
414 int lx = cx - 2;
415 int ly = cy - 4; /* sit above the disc */
416 if (ly < (int)plot_y + 6) ly = cy + 11; /* flip below if cramped */
417 lx = SEQ_CLAMP_INT(lx, (int)plot_x, (int)(plot_x + plot_w - 5));
418 if (i == p->cursor) {
419 /* Emphasise the selected label with an inverted pad. */
420 u8g2_DrawBox(u8g2, (uint8_t)(lx - 1), (uint8_t)(ly - 6), 7, 8);
421 u8g2_SetDrawColor(u8g2, 0);
422 u8g2_DrawStr(u8g2, (uint8_t)lx, (uint8_t)ly, lb);
423 u8g2_SetDrawColor(u8g2, 1);
424 } else {
425 u8g2_DrawStr(u8g2, (uint8_t)lx, (uint8_t)ly, lb);
426 }
427 }
428
429 if (i == p->cursor) {
430 /* Selected point: filled disc; when adjusting, ring around it. */
431 u8g2_DrawDisc(u8g2, cx, cy, 2, U8G2_DRAW_ALL);
432 if (p->editing_value) {
433 u8g2_DrawCircle(u8g2, cx, cy, 4, U8G2_DRAW_ALL);
434 }
435 } else if (!adsr || i > 0) {
436 /* Skip a marker on the fixed origin in ADSR style. */
437 u8g2_DrawFrame(u8g2, cx - 1, cy - 1, 3, 3);
438 }
439 }
440
441 /* Value readout while adjusting (PLAIN only; in ADSR style the host top
442 * bar carries the real ms/percent readout). */
443 if (!adsr && p->editing_value && p->cursor < p->num_points) {
444 char buf[16];
445 const gpopup_point_t *pt = &p->points[p->cursor];
446 /* uint8_t so the percentages are provably three digits: the 0..100
447 * float bound does not survive the float->int conversion, and
448 * -Wformat-truncation then assumes a full-width int per %d. */
449 uint8_t rx = SEQ_CLAMP_U8((int)(clampf01(pt->x) * 100.0f + 0.5f), 0, 100);
450 uint8_t ry = SEQ_CLAMP_U8((int)(clampf01(pt->y) * 100.0f + 0.5f), 0, 100);
451 snprintf(buf, sizeof(buf), "%c %d,%d",
452 p->adjust_axis_y ? 'V' : 'H', rx, ry);
453 u8g2_SetFont(u8g2, u8g2_font_5x7_tr);
454 /* Bottom-right inside the box. */
455 uint8_t tw = (uint8_t)u8g2_GetStrWidth(u8g2, buf);
456 u8g2_DrawStr(u8g2,
457 (uint8_t)(p->x + p->w - tw - 3),
458 (uint8_t)(p->y + p->h - 3),
459 buf);
460 }
461 }
462}

466gpopup_result_t graph_popup_handle_encoder(gpopup_t *p, long delta)
467{
468 if (!p || !p->active || delta == 0) return GPOPUP_RESULT_NONE;
469 if (p->mode != GPOPUP_MODE_EDIT || p->num_points == 0) {
470 return GPOPUP_RESULT_NONE;
471 }
472
473 /* In ADSR style the origin (index 0) is fixed and not selectable. */
474 long min_idx = (p->style == GPOPUP_STYLE_ADSR) ? 1 : 0;
475
476 if (!p->editing_value) {
477 /* Selecting: move the cursor between points, clamped to ends. */
478 long idx = (long)p->cursor + delta;
479 idx = SEQ_CLAMP_INT(idx, min_idx, (long)p->num_points - 1);
480 p->cursor = (uint8_t)idx;
481 } else {
482 /* Adjusting along the point's editable axis (auto-selected in ADSR
483 * mode, manual adjust_axis_y flag in the generic editor). */
484 gpopup_point_t *pt = &p->points[p->cursor];
485 float step = (float)delta * GPOPUP_ADJUST_STEP;
486 if (adsr_effective_axis_is_y(p, p->cursor)) {
487 /* Role-fixed levels (A peak, R end) ignore Y nudges in ADSR mode. */
488 if (adsr_y_editable(p, p->cursor)) {
489 pt->y = clampf01(pt->y + step);
490 }
491 } else {
492 /* Locked points (origin, host-owned S.x) ignore X nudges. An
493 * installed xstep (audio-taper time stride) wins over the fixed
494 * normalised step. */
495 if (adsr_x_editable(p, p->cursor)) {
496 if (p->xstep) {
497 pt->x = clampf01(p->xstep(p->cursor, pt->x, delta));
498 } else {
499 pt->x = clampf01(pt->x + step);
500 }
501 }
502 }
503 /* Re-impose ADSR validity (monotonic time, role-fixed levels). */
504 adsr_apply_constraints(p);
505 }
506 return GPOPUP_RESULT_NONE;
507}

509gpopup_result_t graph_popup_handle_button(gpopup_t *p)
510{
511 if (!p || !p->active) return GPOPUP_RESULT_NONE;
512 if (p->mode != GPOPUP_MODE_EDIT) {
513 /* VIEW mode: short press confirms (dismiss the preview). */
514 return GPOPUP_RESULT_CONFIRMED;
515 }
516 /* EDIT mode. A dual-axis point carries two values on one marker, so a
517 * single toggle can't reach both: cycle select -> level (Y) -> decay time
518 * (X) -> select. Single-axis points keep the plain select/adjust toggle. */
519 if (adsr_point_dual_free(p, p->cursor)) {
520 if (!p->editing_value) {
521 p->editing_value = true;
522 p->adjust_axis_y = true; /* start on the sustain level */
523 } else if (p->adjust_axis_y) {
524 p->adjust_axis_y = false; /* switch to the decay time */
525 } else {
526 p->editing_value = false; /* back to point selection */
527 p->adjust_axis_y = true; /* reset default for the next entry */
528 }
529 return GPOPUP_RESULT_NONE;
530 }
531 p->editing_value = !p->editing_value;
532 return GPOPUP_RESULT_NONE;
533}
