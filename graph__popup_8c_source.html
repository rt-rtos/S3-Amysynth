13static inline float clampf01(float v)
14{
15 if (v < 0.0f) return 0.0f;
16 if (v > 1.0f) return 1.0f;
17 return v;
18}

20/* Inset (in px) of the plot region inside the box border. Leaves room for the
21 * frame and an optional title strip at the top. */

24#define GPOPUP_PAD_BOT 4

30/* Compute the inner plot rectangle (in absolute screen pixels). */
31static void plot_rect(const gpopup_t *p,
32 uint8_t *px, uint8_t *py, uint8_t *pw, uint8_t *ph)
33{
34 /* ADSR style has no internal title strip (host draws the top bar) and uses
35 * tighter insets to maximise the full-screen plot; it reserves 3px at the
36 * bottom for the margin tick marks. */
37 bool adsr = (p->style == GPOPUP_STYLE_ADSR);
38 uint8_t top_pad = adsr ? 2 : (GPOPUP_PAD_TOP + (p->title ? GPOPUP_TITLE_H : 0));
39 uint8_t bot_pad = adsr ? (GPOPUP_PAD_BOT + 3) : GPOPUP_PAD_BOT;
40 uint8_t side_pad = adsr ? 2 : GPOPUP_PAD_X;
41 *px = (uint8_t)(p->x + side_pad);
42 *py = (uint8_t)(p->y + top_pad);
43 *pw = (uint8_t)(p->w - 2 * side_pad);
44 *ph = (uint8_t)(p->h - top_pad - bot_pad);
45}

47/* Map a normalised point to absolute screen pixels within the plot rect.
48 * Y is inverted (screen y grows downward). */
49static void point_to_px(const gpopup_t *p, const gpopup_point_t *pt,
50 uint8_t plot_x, uint8_t plot_y,
51 uint8_t plot_w, uint8_t plot_h,
52 int *out_x, int *out_y)
53{
54 float xn = clampf01(pt->x);
55 float yn = clampf01(pt->y);
56 *out_x = (int)(plot_x + xn * (plot_w - 1));
57 *out_y = (int)(plot_y + (1.0f - yn) * (plot_h - 1));
58}

143/* ── ADSR role-aware constraints ─────────────────────────────────────────────
144 * Points are [origin, A(attack peak), D(decay end / sustain level), R(release
145 * end)]. Enforce a musically-valid shape so the user cannot create nonsensical
146 * curves:
147 * - X is monotonic with a tiny minimum gap so points can't cross or collapse.
148 * - origin pinned to (0,0); A.y pinned to 1.0; R.y pinned to 0.0.
149 * - only the D point's Y (the sustain level) is freely movable.
150 * Index roles assume the standard 4-point ADSR seed used by the host. */
151/* Minimum normalised spacing between points. Sub-pixel on a 128px plot: only
152 * prevents exact crossing/collapse, so a fine-grained (audio-taper) host step
153 * can bring segments down to a few ms without the spacing rule fighting it. */
154#define GPOPUP_MIN_X_GAP 0.004f
155
156static void adsr_apply_constraints(gpopup_t *p)
157{
158 if (p->style != GPOPUP_STYLE_ADSR || p->num_points < 4) return;
159
160 gpopup_point_t *pt = p->points;
161
162 /* Role-fixed Y levels. */
163 pt[0].x = 0.0f; pt[0].y = 0.0f; /* origin */
164 pt[1].y = 1.0f; /* attack peak */
165 pt[3].y = 0.0f; /* release end */
166 /* pt[2].y is the sustain level — left as edited, just bounded 0..1. */
167 pt[2].y = clampf01(pt[2].y);
168
169 /* Monotonic X with a minimum gap, walking forward from the fixed origin.
170 * Each editable point is bounded below by its predecessor + gap and above
171 * by 1.0; a second backward pass keeps an earlier point from being pushed
172 * past a later one when the later point was dragged left. */
173 for (uint8_t i = 1; i < p->num_points; ++i) {
174 float lo = pt[i - 1].x + GPOPUP_MIN_X_GAP;
175 if (pt[i].x < lo) pt[i].x = lo;
176 if (pt[i].x > 1.0f) pt[i].x = 1.0f;
177 }
178 for (int i = (int)p->num_points - 2; i >= 1; --i) {
179 float hi = pt[i + 1].x - GPOPUP_MIN_X_GAP;
180 if (pt[i].x > hi) pt[i].x = hi;
181 if (pt[i].x < 0.0f) pt[i].x = 0.0f;
182 }
183}

185/* Whether the selected point's Y is editable in ADSR style. Only the sustain
186 * point (index 2) has a free level; A/R levels are role-fixed. */
187static bool adsr_y_editable(const gpopup_t *p, uint8_t idx)
188{
189 if (p->style != GPOPUP_STYLE_ADSR) return true;
190 return (idx == 2);
191}
192
193/* Whether the selected point's X (time) is editable in ADSR style. When
194 * adsr_lock_sx is set, the sustain point (index 2) is Y-only: its X is owned by
195 * the host (auto-derived decay time), so user X nudges are ignored. A and R
196 * remain X-editable; the origin (0) is fixed regardless. */
197static bool adsr_x_editable(const gpopup_t *p, uint8_t idx)
198{
199 if (p->style != GPOPUP_STYLE_ADSR) return true;
200 if (idx == 0) return false; /* origin is pinned */
201 if (p->adsr_lock_sx && idx == 2) return false; /* S.x host-owned */
202 return true;
203}
204
205/* True when a point can move on BOTH axes: the explicit-decay sustain point
206 * (lock_sx off) is X (decay time) and Y (sustain level) at once. In the derived
207 * decay mode (lock_sx on) no point is ever dual-free, so all the two-axis
208 * handling below stays dormant and behaviour is byte-identical to before. */
209static bool adsr_point_dual_free(const gpopup_t *p, uint8_t idx)
210{
211 return p->style == GPOPUP_STYLE_ADSR
212 && adsr_x_editable(p, idx) && adsr_y_editable(p, idx);
213}
214
215/* In ADSR style each single-axis point moves on exactly one axis (A/R = time/X,
216 * S = level/Y when its X is host-owned), so there is nothing to toggle: the
217 * editor auto-selects the axis the selected point can actually move on. Returns
218 * true if the encoder should adjust Y for this point, false for X. For the
219 * dual-free sustain point (explicit decay) the manual axis flag decides, so the
220 * user can switch between sustain level (Y) and decay time (X). Also the
221 * fallback for the generic (non-ADSR) curve editor. */
222static bool adsr_effective_axis_is_y(const gpopup_t *p, uint8_t idx)
223{
224 if (p->style != GPOPUP_STYLE_ADSR) return p->adjust_axis_y;
225 if (adsr_y_editable(p, idx) && !adsr_x_editable(p, idx)) return true;
226 if (adsr_x_editable(p, idx) && !adsr_y_editable(p, idx)) return false;
227 /* Both free (explicit-decay sustain): honour the axis flag. Neither free
228 * (origin): value is irrelevant, default to Y. */
229 return p->adjust_axis_y;
230}

257/* Interpolate the curve's screen-Y at a given screen column, walking the
258 * point segments. Returns the baseline if the column is outside the curve. */
259static int curve_y_at_col(const gpopup_t *p, int col,
260 uint8_t plot_x, uint8_t plot_y,
261 uint8_t plot_w, uint8_t plot_h)
262{
263 int baseline = (int)(plot_y + plot_h - 1);
264 int px0, py0, px1, py1;
265 point_to_px(p, &p->points[0], plot_x, plot_y, plot_w, plot_h, &px0, &py0);
266 if (col <= px0) return py0;
267 /* Hold the first point's level if there is no segment to walk (<=1 point);
268 * updated to each segment's far end as we iterate. */
269 py1 = py0;
270 for (uint8_t i = 1; i < p->num_points; ++i) {
271 point_to_px(p, &p->points[i], plot_x, plot_y, plot_w, plot_h, &px1, &py1);
272 if (col <= px1) {
273 int span = px1 - px0;
274 if (span <= 0) return py1;
275 if (p->shape) {
276 /* Shaped segment: evaluate in normalised level space so the
277 * callback sees the same v0/v1/t domain the synth engine uses,
278 * then map the result back to a pixel row. */
279 float t = (float)(col - px0) / (float)span;
280 float yn = p->shape(clampf01(p->points[i - 1].y),
281 clampf01(p->points[i].y), t);
282 return (int)(plot_y + (1.0f - clampf01(yn)) * (plot_h - 1));
283 }
284 /* Linear interpolation between the two bracketing points. */
285 return py0 + ((py1 - py0) * (col - px0)) / span;
286 }
287 px0 = px1; py0 = py1;
288 }
289 (void)baseline;
290 return py1; /* past the last point: hold its level */
291}

293void graph_popup_draw(u8g2_t *u8g2, const gpopup_t *p)
294{
295 if (!u8g2 || !p || !p->active) return;
296
297 const bool adsr = (p->style == GPOPUP_STYLE_ADSR);
298
299 /* Background clear of the box area. ADSR style is full-screen and framed by
300 * the host's top bar, so it skips the rounded border for max plot area. */
301 u8g2_SetDrawColor(u8g2, 0);
302 u8g2_DrawBox(u8g2, p->x, p->y, p->w, p->h);
303 u8g2_SetDrawColor(u8g2, 1);
304 if (!adsr) {
305 u8g2_DrawRFrame(u8g2, p->x, p->y, p->w, p->h, 2);
306 }
307
308 /* Optional title (PLAIN style only; ADSR uses the host top bar). */
309 if (p->title && !adsr) {
310 u8g2_SetFont(u8g2, u8g2_font_5x7_tr);
311 u8g2_DrawStr(u8g2, (uint8_t)(p->x + 3), (uint8_t)(p->y + 7), p->title);
312 }
313
314 uint8_t plot_x, plot_y, plot_w, plot_h;
315 plot_rect(p, &plot_x, &plot_y, &plot_w, &plot_h);
316
317 int baseline = (int)(plot_y + plot_h - 1);
318
319 /* Axes: bottom + left of the plot region. */
320 u8g2_DrawHLine(u8g2, plot_x, (uint8_t)baseline, plot_w);
321 u8g2_DrawVLine(u8g2, plot_x, plot_y, plot_h);
322
323 if (p->num_points == 0) return;
324
325 /* ADSR: fill the area under the curve before the line/markers so points and
326 * the crisp curve edge sit on top. One VLine per column from the curve down
327 * to the baseline. */
328 if (adsr) {
329 int last_x;
330 {
331 int lx, ly;
332 point_to_px(p, &p->points[p->num_points - 1], plot_x, plot_y,
333 plot_w, plot_h, &lx, &ly);
334 last_x = lx;
335 }
336 for (int cx = (int)plot_x; cx <= last_x; ++cx) {
337 int cy = curve_y_at_col(p, cx, plot_x, plot_y, plot_w, plot_h);
338 if (cy < (int)plot_y) cy = (int)plot_y;
339 if (cy > baseline) cy = baseline;
340 if (baseline - cy > 0) {
341 u8g2_DrawVLine(u8g2, (uint8_t)cx, (uint8_t)cy,
342 (uint8_t)(baseline - cy));
343 }
344 }
345 }
346
347 /* Curve: drawn on top of any fill. With a shape callback the path is
348 * evaluated per column so non-linear segments render their true profile;
349 * adjacent columns are joined with short lines to keep steep runs solid.
350 * Without one, straight segments between points as before. */
351 if (p->shape && p->num_points > 1) {
352 int fx, fy, lx, ly;
353 point_to_px(p, &p->points[0], plot_x, plot_y, plot_w, plot_h, &fx, &fy);
354 point_to_px(p, &p->points[p->num_points - 1], plot_x, plot_y,
355 plot_w, plot_h, &lx, &ly);
356 int prev_cx = fx, prev_cy = fy;
357 for (int cx = fx; cx <= lx; ++cx) {
358 int cy = curve_y_at_col(p, cx, plot_x, plot_y, plot_w, plot_h);
359 if (cy < (int)plot_y) cy = (int)plot_y;
360 if (cy > baseline) cy = baseline;
361 u8g2_DrawLine(u8g2, prev_cx, prev_cy, cx, cy);
362 prev_cx = cx;
363 prev_cy = cy;
364 }
365 } else {
366 int prev_x = 0, prev_y = 0;
367 for (uint8_t i = 0; i < p->num_points; ++i) {
368 int cx, cy;
369 point_to_px(p, &p->points[i], plot_x, plot_y, plot_w, plot_h, &cx, &cy);
370 if (i > 0) {
371 u8g2_DrawLine(u8g2, prev_x, prev_y, cx, cy);
372 }
373 prev_x = cx;
374 prev_y = cy;
375 }
376 }
377
378 /* ADSR: minimal time tick marks in the bottom margin (just under the axis).
379 * Spacing reflects the host's (possibly non-linear) time->X mapping. */
380 if (adsr && p->num_ticks > 0) {
381 int ty = baseline + 1;
382 for (uint8_t i = 0; i < p->num_ticks; ++i) {
383 int tx = (int)(plot_x + clampf01(p->ticks[i]) * (plot_w - 1));
384 u8g2_DrawVLine(u8g2, (uint8_t)tx, (uint8_t)ty, 2);
385 }
386 }
387
388 /* ADSR: level tick marks on the left (Y) axis at the quarter divisions, so
389 * both box edges carry a scale reference — time along the bottom, amplitude
390 * up the side (0%/100% are the axis ends). Drawn in the 2px left margin. */
391 if (adsr && plot_x >= 2) {
392 for (uint8_t i = 1; i < 4; ++i) { /* 25%, 50%, 75% */
393 int tyy = (int)plot_y + (int)((4u - i) * (plot_h - 1)) / 4;
394 u8g2_DrawHLine(u8g2, (uint8_t)(plot_x - 2), (uint8_t)tyy, 2);
395 }
396 }
397
398 /* Edit-mode markers + cursor highlight. */
399 if (p->mode == GPOPUP_MODE_EDIT) {
400 /* ADSR letter labels for the editable points (A, D/S, R). */
401 static const char adsr_letters[4] = { 0, 'A', 'S', 'R' };
402
403 for (uint8_t i = 0; i < p->num_points; ++i) {
404 int cx, cy;
405 point_to_px(p, &p->points[i], plot_x, plot_y, plot_w, plot_h, &cx, &cy);
406
407 /* Label above-left of the point (ADSR style, skip the origin).
408 * The sustain marker reads 'D' while its decay-time (X) axis is the
409 * one being adjusted, 'S' otherwise, so the dual-axis point shows
410 * which value the encoder currently moves. */
411 char letter = adsr_letters[i];
412 if (i == 2 && i == p->cursor && p->editing_value
413 && adsr_point_dual_free(p, i) && !adsr_effective_axis_is_y(p, i)) {
414 letter = 'D';
415 }
416 if (adsr && i > 0 && i < 4 && letter) {
417 char lb[2] = { letter, 0 };
418 u8g2_SetFont(u8g2, u8g2_font_5x7_tr);
419 int lx = cx - 2;
420 int ly = cy - 4; /* sit above the disc */
421 if (ly < (int)plot_y + 6) ly = cy + 11; /* flip below if cramped */
422 if (lx < (int)plot_x) lx = (int)plot_x;
423 if (lx > (int)(plot_x + plot_w - 5)) lx = (int)(plot_x + plot_w - 5);
424 if (i == p->cursor) {
425 /* Emphasise the selected label with an inverted pad. */
426 u8g2_DrawBox(u8g2, (uint8_t)(lx - 1), (uint8_t)(ly - 6), 7, 8);
427 u8g2_SetDrawColor(u8g2, 0);
428 u8g2_DrawStr(u8g2, (uint8_t)lx, (uint8_t)ly, lb);
429 u8g2_SetDrawColor(u8g2, 1);
430 } else {
431 u8g2_DrawStr(u8g2, (uint8_t)lx, (uint8_t)ly, lb);
432 }
433 }
434
435 if (i == p->cursor) {
436 /* Selected point: filled disc; when adjusting, ring around it. */
437 u8g2_DrawDisc(u8g2, cx, cy, 2, U8G2_DRAW_ALL);
438 if (p->editing_value) {
439 u8g2_DrawCircle(u8g2, cx, cy, 4, U8G2_DRAW_ALL);
440 }
441 } else if (!adsr || i > 0) {
442 /* Skip a marker on the fixed origin in ADSR style. */
443 u8g2_DrawFrame(u8g2, cx - 1, cy - 1, 3, 3);
444 }
445 }
446
447 /* Value readout for the selected point while adjusting (PLAIN style).
448 * In ADSR style the host top bar shows the real ms/percent readout, so
449 * the in-plot text is suppressed to keep the plot clean. */
450 if (!adsr && p->editing_value && p->cursor < p->num_points) {
451 char buf[16];
452 const gpopup_point_t *pt = &p->points[p->cursor];
453 snprintf(buf, sizeof(buf), "%c %d,%d",
454 p->adjust_axis_y ? 'V' : 'H',
455 (int)(clampf01(pt->x) * 100.0f + 0.5f),
456 (int)(clampf01(pt->y) * 100.0f + 0.5f));
457 u8g2_SetFont(u8g2, u8g2_font_5x7_tr);
458 /* Bottom-right inside the box. */
459 uint8_t tw = (uint8_t)u8g2_GetStrWidth(u8g2, buf);
460 u8g2_DrawStr(u8g2,
461 (uint8_t)(p->x + p->w - tw - 3),
462 (uint8_t)(p->y + p->h - 3),
463 buf);
464 }
465 }
466}

470gpopup_result_t graph_popup_handle_encoder(gpopup_t *p, long delta)
471{
472 if (!p || !p->active || delta == 0) return GPOPUP_RESULT_NONE;
473 if (p->mode != GPOPUP_MODE_EDIT || p->num_points == 0) {
474 return GPOPUP_RESULT_NONE;
475 }
476
477 /* In ADSR style the origin (index 0) is fixed and not selectable. */
478 long min_idx = (p->style == GPOPUP_STYLE_ADSR) ? 1 : 0;
479
480 if (!p->editing_value) {
481 /* Selecting: move the cursor between points, clamped to ends. */
482 long idx = (long)p->cursor + delta;
483 if (idx < min_idx) idx = min_idx;
484 if (idx > (long)p->num_points - 1) idx = (long)p->num_points - 1;
485 p->cursor = (uint8_t)idx;
486 } else {
487 /* Adjusting: change the selected point along its editable axis. In ADSR
488 * mode the axis is auto-selected per point (no X/Y toggle needed); the
489 * generic curve editor still honours the manual adjust_axis_y flag. */
490 gpopup_point_t *pt = &p->points[p->cursor];
491 float step = (float)delta * GPOPUP_ADJUST_STEP;
492 if (adsr_effective_axis_is_y(p, p->cursor)) {
493 /* Role-fixed levels (A peak, R end) ignore Y nudges in ADSR mode. */
494 if (adsr_y_editable(p, p->cursor)) {
495 pt->y = clampf01(pt->y + step);
496 }
497 } else {
498 /* Locked points (origin, host-owned S.x) ignore X nudges. The host
499 * xstep override (audio-taper time stride) wins over the fixed
500 * normalised step when installed. */
501 if (adsr_x_editable(p, p->cursor)) {
502 if (p->xstep) {
503 pt->x = clampf01(p->xstep(p->cursor, pt->x, delta));
504 } else {
505 pt->x = clampf01(pt->x + step);
506 }
507 }
508 }
509 /* Re-impose ADSR validity (monotonic time, role-fixed levels). */
510 adsr_apply_constraints(p);
511 }
512 return GPOPUP_RESULT_NONE;
513}

522 /* EDIT mode. A dual-axis point (the explicit-decay sustain point) carries
523 * two independent values on one marker, so a single toggle can't reach both.
524 * Cycle it: select -> adjust level (Y) -> adjust decay time (X) -> select.
525 * Single-axis points keep the plain select <-> adjust toggle. */
526 if (adsr_point_dual_free(p, p->cursor)) {
527 if (!p->editing_value) {
528 p->editing_value = true;
529 p->adjust_axis_y = true; /* start on the sustain level */
530 } else if (p->adjust_axis_y) {
531 p->adjust_axis_y = false; /* switch to the decay time */
532 } else {
533 p->editing_value = false; /* back to point selection */
534 p->adjust_axis_y = true; /* reset default for the next entry */
535 }
536 return GPOPUP_RESULT_NONE;
537 }
538 p->editing_value = !p->editing_value;
539 return GPOPUP_RESULT_NONE;
540}
