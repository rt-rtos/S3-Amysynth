2/*
3 * graph_popup — reusable pop-up graph box for the U8g2 (SSD1315 128x64) UI.
4 *
5 * Purpose:
6 * A single, self-contained widget for drawing and (optionally) editing a
7 * curve inside a framed pop-up box that overlays the main screen. It is used
8 * for AD/ADSR envelopes, filter curves, and waveform visualisation without
9 * re-implementing the graphing/editing logic each time.
10 *
11 * Design:
12 * - The widget is AMY-agnostic and sequencer-agnostic. It operates purely on
13 * a caller-owned array of *normalised* points (x and y each in 0.0..1.0).
14 * - The host decides WHEN to open it, seeds the points, routes input to it,
15 * and reads the (possibly edited) points back. Mapping domain values
16 * (ms, levels, log-freq) to/from the normalised 0..1 space lives in
17 * adapters outside this file (e.g. graph_popup_amy.c in synth_core).
18 *
19 * Rendering contract:
20 * graph_popup_draw() draws ONLY the pop-up box (background, frame, axes,
21 * curve and — in edit mode — point markers/cursor). The host is responsible
22 * for u8g2_ClearBuffer(), any context frame around the box, and
23 * u8g2_SendBuffer().
24 *
25 * Threading:
26 * Plain state, no internal locking. If input and render run on different
27 * tasks, the host must serialise access to gpopup_t the same way it does for
28 * its other shared UI state.
29 */

55typedef enum {

61/* Optional rendering / editing style. PLAIN keeps the original generic widget
62 * behaviour. ADSR turns on envelope-specific behaviour: the 4 points are treated
63 * as [origin, Attack-peak, Decay/Sustain, Release-end], which enables
64 * - role-aware clamping (monotonic time; A.y pinned to 1.0, R.y to 0.0; only
65 * the sustain level is freely movable in Y) so only valid, musical envelopes
66 * are reachable;
67 * - filled area under the curve;
68 * - A/D/S/R letter labels over the points;
69 * - minimal time tick marks in the bottom margin.
70 * The widget stays AMY-agnostic: it only knows the ADSR point ROLES, not ms. */
71typedef enum {
72 GPOPUP_STYLE_PLAIN = 0,
73 GPOPUP_STYLE_ADSR = 1,
74} gpopup_style_t;
75
76/* Optional per-segment curve shaping. Called per plot column with the segment's
77 * endpoint levels v0/v1 (normalised 0..1) and the position t (0..1) within the
78 * segment; returns the curve level at t. NULL = straight-line segments. Lets
79 * the host render the true shape of a non-linear envelope (exponential, DX7)
80 * while the widget itself stays domain-agnostic: points remain the editable
81 * anchors, only the drawn path between them changes. */
82typedef float (*gpopup_shape_fn)(float v0, float v1, float t);
83
84/* Optional X-axis stepping override for point editing. Called with the point
85 * index, its current normalised X and the encoder detent count; returns the new
86 * normalised X (clamped to 0..1 by the widget). Lets the host implement a
87 * non-linear (e.g. audio-taper time) stride in its own domain units instead of
88 * the widget's fixed normalised step. NULL = fixed linear step. */
89typedef float (*gpopup_xstep_fn)(uint8_t idx, float x, long delta);

110 bool adsr_lock_sx;

112 gpopup_shape_fn shape;
113 /* Optional X-axis stepping override for edits (see gpopup_xstep_fn). */
114 gpopup_xstep_fn xstep;
115} gpopup_t;
