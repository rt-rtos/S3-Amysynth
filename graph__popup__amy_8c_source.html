1/*
2 * graph_popup_amy — AMY <-> graph_popup adapter.
3 *
4 * Converts between AMY envelope breakpoint arrays (times in milliseconds,
5 * values 0..1) and the normalised point model used by the graph_popup widget.
6 * This lives in synth_core (which already REQUIRES amy) so the display
7 * component (priv_i2c_u8g2) stays free of any AMY dependency.
8 *
9 * Normalisation:
10 * x = cumulative time / total time (so the curve spans the full width)
11 * y = breakpoint value (already 0..1)
12 * An implicit origin point (0,0) is prepended so the curve starts at the
13 * bottom-left of the plot, matching how an envelope rises from silence.
14 */
15

27uint8_t gpopup_points_from_envelope(gpopup_point_t *out, uint8_t max_out,
28 const uint32_t *times_ms,
29 const float *values,
30 uint8_t num_bps)
31{
32 if (!out || max_out == 0) return 0;
33
34 /* Total time = sum of all breakpoint times. Guard against zero. */
35 uint32_t total = 0;
36 for (uint8_t i = 0; i < num_bps; ++i) {
37 total += times_ms ? times_ms[i] : 0;
38 }
39 if (total == 0) total = 1; /* avoid divide-by-zero; degenerate flat line */
40
41 uint8_t n = 0;
42
43 /* Implicit origin at (0,0). */
44 out[n].x = 0.0f;
45 out[n].y = 0.0f;
46 n++;
47
48 uint32_t cumulative = 0;
49 for (uint8_t i = 0; i < num_bps && n < max_out; ++i) {
50 cumulative += times_ms ? times_ms[i] : 0;
51 out[n].x = (float)cumulative / (float)total;
52 out[n].y = values ? clampf01(values[i]) : 0.0f;
53 n++;
54 }
55 return n;
56}

58uint8_t gpopup_points_to_envelope(const gpopup_point_t *pts, uint8_t n,
59 uint32_t total_ms,
60 uint32_t *out_times_ms,
61 float *out_values)
62{
63 if (!pts || n == 0 || !out_times_ms || !out_values) return 0;
64
65 /* Skip the implicit origin point at index 0; each subsequent point becomes
66 * a breakpoint whose time is the delta from the previous point's x. */
67 uint8_t bps = 0;
68 float prev_x = pts[0].x;
69 for (uint8_t i = 1; i < n; ++i) {
70 float dx = pts[i].x - prev_x;
71 if (dx < 0.0f) dx = 0.0f;
72 out_times_ms[bps] = (uint32_t)(dx * (float)total_ms + 0.5f);
73 out_values[bps] = clampf01(pts[i].y);
74 prev_x = pts[i].x;
75 bps++;
76 }
77 return bps;
78}
