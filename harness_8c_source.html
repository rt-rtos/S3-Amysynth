38/* Responses go through printf like the seq dump does: raw lines on the
39 * console UART, no ESP_LOG prefix, so the host parser keys on "H<" columns
40 * regardless of interleaved log output. */
41#define REPLY(fmt, ...) printf("H< " fmt "\n", ##__VA_ARGS__)

43/* Return the value of "key=" in args, or NULL. Args is the mutable tail of
44 * the command line; values are space-delimited. */
45static char *arg_value(char *args, const char *key)
46{
47 size_t klen = strlen(key);
48 for (char *tok = strtok(args, " "); tok; tok = strtok(NULL, " ")) {
49 if (strncmp(tok, key, klen) == 0 && tok[klen] == '=') {
50 return tok + klen + 1;
51 }
52 }
53 return NULL;
54}

97static void cmd_in_btn(char *args)
98{
99 if (s_hooks == NULL || s_hooks->inject_button == NULL) {
100 REPLY("err no button hook");
101 return;
102 }
103 /* arg_value consumes args via strtok; grab both keys in one pass by
104 * scanning the copies. */
105 char copy[HARNESS_LINE_MAX];
106 strlcpy(copy, args, sizeof(copy));
107 char *id_s = arg_value(copy, "id");
108 char copy2[HARNESS_LINE_MAX];
109 strlcpy(copy2, args, sizeof(copy2));
110 char *act_s = arg_value(copy2, "act");
111 if (id_s == NULL || act_s == NULL) {
112 REPLY("err usage: in.btn id=<n> act=down|up|click|long");
113 return;
114 }
115 harness_btn_act_t act;
116 if (strcmp(act_s, "down") == 0) act = HARNESS_BTN_DOWN;
117 else if (strcmp(act_s, "up") == 0) act = HARNESS_BTN_UP;
118 else if (strcmp(act_s, "click") == 0) act = HARNESS_BTN_CLICK;
119 else if (strcmp(act_s, "long") == 0) act = HARNESS_BTN_LONG;
120 else { REPLY("err bad act '%s'", act_s); return; }
121
122 s_hooks->inject_button(atoi(id_s), act);
123 REPLY("ok");
124}

151static void handle_line(char *line)
152{
153 /* Strip the command sentinel; tolerate its absence so hand-typed
154 * commands in a terminal work too. */
155 if (strncmp(line, "H>", 2) == 0) line += 2;
156 while (*line == ' ') line++;
157 if (*line == '\0') return; /* blank/log echo - ignore silently */
158
159 char *args = strchr(line, ' ');
160 if (args != NULL) { *args++ = '\0'; } else { args = line + strlen(line); }
161
162 if (strcmp(line, "sys.echo") == 0) REPLY("ok %s", args);
163 else if (strcmp(line, "sys.build") == 0) cmd_sys_build();
164 else if (strcmp(line, "st.heap") == 0) cmd_st_heap();
165 else if (strcmp(line, "st.drop") == 0) cmd_st_drop();
166 else if (strcmp(line, "st.uacpull") == 0) {
167#if CONFIG_AMYSYNTH_DROPOUT_TS
168 uint32_t c[4];
169 int64_t t_us;
170 if (uac_device_get_pull_stats(c, &t_us) == ESP_OK) {
171 REPLY("ok t_us=%lld preload=%" PRIu32 " skip_room=%" PRIu32
172 " pull=%" PRIu32 " bytes=%" PRIu32,
173 (long long)t_us, c[0], c[1], c[2], c[3]);
174 } else {
175 REPLY("err pull stats unavailable");
176 }
177#else
178 REPLY("err dropout ts not compiled in");
179#endif
180 }
181 else if (strcmp(line, "st.dropts") == 0) {
182#if CONFIG_AMYSYNTH_DROPOUT_TS
183 cmd_st_dropts();
184#else
185 REPLY("err dropout ts not compiled in");
186#endif
187 }
188 else if (strcmp(line, "st.render") == 0) {
189#if CONFIG_AMYSYNTH_RENDER_STATS
190 render_stats_report(); /* prints its own block */
191 REPLY("ok");
192#else
193 REPLY("err render_stats not compiled in");
194#endif
195 }
196 else if (strcmp(line, "st.seqdump") == 0) {
197#if CONFIG_SYNTH_DEV_MENU
198 sequencer_core_dump_state(); /* prints SEQDUMP BEGIN/END */
199 REPLY("ok");
200#else
201 REPLY("err dev menu not compiled in");
202#endif
203 }
204 else if (strcmp(line, "tr.play") == 0) { sequencer_core_set_playing(true); REPLY("ok"); }
205 else if (strcmp(line, "tr.stop") == 0) { sequencer_core_set_playing(false); REPLY("ok"); }
206 else if (strcmp(line, "tr.bpm") == 0) cmd_tr_bpm(args);
207 else if (strcmp(line, "in.btn") == 0) cmd_in_btn(args);
208 else if (strcmp(line, "in.enc") == 0) cmd_in_enc(args);
209 else REPLY("err unknown cmd '%s'", line);
210}

212static void harness_task(void *arg)
213{
214 (void)arg;
215 static char line[HARNESS_LINE_MAX];
216 size_t len = 0;
217
218 REPLY("ok harness ready");
219 for (;;) {
220 uint8_t ch;
221 int n = uart_read_bytes(HARNESS_UART, &ch, 1, pdMS_TO_TICKS(100));
222 if (n != 1) continue;
223 if (ch == '\r') continue;
224 if (ch != '\n') {
225 if (len < sizeof(line) - 1) {
226 line[len++] = (char)ch;
227 } else {
228 len = 0; /* oversize: drop the whole line */
229 REPLY("err line too long");
230 }
231 continue;
232 }
233 line[len] = '\0';
234 len = 0;
235 handle_line(line);
236 }
237}

239esp_err_t harness_init(const harness_hooks_t *hooks)
240{
241 if (hooks == NULL) return ESP_ERR_INVALID_ARG;
242 if (s_hooks != NULL) return ESP_ERR_INVALID_STATE;
243
244 /* RX driver only: console TX (logs, replies via printf) keeps using the
245 * default non-driver path; both coexist on UART0. */
246 esp_err_t err = uart_driver_install(HARNESS_UART, HARNESS_RX_BUF, 0, 0, NULL, 0);
247 if (err != ESP_OK) {
248 ESP_LOGW(TAG, "uart_driver_install failed: %s", esp_err_to_name(err));
249 return err;
250 }
251
252 s_hooks = hooks;
253 if (xTaskCreatePinnedToCore(harness_task, "harness_cmd",
254 HARNESS_TASK_STACK, NULL,
255 HARNESS_TASK_PRIO, NULL, 0) != pdPASS) {
256 uart_driver_delete(HARNESS_UART);
257 s_hooks = NULL;
258 return ESP_ERR_NO_MEM;
259 }
260 ESP_LOGI(TAG, "serial harness listening on UART0");
261 return ESP_OK;
262}
