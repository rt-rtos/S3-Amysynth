11/* Serial test harness: a line-command control channel on UART0 for driving a
12 * headless instance (input injection, transport, state dumps). Compiled in
13 * only under CONFIG_DEV_SERIAL_HARNESS; every symbol here is a no-op stub
14 * otherwise.
15 *
16 * Protocol (line-oriented ASCII on the console UART, coexists with log
17 * output): commands arrive as "H> <cmd> [args]", every command gets exactly
18 * one "H< ok ..." or "H< err <reason>" response line. Bulk dumps triggered by
19 * commands (seq dump, render stats) print their own existing formats between
20 * command and response; hosts key on the H< line for completion. */

31/* Injection hooks, provided by main.c so the harness never reaches into
32 * another component's statics.
33 *
34 * Execution context: both hooks are called on the harness command task
35 * (Core 0, low priority). inject_button must post to the same queue the
36 * physical button callback posts to (same drop-on-full policy); it must not
37 * block. inject_encoder_steps runs the encoder step-routing chain directly
38 * and therefore assumes NO concurrent physical encoder activity - the
39 * harness targets a bare board (contract: harness builds run without
40 * encoders attached; on a populated board, injected and physical encoder
41 * processing may interleave mid-gesture). */
42typedef struct {
43 void (*inject_button)(int button_id, harness_btn_act_t act);
44 void (*inject_encoder_steps)(long steps);
45} harness_hooks_t;
