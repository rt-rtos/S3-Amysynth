20// A note on RTOS synchronization:
21// For maximumum flexibility, AMY offers two flags:
22// - config.platform.multicore - whether to use a second core, if avaliable.
23// - config.platform.multithread - whether to use RTOS multithreading, if avaliable.
24// (ESP32 is the only platform where we've tried RTOS, so it's the only context where multithread matters).
25// When using RTOS threads (tasks), synchronization is handled by xTaskNotifyGive(task_handle) / ulTaskNotifyTake():
26// A service task waits to be notified by calling ulTaskNotifyTake(); that call does not specify where that
27// notification comes from. Another task calls xTaskNotifyGive() naming the particular task to be notified,
28// which then releases the ulTaskNotifyTake() in that task.
29// We use this mechanism as follows:
30// * esp_render_task (the task that runs on the second core):
31// - Waits for notification (from main fill_audio thread)
32// - renders half the oscs
33// - notifies either fill_buffer (multithread) or amy_update (single thread) when done.
34// - loop
35// * esp_fill_audio_buffer_task (launched as one parallel task):
36// - if not using I2S, waits for notification (from amy_update task)
37// - read I2S input (if used)
38// - execute AMY command updates (deltas)
39// - Notify esp_render_task
40// - render the other half of oscs
41// - Wait for notification (indicating that esp_render_task is done)
42// - call amy_fill_buffer (to combine the rendered oscs into output samples)
43// - Notify amy_update_handle that the new block is ready
44// - If I2S enabled, write samples to I2S.
45// - loop
46// * ESP's amy_platform_init
47// - if multicore, launch esp_render_task
48// - if multithread, launch esp_fill_audio_buffer_task
49// - if not I2S, Notify fill_buffer to get it started
50// * ESP's amy_render_audio
51// - If multithread, wait for Notification (from fill_audio_buffer)
52// - If not I2S, Notify fill_buffer
53// - If not multithread
54// - If multicore, Notify amy_render_handle to get task on second core to render
55// - wait for Notification indicating esp_render_task is done
56// - else render all oscs
57// - call amy_fill_buffer

172 // Initialize PCM9211 SPDIF transceiver via I2C.
173 // On the MicroPython AMYBOARD path this is done in Python (amyboard.py).
174 // Uses the new (driver_ng) I2C master API so it coexists with Arduino's Wire,
175 // which also uses driver_ng on ESP32 Arduino core 3.x.
176 {
177 #define PCM9211_I2C_ADDR 0x40
178 #define PCM9211_I2C_SDA 17
179 #define PCM9211_I2C_SCL 18
180 #define PCM9211_I2C_FREQ 400000
181
182 static const uint8_t pcm9211_regs[][2] = {
183 { 0x40, 0x33 }, // Power down ADC, DIR, DIT, OSC
184 { 0x40, 0xC0 }, // Normal operation for all
185 { 0x34, 0x00 }, // Initialize DIR - biphase amps on, input from RXIN0
186 { 0x26, 0x01 }, // Main Out is DIR/ADC if no DIR sync
187 { 0x6B, 0x00 }, // Main output pins are DIR/ADC AUTO
188 { 0x30, 0x04 }, // PLL sends 512fs as SCK
189 { 0x31, 0x0A }, // XTI SCK as 512fs too
190 { 0x60, 0x44 }, // DIT sends SPDIF from AUXIN1 through MPO0
191 { 0x61, 0x20 }, // DIT SCK ratio = 512fs (must match PLL config in 0x30/0x31)
192 { 0x78, 0x3D }, // MPO0 = TXOUT, MPO1 = VOUT
193 { 0x6F, 0x40 }, // MPIO_A = CLKST / MPIO_B = AUXIN2 / MPIO_C = AUXIN1
194 };
195
196 i2c_master_bus_config_t bus_conf = {
197 .i2c_port = I2C_NUM_0,
198 .sda_io_num = PCM9211_I2C_SDA,
199 .scl_io_num = PCM9211_I2C_SCL,
200 .clk_source = I2C_CLK_SRC_DEFAULT,
201 .glitch_ignore_cnt = 7,
202 .flags.enable_internal_pullup = true,
203 };
204 i2c_master_bus_handle_t bus_handle = NULL;
205 i2c_master_dev_handle_t dev_handle = NULL;
206 esp_err_t ret = i2c_new_master_bus(&bus_conf, &bus_handle);
207 if (ret == ESP_OK) {
208 i2c_device_config_t dev_conf = {
209 .dev_addr_length = I2C_ADDR_BIT_LEN_7,
210 .device_address = PCM9211_I2C_ADDR,
211 .scl_speed_hz = PCM9211_I2C_FREQ,
212 };
213 ret = i2c_master_bus_add_device(bus_handle, &dev_conf, &dev_handle);
214 }
215 if (ret != ESP_OK) {
216 fprintf(stderr, "PCM9211: I2C init failed: %s\n", esp_err_to_name(ret));
217 } else {
218 for (int i = 0; i < sizeof(pcm9211_regs) / sizeof(pcm9211_regs[0]); i++) {
219 uint8_t buf[2] = { pcm9211_regs[i][0], pcm9211_regs[i][1] };
220 ret = i2c_master_transmit(dev_handle, buf, 2, 100);
221 if (ret != ESP_OK) {
222 fprintf(stderr, "PCM9211: reg 0x%02x write 0x%02x failed: %s\n",
223 buf[0], buf[1], esp_err_to_name(ret));
224 }
225 }
226 }
227
228 // Tear down the I2C driver so Arduino Wire can use bus 0 later
229 if (dev_handle) i2c_master_bus_rm_device(dev_handle);
230 if (bus_handle) i2c_del_master_bus(bus_handle);
231 }
232#endif // AMYBOARD_ARDUINO
233
234 return AMY_OK;
235}

260// Render the second core
261void esp_render_task( void * pvParameters) {
262 while(1) {
263 ulTaskNotifyTake(pdTRUE, portMAX_DELAY); // from esp_render_on_cores
264 amy_render(0, AMY_OSCS/2, 1);
265 // Tell (someone) we're done.
266 xTaskNotifyGive(amy_render_task_done_handle); // to esp_render_on_cores
267 }
268}

270void esp_render_on_cores() {
271 // Call amy_render on all the oscs, using multicore if available.
272 if (amy_global.config.platform.multicore) {
273 // Tell the esp_render_task to inform *us* when it's done.
274 amy_render_task_done_handle = xTaskGetCurrentTaskHandle();
275 // Tell the other core to start rendering.
276 xTaskNotifyGive(amy_render_handle); // to esp_render_task
277 // Render me
278 amy_render(AMY_OSCS/2, AMY_OSCS, 0);
279 // Wait for the other core to finish
280 ulTaskNotifyTake(pdTRUE, portMAX_DELAY); // from esp_render_task
281 } else {
282 // We render everything on this core.
283 amy_render(0, AMY_OSCS, 0);
284 }
285}

296// Set by esp_fill_audio_buffer_task, cleared when returned by amy_render_audio (if used).

300void esp_read_i2s_input() {
301 // Read a block of i2s input. Separated to isolate noise from different i2s formats.
302 size_t read = 0;
303#ifdef I2S_32BIT
304 i2s_channel_read(rx_handle, block32, AMY_BLOCK_SIZE * AMY_NCHANS * sizeof(int32_t), &read, portMAX_DELAY);
305 for (int i = 0; i < AMY_BLOCK_SIZE * AMY_NCHANS; ++i)
306 amy_in_block[i] = (block32[i] >> 16);
307#else
308 i2s_channel_read(rx_handle, amy_in_block, AMY_BLOCK_SIZE * AMY_NCHANS * sizeof(output_sample_type), &read, portMAX_DELAY);
309#endif
310 if(read != AMY_BLOCK_SIZE * AMY_NCHANS * I2S_BYTES_PER_SAMPLE) {
311 fprintf(stderr,"i2s input underrun: %d vs %d\n", read, AMY_BLOCK_SIZE * AMY_NCHANS * I2S_BYTES_PER_SAMPLE);
312 }
313}

315// Make AMY's FABT run forever , as a FreeRTOS task
316// LOCAL EDIT (S3-Amysynth, 2026-06): IDF 6.0 requires void* param for xTaskCreate.
317void esp_fill_audio_buffer_task(void *pvParameters) {
318 (void)pvParameters;
319 while(1) {
320 AMY_PROFILE_START(AMY_ESP_FILL_BUFFER)
321 if(AMY_HAS_I2S && AMY_HAS_AUDIO_IN) {
322 esp_read_i2s_input();
323 }
324 // Get ready to render
325 amy_execute_deltas();
326
327 // Render on whichever cores we have available.
328 esp_render_on_cores();
329
330 // Write to i2s
331 output_sample_type *block = amy_fill_buffer();
332 AMY_PROFILE_STOP(AMY_ESP_FILL_BUFFER)
333
334 last_audio_buffer = block;
335
336 // Notify amy_update() that a block is ready (so it can return from amy_render_audio).
337 if (amy_update_handle)
338 xTaskNotifyGive(amy_update_handle); // to amy_render_audio
339
340 if (AMY_HAS_I2S) {
341 amy_i2s_write((uint8_t *)block, AMY_BLOCK_SIZE * AMY_NCHANS * sizeof(int16_t));
342 } else {
343 // Wait for update sync.
344 ulTaskNotifyTake(pdTRUE, portMAX_DELAY); // from amy_render_audio:!AMY_HAS_I2S
345 }
346 }
347}

349// init AMY from the esp. wraps some amy funcs in a task to do multicore rendering on the ESP32
350void amy_platform_init() {
351 // If we're running amy_update, this should be the task we need to return to.
352 // However, if we're not using amy_update (e.g., Tulip/AMYboard native), we don't
353 // want to do this - the un-handled xTaskNotifyGives cause Tulip to crash when it
354 // turns on WiFi.
355#ifdef ARDUINO
356 amy_update_handle = xTaskGetCurrentTaskHandle();
357#endif
358 if (AMY_HAS_I2S) {
359 // Start i2s
360 esp32_setup_i2s();
361 }
362 if (amy_global.config.platform.multicore) {
363 // On ESP, multicore starts a second thread even if multithread is not requested.
364 // Create the second core rendering task
365 xTaskCreatePinnedToCore(&esp_render_task, AMY_RENDER_TASK_NAME, AMY_RENDER_TASK_STACK_SIZE, NULL, AMY_RENDER_TASK_PRIORITY, &amy_render_handle, AMY_RENDER_TASK_COREID);
366 }
367 if (amy_global.config.platform.multithread) {
368 // Create the fill audio buffer thread, combines, does volume & filters
369 xTaskCreatePinnedToCore(&esp_fill_audio_buffer_task, AMY_FILL_BUFFER_TASK_NAME, AMY_FILL_BUFFER_TASK_STACK_SIZE, NULL, AMY_FILL_BUFFER_TASK_PRIORITY, &amy_fill_buffer_handle, AMY_FILL_BUFFER_TASK_COREID);
370 // Let amy_update know we have I2S covered.
371 if (AMY_HAS_I2S) {
372 amy_global.i2s_is_in_background = 1;
373 }
374 }
375}

405int16_t *amy_render_audio() {
406 // Called by api.amy_update() to render the audio. Not used for non-Arduino.
407 int16_t *buf = NULL;
408 if (amy_global.config.platform.multithread) {
409 // Wait for esp_fill_audio_buffer_task to indicate a buffer is ready.
410 ulTaskNotifyTake(pdTRUE, portMAX_DELAY); // from esp_fill_audio_buffer_task

412 // Allow the FABT to generate another block
413 if (!AMY_HAS_I2S) {
414 xTaskNotifyGive(amy_fill_buffer_handle); // to esp_fill_audio_buffer_task:!AMY_HAS_I2S

416 } else {
417 // No multithread, we have to render here.
418 esp_render_on_cores();
419 buf = amy_fill_buffer();
420 }
421 return buf;
422}

424size_t amy_i2s_write(const uint8_t *buffer, size_t nbytes) {
425 size_t written = 0;
426 int16_t *block = (int16_t *)buffer;
427
428#ifdef I2S_32BIT // including AMYBOARD
429 // Convert to 32 bits
430 for (int i = 0; i < AMY_BLOCK_SIZE * AMY_NCHANS; ++i)
431 block32[i] = ((int32_t)block[i]) << 16;
432 i2s_channel_write(tx_handle, block32, AMY_BLOCK_SIZE * AMY_NCHANS * I2S_BYTES_PER_SAMPLE, &written, portMAX_DELAY);
433#else // 16 bit I2S
434 i2s_channel_write(tx_handle, block, AMY_BLOCK_SIZE * AMY_NCHANS * I2S_BYTES_PER_SAMPLE, &written, portMAX_DELAY);
435#endif
436
437 if(written != AMY_BLOCK_SIZE * AMY_NCHANS * I2S_BYTES_PER_SAMPLE) {
438 fprintf(stderr,"i2s output underrun: %d vs %d\n", written, AMY_BLOCK_SIZE * AMY_NCHANS * I2S_BYTES_PER_SAMPLE);
439 }
440 return 1;
441}
