15struct voice_fifo {
16 const char *name;
17 uint8_t length;
18 uint8_t entries[MAX_VOICES_PER_INSTRUMENT + 1]; // fifo has one extra element to distinguish empty and full.
19 uint8_t head;
20 uint8_t tail;
21};

25void voice_fifo_debug(struct voice_fifo *fifo) {
26 fprintf(stderr, "fifo %s entries 0x%lx len %d head %d tail %d: ", fifo->name, (long)fifo->entries, fifo->length, fifo->head, fifo->tail);
27 uint8_t current = fifo->tail;
28 while(current != fifo->head) {
29 fprintf(stderr, "%d ", fifo->entries[current]);
30 current = (current + 1) % fifo->length;
31 }
32 fprintf(stderr, "\n");
33}

35struct voice_fifo *voice_fifo_init(int size, const char *name) {
36 if (size <=0 || size > MAX_VOICES_PER_INSTRUMENT) {
37 fprintf(stderr, "init_voice_fifo: size %d value error (max size is %d)\n", size, MAX_VOICES_PER_INSTRUMENT);
38 return NULL;
39 }
40 struct voice_fifo *result = (struct voice_fifo *)malloc_caps(sizeof(struct voice_fifo), amy_global.config.ram_caps_synth);
41 result->head = 0;
42 result->tail = 0;
43 result->length = MAX_VOICES_PER_INSTRUMENT + 1; // One more than max size. fixed in this implementation.
44 result->name = name; // very dicey - only works because I'm passing statically-allocated names.
45 return result;
46}

52bool voice_fifo_empty(struct voice_fifo *fifo) {

56void voice_fifo_put(struct voice_fifo *fifo, uint8_t val) {
57 fifo->entries[fifo->head++] = val;
58 if (fifo->head == fifo->length) fifo->head = 0;
59 if (fifo->head == fifo->tail) {
60 fprintf(stderr, "**fifo %s: overflow\n", fifo->name);
61 // Drop the entry we just overwrote.
62 fifo->tail = (fifo->tail + 1) % fifo->length;
63 }
64 //fprintf(stderr, "I: fifo %s: put %d\n", fifo->name, val);
65}
66
67uint8_t voice_fifo_get(struct voice_fifo *fifo) {
68 if (voice_fifo_empty(fifo)) {
69 fprintf(stderr, "**fifo %s: get on empty\n", fifo->name);
70 return 0;
71 }
72 uint8_t val = fifo->entries[fifo->tail++];
73 if (fifo->tail == fifo->length) fifo->tail = 0;
74 //fprintf(stderr, "I: fifo %s: get %d\n", fifo->name, val);
75 return val;
76}

78void voice_fifo_remove(struct voice_fifo *fifo, uint8_t val) {
79 // Remove the oldest instance of val in the fifo and close up.
80 //fprintf(stderr, "I: fifo %s: remove %d\n", fifo->name, val);
81 uint8_t index;
82 for (index = fifo->tail; index != fifo->head; index = (index + 1) % fifo->length) {
83 if (fifo->entries[index] == val) {
84 // Copy entries beyond the value to remove back one position.
85 uint8_t new_head = (fifo->head + fifo->length - 1) % fifo->length;
86 while (index != new_head) {
87 uint8_t next_index = (index + 1) % fifo->length;
88 fifo->entries[index] = fifo->entries[next_index];
89 index = next_index;
90 }
91 fifo->head = new_head;
92 return;
93 }
94 }
95 fprintf(stderr, "**fifo %s: remove did not find %d (#entries=%d)\n", fifo->name, val, (fifo->head - fifo->tail) % fifo->length);
96}

98// How many forgotten note-ons (due to repeated onsets, or note stealing) do we remember
99// (to match their note-offs)
100#define FORGOTTEN_POOL_SIZE 16
101
102struct instrument_info {
103 struct voice_fifo *released_voices;
104 struct voice_fifo *active_voices;
105 uint8_t num_voices;
106 uint8_t oscs_per_voice; // How many oscs each voice uses. Stored for convenience.
107 uint8_t id; // synth number assigned by client.
108 uint8_t bus; // which bus this instrument ends up on.
109 uint16_t patch_number; // What patch this instrument is currently set to. Stored for convenience.
110 int16_t bank_number; // Optional top-7-bit word of Program, set by MIDI CC 0 (-1 if not set).
111 uint32_t flags; // Bitmask of special instrument properties (for MIDI Drums translation).
112 // AMY "voice" index for each of the num_voices allocated voices.
113 uint16_t amy_voices[MAX_VOICES_PER_INSTRUMENT];
114 // Track which note each voice is sounding. We use int16 so we can store PCM_PRESET *127 + midi_note
115 uint16_t note_per_voice[MAX_VOICES_PER_INSTRUMENT];
116 // A buffer of forgotten notes, so we don't complain about their note-offs.
117 uint16_t forgotten_notes[FORGOTTEN_POOL_SIZE]; // We have a max capacity for forgotten notes.
118 uint16_t forgotten_note_count[FORGOTTEN_POOL_SIZE]; // Count mulitple onsets per forgotten note.
119 // Delay added to note-ons. Permits some decay for voice stealing.
120 uint16_t noteon_delay_ms;
121 // Sustain tracking
122 bool in_sustain; // Pedal is down.
123 bool pending_release[MAX_VOICES_PER_INSTRUMENT];
124 bool grab_midi_notes; // Flag to automatically play midi notes.
125};

135 voice_fifo_debug(instrument->released_voices);

144void _instrument_reset_forgotten_pool(struct instrument_info *instrument) {
145 for (int i = 0; i < FORGOTTEN_POOL_SIZE; ++i) {
146 instrument->forgotten_notes[i] = _INSTRUMENT_NO_NOTE;
147 instrument->forgotten_note_count[i] = 0;
148 }

151struct instrument_info *instrument_init(int id, int num_voices, uint16_t* amy_voices, uint16_t patch_number, uint16_t oscs_per_voice, uint8_t bus, uint32_t flags) {
152 struct instrument_info *instrument = (struct instrument_info *)malloc_caps(sizeof(struct instrument_info), amy_global.config.ram_caps_synth);
153 instrument->id = id;
154 if (num_voices <= 0 || num_voices > MAX_VOICES_PER_INSTRUMENT) {
155 fprintf(stderr, "num_voices %d not within 1 .. MAX_VOICES_PER_INSTRUMENT %d\n", num_voices, MAX_VOICES_PER_INSTRUMENT);
156 abort();
157 return NULL;
158 }
159 instrument->num_voices = num_voices;
160 instrument->patch_number = patch_number;
161 instrument->oscs_per_voice = oscs_per_voice;
162 instrument->bank_number = -1;
163 instrument->bus = bus;
164 instrument->flags = flags;
165 instrument->noteon_delay_ms = 0;
166 instrument->in_sustain = false;
167 instrument->grab_midi_notes = true;
168 instrument->released_voices = voice_fifo_init(num_voices, "released");
169 instrument->active_voices = voice_fifo_init(num_voices, "active");
170 for (uint8_t voice = 0; voice < num_voices; ++voice) {
171 instrument->amy_voices[voice] = amy_voices[voice];
172 voice_fifo_put(instrument->released_voices, voice);
173 instrument->note_per_voice[voice] = _INSTRUMENT_NO_NOTE;
174 instrument->pending_release[voice] = false;
175 }
176 _instrument_reset_forgotten_pool(instrument);
177 return instrument;
178}

183 free(instrument);

186void _instrument_push_note_forgotten(struct instrument_info *instrument, uint16_t note) {
187 int available_index = -1;
188 for (int i = 0; i < FORGOTTEN_POOL_SIZE; ++i) {
189 if (instrument->forgotten_notes[i] == note) {
190 ++instrument->forgotten_note_count[i];
191 //fprintf(stderr, "synth %d: caching existing forgotten note %d/%d (%d, count %d)\n",
192 // instrument->id, note / 128, note & 0x7F, i, instrument->forgotten_note_count[i]);
193 return;
194 }
195 if (available_index < 0 && instrument->forgotten_notes[i] == _INSTRUMENT_NO_NOTE) {
196 available_index = i;
197 }
198 }
199 // We didn't find the note, hopefully we saw an empty slot.
200 if (available_index >= 0) {
201 instrument->forgotten_notes[available_index] = note;
202 instrument->forgotten_note_count[available_index] = 1;
203 //fprintf(stderr, "synth %d: caching new forgotten note %d/%d (%d, count %d)\n",
204 // instrument->id, note / 128, note & 0x7F, available_index, instrument->forgotten_note_count[available_index]);
205 } else {
206 fprintf(stderr, "**_instrument_push_forgotten_note: forgotten pool overflow synth %d note %d/%d\n",
207 instrument->id, note / 128, note & 0x7F);
208 }
209}

211bool _instrument_pop_note_forgotten(struct instrument_info *instrument, uint16_t note) {
212 // Checks forgotten_notes for this note; if found, remove it and return true; else false.
213 for (int i = 0; i < FORGOTTEN_POOL_SIZE; ++i) {
214 if (instrument->forgotten_notes[i] == note) {
215 //fprintf(stderr, "synth %d: uncaching forgotten note %d/%d (%d, count %d)\n",
216 // instrument->id, note / 128, note & 0x7F, i, instrument->forgotten_note_count[i]);
217 --instrument->forgotten_note_count[i];
218 if (instrument->forgotten_note_count[i] == 0)
219 instrument->forgotten_notes[i] = _INSTRUMENT_NO_NOTE;
220 return true;
221 }
222 }
223 return false;
224}

226uint16_t _instrument_get_voice(struct instrument_info *instrument, bool *pstolen) {
227 if (!voice_fifo_empty(instrument->released_voices)) {
228 return voice_fifo_get(instrument->released_voices);
229 }
230 // No released voices, have to steal.
231 if (pstolen) *pstolen = true; // *pstolen is set if steal occurs, otherwise not touched.
232 // Mark this note as one that got stolen, so we can absorb its eventual note-off.
233 uint16_t voice = voice_fifo_get(instrument->active_voices);
234 _instrument_push_note_forgotten(instrument, instrument->note_per_voice[voice]);
235 return voice;
236}

238uint16_t _instrument_voice_for_note(struct instrument_info *instrument, uint16_t note) {
239 // Linear search through note_per_voice to see if this note is present.
240 for (uint16_t voice = 0; voice < instrument->num_voices; ++voice)
241 if (instrument->note_per_voice[voice] == note)
242 return voice;
243 return _INSTRUMENT_NO_VOICE;
244}

246uint16_t _instrument_voice_off(struct instrument_info *instrument, uint16_t voice) {
247 instrument->note_per_voice[voice] = _INSTRUMENT_NO_NOTE;
248 instrument->pending_release[voice] = false;
249 voice_fifo_remove(instrument->active_voices, voice);
250 voice_fifo_put(instrument->released_voices, voice);
251 return instrument->amy_voices[voice];
252}

254uint16_t instrument_note_off(struct instrument_info *instrument, uint16_t note) {
255 uint16_t voice = _instrument_voice_for_note(instrument, note);
256 if (voice == _INSTRUMENT_NO_VOICE) {
257 // Don't report an unmatched note-off if it was a victim of stealing.
258 if (!_instrument_pop_note_forgotten(instrument, note))
259 fprintf(stderr, "note off for %d/%d does not match note on\n", note / 128, note & 0x7F);
260 //instrument_debug(instrument);
261 return _INSTRUMENT_NO_VOICE; // We could just fall through, but this is more explicit.
262 }
263 if (instrument->in_sustain) {
264 instrument->pending_release[voice] = true;
265 return _INSTRUMENT_NO_VOICE;
266 }
267 //fprintf(stderr, "instr 0x%lx: patch %d voice %d note %d/%d off\n", (unsigned long)instrument, instrument->patch_number, instrument->amy_voices[voice], note / 128, note & 0x7F);
268 return _instrument_voice_off(instrument, voice);
269}

271int _instrument_all_notes_off(struct instrument_info *instrument, uint16_t *amy_voices) {
272 // Register any active voices as inactive; return those voices.
273 int num_voices_turned_off = 0;
274 for (uint16_t voice = 0; voice < instrument->num_voices; ++voice)
275 if (instrument->note_per_voice[voice] != _INSTRUMENT_NO_NOTE) {
276 //fprintf(stderr, "voice %d note %d/%d all-off\n", instrument->amy_voices[voice], instrument->note_per_voice[voice] / 128, instrument->note_per_voice[voice] & 0x7F);
277 _instrument_voice_off(instrument, voice);
278 *amy_voices++ = instrument->amy_voices[voice];
279 ++num_voices_turned_off;
280 }
281 _instrument_reset_forgotten_pool(instrument);
282 return num_voices_turned_off;
283}

285uint16_t instrument_note_on(struct instrument_info *instrument, uint16_t note, bool *pstolen) {
286 if ((note & 0x7F) == 0) {
287 // note == 0 is for all-notes-off, it's not allowed for note-on (sorry, C-1).
288 fprintf(stderr, "note-on for note 0: ignored.\n");
289 return _INSTRUMENT_NO_VOICE;
290 }
291 uint16_t voice = _instrument_voice_for_note(instrument, note);
292 if (voice == _INSTRUMENT_NO_VOICE) {
293 // Not a re-onset, need to allocate a new voice.
294 voice = _instrument_get_voice(instrument, pstolen);
295 voice_fifo_put(instrument->active_voices, voice);
296 } else {
297 // If re-onset, push the previous onset
298 _instrument_push_note_forgotten(instrument, note);
299 }
300 instrument->note_per_voice[voice] = note;
301 instrument->pending_release[voice] = false;
302 //fprintf(stderr, "instr 0x%lx: patch %d voice %d note %d/%d on\n", (unsigned long)instrument, instrument->patch_number, instrument->amy_voices[voice], note);
303 //instrument_debug(instrument);
304 return instrument->amy_voices[voice];
305}

400uint16_t instrument_voice_for_note_event(int instrument_number, int note, bool is_note_off, bool *pstolen) {
401 // Called from patches_event_has_voices for events including an instrument, velocity, and note (note-on/note-off).
402 // *pstolen is set (if pstolen is nonnull) if the note is stolen, otherwise not touched.
403 if (!instrument_number_exists(instrument_number, "voice_for_event")) return _INSTRUMENT_NO_VOICE;
404 struct instrument_info *instrument = instruments[instrument_number];
405 if (instrument->num_voices == 1) {
406 // If there's only one voice, we can skip the voice management.
407 return instrument->amy_voices[0];
408 }
409 if (is_note_off) {
410 // Note off.
411 if (note == 0) {
412 uint16_t amy_voices[MAX_VOICES_PER_INSTRUMENT];
413 _instrument_all_notes_off(instrument, amy_voices);
414 return _INSTRUMENT_NO_VOICE;
415 }
416 return instrument_note_off(instrument, note);
417 } else {
418 // Note on.
419 return instrument_note_on(instrument, note, pstolen);
420 }
421}

429int instrument_sustain(int instrument_number, bool sustain, uint16_t *amy_voices) {
430 // Will return nonzero voices if the result is to release multiple notes.
431 if (!instrument_number_exists(instrument_number, "sustain")) return 0;
432 struct instrument_info *instrument = instruments[instrument_number];
433 if (sustain) {
434 instrument->in_sustain = true;
435 return 0;
436 }
437 // Sustain pedal released - return multiple note-offs. Like all notes off, but a different criterion.
438 instrument->in_sustain = false;
439 int num_voices_turned_off = 0;
440 for (uint16_t voice = 0; voice < instrument->num_voices; ++voice)
441 if (instrument->pending_release[voice]) {
442 //fprintf(stderr, "voice %d note %d pedal-released\n", instrument->amy_voices[voice], instrument->note_per_voice[voice]);
443 _instrument_voice_off(instrument, voice);
444 *amy_voices++ = instrument->amy_voices[voice];
445 ++num_voices_turned_off;
446 }
447 return num_voices_turned_off;
448}
