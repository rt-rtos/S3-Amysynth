43// choose a preset from the .h file
44void partials_note_on(uint16_t osc) {
45 int num_partials = synth[osc]->preset;
46 for (int i = 0; i < num_partials; ++i) {
47 int o = osc + 1 + i;
48 // On OOM this partial stays silent.
49 if (!ensure_osc_allocd(o, NULL)) continue;
50 // Mark this PARTIAL as part of a build-your own with a flag value in its preset field.
51 // This is used I think only at envelope.c:121 to avoid the normal partial preset special-case for PARTIALs.
52 synth[o]->preset = synth[osc]->preset;
53 synth[o]->logfreq_coefs[COEF_BEND] = 0; // Each PARTIAL will receive pitch bend via the midi_note modulation from the parent osc, don't add it twice.
54 synth[o]->role = SYNTH_IS_ALGO_SOURCE;
55 synth[o]->note_on_clock = amy_global.total_blocks*AMY_BLOCK_SIZE;
56 AMY_UNSET(synth[o]->note_off_clock);
57 msynth[o]->logfreq = synth[o]->logfreq_coefs[COEF_CONST] + msynth[osc]->logfreq;
58 partial_note_on(o);
59 }
60 // Squirrel away num_oscs
61 synth[osc]->last_two[0] = synth[osc]->preset;
62}

64void partials_note_off(uint16_t osc) {
65 int num_oscs = synth[osc]->preset;
66 for(uint16_t i = osc + 1; i < osc + 1 + num_oscs; i++) {
67 uint16_t o = i % AMY_OSCS;
68 // The partial may have failed to alloc at note-on.
69 if (synth[o] == NULL) continue;

72 }

75static inline float p_combine_controls(float *controls, float *coefs) {
76 float result = 0;
77 for (int i = 0; i <= COEF_EG0; ++i)
78 result += coefs[i] * controls[i];
79 return result;
80}

237void _cumulate_scaled_harmonic_params(float *harm_param, int harmonic_index, float alpha, const interp_partials_voice_t *partials_voice) {
238 int num_bps = partials_voice->num_sample_times_ms;
239 // Pitch
240 harm_param[0] += alpha * partials_voice->harmonics_freq[harmonic_index];
241 // Envelope magnitudes
242 for (int i = 0; i < num_bps; ++i)
243 harm_param[1 + i] += alpha * partials_voice->harmonics_mags[harmonic_index * num_bps + i];
244}

246int _harmonic_base_index_for_pitch_vel(int pitch_index, int vel_index, const interp_partials_voice_t *partials_voice) {
247 int note_number = partials_voice->num_velocities * pitch_index + vel_index;
248 int harmonic_index = 0;
249 for (int i = 0; i < note_number; ++i)
250 harmonic_index += partials_voice->num_harmonics[i];
251 return harmonic_index;
252}

254float _logfreq_of_midi_cents(float midi_cents) {
255 // Frequency is already log scaled, but need to re-center and change from 1200/oct to 1.0/oct.
256 return (midi_cents - (100 * ZERO_MIDI_NOTE)) / 1200.f;
257}

259float _env_lin_of_db(float db) {
260 float lin = powf(10.f, MIN(20.f, (db - 100.f)) / 20.f) - 0.001f;
261 if (lin < 0) return 0;
262 return lin;
263}

265void _osc_on_with_harm_param(uint16_t o, float *harm_param, const interp_partials_voice_t *partials_voice) {
266 // We coerce this voice into being a partial, regardless of user wishes.
267 synth[o]->wave = PARTIAL;
268 synth[o]->preset = 1; // Flag that this is an envelope-based partial
269 // Setup the specified frequency.
270 synth[o]->logfreq_coefs[COEF_CONST] = _logfreq_of_midi_cents(harm_param[0]);
271 // Setup envelope.
272 //synth[o]->eg_type[0] = ENVELOPE_DB;
273 synth[o]->breakpoint_times[0][0] = 0;
274 synth[o]->breakpoint_values[0][0] = 0;
275 int last_time = 0;
276 for (int bp = 0; bp < partials_voice->num_sample_times_ms; ++bp) {
277 synth[o]->breakpoint_times[0][bp + 1] = (partials_voice->sample_times_ms[bp] - last_time) * AMY_SAMPLE_RATE / 1000;
278 synth[o]->breakpoint_values[0][bp + 1] = _env_lin_of_db(harm_param[bp + 1]);
279 last_time = partials_voice->sample_times_ms[bp];
280 }
281 // Final release
282 synth[o]->breakpoint_times[0][partials_voice->num_sample_times_ms + 1] = 200 * AMY_SAMPLE_RATE / 1000;
283 synth[o]->breakpoint_values[0][partials_voice->num_sample_times_ms + 1] = 0;
284 // Decouple osc freq and amp from note and amp.
285 synth[o]->logfreq_coefs[COEF_NOTE] = 0;
286 synth[o]->amp_coefs[COEF_VEL] = 1.0; // velocity is modified on-the-fly by the control osc to vary global amplitude.
287 // Other osc params.
288 synth[o]->role = SYNTH_IS_ALGO_SOURCE;
289 synth[o]->note_on_clock = amy_global.total_blocks*AMY_BLOCK_SIZE;
290 AMY_UNSET(synth[o]->note_off_clock);
291 partial_note_on(o);
292}

294// HOW DOES INTERP_PARTIALS (e.g. DPWE_PIANO) WORK?
295// The special thing about interp_partials is that the harmonic envelopes depend on the note velocity,
296// so they all have to be recomputed in response at note_on time. Then, because their values have been
297// determined to reflect velocity via the note_on calculation, the parent osc should not use velocity
298// as part of its overall scaling calculation, since it would otherwise be applied twice.
299// Thus, when setting up a control osc for piano, we set amp_coef[COEF_VELOCITY] = 0.
300
301void interp_partials_note_on(uint16_t osc) {
302 // Choose the interp_partials preset.
303 const interp_partials_voice_t *partials_voice = &interp_partials_map[synth[osc]->preset % NUM_INTERP_PARTIALS_PRESETS];
304 float midi_note = synth[osc]->midi_note;
305 float midi_vel = (int)roundf(synth[osc]->velocity * 127.f);
306 // Clip velocity to the range covered by the tables. Pitch is deliberately not clipped:
307 // notes outside the table range are linearly extrapolated from the edge rows (pitch_alpha
308 // outside [0, 1]); the index search below is bounded so table reads stay in range.
309 if (midi_vel < partials_voice->velocities[0]) midi_vel = partials_voice->velocities[0];
310 if (midi_vel > partials_voice->velocities[partials_voice->num_velocities - 1]) midi_vel = partials_voice->velocities[partials_voice->num_velocities - 1];
311 // Find the lower bound pitch/velocity indices.
312 uint8_t pitch_index = 0, vel_index = 0;
313 while(pitch_index < partials_voice->num_pitches - 2 // We're going to inspect pitch_index + 1, so make sure that's in the table.
314 && partials_voice->pitches[pitch_index + 1] < midi_note)
315 ++pitch_index;
316 while(vel_index < partials_voice->num_velocities - 1
317 && partials_voice->velocities[vel_index + 1] < midi_vel)
318 ++vel_index;
319 // Interp weights
320 float pitch_alpha = (midi_note - partials_voice->pitches[pitch_index])
321 / (float)(partials_voice->pitches[pitch_index + 1] - partials_voice->pitches[pitch_index]);
322 float vel_alpha = (midi_vel - partials_voice->velocities[vel_index])
323 / (float)(partials_voice->velocities[vel_index + 1] - partials_voice->velocities[vel_index]);
324 float harm_param[MAX_NUM_MAGNITUDES + 1]; // frequency + harmonic magnitudes.
325 int note_number = partials_voice->num_velocities * pitch_index + vel_index;
326 // Find the least number of harmonics across everything we're interpolating.
327 int num_harmonics = MIN(MAX_NUM_HARMONICS, partials_voice->num_harmonics[note_number]); // pl_vl note
328 num_harmonics = MIN(num_harmonics, partials_voice->num_harmonics[note_number + 1]); // pl_vh note
329 num_harmonics = MIN(num_harmonics, partials_voice->num_harmonics[note_number + partials_voice->num_velocities]); // ph_vl note
330 num_harmonics = MIN(num_harmonics, partials_voice->num_harmonics[note_number + partials_voice->num_velocities + 1]); // ph_vh note
331 // Interpolate the 4 notes.
332 int harmonic_base_index_pl_vl =
333 _harmonic_base_index_for_pitch_vel(pitch_index, vel_index, partials_voice);
334 float alpha_pl_vl = (1.f - pitch_alpha) * (1.f - vel_alpha);
335 int harmonic_base_index_pl_vh =
336 _harmonic_base_index_for_pitch_vel(pitch_index, vel_index + 1, partials_voice);
337 float alpha_pl_vh = (1.f - pitch_alpha) * (vel_alpha);
338 int harmonic_base_index_ph_vl =
339 _harmonic_base_index_for_pitch_vel(pitch_index + 1, vel_index, partials_voice);
340 float alpha_ph_vl = (pitch_alpha) * (1.f - vel_alpha);
341 int harmonic_base_index_ph_vh =
342 _harmonic_base_index_for_pitch_vel(pitch_index + 1, vel_index + 1, partials_voice);
343 float alpha_ph_vh = (pitch_alpha) * (vel_alpha);
344 //fprintf(stderr, "interp_partials@%u: osc %d note %.1f vel %.1f pitch_x %d vel_x %d numh %d harm_bi_ll %d pitch_a %.3f vel_a %.3f alphas %.2f %.2f %.2f %.2f\n",
345 // amy_global.total_blocks*AMY_BLOCK_SIZE, osc, midi_note, midi_vel, pitch_index, vel_index, num_harmonics,
346 // harmonic_base_index_pl_vl, pitch_alpha, vel_alpha,
347 // alpha_pl_vl, alpha_pl_vh, alpha_ph_vl, alpha_ph_vh);
348 // Make sure enough oscs are alloc'd in our dynamic osc alloc world.
349 // This has to be enough for any note in this map. Assume num_harmonics[0] is largest (lowest pitch).
350 uint8_t max_num_partials = _max_partials_for_partials_voice(partials_voice);
351 uint8_t max_num_breakpoints[MAX_BREAKPOINT_SETS] = {2 + partials_voice->num_sample_times_ms, DEFAULT_NUM_BREAKPOINTS};
352 for (int o = 0; o < max_num_partials; ++o) {
353 // On OOM drop the note rather than render under-sized partials.
354 if (!ensure_osc_allocd(osc + 1 + o, max_num_breakpoints)) return;
355 }
356 int partial_osc = osc;
357 for (int h = 0; h < num_harmonics; ++h) {
358 if (use_this_partial_map[h]) {
359 for (int i = 0; i < MAX_NUM_MAGNITUDES + 1; ++i) harm_param[i] = 0;
360 _cumulate_scaled_harmonic_params(harm_param, harmonic_base_index_pl_vl + h,
361 alpha_pl_vl, partials_voice);
362 _cumulate_scaled_harmonic_params(harm_param, harmonic_base_index_pl_vh + h,
363 alpha_pl_vh, partials_voice);
364 _cumulate_scaled_harmonic_params(harm_param, harmonic_base_index_ph_vl + h,
365 alpha_ph_vl, partials_voice);
366 _cumulate_scaled_harmonic_params(harm_param, harmonic_base_index_ph_vh + h,
367 alpha_ph_vh, partials_voice);
368 //fprintf(stderr, "harm %d freq %.2f bps %.3f %.3f %.3f %.3f\n", h, harm_param[0], harm_param[1], harm_param[2], harm_param[3], harm_param[4]);
369 ++partial_osc;
370 _osc_on_with_harm_param(partial_osc, harm_param, partials_voice);
371 }
372 }
373 // Squirrel away num_oscs
374 synth[osc]->last_two[0] = partial_osc - osc;
375 // Make sure any remaining oscs are still marked as ALGO_SOURCE
376 while(partial_osc < osc + 1 + max_num_partials) { synth[partial_osc]->role = SYNTH_IS_ALGO_SOURCE; synth[partial_osc]->status = SYNTH_OFF; ++partial_osc; }
377}
