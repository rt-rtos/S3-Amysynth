1/* MIDI on Linux, over ALSA raw MIDI.
2 *
3 * The host layer for __linux__, the way macos_midi.m is the one for
4 * macOS: run_midi / stop_midi / midi_out plus the port chooser
5 * amy_midi.h declares. Started from A. Doinel's contribution (#563),
6 * which is where the virtual-port arrangement below comes from.
7 *
8 * WHAT -1 MEANS IS NOT THE SAME ON THE TWO DESKTOPS, and it cannot be.
9 * On macOS every source and every destination can be held open at once,
10 * so -1 is "all of them". ALSA raw MIDI is one handle per device — there
11 * is no listening to everything — so here -1 is the VIRTUAL PORT: amy
12 * appears in the MIDI graph under its own name and anything is free to
13 * connect to it with aconnect or a patchbay. That is the Linux idiom
14 * for "no particular device", it is what a soft synth normally does,
15 * and it is what #563 did before there was a choice at all. Picking a
16 * device instead opens that hardware port directly.
17 *
18 * THE READ THREAD OWNS THE HANDLES. A selection made from the host
19 * thread only sets what is WANTED and raises a flag; the thread notices
20 * within one poll timeout, closes what it has and opens what was asked
21 * for. Nothing else may touch the handles — snd_rawmidi_read blocks in
22 * one thread while snd_rawmidi_close runs in another is exactly the
23 * shape that crashes rarely and unreproducibly. midi_out is the one
24 * crossing and takes the lock.
25 */
26/* AMY_ALSA IS THE SWITCH, AND THE WHOLE FILE IS BEHIND IT — the same
27 * shape libminiaudio-audio.c and macos_midi.m already have, and for a
28 * blunter reason than portability taste: the Arduino builds compile
29 * every .c in src/ whatever the target is, so without this the Pico and
30 * the Teensy try to include <poll.h> and stop. emscripten defines
31 * __linux__ too, which is why the platform macro is not the guard.
32 *
33 * The Makefile sets it only when libasound is actually present, so a
34 * Linux box without the dev package builds amy exactly as it always
35 * did, minus MIDI — a synth library may not make itself unbuildable
36 * over an optional device layer. */

51/* How long the reader waits on poll() before looking at its flags. Long
52 * enough to cost nothing, short enough that a menu click and a quit both
53 * feel immediate. #563 slept 500us in a tight read loop; polling on the
54 * descriptors ALSA hands out does the same job without spinning. */
55#define POLL_MS 100

81/* ---- what devices are there ---------------------------------------
82 *
83 * Walked in one fixed order — card, then device, then subdevice — so an
84 * index means the same thing from one call to the next, which is the
85 * whole contract of the count/name pair. */
86
87typedef bool (*port_fn)(int card, int dev, int sub, const char *name, void *ctx);
88
89static void each_port(uint8_t dir, port_fn fn, void *ctx)
90{
91 snd_rawmidi_stream_t stream = (dir == AMY_MIDI_PORT_OUT)
92 ? SND_RAWMIDI_STREAM_OUTPUT
93 : SND_RAWMIDI_STREAM_INPUT;
94 int card = -1;
95 while (snd_card_next(&card) >= 0 && card >= 0) {
96 char ctlname[DEV_MAX];
97 snprintf(ctlname, sizeof(ctlname), "hw:%d", card);
98 snd_ctl_t *ctl;
99 if (snd_ctl_open(&ctl, ctlname, 0) < 0) continue;
100
101 char *cardname = NULL;
102 if (snd_card_get_name(card, &cardname) < 0) cardname = NULL;
103
104 int dev = -1;
105 while (snd_ctl_rawmidi_next_device(ctl, &dev) >= 0 && dev >= 0) {
106 snd_rawmidi_info_t *info;
107 snd_rawmidi_info_alloca(&info);
108 snd_rawmidi_info_set_device(info, dev);
109 snd_rawmidi_info_set_stream(info, stream);
110 snd_rawmidi_info_set_subdevice(info, 0);
111 /* A device that has no stream in this direction is not an
112 * error, it is an input-only or output-only interface. */
113 if (snd_ctl_rawmidi_info(ctl, info) < 0) continue;
114
115 unsigned int subs = snd_rawmidi_info_get_subdevices_count(info);
116 for (unsigned int sub = 0; sub < subs; sub++) {
117 snd_rawmidi_info_set_subdevice(info, sub);
118 if (snd_ctl_rawmidi_info(ctl, info) < 0) continue;
119 const char *devname = snd_rawmidi_info_get_name(info);
120 const char *subname = snd_rawmidi_info_get_subdevice_name(info);
121
122 char name[NAME_MAX];
123 /* The card is what somebody recognises ("UMC404HD"), the
124 * device is what it does ("MIDI 1"), and the subdevice
125 * only earns its room when there is more than one. */
126 if (subs > 1 && subname && subname[0])
127 snprintf(name, sizeof(name), "%s: %s (%s)",
128 cardname ? cardname : ctlname,
129 devname ? devname : "MIDI", subname);
130 else
131 snprintf(name, sizeof(name), "%s: %s",
132 cardname ? cardname : ctlname,
133 devname ? devname : "MIDI");
134
135 if (!fn(card, dev, (int)sub, name, ctx)) {
136 if (cardname) free(cardname);
137 snd_ctl_close(ctl);
138 return;
139 }
140 }
141 }
142 if (cardname) free(cardname);
143 snd_ctl_close(ctl);
144 }
145}

148static bool count_one(int card, int dev, int sub, const char *name, void *ctx)
149{
150 (void)card; (void)dev; (void)sub; (void)name;

155struct finder {
156 uint32_t want; /* by index... */
157 const char *want_name; /* ...or by name, whichever is set */
158 uint32_t seen;
159 bool found;
160 char dev[DEV_MAX];
161 char name[NAME_MAX];
162 uint32_t index;
163};
164
165static bool find_one(int card, int dev, int sub, const char *name, void *ctx)
166{
167 struct finder *f = (struct finder *)ctx;
168 bool hit = f->want_name ? (strcmp(name, f->want_name) == 0)
169 : (f->seen == f->want);
170 if (hit) {
171 snprintf(f->dev, sizeof(f->dev), "hw:%d,%d,%d", card, dev, sub);
172 snprintf(f->name, sizeof(f->name), "%s", name);
173 f->index = f->seen;
174 f->found = true;
175 return false; /* stop walking */
176 }
177 f->seen++;
178 return true;
179}

190uint32_t amy_midi_port_name(uint8_t dir, uint32_t index, char *buf, uint32_t buflen)
191{
192 if (buflen == 0) return 0;
193 buf[0] = 0;
194 if (dir > AMY_MIDI_PORT_OUT) return 0;
195 struct finder f = {0};
196 f.want = index;
197 each_port(dir, find_one, &f);
198 if (!f.found) return 0;
199 snprintf(buf, buflen, "%s", f.name);
200 return (uint32_t)strlen(buf);
201}

203int32_t amy_midi_port_selected(uint8_t dir)
204{
205 if (dir > AMY_MIDI_PORT_OUT) return -1;
206 pthread_mutex_lock(&midi_lock);
207 bool none = want_name[dir][0] == 0;
208 struct finder f = {0};
209 char name[NAME_MAX];
210 snprintf(name, sizeof(name), "%s", want_name[dir]);
211 pthread_mutex_unlock(&midi_lock);
212 if (none) return -1;
213 f.want_name = name;
214 each_port(dir, find_one, &f);
215 /* Chosen, and not plugged in right now. Deliberately not -1: the
216 * choice stands and comes back with the cable. */
217 return f.found ? (int32_t)f.index : -2;
218}

220void amy_midi_port_select(uint8_t dir, int32_t index)
221{
222 if (dir > AMY_MIDI_PORT_OUT) return;
223 char dev[DEV_MAX] = {0};
224 char name[NAME_MAX] = {0};
225 if (index >= 0) {
226 struct finder f = {0};
227 f.want = (uint32_t)index;
228 each_port(dir, find_one, &f);
229 /* An index past the end is "no particular device" rather than an
230 * error: the list a caller picked from is a snapshot, and
231 * unplugging something between the two is nobody's bug. */
232 if (f.found) {
233 snprintf(dev, sizeof(dev), "%s", f.dev);
234 snprintf(name, sizeof(name), "%s", f.name);
235 }
236 }
237 pthread_mutex_lock(&midi_lock);
238 memcpy(want_dev[dir], dev, sizeof(dev));
239 memcpy(want_name[dir], name, sizeof(name));
240 reopen_wanted = true;
241 pthread_mutex_unlock(&midi_lock);
242}

245/* ---- opening what was asked for ------------------------------------ */
246
247/* Where a remembered port is NOW, which is not where it was.
248 * "hw:1,0,0" is a position in a list, not an identity — plug a second
249 * interface in and yesterday's string is somebody else's device. So the
250 * stored string is only a HINT: it is used when the name at that
251 * position still matches, and otherwise the name is looked for
252 * elsewhere. Same guarantee macOS gets from CoreMIDI's unique ids,
253 * which ALSA has no equivalent of. */
254static bool resolve_port(uint8_t dir, char *out, size_t outlen)
255{
256 if (want_name[dir][0] == 0) return false; /* the virtual port */
257 struct finder f = {0};
258 f.want_name = want_name[dir];
259 each_port(dir, find_one, &f);
260 if (!f.found) return false; /* gone: fall back */
261 snprintf(out, outlen, "%s", f.dev);
262 return true;
263}

265/* Called on the read thread ONLY, with the lock held. */
266static void reopen_ports(void)
267{
268 if (midi_rx_port) { snd_rawmidi_close(midi_rx_port); midi_rx_port = NULL; }
269 if (midi_tx_port) { snd_rawmidi_close(midi_tx_port); midi_tx_port = NULL; }
270
271 char in_dev[DEV_MAX], out_dev[DEV_MAX];
272 bool have_in = resolve_port(AMY_MIDI_PORT_IN, in_dev, sizeof(in_dev));
273 bool have_out = resolve_port(AMY_MIDI_PORT_OUT, out_dev, sizeof(out_dev));
274
275 int err;
276 if (!have_in && !have_out) {
277 /* BOTH DEFAULT: one duplex virtual port, which is exactly the
278 * call #563 shipped and the only arrangement anybody has run.
279 * Kept as its own case rather than opening two halves, because
280 * the known-working path should stay the common one. */
281 err = snd_rawmidi_open(&midi_rx_port, &midi_tx_port, "virtual",
282 SND_RAWMIDI_NONBLOCK);
283 if (err < 0) {
284 /* NOT exit(1). This is a library inside somebody's
285 * application: no MIDI is a machine that still makes sound,
286 * and taking the process down over it is not amy's call. */
287 fprintf(stderr, "amy: could not open the virtual MIDI port: %s\n",
288 snd_strerror(err));
289 midi_rx_port = midi_tx_port = NULL;
290 }
291 return;
292 }
293
294 err = snd_rawmidi_open(&midi_rx_port, NULL, have_in ? in_dev : "virtual",
295 SND_RAWMIDI_NONBLOCK);
296 if (err < 0) {
297 fprintf(stderr, "amy: could not open MIDI in (%s): %s\n",
298 have_in ? in_dev : "virtual", snd_strerror(err));
299 midi_rx_port = NULL;
300 }
301 err = snd_rawmidi_open(NULL, &midi_tx_port, have_out ? out_dev : "virtual", 0);
302 if (err < 0) {
303 fprintf(stderr, "amy: could not open MIDI out (%s): %s\n",
304 have_out ? out_dev : "virtual", snd_strerror(err));
305 midi_tx_port = NULL;
306 }
307}

310void midi_out(uint8_t *bytes, uint16_t len)
311{
312 /* EVERY midi_out must call this (amy_midi.h says so, and macOS
313 * does): it is how amy_config's external output hook is reached, so
314 * an embedding host that mirrors outgoing MIDI hears nothing without
315 * it. #563 left it out. */
316 midi_out_external_hook(bytes, len);
317 pthread_mutex_lock(&midi_lock);
318 /* A write to a port that was never opened, or has just been closed
319 * under a selection change, is a crash rather than a dropped
320 * message — the same guard macOS keeps on out_port. */
321 if (midi_tx_port) snd_rawmidi_write(midi_tx_port, bytes, len);
322 pthread_mutex_unlock(&midi_lock);
323}

326void *run_midi_linux(void *argp)
327{
328 (void)argp;
329 uint8_t data[MAX_MIDI_BYTES_TO_PARSE];
330
331 while (!midi_linux_should_exit) {
332 pthread_mutex_lock(&midi_lock);
333 if (reopen_wanted) {
334 reopen_wanted = false;
335 reopen_ports();
336 }
337 snd_rawmidi_t *rx = midi_rx_port;
338 pthread_mutex_unlock(&midi_lock);
339
340 if (rx == NULL) {
341 /* Nothing open — usually a device that would not open. Wait
342 * out a poll interval so a later selection still gets in. */
343 usleep(POLL_MS * 1000);
344 continue;
345 }
346
347 int nfds = snd_rawmidi_poll_descriptors_count(rx);
348 if (nfds <= 0) { usleep(POLL_MS * 1000); continue; }
349 struct pollfd pfds[16];
350 if (nfds > (int)(sizeof(pfds) / sizeof(pfds[0])))
351 nfds = (int)(sizeof(pfds) / sizeof(pfds[0]));
352 snd_rawmidi_poll_descriptors(rx, pfds, (unsigned int)nfds);
353
354 /* The TIMEOUT is what makes a selection change and a quit take
355 * effect: a blocking read would sit here until the next note
356 * arrived, which on an idle keyboard is never. */
357 if (poll(pfds, (nfds_t)nfds, POLL_MS) <= 0) continue;
358
359 unsigned short revents = 0;
360 if (snd_rawmidi_poll_descriptors_revents(rx, pfds, (unsigned int)nfds,
361 &revents) < 0) continue;
362 if (!(revents & POLLIN)) continue;
363
364 pthread_mutex_lock(&midi_lock);
365 /* Re-check: the handle may have been replaced between the poll
366 * and here by a selection made on the host thread. */
367 ssize_t len = midi_rx_port ? snd_rawmidi_read(midi_rx_port, data, sizeof(data)) : -1;
368 pthread_mutex_unlock(&midi_lock);
369
370 if (len > 0) convert_midi_bytes_to_messages(data, (size_t)len, 0);
371 }
372
373 pthread_mutex_lock(&midi_lock);
374 if (midi_rx_port) { snd_rawmidi_close(midi_rx_port); midi_rx_port = NULL; }
375 if (midi_tx_port) { snd_rawmidi_close(midi_tx_port); midi_tx_port = NULL; }
376 pthread_mutex_unlock(&midi_lock);
377 return NULL;
378}
