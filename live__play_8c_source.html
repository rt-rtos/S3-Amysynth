43 * Defaults land on first ensure_ready so amp_trim starts at unity, not 0. */

47/* Out-of-the-box editor values, voiced for held keys rather than a running
48 * sequence: high sustain, controlled release. Everything stays unauthored
49 * until the user commits, except the EG0 shape that live_apply_authored()
50 * force-pushes for wave patches - a raw wave with no envelope reads AMY's
51 * empty breakpoint set as a permanently open gate and rings forever after
52 * note-off (the melodic force_wave rule). */
53static void live_seed_defaults(void)
54{
55 voice_params_init_defaults(&s_vp);
56 s_vp.env.attack_ms = 4; /* tiny ramp kills the note-on click */
57 s_vp.env.decay_ms = 250;
58 s_vp.env.sustain_pct = 75; /* held keys keep their body */
59 s_vp.env.release_ms = 250;
60 s_vp.env.eg_type = 0; /* ENVELOPE_NORMAL */
61 /* EG1 (filter sweep once a filter is authored): the arp's slower-tail pair. */
62 s_vp.env1.attack_ms = 15;
63 s_vp.env1.decay_ms = 400;
64 s_vp.env1.sustain_pct = 20;
65 s_vp.env1.release_ms = 300;
66 s_vp.env1.eg_type = 0;
67 /* Filter/LFO: bypass until committed, but seeded so the editors open on
68 * something musical instead of zeros. */
69 s_vp.filter.filter_type = 0; /* SEQ_FILTER_NONE */
70 s_vp.filter.cutoff_hz = 800.0f;
71 s_vp.filter.resonance = 1.0f;
72 s_vp.lfo.wave = LFO_WAVE_SINE;
73 s_vp.lfo.rate = LFO_RATE_1BAR;
74 s_vp.lfo.depth = 50;
75 s_vp.lfo.targets = LFO_TGT_BIT(LFO_TARGET_FILTER);
76}

78/* The editors can open before ensure_ready() installs the defaults, and a
79 * zero-initialised block has amp_trim 0.0 - which the graph editor reads as 0%
80 * and commits as silence. Every accessor goes through this. UI task only, so
81 * no guard needed; live_note() deliberately skips it, running on the transport
82 * task where ensure_ready() always precedes any note. */
83static voice_params_t *live_vp(void)
84{

86 live_seed_defaults();

92/* Held-note bitmap so all_notes_off releases exactly what is sounding. Written
93 * from the transport task, drained from synth_ui / NimBLE host on
94 * stop/disconnect; per-bit races are harmless - a spurious note-off to an idle
95 * voice is a no-op in AMY. */

98static void live_note(uint8_t note, float velocity)
99{
100 amy_event *e = amy_helpers_event_begin();
101 e->synth = LIVE_SYNTH;
102 e->midi_note = (float)note;
103 /* Amp trim rides the note velocity (the graph editor's amp mode). Note-offs
104 * pass velocity 0 and must stay 0. */
105 e->velocity = (velocity > 0.0f) ? velocity * s_vp.amp_trim : 0.0f;
106 amy_helpers_event_send(e);
107}

109/* Push the stored filter including the bipolar EG1->cutoff sweep depth.
110 * Mirrors arp_apply_filter() minus its KS branch (the live slot is always a
111 * patch voice). EG1 breakpoints go out alongside a nonzero sweep so
112 * filter_freq_coefs[COEF_EG1] never reads AMY's never-configured
113 * always-open unity gate. */
114static void live_apply_filter(const seq_filter_t *f)
115{
116 if (!f) return;
117 amy_event *e = amy_helpers_event_begin();
118 e->synth = LIVE_SYNTH;
119 if (f->enabled) {
120 e->filter_type = f->filter_type;
121 e->filter_freq_coefs[COEF_CONST] = f->cutoff_hz;
122 e->resonance = f->resonance;
123 if (f->filter_env_amount != 0.0f) {
124 e->filter_freq_coefs[COEF_EG1] = f->filter_env_amount;
125 }
126 } else {
127 e->filter_type = FILTER_NONE;
128 }
129 amy_helpers_event_send(e);
130
131 if (f->enabled && f->filter_env_amount != 0.0f) {
132 sequencer_core_push_envelope_eg1(LIVE_SYNTH, 0, &s_vp.env1);
133 }
134}

146/* Apply or park the native LFO on any patch with a reserved carrier pair
147 * (sequencer_core_lfo_native_layout); patch strings own their whole osc layout
148 * and get the software stepper below instead. Parking a never-authored voice
149 * is a coupled-osc-only clear - the carrier pair stays unmaterialized
150 * (voice_config.h). */
151static void live_apply_lfo(void)
152{
153 uint8_t carrier, coupled;
154 if (!sequencer_core_lfo_native_layout(s_patch, &carrier, &coupled)) return;
155 bool on = s_vp.lfo_authored && s_vp.lfo.enabled;
156 voice_apply_native_lfo_topo(LIVE_SYNTH, on ? &s_vp.lfo : NULL,
157 sequencer_core_get_bpm(), carrier, coupled);
158}

160/* Re-push whatever the user has authored. Called after every slot configure,
161 * since a patch load rebuilds the voice and drops our overrides. */
162static void live_apply_authored(void)
163{
164 bool wave = sequencer_core_is_wave_patch(s_patch);
165 /* Wave patches carry no envelope of their own: with nothing pushed a
166 * note-off never releases, so force the default shape even unauthored.
167 * Patch strings keep their built-in envelope until the user commits. */
168 if (s_vp.env_authored || wave)
169 sequencer_core_push_envelope(LIVE_SYNTH, &s_vp.env);
170 if (s_vp.env1_authored) sequencer_core_push_envelope_eg1(LIVE_SYNTH, 0, &s_vp.env1);
171 if (s_vp.filter_authored) live_apply_filter(&s_vp.filter);
172 if (s_vp.dist_authored) voice_apply_dist(LIVE_SYNTH, &s_vp.dist);
173 live_apply_lfo(); /* internal layout guard: no-op for patch strings */
174 /* Patch loads reset per-osc portamento_alpha; reassert unconditionally
175 * (0 is a valid "off" reassert) - the arp_rebuild discipline. */
176 live_push_glide();
177}

193void live_play_note_on(uint8_t channel, uint8_t note, uint8_t velocity)
194{
195 (void)channel;
196 if (!s_ready || note > 127) return;
197 s_held[note >> 5] |= 1u << (note & 31u);
198 live_note(note, (float)velocity * (1.0f / 127.0f));
199}

201void live_play_note_off(uint8_t channel, uint8_t note)
202{
203 (void)channel;
204 if (!s_ready || note > 127) return;
205 s_held[note >> 5] &= ~(1u << (note & 31u));
206 live_note(note, 0.0f);
207}

209void live_play_all_notes_off(void)
210{
211 if (!s_ready) return;
212 for (uint8_t w = 0; w < 4; w++) {
213 uint32_t bits = s_held[w];
214 s_held[w] = 0;
215 while (bits) {
216 uint8_t bit = (uint8_t)__builtin_ctz(bits);
217 bits &= bits - 1u;
218 live_note((uint8_t)(w * 32u + bit), 0.0f);
219 }
220 }

285/* ── PATCH-mode software LFO fallback ────────────────────────────────────
286 * Wave patches get the AMY-native voice-local LFO (live_apply_lfo). A patch
287 * string owns its whole osc layout, so it runs the same 20 Hz software stepper
288 * as non-wave melodic tracks and the arp's PATCH source (mirrors
289 * arp_swlfo_service), modulating each checked target's COEF_CONST rail.
290 * WOBBLE and SCAN have no software analog and are ignored. */
291static float s_swlfo_phase = 0.0f;
292static float s_swlfo_rnd = 0.0f;
293static bool s_swlfo_active = false;
294static uint8_t s_swlfo_targets = 0; /* rails driven while active - the set
295 * to restore, even if edited since */
296
297/* seq_core_editors.c internals shared with this stepper. Mirrored prototypes:
298 * seq_core_internal.h cannot be included here (it defines a TU-local TAG). */
299float lfo_next_rand(void);
300void lfo_push_target_neutral(uint8_t synth_id, lfo_target_t target);
301
302static inline float live_swlfo_hz(lfo_rate_t rate, uint16_t bpm)
303{
304 float hz = lfo_rate_to_hz(rate, bpm);
305 return (hz > SEQ_LFO_SW_MAX_HZ) ? SEQ_LFO_SW_MAX_HZ : hz;
306}

308static float live_swlfo_eval(lfo_wave_t wave, float ph)
309{
310 switch (wave) {
311 case LFO_WAVE_SINE: return sinf(2.0f * 3.14159265f * ph);
312 case LFO_WAVE_TRIANGLE: return (ph < 0.5f) ? (4.0f * ph - 1.0f)
313 : (3.0f - 4.0f * ph);
314 case LFO_WAVE_SAW_UP: return 2.0f * ph - 1.0f;
315 case LFO_WAVE_SAW_DOWN: return 1.0f - 2.0f * ph;
316 case LFO_WAVE_SQUARE: return (ph < 0.5f) ? 1.0f : -1.0f;
317 case LFO_WAVE_RANDOM: return s_swlfo_rnd;
318 default: return 0.0f;
319 }
320}

322void live_play_lfo_service(void)
323{
324 const seq_lfo_t *lfo = &s_vp.lfo;
325 /* DIST has COEF_MOD rails now (dist_*_coefs), so on native-eligible patches
326 * it rides the carrier like every other target and the stepper is fully off
327 * there (mirrors the arp/sequencer steppers). */
328 bool native = live_play_lfo_native_eligible();
329 bool want = s_ready && s_vp.lfo_authored && lfo->enabled &&
330 lfo->targets != 0 && !native;
331
332 if (!want) {
333 if (s_swlfo_active) {
334 /* Restore every rail the stepper drove (melodic
335 * lfo_restore_target_neutrals rule: FILTER re-pushes the stored
336 * filter, the rest a neutral constant). */
337 s_swlfo_active = false;
338 for (int t = 0; t < LFO_TARGET_COUNT; t++) {
339 if (!(s_swlfo_targets & LFO_TGT_BIT(t))) continue;
340 if (t == LFO_TARGET_FILTER)
341 live_apply_filter(&s_vp.filter);
342 else if (t == LFO_TARGET_DIST_DRIVE || t == LFO_TARGET_DIST_MIX) {
343 /* One restore event covers both dist bits. */
344 if (t == LFO_TARGET_DIST_DRIVE ||
345 !(s_swlfo_targets & LFO_TGT_BIT(LFO_TARGET_DIST_DRIVE)))
346 voice_apply_dist(LIVE_SYNTH, &s_vp.dist);
347 }
348 else
349 lfo_push_target_neutral(LIVE_SYNTH, (lfo_target_t)t);
350 }
351 }
352 return;
353 }
354
355 if (!s_swlfo_active) {
356 s_swlfo_active = true;
357 s_swlfo_phase = 0.0f;
358 }
359 /* want requires !native, so the stepper only runs on PATCH-mode voices -
360 * record the full target set for the deactivation restore. */
361 s_swlfo_targets = lfo->targets;
362
363 float ph = s_swlfo_phase +
364 live_swlfo_hz(lfo->rate, sequencer_core_get_bpm()) * 0.05f;
365 if (ph >= 1.0f) {
366 ph -= 1.0f;
367 if (lfo->wave == LFO_WAVE_RANDOM) s_swlfo_rnd = lfo_next_rand();
368 }
369 s_swlfo_phase = ph;
370
371 float val = live_swlfo_eval(lfo->wave, ph);
372 float d = (float)lfo->depth / 100.0f;
373
374 if (!native) {
375 amy_event *e = amy_helpers_event_begin();
376 e->synth = LIVE_SYNTH;
377 if (LFO_HAS_TGT(lfo, LFO_TARGET_FILTER)) {
378 float base = (s_vp.filter.enabled && s_vp.filter.cutoff_hz > 0.0f)
379 ? s_vp.filter.cutoff_hz : 1000.0f;
380 e->filter_freq_coefs[COEF_CONST] =
381 base * powf(2.0f, voice_lfo_filter_octaves(lfo) * val);
382 }
383 if (LFO_HAS_TGT(lfo, LFO_TARGET_AMP))
384 e->amp_coefs[COEF_CONST] = 1.0f - d * (0.5f - 0.5f * val);
385 if (LFO_HAS_TGT(lfo, LFO_TARGET_PAN))
386 e->pan_coefs[COEF_CONST] = 0.5f + d * 0.5f * val;
387 /* SCAN needs a wavetable voice - wave patches / native only. */
388 amy_helpers_event_send(e);
389 } /* !native */
390
391 /* DIST targets on a PATCH-mode voice (no carrier): step around the committed
392 * dist block, same law as the native rail (voice_push_dist_lfo). Native
393 * voices drive it via COEF_MOD, so !native excludes them like the rest. */
394 if (!native && (lfo->targets & LFO_TGT_DIST_MASK))
395 voice_push_dist_lfo(LIVE_SYNTH, &s_vp.dist, lfo, val);
396
397 if (!native && LFO_HAS_TGT(lfo, LFO_TARGET_PITCH)) {
398 /* Absolute Hz: see SEQ_LFO_PITCH_BASE_HZ - a bare ratio here lands
399 * ~8.8 octaves down. Osc 0 only: a synth-wide push would rewrite
400 * patch-internal modulator oscs' rates (stopgap - revisit, see the
401 * sequencer stepper). */
402 amy_event *pe = amy_helpers_event_begin();
403 pe->synth = LIVE_SYNTH;
404 pe->osc = 0;
405 pe->freq_coefs[COEF_CONST] = SEQ_LFO_PITCH_BASE_HZ * powf(2.0f, d * VOICE_LFO_DEPTH_PITCH * val);
406 amy_helpers_event_send(pe);
407 }
408}
