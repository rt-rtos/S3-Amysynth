156static void log_rtos_stats(void)
157{
158 UBaseType_t num_tasks = uxTaskGetNumberOfTasks();
159 TaskStatus_t *tasks = malloc(num_tasks * sizeof(TaskStatus_t));
160 if (tasks == NULL) {
161 ESP_LOGW(TAG, "rtos stats: malloc(%u tasks) failed", (unsigned)num_tasks);
162 return;
163 }
164
165 uint32_t total_runtime = 0;
166 num_tasks = uxTaskGetSystemState(tasks, num_tasks, &total_runtime);
167 if (num_tasks == 0 || total_runtime == 0) {
168 ESP_LOGW(TAG, "rtos stats: snapshot unavailable");
169 free(tasks);
170 return;
171 }
172
173 // Per-task table + accumulate cumulative idle time per core for the
174 // per-core busy % (computed over the interval since the previous dump).
175 uint64_t idle_now[portNUM_PROCESSORS] = {0};
176 ESP_LOGI(TAG, "RTOS tasks: name core prio stack_hwm cpu%%(cumulative)");
177 for (UBaseType_t i = 0; i < num_tasks; i++) {
178 const TaskStatus_t *t = &tasks[i];
179 uint32_t cpu_pct = (uint32_t)(((uint64_t)t->ulRunTimeCounter * 100ULL) / total_runtime);
180 int core = (int)t->xCoreID; // tskNO_AFFINITY shows as a large value
181 ESP_LOGI(TAG, " %-16s %4d %4u %9u %3u%%",
182 t->pcTaskName,
183 core,
184 (unsigned)t->uxCurrentPriority,
185 (unsigned)t->usStackHighWaterMark,
186 (unsigned)cpu_pct);
187 // The IDLE tasks are named "IDLE0"/"IDLE1" on the IDF kernel.
188 if (strncmp(t->pcTaskName, "IDLE", 4) == 0) {
189 int c = t->pcTaskName[4] - '0';
190 if (c >= 0 && c < portNUM_PROCESSORS) {
191 idle_now[c] = (uint64_t)t->ulRunTimeCounter;
192 }
193 }
194 }
195
196 // Per-core busy %, diffing each core's IDLE-task counter against wall time.
197 // RUN_TIME_STATS_USING_ESP_TIMER=y => counters are in microseconds and
198 // share the esp_timer time base, so the interval == esp_timer delta.
199 static uint64_t s_prev_wall_us = 0;
200 static uint64_t s_prev_idle_us[portNUM_PROCESSORS] = {0};
201 uint64_t now_us = (uint64_t)esp_timer_get_time();
202 uint64_t wall_delta = now_us - s_prev_wall_us;
203 for (int core = 0; core < portNUM_PROCESSORS; core++) {
204 uint64_t idle_delta = idle_now[core] - s_prev_idle_us[core];
205 if (s_prev_wall_us != 0 && wall_delta > 0) {
206 uint32_t idle_pct = (uint32_t)((idle_delta * 100ULL) / wall_delta);
207 if (idle_pct > 100) idle_pct = 100; // clamp sampling skew
208 ESP_LOGI(TAG, "core %d load: busy=%u%% idle=%u%% (interval %u ms)",
209 core, (unsigned)(100 - idle_pct), (unsigned)idle_pct,
210 (unsigned)(wall_delta / 1000ULL));
211 } else {
212 ESP_LOGI(TAG, "core %d load: (baseline captured)", core);
213 }
214 s_prev_idle_us[core] = idle_now[core];
215 }
216 s_prev_wall_us = now_us;
217
218 free(tasks);
219
220 // Heap snapshot: internal vs PSRAM free + largest free block.
221 ESP_LOGI(TAG,
222 "heap: free=%u min_free=%u | internal free=%u largest=%u | psram free=%u largest=%u",
223 (unsigned)esp_get_free_heap_size(),
224 (unsigned)esp_get_minimum_free_heap_size(),
225 (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL),
226 (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_INTERNAL),
227 (unsigned)heap_caps_get_free_size(MALLOC_CAP_SPIRAM),
228 (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_SPIRAM));
229}

233static void amy_usb_render_task(void *arg) {
234 (void)arg;
235
236 // Master clock: by default a GPTimer fires every block period (256
237 // samples @ 48 kHz = 5333.33 us), tick-rate-independent (the old
238 // vTaskDelay(pdMS_TO_TICKS(5)) floored to 0 ticks at FREERTOS_HZ=100 and
239 // busy-spun the core). CONFIG_RENDER_CLOCK_I2S_ENABLE (default off) swaps
240 // in an I2S-DMA-paced backend instead; see render_clock.h. Started from
241 // inside the task so either backend's ISR registers on the render core.
242 if (render_clock_start(AMY_BLOCK_SIZE, AMY_SAMPLE_RATE) != ESP_OK) {
243 ESP_LOGE(TAG, "render_clock_start failed; render task aborting");
244 vTaskDelete(NULL);
245 return;
246 }
247
248 while (1) {
249 // Block until the next tick. Returns the accumulated tick count; >1 means
250 // the previous block overran its period. STRICT 1:1: we render exactly
251 // ONE block regardless, so AMY's total_blocks stays locked to realtime
252 // (sequencer tempo, derived from amy_sysclock(), cannot drift). The
253 // 170 ms USB ring buffer absorbs transient jitter, not a catch-up queue.
254 uint32_t ticks = render_clock_wait();
255 if (unlikely(ticks > 1)) {
256 s_render_overruns += (ticks - 1); // diagnostic only
257 }
258
259 int16_t *block = amy_update(); // synthesizes everything / advances AMY sample clock
260 if (likely(block != NULL)) {
261 s_render_block_count++;
262
263 // Runtime PCM sampler (custompatches/sample_rec): alloc-free/non-
264 // blocking no-op unless a recording is actually armed/in-progress.
265 // Must run after amy_update() so amy_queue_lock is already released
266 // (see amy-internals.md's lock-ownership section).
267 sample_rec_render_tick(block, AMY_BLOCK_SIZE);
268
269#if CONFIG_USB_AUDIO_BLOCKING_WRITE
270 // Resilient path: retry until the host consumes the data, slaving the
271 // synth to the PC's consumption rate. NOTE: vTaskDelay(1) waits one
272 // whole tick (>=1, never floors to 0 like pdMS_TO_TICKS(1) does at
273 // FREERTOS_HZ=100) so this cannot degrade into a busy-spin.
274 while (usb_audio_write_stereo(block, AMY_BLOCK_SIZE) == ESP_ERR_NO_MEM) {
275 vTaskDelay(1);
276 }
277#else
278 // Real-time path: the write is all-or-nothing. If the ring buffer is
279 // full the whole block is dropped to keep AMY's clock phase-aligned;
280 // we do NOT delay here (the strict-1:1 GPTimer already paces us, and
281 // the previous pdMS_TO_TICKS(1) backoff floored to 0 ticks and did
282 // nothing anyway). Just count the drop for diagnostics.
283 // Only a real drop if a host is consuming; write_stereo returns
284 // ESP_OK (skips the ring) when idle, so this is belt-and-suspenders.
285 if (unlikely(usb_audio_write_stereo(block, AMY_BLOCK_SIZE) == ESP_ERR_NO_MEM)
286 && usb_audio_consumer_active()) {
287 s_usb_drops++;
288 }
289#endif
290 }
291 }
292}

295// Drains the button queue and runs the full per-screen dispatch in task
296// context. iot_button delivers events on the esp_timer task — the same task
297// that runs the 500 µs sequencer tick callback — so the heavy editor/screen
298// branches must not execute there: a long handler would delay sequencer ticks,
299// and the esp_timer stack (CONFIG_ESP_TIMER_TASK_STACK_SIZE) is not budgeted
300// for them. The callback below only enqueues; all handling lands here.
301static void button_handler_task(void *pvParameters)
302{
303 (void)pvParameters;
304 button_msg_t msg;
305 for (;;) {
306 if (xQueueReceive(s_button_queue, &msg, portMAX_DELAY) == pdTRUE) {
307 dispatch_button_event(msg.id, msg.event);
308 }
309 }
310}

312// Thin enqueue shim running on the esp_timer task (iot_button's dispatch
313// context). Filters to the event types the dispatcher consumes and posts to
314// button_task. Inline fallback only when the queue never got created — never
315// on a full queue, since handling one event ahead of already-queued ones
316// would reorder presses (a drop is less harmful than reordering).
317static void main_button_event_cb(my_button_id_t button_id, button_event_t event, void *user_data)
318{
319 (void)user_data;
320
321 if (event != BUTTON_PRESS_DOWN && event != BUTTON_PRESS_UP &&
322 event != BUTTON_SINGLE_CLICK && event != BUTTON_LONG_PRESS_START) {
323 return;
324 }
325
326 if (s_button_queue != NULL) {
327 button_msg_t msg = { .id = button_id, .event = event };
328 (void)xQueueSend(s_button_queue, &msg, 0);
329 } else {
330 dispatch_button_event(button_id, event);
331 }
332}

334// Full button → sequencer/UI action dispatch. Runs on button_task (or inline
335// on the esp_timer task only if queue creation failed at startup).
336static void dispatch_button_event(my_button_id_t button_id, button_event_t event)
337{
338 /* MY_BUTTON_SHOULDER (GPIO17, original shoulder): per-view bindings. On the
339 * sequencer grid a press toggles the step under the cursor, so one hand
340 * rides the encoder while the other enters steps (tracker-style two-handed
341 * entry); on the envelope editor it flips the EG1 sweep polarity. Fires on
342 * PRESS_DOWN for zero tap latency; all events are consumed here so they
343 * never leak into screen logic. */
344 if (button_id == MY_BUTTON_SHOULDER) {
345 if (event == BUTTON_PRESS_DOWN) {
346 ui_view_id_t sv = synth_ui_active_view();
347 if (sv == UI_VIEW_SEQ) {
348 synth_ui_toggle_step_at_cursor();
349 } else if (sv == UI_VIEW_GRAPH) {
350 /* Envelope editor: flip the melodic EG1 sweep polarity
351 * (no-op on the EG0 page and for arp/drone targets). */
352 synth_ui_graph_flip_eg1_polarity();
353 }
354 }
355 return;
356 }
357
358 /* MY_BUTTON_SHIFT (GPIO47, new shoulder): hold-modifier layer. Track the
359 * held state here; the chord interception just below turns SHIFT + digit
360 * into the "open editor" gestures that used to live on long-presses. A bare
361 * SHIFT tap does nothing. */
362 if (button_id == MY_BUTTON_SHIFT) {
363 if (event == BUTTON_PRESS_DOWN) s_shift_held = true;
364 else if (event == BUTTON_PRESS_UP) s_shift_held = false;
365 return;
366 }
367
368 /* SHIFT chords (replace the former open-editor long-presses):
369 * SHIFT + MY_BUTTON_1 -> from a mode screen: open the ADSR/graph editor;
370 * inside an editor: close+commit it (graph/LFO/filter)
371 * SHIFT + MY_BUTTON_2 -> toggle the step probability / trig editor
372 * (open, or close+commit if already open;
373 * open was the former MY_BUTTON_3 long-press)
374 * SHIFT + MY_BUTTON_3 -> inside the envelope/LFO editor: apply-to-whole-
375 * layer scope toggle (moved off SHIFT+1 so that
376 * combo can both open and close the editor); a
377 * no-op elsewhere
378 * The chord fires on the digit's PRESS_DOWN and latches that button until
379 * its NEXT press-down, swallowing the rest of the chorded press (PRESS_UP
380 * plus the trailing SINGLE_CLICK / LONG_PRESS_START) so the button's normal
381 * gesture (patch-hold on 1, drum-select on 2, menu toggle / editor-page
382 * cycle on 3) never runs. synth_ui_stepedit_open() self-gates to the
383 * sequencer screen, so SHIFT+2 elsewhere is a no-op. */
384 if (s_shift_chord_latched[button_id]) {
385 if (event == BUTTON_PRESS_DOWN) {
386 s_shift_chord_latched[button_id] = false; /* fresh press: re-evaluate */
387 } else {
388 return; /* swallow the chorded press */
389 }
390 }
391 if (s_shift_held && event == BUTTON_PRESS_DOWN &&
392 (button_id == MY_BUTTON_1 || button_id == MY_BUTTON_2 ||
393 button_id == MY_BUTTON_3)) {
394 s_shift_chord_latched[button_id] = true;
395 ui_view_id_t sv = synth_ui_active_view();
396 if (button_id == MY_BUTTON_1) {
397 /* From a mode screen SHIFT+1 opens the ADSR/graph editor; inside an
398 * open editor the same chord closes+commits it (mirrors the
399 * MY_BUTTON_0 tap). The former in-editor apply-scope toggle moved to
400 * SHIFT+3 so this one combo both opens and closes the editor. */
401 if (sv == UI_VIEW_SEQ || sv == UI_VIEW_ARP ||
402 sv == UI_VIEW_DRONE || sv == UI_VIEW_DRONE_VIS ||
403 sv == UI_VIEW_DRONE_STD) {
404 synth_ui_graph_open_envelope();
405 } else if (sv == UI_VIEW_GRAPH) {
406 synth_ui_graph_close_commit();
407 } else if (sv == UI_VIEW_LFO) {
408 synth_ui_lfo_close_commit();
409 } else if (sv == UI_VIEW_FILTER) {
410 synth_ui_filter_close_commit();
411 }
412 } else if (button_id == MY_BUTTON_2) {
413 /* Same chord toggles: SHIFT+2 closes the step editor when it is
414 * already open, in addition to the existing MY_BUTTON_0 tap /
415 * MY_BUTTON_3 close. stepedit has no discard path, so close ==
416 * commit. */
417 if (synth_ui_stepedit_is_active()) synth_ui_stepedit_close();
418 else if (!synth_ui_menu_is_active()) synth_ui_stepedit_open();
419 } else { /* MY_BUTTON_3 */
420 /* SHIFT+3: apply-to-whole-layer scope toggle inside the envelope/LFO
421 * editor (moved off SHIFT+1). Kept a chord so it can't be hit by an
422 * accidental bare press and overwrite the whole layer's config. */
423 if (sv == UI_VIEW_GRAPH || sv == UI_VIEW_LFO) {
424 synth_ui_toggle_editor_apply_scope();
425 }
426 }
427 return;
428 }
429
430 /* Project rename editor: while a project name is being typed, MY_BUTTON_1
431 * saves it and MY_BUTTON_2 discards it (MY_BUTTON_3 still closes the whole
432 * menu). synth_ui_menu_rename_active() is the authoritative gate (menu open
433 * on the projects page mid-rename) and short-circuits instantly otherwise,
434 * so this runs before the normal patch-hold / drum-select gestures without
435 * disturbing them elsewhere. Fires on PRESS_DOWN and swallows the rest. */
436 if ((button_id == MY_BUTTON_1 || button_id == MY_BUTTON_2) &&
437 synth_ui_menu_rename_active()) {
438 if (event == BUTTON_PRESS_DOWN) {
439 if (button_id == MY_BUTTON_1) synth_ui_menu_rename_save();
440 else synth_ui_menu_rename_discard();
441 }
442 return;
443 }
444
445 // MY_BUTTON_1 is the patch-select hold button, repurposed per editor:
446 // filter editor → enabled on/off toggle (single press)
447 // envelope editor → cycle EG curve type (Normal/Linear/DX7/Exp)
448 // LFO editor → unused (apply-scope moved to SHIFT+3)
449 // The apply-to-whole-layer scope toggle that used to live here is now
450 // SHIFT+3 (handled in the chord block above) so it can't be pressed by
451 // accident and overwrite the whole layer's envelope/filter config.
452 if (button_id == MY_BUTTON_1) {
453 if (synth_ui_filter_is_active()) {
454 if (event == BUTTON_PRESS_DOWN) {
455 synth_ui_filter_toggle_enabled();
456 }
457 return;
458 }
459 if (synth_ui_graph_is_active()) {
460 if (event == BUTTON_PRESS_DOWN) {
461 synth_ui_graph_cycle_eg_type();
462 }
463 return;
464 }
465 if (synth_ui_lfo_is_active()) {
466 return; /* scope is now SHIFT+3; bare press is a no-op here */
467 }
468 /* PROG screen: MY_BUTTON_1 deletes the entry at the cursor (the patch-hold
469 * gesture has no meaning here). */
470 if (synth_ui_prog_is_active()) {
471 if (event == BUTTON_PRESS_DOWN) {
472 synth_ui_prog_delete_entry();
473 }
474 return;
475 }
476 if (event == BUTTON_PRESS_DOWN) {
477 s_patch_held = true;
478 synth_ui_set_patch_select_mode(true);
479 } else if (event == BUTTON_PRESS_UP) {
480 s_patch_held = false;
481 synth_ui_set_patch_select_mode(false);
482 }
483 return;

486 /* Arp screen isolation: while the arp screen is showing (and neither the
487 * graph editor nor the menu is up), the sequencer's editing gestures must
488 * NOT leak through and mutate sequencer state behind the hidden grid. The
489 * arp screen's own input (encoder nav + MY_BUTTON_ENC field edit + the
490 * MY_BUTTON_1 patch hold/turn) is handled elsewhere and is unaffected; the
491 * menu toggle (MY_BUTTON_3) and global play/pause (MY_BUTTON_0 long-press)
492 * also stay live. Everything else is suppressed here. */
493 if (synth_ui_arp_is_active()) {
494 switch (button_id) {
495 case MY_BUTTON_2:
496 /* Drum-sound-select hold: pure sequencer state. Block, and make
497 * sure we never leave the latch stuck on if it was held when the
498 * screen switched. */
499 s_drum_select_held = false;
500 synth_ui_set_drum_select_mode(false);
501 return;
502 case MY_BUTTON_0:
503 /* Layer cycle (single-click) is sequencer-only; suppress it.
504 * Keep global play/pause on long-press. */
505 if (event == BUTTON_LONG_PRESS_START) {
506 synth_ui_toggle_playing();
507 }
508 return;
509 default:
510 break;
511 }

514 /* Prog screen isolation: suppress sequencer-only gestures (MY_BUTTON_2
515 * drum-select hold, MY_BUTTON_0 layer-cycle) while the PROG screen is up.
516 * MY_BUTTON_2 is repurposed as "+entry"; MY_BUTTON_1 as "del" (handled above). */
517 if (synth_ui_prog_is_active()) {
518 switch (button_id) {
519 case MY_BUTTON_2:
520 s_drum_select_held = false;
521 synth_ui_set_drum_select_mode(false);
522 if (event == BUTTON_PRESS_DOWN) {
523 synth_ui_prog_add_entry();
524 }

526 case MY_BUTTON_0:

760static void encoder_task(void *pvParameters)
761{
762 rotary_encoder_handle_t enc = (rotary_encoder_handle_t)pvParameters;
763 long prev = rotary_encoder_get_count(enc);
764 static long enc_accum = 0; // sub-step accumulator (2 raw ticks = 1 action)
765 for (;;) {
766 long cur = rotary_encoder_get_count(enc);
767
768 if (cur != prev) {
769 long delta = cur - prev;
770#if CONFIG_AMYSYNTH_INPUT_DIAGNOSTICS
771 ESP_LOGI(TAG,
772 "encoder count=%ld (delta=%ld)",
773 (long)cur, (long)(delta));
774#endif
775 prev = cur;
776
777 enc_accum += delta;
778 long steps = enc_accum / 2; // require 2 raw ticks per action
779 enc_accum %= 2;
780 if (steps == 0) goto next_poll;
781
782 /* One precedence resolver drives both input routers and the draw
783 * switch, so input and the OLED can never target different views.
784 * The five overlays (canonical order FILTER>LFO>STEPEDIT>GRAPH>MENU)
785 * capture the encoder outright; below them the mode screens are
786 * dispatched by resolved view, with the patch-hold / drum-select
787 * modifiers taking over exactly as before. */
788 ui_view_id_t v = synth_ui_active_view();
789 switch (v) {
790 case UI_VIEW_FILTER: synth_ui_filter_handle_encoder(steps); goto next_poll;
791 case UI_VIEW_LFO: synth_ui_lfo_handle_encoder(steps); goto next_poll;
792 case UI_VIEW_STEPEDIT: synth_ui_stepedit_handle_encoder(steps); goto next_poll;
793 case UI_VIEW_GRAPH: synth_ui_graph_handle_encoder(steps); goto next_poll;
794 case UI_VIEW_MENU: synth_ui_menu_handle_encoder(steps); goto next_poll;
795 default: break; /* fall through to the mode-tail */
796 }
797
798 if (s_patch_held) {
799 // Plain patch hold+turn. On the arp screen this cycles the
800 // arp's own patch; on the drone screen the drone's PATCH-mode
801 // preset. On the sequencer screen it cycles the active layer's
802 // patch: the melodic layer's shared patch, or — for a drum
803 // layer — the SELECTED drum track's own patch (drums are
804 // per-track Juno patches now).
805 if (v == UI_VIEW_DRONE || v == UI_VIEW_DRONE_VIS) {
806 synth_ui_drone_cycle_patch((int)steps);
807 } else if (v == UI_VIEW_DRONE_STD) {
808 synth_ui_drone_std_cycle_patch((int)steps);
809 } else if (v == UI_VIEW_ARP) {
810 synth_ui_arp_cycle_patch((int)steps);
811 } else if (sequencer_core_get_layer_type(seq_get_active_layer_idx())
812 == SEQ_LAYER_DRUM) {
813 synth_ui_cycle_drum_patch((int)steps);
814 } else {
815 synth_ui_cycle_melodic_patch((int)steps);
816 }
817 } else if (v == UI_VIEW_DRONE || v == UI_VIEW_DRONE_VIS) {
818 // Drone screen: encoder moves the cursor / edits the focused row.
819 synth_ui_drone_handle_encoder(steps);
820 } else if (v == UI_VIEW_DRONE_STD) {
821 synth_ui_drone_std_handle_encoder(steps);
822 } else if (s_drum_select_held) {
823 // Pitch-edit mode: hold MY_BUTTON_2 + turn encoder edits the
824 // selected track's pitch (works for both drum and melodic).
825 synth_ui_adjust_track_note((int)steps);
826 } else if (v == UI_VIEW_ARP) {
827 // Arp screen: encoder moves the cursor / edits the focused field.
828 synth_ui_arp_handle_encoder(steps);
829 } else if (v == UI_VIEW_PROG) {
830 // Prog screen: encoder scrolls cursor / edits the focused entry field.
831 synth_ui_prog_handle_encoder((int)steps);
832 } else if (v == UI_VIEW_TRACKOPTS) {
833 // Track Options screen: encoder scrolls rows / edits focused value.
834 synth_ui_trackopts_handle_encoder((int)steps);
835#if CONFIG_SYNTH_CUSTOM_FM
836 } else if (v == UI_VIEW_FM) {
837 // FM screen: encoder scrolls rows / edits focused value.
838 synth_ui_fm_handle_encoder((int)steps);
839#endif
840 } else {
841 synth_ui_handle_encoder(steps);
842 }
843 }
844
845next_poll:
846 vTaskDelay(pdMS_TO_TICKS(20)); // Poll at 50Hz
847 }
848}

850static void encoder_init_task(void *pvParameters)
851{
852 (void)pvParameters;
853 // Delay init to avoid early-boot conflicts (PSRAM, console pins, etc.)
854 vTaskDelay(pdMS_TO_TICKS(1000));
855
856 ESP_LOGI(TAG, "[encoder_init] starting after delay");
857 // Button GPIO (GPIO16) is now managed by my_buttons/iot_button.
858
859 rotary_encoder_config_t enc_cfg = rotary_encoder_default_config(ENCODER_PIN_A, ENCODER_PIN_B);
860 rotary_encoder_handle_t enc = NULL;
861 esp_err_t err = rotary_encoder_new_with_config(&enc_cfg, &enc);
862 ESP_LOGI(TAG, "[encoder_init] rotary_encoder_new_with_config returned %d", err);
863 if (err == ESP_OK && enc) {
864 // 8192: the patch-toggle gesture (MY_BUTTON_1 held + encoder turn)
865 // runs the AMY patch-string parse INLINE on this task stack —
866 // synth_ui_cycle_*_patch → amy_send_patch → patches_load_patch →
867 // parse_patch_string_to_queue, whose amy_event frame (~828 B) plus the
868 // nested amy_parse_message buffers drive peak usage to ~3.3-3.6 KB.
869 // A 4096 stack could not absorb that inline frame plus an ISR frame
870 // landing at the peak, causing intermittent patch-toggle stack
871 // overflow (regression from the 8192->4096 trim in 4258556). Restored
872 // to 8192 (~8 KB DRAM back, the amount the trim reclaimed). The
873 // DRAM-preserving alternative is to port the amy_ingest pump task so
874 // the parse runs on its own stack instead of the caller's.
875 // Pin to Core 0 so the encoder poll never migrates onto Core 1 and
876 // jitters the audio DSP now running there.
877 xTaskCreatePinnedToCore(encoder_task,
878 "encoder_task",
879 8192,
880 enc,
881 5,
882 NULL,
883 0);
884 }
885
886 vTaskDelete(NULL);
887}

890// Measure the AMY profiler's own per-timestamp cost on-target so the per-stage
891// numbers can be corrected for instrumentation overhead. AMY's profiler uses
892// amy_get_us() == esp_timer_get_time() on ESP; each START/STOP is two reads.
893// Coarse mode does ~8 reads/block; full mode does 2*(sum of tag 'calls')/block,
894// which the dump itself reports per tag.
895static void amy_profiler_overhead_selftest(void)
896{
897 const int N = 20000;
898 volatile int64_t sink = 0;
899 int64_t t0 = esp_timer_get_time();
900 for (int i = 0; i < N; ++i) sink ^= esp_timer_get_time();
901 int64_t t1 = esp_timer_get_time();
902 (void)sink;
903 // Subtract one read (t1) negligibly; report ns/read averaged over N reads.
904 double ns_per_read = ((double)(t1 - t0) * 1000.0) / (double)N;
905 double coarse_us_per_block = (ns_per_read * 8.0) / 1000.0; // 4 START/STOP pairs
906 const uint32_t block_us = (uint32_t)(((uint64_t)AMY_BLOCK_SIZE * 1000000ULL)
907 / (uint64_t)AMY_SAMPLE_RATE);
908 ESP_LOGW(TAG,
909 "[amy-profile] esp_timer_get_time ~%.1f ns/read | coarse overhead ~%.2f us/block "
910 "(~%.3f%% of %u us budget) | full overhead = 2*sum(tag.calls)/block * read_cost",
911 ns_per_read, coarse_us_per_block,
912 (coarse_us_per_block / (double)block_us) * 100.0, block_us);
913}

916#ifdef GAMMA9001
917/* Feed AMY the gamma9001 drum-bank blob (PCM presets 256-391) from the
918 * 'drums' data partition. Preferred path is a read-only flash mmap of
919 * exactly the blob's size, but PSRAM XIP (SPIRAM_FETCH_INSTRUCTIONS +
920 * SPIRAM_RODATA) leaves too few data-cache MMU pages for ~3.6 MB on this
921 * config, so the fallback copies the blob into PSRAM heap instead (~3.6 MB
922 * of the ~7 MB free; same placement class as the FX delay lines). Every
923 * failure path is non-fatal: AMY keeps those presets disabled. */
924static void gamma9001_pcm_mount(void)
925{
926 const esp_partition_t *part = esp_partition_find_first(
927 ESP_PARTITION_TYPE_DATA, ESP_PARTITION_SUBTYPE_ANY, "drums");
928 if (part == NULL) {
929 ESP_LOGW(TAG, "gamma9001: no 'drums' partition; PCM presets 256+ unavailable");
930 return;
931 }
932
933 /* A never-flashed partition reads as erased flash (all 0xFF); don't hand
934 * AMY full-scale noise as samples. */
935 uint16_t probe[4];
936 if (esp_partition_read(part, 0, probe, sizeof(probe)) != ESP_OK ||
937 (probe[0] == 0xFFFFu && probe[1] == 0xFFFFu &&
938 probe[2] == 0xFFFFu && probe[3] == 0xFFFFu)) {
939 ESP_LOGW(TAG, "gamma9001: 'drums' partition is blank (flash drums.bin); "
940 "PCM presets 256+ unavailable");
941 return;
942 }
943
944 size_t bytes = amy_gamma9001_pcm_bytes();
945 if (bytes == 0 || bytes > part->size) {
946 ESP_LOGW(TAG, "gamma9001: blob size %u exceeds partition (%u); "
947 "PCM presets 256+ unavailable",
948 (unsigned)bytes, (unsigned)part->size);
949 return;
950 }
951
952 const void *map = NULL;
953 esp_partition_mmap_handle_t handle;
954 if (esp_partition_mmap(part, 0, bytes, ESP_PARTITION_MMAP_DATA,
955 &map, &handle) == ESP_OK) {
956 amy_set_gamma9001_pcm((const int16_t *)map);
957 ESP_LOGI(TAG, "gamma9001: drum banks flash-mapped (%u KB)",
958 (unsigned)(bytes / 1024u));
959 return;
960 }
961
962 /* MMU pages exhausted: stage the blob in PSRAM instead. */
963 int16_t *buf = heap_caps_malloc(bytes, MALLOC_CAP_SPIRAM);
964 if (buf == NULL) {
965 ESP_LOGW(TAG, "gamma9001: mmap failed and PSRAM alloc of %u KB failed; "
966 "PCM presets 256+ unavailable", (unsigned)(bytes / 1024u));
967 return;
968 }
969 if (esp_partition_read(part, 0, buf, bytes) != ESP_OK) {
970 ESP_LOGW(TAG, "gamma9001: partition read failed; PCM presets 256+ unavailable");
971 free(buf);
972 return;
973 }
974 amy_set_gamma9001_pcm(buf);
975 ESP_LOGI(TAG, "gamma9001: drum banks copied to PSRAM (%u KB; flash mmap unavailable)",
976 (unsigned)(bytes / 1024u));
977}
