117static void amy_usb_render_task(void *arg) {
118 (void)arg;
119
120 // Master clock: a GPTimer fires every block period (256 samples @ 48 kHz
121 // = 5333.33 us); CONFIG_RENDER_CLOCK_I2S_ENABLE swaps in an I2S-DMA-paced
122 // backend (render_clock.h). Started inside the task so either backend's
123 // ISR registers on the render core.
124 if (render_clock_start(AMY_BLOCK_SIZE, AMY_SAMPLE_RATE) != ESP_OK) {
125 ESP_LOGE(TAG, "render_clock_start failed; render task aborting");
126 vTaskDelete(NULL);
127 return;
128 }
129
130 while (1) {
131 // STRICT 1:1: render exactly one block even after an overrun (ticks
132 // > 1), so amy_sysclock-derived tempo can't drift. The USB ring
133 // absorbs jitter; it is not a catch-up queue.
134 uint32_t ticks = render_clock_wait();
135 if (unlikely(ticks > 1)) {
136 dropout_count_render_overrun(ticks - 1); // diagnostic only
137 }
138
139 render_stats_block_begin();
140 uint32_t tick_hooks_pre = s_seq_tick_hook_count;
141
142 int16_t *block = amy_update(); // synthesizes everything / advances AMY sample clock
143 if (likely(block != NULL)) {
144 s_render_block_count++;
145
146 // Runtime PCM sampler: non-blocking no-op unless a recording is
147 // armed. Must run after amy_update() releases amy_queue_lock.
148 sample_rec_render_tick(block, AMY_BLOCK_SIZE);
149
150#if CONFIG_FILTER_SCOPE
151 // Filter-editor overlay: reads msynth[]/synth[], so it too must
152 // run after amy_update() releases amy_queue_lock. Armed only
153 // while that editor is open.
154 filter_scope_render_tick();
155#endif
156
157#if CONFIG_USB_AUDIO_BLOCKING_WRITE
158 // Resilient path: retry until the host drains, slaving the synth
159 // to the PC's consumption rate. vTaskDelay(1) waits a full tick
160 // (never floors to 0), so this cannot busy-spin.
161 while (usb_audio_write_stereo(block, AMY_BLOCK_SIZE) == ESP_ERR_NO_MEM) {
162 vTaskDelay(1);
163 }
164#else
165 // Real-time path: all-or-nothing write; a full ring drops the
166 // whole block to keep AMY phase-aligned, with no backoff (the
167 // GPTimer already paces us). Counted as a drop only while a host
168 // is actually consuming.
169 if (unlikely(usb_audio_write_stereo(block, AMY_BLOCK_SIZE) == ESP_ERR_NO_MEM)
170 && usb_audio_consumer_active()) {
171 dropout_count_ring_overrun();
172 }
173#endif
174 }
175 // Measures the full block's work (DSP + sampler/scope hooks + USB
176 // ring write); the seq-tick flag feeds the worst-block snapshot.
177 render_stats_block_end(s_seq_tick_hook_count != tick_hooks_pre);
178 }
179}

248 /* MY_BUTTON_SHOULDER, per view: SEQ toggles the step under the cursor
249 * (two-handed tracker-style entry), GRAPH flips the EG1 sweep polarity,
250 * LFO flips the target checklist tab. Step entry and the polarity flip
251 * take PRESS_DOWN for zero tap latency; the tab flip is a deliberate
252 * navigation gesture, so it waits for the click. All events consumed. */
253 if (button_id == MY_BUTTON_SHOULDER) {
254 ui_view_id_t sv = synth_ui_active_view();
255 if (event == BUTTON_PRESS_DOWN) {
256 if (sv == UI_VIEW_SEQ) {
257 synth_ui_toggle_step_at_cursor();
258 } else if (sv == UI_VIEW_GRAPH) {
259 /* No-op on the EG0 page and for arp/drone targets. */
260 synth_ui_graph_flip_eg1_polarity();
261#if CONFIG_SYNTH_CUSTOM_FM
262 } else if (sv == UI_VIEW_FM) {
263 synth_ui_fm_toggle_feedback();
264#endif
265 }
266 } else if (event == BUTTON_SINGLE_CLICK && sv == UI_VIEW_LFO) {
267 synth_ui_lfo_toggle_target_tab();
268 }
269 return;
270 }
271
272 /* MY_BUTTON_SHIFT: hold-modifier layer; a bare tap does nothing. */
273 if (button_id == MY_BUTTON_SHIFT) {
274 if (event == BUTTON_PRESS_DOWN) s_shift_held = true;
275 else if (event == BUTTON_PRESS_UP) s_shift_held = false;
276 return;
277 }

279 /* SHIFT chords, fired on the digit's PRESS_DOWN:
280 * SHIFT+1 -> open the ADSR/graph editor, or close+commit an open one
281 * SHIFT+2 -> toggle the step probability/trig editor; inside an
282 * effects editor: release the open tab to the patch
283 * SHIFT+3 -> inside an effects editor: flip the row between its own
284 * voice block and the layer's shared one
285 * The chord latches the button until its next PRESS_DOWN, swallowing the
286 * rest of the press so the button's normal gesture never runs. */
287 if (s_shift_chord_latched[button_id]) {
288 if (event == BUTTON_PRESS_DOWN) {
289 s_shift_chord_latched[button_id] = false; /* fresh press: re-evaluate */
290 } else {
291 return; /* swallow the chorded press */
292 }
293 }
294 if (s_shift_held && event == BUTTON_PRESS_DOWN &&
295 (button_id == MY_BUTTON_1 || button_id == MY_BUTTON_2 ||
296 button_id == MY_BUTTON_3)) {
297 s_shift_chord_latched[button_id] = true;
298 ui_view_id_t sv = synth_ui_active_view();
299 if (button_id == MY_BUTTON_1) {
300 bool open_from_screen = (sv == UI_VIEW_SEQ || sv == UI_VIEW_ARP ||
301 sv == UI_VIEW_DRONE || sv == UI_VIEW_DRONE_VIS ||
302 sv == UI_VIEW_DRONE_STD || sv == UI_VIEW_FM);
303#if CONFIG_SYNTH_WIRELESS
304 /* Wireless overlay page: same chord, bound to the BLE live-play
305 * voice (synth_ui_graph_open_envelope picks the target). */
306 open_from_screen = open_from_screen ||
307 (sv == UI_VIEW_MENU && synth_ui_wireless_page_is_open());
308#endif
309 if (open_from_screen) {
310 synth_ui_graph_open_envelope();
311 } else if (sv == UI_VIEW_GRAPH) {
312 synth_ui_graph_close_commit();
313 } else if (sv == UI_VIEW_LFO) {
314 synth_ui_lfo_close_commit();
315 } else if (sv == UI_VIEW_DIST) {
316 synth_ui_dist_close_commit();
317 } else if (sv == UI_VIEW_FILTER) {
318 synth_ui_filter_close_commit();
319 }
320 } else if (button_id == MY_BUTTON_2) {
321 bool editor_open = (sv == UI_VIEW_GRAPH || sv == UI_VIEW_LFO ||
322 sv == UI_VIEW_DIST || sv == UI_VIEW_FILTER);
323 if (editor_open) {
324 /* Inside an editor the chord hands the open tab back to the
325 * patch - chorded, like the source flip, so a stray press
326 * cannot drop an authored group. */
327 synth_ui_editor_release_to_patch();
328 } else if (synth_ui_stepedit_is_active()) {
329 /* stepedit has no discard path, so close == commit.
330 * synth_ui_stepedit_open() self-gates to the sequencer screen. */
331 synth_ui_stepedit_close();
332 } else if (!synth_ui_menu_is_active()) {
333 synth_ui_stepedit_open();
334 }
335 } else { /* MY_BUTTON_3 */
336 /* Chorded so an accidental bare press can't flip which voice
337 * block the row reads. */
338 if (sv == UI_VIEW_GRAPH || sv == UI_VIEW_LFO || sv == UI_VIEW_DIST ||
339 sv == UI_VIEW_FILTER) {
340 synth_ui_toggle_editor_source();
341 }
342 }
343 return;

701static void encoder_task(void *pvParameters)
702{
703 rotary_encoder_handle_t enc = (rotary_encoder_handle_t)pvParameters;
704 long prev = rotary_encoder_get_count(enc);
705 static long enc_accum = 0; // sub-step accumulator (2 raw ticks = 1 action)
706 for (;;) {
707 long cur = rotary_encoder_get_count(enc);
708
709 if (cur != prev) {
710 long delta = cur - prev;
711#if CONFIG_AMYSYNTH_INPUT_DIAGNOSTICS
712 ESP_LOGI(TAG,
713 "encoder count=%ld (delta=%ld)",
714 (long)cur, (long)(delta));
715#endif
716 prev = cur;
717
718 enc_accum += delta;
719 long steps = enc_accum / 2; // require 2 raw ticks per action
720 enc_accum %= 2;
721 if (steps != 0) {
722 encoder_process_steps(steps);
723 }
724 }
725
726 vTaskDelay(pdMS_TO_TICKS(20)); // Poll at 50Hz
727 }
728}

779#ifdef GAMMA9001
780/* Feed AMY the gamma9001 drum-bank blob (PCM presets 256-391) from the
781 * 'drums' partition. Flash mmap preferred, but PSRAM XIP leaves too few MMU
782 * pages for ~3.6 MB, so the fallback copies the blob into PSRAM heap. All
783 * failure paths are non-fatal: those presets just stay disabled. */
784static void gamma9001_pcm_mount(void)
785{
786 const esp_partition_t *part = esp_partition_find_first(
787 ESP_PARTITION_TYPE_DATA, ESP_PARTITION_SUBTYPE_ANY, "drums");
788 if (part == NULL) {
789 ESP_LOGW(TAG, "gamma9001: no 'drums' partition; PCM presets 256+ unavailable");
790 return;
791 }
792
793 /* A never-flashed partition reads all 0xFF; don't play noise. */
794 uint16_t probe[4];
795 if (esp_partition_read(part, 0, probe, sizeof(probe)) != ESP_OK ||
796 (probe[0] == 0xFFFFu && probe[1] == 0xFFFFu &&
797 probe[2] == 0xFFFFu && probe[3] == 0xFFFFu)) {
798 ESP_LOGW(TAG, "gamma9001: 'drums' partition is blank (flash drums.bin); "
799 "PCM presets 256+ unavailable");
800 return;
801 }
802
803 size_t bytes = amy_gamma9001_pcm_bytes();
804 if (bytes == 0 || bytes > part->size) {
805 ESP_LOGW(TAG, "gamma9001: blob size %u exceeds partition (%u); "
806 "PCM presets 256+ unavailable",
807 (unsigned)bytes, (unsigned)part->size);
808 return;
809 }
810
811 const void *map = NULL;
812 esp_partition_mmap_handle_t handle;
813 if (esp_partition_mmap(part, 0, bytes, ESP_PARTITION_MMAP_DATA,
814 &map, &handle) == ESP_OK) {
815 amy_set_gamma9001_pcm((const int16_t *)map);
816 diag_mem_track("drums-pcm", map, bytes);
817 ESP_LOGI(TAG, "gamma9001: drum banks flash-mapped (%u KB)",
818 (unsigned)(bytes / 1024u));
819 return;
820 }
821
822 /* MMU pages exhausted: stage the blob in PSRAM instead. */
823 int16_t *buf = heap_caps_malloc(bytes, MALLOC_CAP_SPIRAM);
824 if (buf == NULL) {
825 ESP_LOGW(TAG, "gamma9001: mmap failed and PSRAM alloc of %u KB failed; "
826 "PCM presets 256+ unavailable", (unsigned)(bytes / 1024u));
827 return;
828 }
829 if (esp_partition_read(part, 0, buf, bytes) != ESP_OK) {
830 ESP_LOGW(TAG, "gamma9001: partition read failed; PCM presets 256+ unavailable");
831 free(buf);
832 return;
833 }
834 amy_set_gamma9001_pcm(buf);
835 diag_mem_track("drums-pcm", buf, bytes);
836 ESP_LOGI(TAG, "gamma9001: drum banks copied to PSRAM (%u KB; flash mmap unavailable)",
837 (unsigned)(bytes / 1024u));
838}
839#endif /* GAMMA9001 */
