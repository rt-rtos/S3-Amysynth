40{
41 for (size_t i = 0; i < len; i++) {
42 uint8_t b = data[i];
43
44 if (b >= 0xF8u) continue; /* realtime: transparent, keeps
45 * running status intact */
46 if (b & 0x80u) {
47 if (b == 0xF0u) { /* sysex start */
48 s_in_sysex = true;
49 s_status = 0;
50 s_have_d1 = false;
51 } else if (b == 0xF7u) { /* sysex end */
52 s_in_sysex = false;
53 } else if (b >= 0xF0u) { /* system common: cancels
54 * running status per spec */
55 s_status = 0;
56 s_have_d1 = false;
57 } else { /* channel status */
58 s_status = b;
59 s_have_d1 = false;
60 }
61 continue;
62 }
63
64 /* Data byte. */
65 if (s_in_sysex || s_status == 0) continue;
66
67 if (status_data_len(s_status) == 1) continue; /* 1-data messages: drop */
68
69 if (!s_have_d1) {
70 s_d1 = b;
71 s_have_d1 = true;
72 continue;
73 }
74 s_have_d1 = false; /* message complete */
75
76 uint8_t hi = s_status & 0xF0u;
77 uint8_t ch = s_status & 0x0Fu;
78 if (!s_sink) continue;
79
80 if (hi == 0x90u) {

82 if (s_sink->note_off) s_sink->note_off(ch, s_d1);
