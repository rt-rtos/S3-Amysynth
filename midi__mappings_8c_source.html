8void juno_filter_midi_handler(uint8_t * bytes, uint16_t len, uint8_t is_sysex_unused) {
9 // An example of adding a handler for MIDI CCs. Can't really build this in because it depends on your synth/patch config, controllers, wishes...
10 // Here, we use MIDI CC 70 to modify the Juno VCF center freq, and 71 for resonance.
11 amy_event e;
12 if (bytes[0] == 0xB0) { // Channel 1 CC
13 if (bytes[1] == 70) {
14 // Modify Synth 0 filter_freq.
15/* def to_filter_freq(val): */
16/* # filter_freq goes from ? 100 to 6400 Hz with 18 steps/octave */
17/* return float("%.3f" % (13 * exp2(0.0938 * val * 127))) */
18 e = amy_default_event();
19 e.synth = 1;
20 e.filter_freq_coefs[COEF_CONST] = exp2f(0.0938f * (float)bytes[2]);
21 amy_add_event(&e);
22 } else if (bytes[1] == 71) {
23/* def to_resonance(val): */
24/* # Q goes from 0.5 to 16 exponentially */
25/* return float("%.3f" % (0.7 * exp2(4.0 * val))) */
26 e = amy_default_event();
27 e.synth = 1;
28 e.resonance = 0.7f * exp2f(0.03125f * (float)bytes[2]);
29 amy_add_event(&e);
30 }
31 }
32}

50// Mappings are indexed by channel, but "channel" here means a synth number as
51// well as a MIDI channel -- patches.c routes synth note-ons through the mapping
52// machinery, and synth numbers run up to config.max_synths, well past the 16
53// channels a MIDI cable can carry. So these are sized at init, not compiled in.
54// Channel 0 is a valid synth (it just isn't reachable from a MIDI cable, whose
55// channels are numbered from 1), so valid channels are 0..num_mapping_channels.

84struct midi_mapping *midi_mapping_init(int channel, int type, int code, int is_log, float min_val, float max_val, float offset_val, const char *message_template, int message_len) {
85 if (!mapping_channel_ok(channel)) return NULL;
86 struct midi_mapping **p_root;
87 if (type == MIDI_MAP_TYPE_CC)
88 p_root = &midi_cc_mapping_root_by_chan[channel];
89 else
90 p_root = &midi_note_mapping_root_by_chan[channel];
91 struct midi_mapping *result = (struct midi_mapping *)malloc_caps(sizeof(struct midi_mapping) + message_len + 1, amy_global.config.ram_caps_synth);
92 result->message_template = ((char *)result) + sizeof(struct midi_mapping);
93 result->channel = channel;
94 result->type = type;
95 result->code = code;
96 result->is_log = is_log;
97 result->min_val = min_val;
98 result->max_val = max_val;
99 result->offset_val = offset_val;
100 strncpy(result->message_template, message_template, message_len);
101 result->message_template[message_len] = '\0';
102 // Insert into the linked list at the head.
103 result->next = *p_root;
104 *p_root = result;
105 return result;
106}

138void midi_mappings_init(void) {
139 midi_mappings_deinit(); // Release any earlier allocation; init is called more than once.
140 // A mapping channel is a synth number, so cover every synth, but never fewer
141 // than the 16 channels that can arrive over a MIDI cable.
142 num_mapping_channels = AMY_NUM_MIDI_CHANNELS;
143 if ((int)amy_global.config.max_synths > num_mapping_channels)
144 num_mapping_channels = (int)amy_global.config.max_synths;
145 size_t num_bytes = sizeof(struct midi_mapping *) * (num_mapping_channels + 1);
146 midi_cc_mapping_root_by_chan = (struct midi_mapping **)malloc_caps(num_bytes, amy_global.config.ram_caps_synth);
147 midi_note_mapping_root_by_chan = (struct midi_mapping **)malloc_caps(num_bytes, amy_global.config.ram_caps_synth);
148 if (midi_cc_mapping_root_by_chan == NULL || midi_note_mapping_root_by_chan == NULL) {
149 fprintf(stderr, "unable to alloc midi mappings for %d channels\n", num_mapping_channels);
150 free(midi_cc_mapping_root_by_chan);
151 midi_cc_mapping_root_by_chan = NULL;
152 free(midi_note_mapping_root_by_chan);
153 midi_note_mapping_root_by_chan = NULL;
154 num_mapping_channels = 0;
155 return;
156 }
157 for (int channel = 0; channel < num_mapping_channels + 1; ++channel) {
158 midi_cc_mapping_root_by_chan[channel] = NULL;
159 midi_note_mapping_root_by_chan[channel] = NULL;
160 }
161 mappings_inited = true;
162}

179void midi_clear_channel_mappings(int channel, int type) {
180 if (!mapping_channel_ok(channel)) return;
181 // Each root list holds only mappings of its own type on this one channel, so
182 // clearing a type is just emptying its list -- no filtering, no recursion.
183 if (type == MIDI_MAP_TYPE_ANY || type == MIDI_MAP_TYPE_CC)
184 midi_mappings_free(&midi_cc_mapping_root_by_chan[channel]);
185 if (type == MIDI_MAP_TYPE_ANY || type == MIDI_MAP_TYPE_NOTE)
186 midi_mappings_free(&midi_note_mapping_root_by_chan[channel]);
187 // Stop listening to this MIDI channel unless there's a synth on it.
188 if (!instrument_number_exists(channel, NULL))
189 midi_active_channel_set(channel, false);
190}

192struct midi_mapping **midi_mapping_find(int channel, int type, int code) {
193 if (!mapping_channel_ok(channel)) return NULL;
194 // Retrieve the mapping associated with a midi channel + code, if any.
195 struct midi_mapping **result;
196 if (type == MIDI_MAP_TYPE_ANY) {
197 result = midi_mapping_find(channel, MIDI_MAP_TYPE_CC, code);
198 if (result == NULL)
199 result = midi_mapping_find(channel, MIDI_MAP_TYPE_NOTE, code);
200 return result;
201 }
202 struct midi_mapping **p_mapping;
203 if (type == MIDI_MAP_TYPE_CC)
204 p_mapping = &midi_cc_mapping_root_by_chan[channel];
205 else
206 p_mapping = &midi_note_mapping_root_by_chan[channel];
207 while (*p_mapping != NULL) {
208 if ((*p_mapping)->channel == channel && ((type == MIDI_MAP_TYPE_ANY) || (*p_mapping)->type == type)) {
209 if ((code == MIDI_MAP_CODE_ANY) || ((*p_mapping)->code == MIDI_MAP_CODE_ANY) || ((*p_mapping)->code == code))
210 return p_mapping;
211 }
212 p_mapping = &((*p_mapping)->next);
213 }
214 return NULL;
215}

217int midi_clear_mapping(int channel, int type, int code) {
218 if (!mapping_channel_ok(channel)) return 0;
219 // Backwards compatibility
220 if (code == 255) code = MIDI_MAP_CODE_ANY;
221 if (code == MIDI_MAP_CODE_ANY) {
222 // Magic value means clear all MIDI CCs for this channel
223 midi_clear_channel_mappings(channel, type);
224 return 1;
225 }
226 struct midi_mapping **p_mapping = midi_mapping_find(channel, type, code);
227 if (p_mapping) {
228 midi_mapping_free(p_mapping);
229 // We just deleted a mapping on this channel, was it the last one?
230 midi_active_channel_set(channel, midi_mappings_exist_for_channel(channel) || instrument_number_exists(channel, NULL));
231 return 1;
232 }
233 return 0; // nothing found.
234}

293float map_midi_value(struct midi_mapping *mapping, uint8_t value) {
294 float ret_val = 0;
295 if (mapping->is_log != 0) {
296 ret_val = (mapping->min_val + mapping->offset_val)
297 * expf(
298 logf((mapping->max_val + mapping->offset_val)
299 / (mapping->min_val + mapping->offset_val))
300 * (float)value / 127.0f
301 )
302 - mapping->offset_val;
303 } else { // Linear.
304 ret_val = mapping->min_val
305 + (mapping->max_val - mapping->min_val)
306 * (float)value / 127.0f;
307 }
308 return ret_val;
309}

311void substitute_midi_special_values(char *dest, const char *src, int channel, int code, float value) {
312 // Copy src string to dest, but replace "%i" with channel and "%v" with value.
313 const char *s;
314 const char *entry_src = src;
315 int n_remain = AMY_WIRE_COMMAND_LEN - 1;
316 while((s = strchr(src, '%')) != NULL && n_remain > (int)strlen(src)) {
317 // Copy up to the %.
318 int nchars = s - src;
319 strncpy(dest, src, nchars);
320 dest += nchars;
321 n_remain -= nchars;
322 src += nchars;
323 ++src; // skip over the '%'
324 dest[0] = '\0';
325 if (src[0] == 'v') {
326 sprintf(dest, "%.3f", value);
327 } else if (src[0] == 'V') {
328 sprintf(dest, "%d", (int)value);
329 } else if (src[0] == 'i') {
330 sprintf(dest, "%d", channel);
331 } else if (src[0] == 'n') { // 'n' is for note.
332 sprintf(dest, "%d", code);
333 } else {
334 fprintf(stderr, "substitute_midi: unrecognized '%%%c' in %s\n", src[0], entry_src);
335 }
336 ++src; // skip over the code char
337 nchars = strlen(dest);
338 dest += nchars;
339 n_remain -= nchars;
340 }
341 // Copy anything left in the string.
342 if (n_remain > (int)strlen(src)) strcpy(dest, src);
343}

350void *yield_midi_message_handler_events(uint8_t status, uint16_t channel, uint8_t * data, uint16_t len, uint32_t time, amy_event *event, void *state) {
351 //fprintf(stderr, "time %.3f midi_msg_handler: status 0x%x chan %d 0x%x 0x%x\n", amy_global.time, status, channel, data[0], data[1]);
352 //fprintf_event_stderr(event);
353 //
354 struct midi_cmd_yield_state *yield_state = (struct midi_cmd_yield_state *)state;
355 if (len < 2) return NULL; // Every status we act on carries two data bytes.
356 if (status == 0xB0
357 || ((!instrument_number_exists(channel, NULL) || instrument_grab_midi_notes(channel))
358 && (status == 0x90 || status == 0x80))) { // CC or note-on with grab_midi set.
359 int type = (status == 0xB0) ? MIDI_MAP_TYPE_CC : MIDI_MAP_TYPE_NOTE;
360 int code = data[0]; // note for note-on events
361 struct midi_mapping **p_mapping = midi_mapping_find(channel, type, code);
362 struct midi_mapping *mapping = &default_note_mapping;
363 if (type == MIDI_MAP_TYPE_NOTE || p_mapping != NULL) {
364 if (p_mapping != NULL)
365 mapping = *p_mapping;
366 if (yield_state == NULL) {
367 // First call to this mapping, allocate state, perform processing.
368 yield_state = malloc_caps(sizeof(struct midi_cmd_yield_state) + AMY_WIRE_COMMAND_LEN, amy_global.config.ram_caps_events);
369 // On OOM drop this midi message.
370 if (yield_state == NULL) {
371 amy_oom("yield_midi_message_handler_events: out of memory\n");
372 return NULL;
373 }
374 char *message = yield_state->message = (char *)(yield_state + 1);
375 yield_state->pos = 0;
376 // And now set up the message
377 float value = map_midi_value(mapping, (data[1] == 0xFF)? 0 : data[1]); // suppress "fake note on" value.
378 if (status == 0x80) { // Translate note-off to note-on with vel 0.
379 status = 0x90;
380 value = 0;
381 }
382 substitute_midi_special_values(message, mapping->message_template, channel, code, value);
383 // Mark the event as already passed through mapping for this
384 // channel, so we don't send it back out again.
385 event->note_source_channel = channel;
386 // If we're given a time, set it in the event.
387 if (AMY_IS_SET(time)) event->time = time;
388 } // If state is non-null, assume we're working through the later yields.
389 // Layer each parsed event on top of the caller's base event, if any.
390 yield_state->pos = yield_event_from_message(yield_state->message, event, yield_state->pos);
391 if (yield_state->pos == 0) {
392 // End of iteration
393 free(yield_state);
394 yield_state = NULL;
395 }
396 }
397 }
398 return (void *)yield_state;
399}
