8void juno_filter_midi_handler(uint8_t * bytes, uint16_t len, uint8_t is_sysex) {
9 // An example of adding a handler for MIDI CCs. Can't really build this in because it depends on your synth/patch config, controllers, wishes...
10 // Here, we use MIDI CC 70 to modify the Juno VCF center freq, and 71 for resonance.
11 amy_event e;
12 if (bytes[0] == 0xB0) { // Channel 1 CC
13 if (bytes[1] == 70) {
14 // Modify Synth 0 filter_freq.
15/* def to_filter_freq(val): */
16/* # filter_freq goes from ? 100 to 6400 Hz with 18 steps/octave */
17/* return float("%.3f" % (13 * exp2(0.0938 * val * 127))) */
18 e = amy_default_event();
19 e.synth = 1;
20 e.filter_freq_coefs[COEF_CONST] = exp2f(0.0938f * (float)bytes[2]);
21 amy_add_event(&e);
22 } else if (bytes[1] == 71) {
23/* def to_resonance(val): */
24/* # Q goes from 0.5 to 16 exponentially */
25/* return float("%.3f" % (0.7 * exp2(4.0 * val))) */
26 e = amy_default_event();
27 e.synth = 1;
28 e.resonance = 0.7f * exp2f(0.03125f * (float)bytes[2]);
29 amy_add_event(&e);
30 }
31 }
32}

185float map_midi_value(struct midi_mapping *mapping, uint8_t value) {
186 float ret_val = 0;
187 if (mapping->is_log != 0) {
188 ret_val = (mapping->min_val + mapping->offset_val)
189 * expf(
190 logf((mapping->max_val + mapping->offset_val)
191 / (mapping->min_val + mapping->offset_val))
192 * (float)value / 127.0f
193 )
194 - mapping->offset_val;
195 } else { // Linear.
196 ret_val = mapping->min_val
197 + (mapping->max_val - mapping->min_val)
198 * (float)value / 127.0f;
199 }
200 return ret_val;
201}

203void substitute_midi_special_values(char *dest, const char *src, int channel, int code, float value) {
204 // Copy src string to dest, but replace "%i" with channel and "%v" with value.
205 const char *s;
206 const char *entry_src = src;
207 int n_remain = AMY_WIRE_COMMAND_LEN - 1;
208 while((s = strchr(src, '%')) != NULL && n_remain > (int)strlen(src)) {
209 // Copy up to the %.
210 int nchars = s - src;
211 strncpy(dest, src, nchars);
212 dest += nchars;
213 n_remain -= nchars;
214 src += nchars;
215 ++src; // skip over the '%'
216 dest[0] = '\0';
217 if (src[0] == 'v') {
218 sprintf(dest, "%.3f", value);
219 } else if (src[0] == 'V') {
220 sprintf(dest, "%d", (int)value);
221 } else if (src[0] == 'i') {
222 sprintf(dest, "%d", channel);
223 } else if (src[0] == 'n') { // 'n' is for note.
224 sprintf(dest, "%d", code);
225 } else {
226 fprintf(stderr, "substitute_midi: unrecognized '%%%c' in %s\n", src[0], entry_src);
227 }
228 ++src; // skip over the code char
229 nchars = strlen(dest);
230 dest += nchars;
231 n_remain -= nchars;
232 }
233 // Copy anything left in the string.
234 if (n_remain > (int)strlen(src)) strcpy(dest, src);
235}

237void midi_msg_handler(uint8_t * bytes, uint16_t len, uint8_t is_sysex, uint32_t time) {
238 //fprintf(stderr, "time %.3f midi_msg_handler: 0x%x 0x%x 0x%x\n", amy_global.time, bytes[0], bytes[1], bytes[2]);
239 uint8_t status = bytes[0] & 0xF0;
240 uint8_t channel = (bytes[0] & 0x0F) + 1;
241 if (status == 0xB0
242 || ((!instrument_number_exists(channel, NULL) || instrument_grab_midi_notes(channel))
243 && (status == 0x90 || status == 0x80))) { // CC or note-on with grab_midi set.
244 int type = (status == 0xB0) ? MIDI_MAP_TYPE_CC : MIDI_MAP_TYPE_NOTE;
245 int code = bytes[1]; // note for note-on events
246 struct midi_mapping **p_mapping = midi_mapping_find(channel, type, code);
247 struct midi_mapping *mapping = &default_note_mapping;
248 if (type == MIDI_MAP_TYPE_NOTE || p_mapping != NULL) {
249 if (p_mapping != NULL)
250 mapping = *p_mapping;
251 float value = map_midi_value(mapping, bytes[2]);
252 if (status == 0x80) { // Translate note-off to note-on with vel 0.
253 status = 0x90;
254 value = 0;
255 }
256 char message[AMY_WIRE_COMMAND_LEN];
257 // Mark message as already passed through mapping for this channel (synth).
258 sprintf(message, "iM%d", channel);
259 int offset = strlen(message);
260 if (AMY_IS_SET(time)) {
261 sprintf(message + offset, "t%" PRId32, time);
262 offset = strlen(message);
263 }
264 substitute_midi_special_values(message + offset, mapping->message_template, channel, code, value);
265 amy_add_message(message);
266 }
267 }
