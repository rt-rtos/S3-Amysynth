18// Button GPIO assignments (active low, internal pull-up). This list is the
19// WIRING: entry N is the pin of the Nth member of my_button_id_t, so reordering
20// that enum is what reassigns roles. Keep the names below in step with it.
21static const int32_t s_button_gpios[MY_BUTTON_MAX] = {
22 15, // MY_BUTTON_SHOULDER → per-view bindings (step toggle on the grid)
23 18, // MY_BUTTON_1 → patch-select hold
24 8, // MY_BUTTON_2
25 42, // MY_BUTTON_3 (avoiding GPIO3 strapping pin)
26 16, // MY_BUTTON_ENC → encoder push button
27 17, // MY_BUTTON_0 → layer cycle / play-stop
28 47, // MY_BUTTON_SHIFT → hold modifier
29};

35// Internal callback that routes events to user callback
36static void button_event_cb(void *button_handle, void *usr_data)
37{
38 if (s_user_cb == NULL) {
39 return;
40 }
41
42 my_button_id_t btn_id = (my_button_id_t)(uintptr_t)usr_data;
43 button_event_t event = iot_button_get_event(button_handle);
44
45#if CONFIG_AMYSYNTH_INPUT_DIAGNOSTICS
46 ESP_LOGI(TAG, "Button %d: %s", btn_id, iot_button_get_event_str(event));
47#endif
48 s_user_cb(btn_id, event, s_user_data);
49}

51esp_err_t my_buttons_init(void)
52{
53 esp_err_t ret = ESP_OK;
54
55 // Common button config (use defaults for timing)
56 button_config_t btn_cfg = {
57 .long_press_time = 0, // Use default (1500ms)
58 .short_press_time = 0, // Use default (180ms)
59 };
60
61 for (int i = 0; i < MY_BUTTON_MAX; i++) {
62 button_gpio_config_t gpio_cfg = {
63 .gpio_num = s_button_gpios[i],
64 .active_level = 0, // Active low
65 .enable_power_save = false,
66 .disable_pull = false, // Enable internal pull-up (for active_level=0)
67 };
68
69 ret = iot_button_new_gpio_device(&btn_cfg, &gpio_cfg, &s_button_handles[i]);
70 if (ret != ESP_OK) {
71 ESP_LOGE(TAG, "Failed to create button %d on GPIO%ld: %s",
72 i, gpio_cfg.gpio_num, esp_err_to_name(ret));
73 // Clean up already created buttons
74 my_buttons_deinit();
75 return ret;
76 }
77
78 // Register callbacks for common events
79 iot_button_register_cb(s_button_handles[i], BUTTON_PRESS_DOWN, NULL,
80 button_event_cb, (void *)(uintptr_t)i);
81 iot_button_register_cb(s_button_handles[i], BUTTON_PRESS_UP, NULL,
82 button_event_cb, (void *)(uintptr_t)i);
83 iot_button_register_cb(s_button_handles[i], BUTTON_SINGLE_CLICK, NULL,
84 button_event_cb, (void *)(uintptr_t)i);
85 iot_button_register_cb(s_button_handles[i], BUTTON_DOUBLE_CLICK, NULL,
86 button_event_cb, (void *)(uintptr_t)i);
87 iot_button_register_cb(s_button_handles[i], BUTTON_LONG_PRESS_START, NULL,
88 button_event_cb, (void *)(uintptr_t)i);
89
90 ESP_LOGI(TAG, "Button %d initialized on GPIO%ld (active low, pull-up)",
91 i, gpio_cfg.gpio_num);
92 }
93
94 ESP_LOGI(TAG, "All %d buttons initialized", MY_BUTTON_MAX);
95 return ESP_OK;
96}

105esp_err_t my_buttons_deinit(void)
106{
107 for (int i = 0; i < MY_BUTTON_MAX; i++) {
108 if (s_button_handles[i] != NULL) {
109 iot_button_delete(s_button_handles[i]);
110 s_button_handles[i] = NULL;
111 }
112 }
113 s_user_cb = NULL;
114 s_user_data = NULL;
115 return ESP_OK;
116}
