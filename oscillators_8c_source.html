22const LUT *choose_from_lutset(float period, const LUT *lutset) {
23 // Select the best entry from a lutset for a given period.
24 //
25 // Args:
26 // period: (float) Target period of waveform in fractional samples.
27 // lutset: Sorted list of LUTs, as generated by create_lutset().
28 //
29 // Returns:
30 // One of the LUTs from the lutset, best suited to interpolating to generate
31 // a waveform of the desired period.
32 // Use the earliest (i.e., longest, most harmonics) LUT that works
33 // (i.e., will not actually cause aliasing).
34 // So start with the highest-bandwidth (and longest) LUTs, but skip them
35 // if they result in aliasing.
36 const LUT *lut_table = NULL;
37 int lut_size = 0;
38 int lut_index = 0;
39 while(lutset[lut_index].table_size > 0) {
40 lut_table = &lutset[lut_index];
41 lut_size = lutset[lut_index].table_size;
42 // What proportion of nyquist does the highest harmonic in this table occupy?
43 float lut_bandwidth = 2 * lutset[lut_index].highest_harmonic / (float)lut_size;
44 // To complete one cycle of <lut_size> points in <period> steps, each step
45 // will need to be this many samples:
46 float lut_hop = lut_size / period;
47 // If we have a signal with a given bandwidth, but then speed it up by
48 // skipping lut_hop samples per sample, its bandwidth will increase
49 // proportionately.
50 float interp_bandwidth = lut_bandwidth * lut_hop;
51 //printf("period=%f freq=%f lut_size=%d interp_bandwidth=%f\n", period, ((float)AMY_SAMPLE_RATE)/period, lut_size, interp_bandwidth);
52 if (interp_bandwidth < 0.9f) {
53 // No aliasing, even with a 10% buffer (i.e., 19.8 kHz).
54 break;
55 }
56 ++lut_index;
57 }
58 // At this point, we either got to the end of the LUT table, or we found a
59 // table we could interpolate without aliasing.
60
61 return lut_table;
62}

96// Miller's optimization -
97// https://github.com/pure-data/pure-data/blob/db777311d808bb3ba728b94ab067f8d333b7d0c2/src/d_array.c#L831C1-L833C76
98// outlet_float(x->x_obj.ob_outlet, b + frac * (
99// cminusb - 0.1666667f * (1.-frac) * (
100// (d - a - 3.0f * cminusb) * frac + (d + 2.0f*a - 3.0f*b))));#
101#define INTERP_CUBIC \
102 SAMPLE a = L2S(lut->table[(base_index - 1) & lut_mask]); \
103 SAMPLE d = L2S(lut->table[(base_index + 2) & lut_mask]); \
104 SAMPLE cminusb = c - b; \
105 SAMPLE fr_d_ma_m3cmb = MUL0_SS(d - a - cminusb - SHIFTL(cminusb, 1), frac); \
106 SAMPLE next_bit = MUL0_SS(fr_d_ma_m3cmb + d + SHIFTL(a - b, 1) - b, MUL0_SS(F2S(1.0f) - frac, F2S(0.16666666666667f))); \
107 sample = b + MUL0_SS(cminusb - next_bit, frac);

111 buf[i] = value; \
112 if (value < 0) value = -value; \
113 if (value > max_value) max_value = value; \
114 current_amp += incremental_amp; \
115 phase = P_WRAPPED_SUM(phase, step);

316AMY_IRAM_ATTR SAMPLE render_lpf_lut(SAMPLE* buf, uint16_t osc, int8_t is_square, int8_t direction, SAMPLE dc_offset) {
317 AMY_PROFILE_START(RENDER_LPF_LUT)
318 // Common function for pulse and saw.
319 float freq = freq_of_logfreq(msynth[osc]->logfreq);
320 PHASOR step = F2P(freq / (float)AMY_SAMPLE_RATE); // cycles per sec / samples per sec -> cycles per sample
321 SAMPLE amp = direction * F2S(msynth[osc]->amp);
322 SAMPLE last_amp = direction * F2S(msynth[osc]->last_amp);
323 PHASOR pwm_phase = synth[osc]->phase;
324 SAMPLE max_value;
325 synth[osc]->phase = render_lut_cub(buf, synth[osc]->phase, step, last_amp, amp, synth[osc]->lut, &max_value);
326 if (is_square) { // For pulse only, add a second delayed negative LUT wave.
327 float duty = msynth[osc]->duty;
328 if (duty < 0.01f) duty = 0.01f;
329 if (duty > 0.99f) duty = 0.99f;
330 pwm_phase = P_WRAPPED_SUM(pwm_phase, F2P(msynth[osc]->last_duty));
331 // Second pulse is given some blockwise-constant FM to maintain phase continuity across blocks.
332 PHASOR delta_phase_per_sample = F2P((duty - msynth[osc]->last_duty) / AMY_BLOCK_SIZE);
333 render_lut_cub(buf, pwm_phase, step + delta_phase_per_sample, -last_amp, -amp, synth[osc]->lut, &max_value);
334 msynth[osc]->last_duty = duty;
335 }
336 // Remember last_amp.
337 msynth[osc]->last_amp = msynth[osc]->amp;
338 AMY_PROFILE_STOP(RENDER_LPF_LUT)
339 return max_value;
340}

592// I think this is to do with the implicit state - it's not repeatable, nor strictly safe,
593// to potentially have two cores accessing the same rand_state memory at the same time.

722#define KS_LOWEST_FREQ 55 // 55Hz (A1) lowest note we can play
723// One period of KS_LOWEST_FREQ, so the ring holds any note we admit at any
724// sample rate. A literal sized for 44100 truncates the ring above it.
725#define MAX_KS_BUFFER_LEN (AMY_SAMPLE_RATE / KS_LOWEST_FREQ + 1)
726SAMPLE ** ks_buffer;
727// Which osc each ring belongs to. A ring is a physical string: two oscs
728// playing one ring damp each other's decay and re-excite each other on
729// note-on, so the mapping has to be per osc rather than a shared cursor.
730// The claim is two-way -- ks_row_owner[r] == osc AND synth[osc]->ks_index == r
731// -- so resetting or freeing an osc releases its ring with no explicit undo.
732static uint16_t *ks_row_owner;
733#define KS_ROW_UNOWNED 0xFFFF
734
735
736/* karplus-strong */
737
738// True if nothing is currently sounding on ring r.
739static bool ks_row_is_free(uint8_t r) {
740 uint16_t owner = ks_row_owner[r];
741 if (owner == KS_ROW_UNOWNED) return true;
742 if (synth[owner] == NULL) return true; // osc was freed
743 if (synth[owner]->ks_index != r) return true; // osc reset, or took another ring
744 if (synth[owner]->wave != KS) return true; // osc switched wave
745 if (synth[owner]->status == SYNTH_OFF) return true;
746 return false;
747}
748
749// Pick the ring this note will play. In order: the one this osc already owns
750// (a retrigger re-excites its own string, which is the common case and never
751// disturbs anyone else), then any idle ring, then -- only if every ring is
752// sounding -- the quietest one, because that is the collision least likely to
753// be heard.
754static uint8_t ks_alloc_row(uint16_t osc) {
755 uint8_t own = synth[osc]->ks_index;
756 if (own < AMY_KS_OSCS && ks_row_owner[own] == osc) return own;
757 for (uint8_t r = 0; r < AMY_KS_OSCS; ++r) {
758 if (ks_row_is_free(r)) return r;
759 }
760 uint8_t quietest = 0;
761 float lowest = msynth[ks_row_owner[0]]->amp;
762 for (uint8_t r = 1; r < AMY_KS_OSCS; ++r) {
763 float amp = msynth[ks_row_owner[r]]->amp;
764 if (amp < lowest) { lowest = amp; quietest = r; }
765 }
766 return quietest;
767}

799void ks_note_on(uint16_t osc, float freq) {
800 // render_ks reads phase as a plain ring index, so start it at the bottom
801 // of the ring. Any other wave leaves a fixed-point phasor here, whose
802 // low 16 bits index far past the buffer.
803 synth[osc]->phase = 0;
804 if(ks_buffer == NULL) { synth[osc]->ks_index = KS_NO_ROW; return; }
805 uint8_t row = ks_alloc_row(osc);
806 synth[osc]->ks_index = row;
807 ks_row_owner[row] = osc;
808 SAMPLE *ring = ks_buffer[row];
809 uint16_t buflen = (uint16_t)(AMY_SAMPLE_RATE / freq);
810 if(buflen > MAX_KS_BUFFER_LEN) buflen = MAX_KS_BUFFER_LEN;
811 // init KS buffer with noise up to max
812 SAMPLE sum = 0;
813 for(uint16_t i = 0; i < buflen; i++) {
814 SAMPLE val = amy_get_random();

816 sum += val;
817 }
818 // Remove dc, to avoid ending up with a dc-offset residual.
819 SAMPLE mean = sum / buflen;
820 for(uint16_t i = 0; i < buflen; i++) {
821 ring[i] -= mean;
822 }
823 //fprintf(stderr, "ks_note_on: osc %d buflen %d row %d\n", osc, buflen, row);

831void ks_init(void) {
832 // One ring per simultaneously sounding KS osc; ks_oscs is the polyphony
833 // ceiling, past which ks_alloc_row() steals the quietest ring.
834 ks_buffer = (SAMPLE**) malloc_caps(sizeof(SAMPLE*)*AMY_KS_OSCS, amy_global.config.ram_caps_synth);
835 if(ks_buffer == NULL) {
836 amy_oom("unable to alloc %d KS rings\n", (int)AMY_KS_OSCS);

838 }
839 // Before anything else can fail: ks_deinit() walks this array.
840 for(int i=0;i<AMY_KS_OSCS;i++) ks_buffer[i] = NULL;
841 ks_row_owner = (uint16_t*) malloc_caps(sizeof(uint16_t)*AMY_KS_OSCS, amy_global.config.ram_caps_synth);
842 if(ks_row_owner == NULL) {
843 amy_oom("unable to alloc %d KS ring owners\n", (int)AMY_KS_OSCS);
844 ks_deinit();
845 return;
846 }
847 for(int i=0;i<AMY_KS_OSCS;i++) ks_row_owner[i] = KS_ROW_UNOWNED;
848 for(int i=0;i<AMY_KS_OSCS;i++) {
849 ks_buffer[i] = (SAMPLE*)malloc_caps(sizeof(SAMPLE)*MAX_KS_BUFFER_LEN, amy_global.config.ram_caps_synth);

851 // All or nothing: a partly-filled table would let ks_alloc_row()
852 // hand out a NULL ring, so KS goes silent instead of half-working.
853 // ks_note_on's NULL check and render_ks's ks_index guard are what
854 // make silence safe.
855 amy_oom("unable to alloc KS ring of %d samples\n", (int)MAX_KS_BUFFER_LEN);
856 ks_deinit();

858 }
