11float atoff(const char *s) {
12 // Returns float value corresponding to parseable prefix of s.
13 // Unlike atof(), it does not recognize scientific format ('e' or 'E')
14 // and will stop parsing there. Needed for message strings that contain
15 // 'e' as a command prefix.
16 float frac = 0;
17 // Skip leading spaces.
18 while (*s == ' ') ++s;
19 float whole = (float)atoi(s);
20 int is_negative = (s[0] == '-'); // Can't use (whole < 0) because of "-0.xx".
21 //const char *s_in = s; // for debug message.
22 s += strspn(s, "-0123456789");
23 if (*s == '.') {
24 // Float with a decimal part.
25 // Step over dp
26 ++s;
27 // Extract fractional part.
28 int fraclen = strspn(s, "0123456789");
29 char fracpart[8];
30 // atoi() will overflow for values larger than 2^31, so only decode a prefix.
31 if (fraclen > 6) {
32 for(int i = 0; i < 7; ++i) {
33 fracpart[i] = s[i];
34 }
35 fracpart[7] = '\0';
36 s = fracpart;
37 fraclen = 7;
38 }
39 frac = (float)atoi(s);
40 frac /= powf(10.f, (float)fraclen);
41 if (is_negative) frac = -frac;
42 }
43 //fprintf(stderr, "input was %s output is %f + %f = %f\n", s_in, whole, frac, whole+frac);
44 return whole + frac;
45}

72#define PARSE_LIST(type) \
73 int parse_list_##type(char *message, type *vals, int max_num_vals, type skipped_val) { \
74 uint16_t c = 0, last_c; \
75 uint16_t stop = strspn(message, PARSE_LIST_STRSPN1(skipped_val)); \
76 int num_vals_received = 0; \
77 while(c < stop && num_vals_received < max_num_vals) { \
78 *vals = PARSE_LIST_ATO(skipped_val)(message + c); \
79 while (message[c] == ' ') ++c; \
80 last_c = c; \
81 c += strspn(message + c, PARSE_LIST_STRSPN2(skipped_val)); \
82 if (last_c == c) \
83 *vals = skipped_val; \
84 while (message[c] != ',' && message[c] != 0 && c < MAX_MESSAGE_LEN) c++; \
85 ++c; \
86 ++vals; \
87 ++num_vals_received; \
88 } \
89 if (c < stop) { \
90 fprintf(stderr, "WARNING: parse__list_##type: More than %d values in \"%s\"\n", \
91 max_num_vals, message); \
92 } else { /* pad to end */ \
93 for (int i = num_vals_received; i < max_num_vals; ++i) { \
94 *vals++ = skipped_val; \
95 } \
96 } \
97 return num_vals_received; \
98 }

119char *copy_with_trim(char *dest, size_t dest_len, const char *src, size_t src_len) {
120 // Copy a string while trimming leading and trailing spaces.
121 const char *s = src;
122 char *d = dest;
123 size_t d_writ = 0;
124 size_t s_read = 0;
125 size_t trimmed_src_len = src_len;
126 // scan for spaces at end
127 while (trimmed_src_len > 0 && isspace((unsigned char)src[trimmed_src_len - 1])) {
128 --trimmed_src_len;
129 }
130 // skip over leading spaces
131 while (s_read < trimmed_src_len && isspace((unsigned char)*s)) {
132 ++s;
133 ++s_read;
134 }
135 while(s_read < trimmed_src_len && d_writ < (dest_len - 1)) {
136 *d++ = *s++;
137 ++s_read;
138 ++d_writ;
139 }
140 *d = '\0'; // terminator.
141 return (char*) (src + src_len);
142}

152uint16_t parse_list_file_params(char *message, uint32_t *preset, char *filename, size_t filename_len, uint32_t *midinote) {
153 // Returns number of characters of message that are consumed.
154 if (filename_len > 0) {
155 filename[0] = '\0';
156 }
157 char *m = message;
158 *preset = strtol(m, &m, 0);
159 if (*m != ',') return m - message;
160 ++m;
161 m = copy_with_trim(filename, filename_len, m, strchrnul_local(m, ',') - m);
162 if (*m != ',') return m - message;
163 ++m;
164 *midinote = strtol(m, &m, 0);
165 return m - message;
166}

195float int_db_to_float_lin(uint32_t db) {
196 // in interp_partials.h, we store amplitudes as integer dB values in range 0..100.
197 float lin = 0;
198 if (AMY_IS_UNSET(db)) return AMY_UNSET_VALUE(lin);
199 lin = powf(10.0f, ((((float)db) - 100.0f) / 20.0f)) - 0.001f;
200 if (lin < 0) return 0;
201 return lin;
202}

218static int parse_breakpoint_event_core_float_lin(char* message, uint32_t *times_ms, float *values) {
219 float vals[2 * MAX_BREAKPOINTS];
220 int num_vals = parse_list_float(message, vals, 2 * MAX_BREAKPOINTS,
221 AMY_UNSET_VALUE(vals[0]));
222 for (int i = 0; i < num_vals; ++i) {
223 int bp_index = (i >> 1);
224 if (bp_index >= MAX_BREAKPOINTS) break;
225 if ((i % 2) == 0) {
226 if (AMY_IS_SET(vals[i])) {
227 int32_t t_ms = (int32_t)vals[i];
228 if (t_ms < 0) t_ms = 0;
229 times_ms[bp_index] = (uint32_t)t_ms; // clamp_bp_time_ms_to_i16((uint32_t)t_ms);
230 } else {
231 AMY_UNSET(times_ms[bp_index]);
232 }
233 } else {
234 values[bp_index] = vals[i];
235 }
236 }
237 return num_vals;
238}

261static void parse_event_breakpoints(char *message, uint32_t *times_ms, float *values) {
262 int num_vals = 0;
263 for (int i = 0; i < MAX_BREAKPOINTS; ++i) {
264 AMY_UNSET(times_ms[i]);
265 AMY_UNSET(values[i]);
266 }
267 if (message[0] == '.' && message[1] == '.') {
268 num_vals = parse_breakpoint_event_core_int_db(message + 2, times_ms, values);
269 } else {
270 num_vals = parse_breakpoint_event_core_float_lin(message, times_ms, values);
271 }
272 for (int i = num_vals; i < 2 * MAX_BREAKPOINTS; ++i) {
273 int bp_index = (i >> 1);
274 if (bp_index < MAX_BREAKPOINTS) {
275 if ((i % 2) == 0) AMY_UNSET(times_ms[bp_index]);
276 else AMY_UNSET(values[bp_index]);
277 }
278 }

334int midi_mapping_from_message(char *message, char cmd, int instr_num, int skip_chars) {
335 // MIDI CC mapping ic<C>,<L>,<N>,<X>,<O>,<CODE>, see https://github.com/shorepine/amy/issues/524
336 // ic255 clears all MIDI CC mappings for this synth (short form, no extra fields needed).
337 size_t pos = 0;
338 size_t mlen = strlen(message);
339 while (pos < mlen) {
340 // Break the mapping on ZZs (for K257).
341 size_t sub_mlen, next_pos;
342 char *end = strstr(message + pos, "ZZ");
343 if (end != NULL) {
344 sub_mlen = end - (message + pos);
345 next_pos = pos + sub_mlen + 2; // Step over "ZZ"
346 } else {
347 sub_mlen = mlen - pos;
348 next_pos = mlen;
349 }
350 while (sub_mlen > 0 && message[pos + sub_mlen - 1] == 'Z') {
351 --sub_mlen;
352 }
353 // Parse the fragment.
354 int32_t code, is_log;
355 float min_val = 0, max_val = 0, offset_val = 0;
356 int type = (cmd == 'c') ? MIDI_MAP_TYPE_CC : MIDI_MAP_TYPE_NOTE;
357 AMY_UNSET(code);
358 AMY_UNSET(is_log);
359 skip_chars = parse_midi_mapping_payload(message + pos, &code, &is_log, &min_val, &max_val, &offset_val);
360 if (*(message + pos + skip_chars) != ',') {
361 if (AMY_IS_UNSET(code) || AMY_IS_SET(is_log)) {
362 // Either parsing bailed without even a CC code, or it got past the is_log, meaning it wasn't a bare ic<NUM> command.
363 fprintf(stderr, "synth_layer: midi mapping payload didn't parse for %s.\n", message - 1);
364 return pos + skip_chars; // maybe the rest will parse?
365 }
366 // Else we got an incomplete message with a valid CC code - clear it
367 midi_clear_mapping(instr_num, type, code); // (handles 255 as special case).
368 return pos + skip_chars;
369 }
370 ++skip_chars; // step over the "," before the wire string template.
371 midi_store_mapping(instr_num, type, code, is_log, min_val, max_val, offset_val, message + pos + skip_chars, sub_mlen - skip_chars);
372 pos = next_pos;
373 if (pos < mlen && message[pos] != 'i') break;
374 cmd = message[pos + 1];
375 pos += 2;
376 }
377 return pos - 1;
378}

380int parse_cv_trigger_payload(char *message, int32_t *p_gate_cv, float *p_thresh_high, float *p_thresh_low, int32_t *p_pitch_cv, float *p_pitch_scale, float *p_pitch_offset) {
381 char *m = message;
382 m += parse_val_int32_t(m, p_gate_cv);
383 if (m[0] != ',') goto end; else ++m;
384 m += parse_val_float(m, p_thresh_high);
385 if (m[0] != ',') goto end; else ++m;
386 m += parse_val_float(m, p_thresh_low);
387 if (m[0] != ',') goto end; else ++m;
388 // The pitch CV args are optional

390 // Next arg is not numeric, looks like the wire code

392 --m;

397 m += parse_val_float(m, p_pitch_scale);

399 m += parse_val_float(m, p_pitch_offset);
400 end:
401 return m - message;
402}

404int cv_trigger_from_message(char *message, int instr_num, int skip_chars) {
405 // i<synth>ig<gate_cv>,<thresh_high>,<thresh_low>,<pitch_cv>,<pitch_scale>,<pitch_offset>,<wire_template>
406 int32_t gate_cv, pitch_cv;
407 uint8_t pitch_cv_uint8;
408 float thresh_high = 0, thresh_low = 0, pitch_scale = 0, pitch_offset = 0;
409 AMY_UNSET(gate_cv);
410 AMY_UNSET(pitch_cv);
411 AMY_UNSET(thresh_high);
412 skip_chars = parse_cv_trigger_payload(message, &gate_cv, &thresh_high, &thresh_low, &pitch_cv, &pitch_scale, &pitch_offset);
413 if (*(message + skip_chars) != ',') {
414 if (AMY_IS_UNSET(gate_cv) || AMY_IS_SET(thresh_high)) {
415 // Either parsing bailed without even a gate CV, or it got past the thresh, meaning it wasn't a bare ic<NUM> command.
416 fprintf(stderr, "cv_trigger: payload didn't parse for %s.\n", message - 1);
417 return skip_chars; // maybe the rest will parse?
418 }
419 // Else we got an incomplete message with a valid gate_cv - clear all triggers for that CV.
420 cv_trigger_clear_mappings(gate_cv);
421 return skip_chars;
422 }
423 ++skip_chars; // step over the "," before the wire string template.
424 if (AMY_IS_SET(pitch_cv))
425 pitch_cv_uint8 = pitch_cv;
426 else
427 AMY_UNSET(pitch_cv_uint8);
428 cv_trigger_new(gate_cv, thresh_high, thresh_low,
429 pitch_cv_uint8, pitch_scale, pitch_offset, message + skip_chars);
430 // Consume rest of message but leave the trailing 'Z' for the outer parser.
431 int remainder = strlen(message);
432 if (remainder > 0 && message[remainder - 1] == 'Z') remainder--;
433 skip_chars = remainder;
434 return skip_chars;
435}

437// Parser for synth-layer ('i') prefix.
438int amy_parse_synth_layer_message(char *message, amy_event *e) {
439 int skip_chars = 1; // default is to skip one extra char.
440 if (message[0] >= '0' && message[0] <= '9') {
441 // It's just the instrument number.
442 e->synth = atoi(message);
443 return 0; // no extra skip.
444 }
445 char cmd = message[0];
446 message++;
447 if (cmd == 'd') e->synth_delay_ms = atoi(message);
448 else if (cmd == 'f') e->synth_flags = atoi(message);
449 else if (cmd == 'g') skip_chars = cv_trigger_from_message(message, e->synth, skip_chars);
450 else if (cmd == 'm') e->grab_midi_notes = atoi(message);
451 else if (cmd == 'M') e->note_source_channel = atoi(message); // To mark MIDI-in notes.
452 else if (cmd == 'n') e->oscs_per_voice = atoi(message);
453 else if (cmd == 'p') e->pedal = atoi(message);
454 else if (cmd == 't') e->to_synth = atoi(message);
455 else if (cmd == 'v') e->num_voices = atoi(message);
456 else if (cmd == 'y') e->bus = atoi(message); // 'i1iy1' is the same as 'i1y1'.
457 else if (cmd == 'c' || cmd == 'o') skip_chars = midi_mapping_from_message(message, cmd, e->synth, skip_chars);
458 else fprintf(stderr, "Unrecognized synth-level command '%s'\n", message - 1);
459 return skip_chars;
460}

462// Parser for transfer-layer ('z') prefix. Returns how much of a message to skip
463uint16_t amy_parse_transfer_layer_message(char *message) {
464
465 if (message[0] >= '0' && message[0] <= '9') {
466 // z: Signal to start loading sample.
467 // Params: preset number, length(frames), samplerate, midinote, loopstart, loopend.
468 uint32_t sm[6]; // preset, length, SR, midinote, loop_start, loopend
469 parse_list_uint32_t(message, sm, 6, 0);
470 if(sm[1]==0) { // remove preset
471 pcm_unload_preset(sm[0]);
472 } else {
473 int16_t * ram = pcm_load(sm[0], sm[1], sm[2], 1, sm[3], sm[4], sm[5]);
474 start_receiving_transfer(sm[1]*2, (uint8_t*)ram);
475 }
476 return 0;
477 }
478 char cmd = message[0];
479 message++;
480 if (cmd == 'T') {
481 // zT: Signal to start loading file.
482 //Params: Destination name, file size.
483 uint32_t file_size = 0;
484 char filename[MAX_FILENAME_LEN];
485 uint16_t len = parse_list_file_transfer_params(message, filename, sizeof(filename), &file_size);
486 if (filename[0] != '\0') {
487 sequencer_midi_stop(); // Stop sequencer (and sketch loop) during file transfer.
488 start_receiving_file_transfer(file_size, filename);
489 }
490 return len;
491 }
492 else if (cmd == 'F') {
493 // zF: setup PCM preset from WAV filename on disk.
494 // Params: Preset number, filename, midi note
495
496 uint32_t preset = 0;
497 uint32_t midinote = 0;
498 char filename[MAX_FILENAME_LEN];
499 uint16_t len = parse_list_file_params(message, &preset, filename, sizeof(filename),
500 &midinote);
501 if (filename[0] != '\0') {
502 amy_global.transfer_stored_bytes = midinote;
503 strncpy(amy_global.transfer_filename, filename, MAX_FILENAME_LEN);
504 amy_global.transfer_file_handle = preset;
505 // For tulip/amyboard we have to load the PCM file from the MP "task"
506 #if (defined AMYBOARD) || (defined TULIP)
507 mp_sched_schedule(MP_OBJ_FROM_PTR(&tulip_pcm_load_file_obj), mp_const_none);
508 #else
509 pcm_load_file();
510 #endif
511
512 }
513 return len;
514 }
515 else if (cmd == 'S') {
516 // zS: sample from BUS[1] to a memorypcm patch.
517 // Params: Preset number, bus, max length in frames,midinote,loopstart,loopend
518 uint32_t sm[6]; // preset, bus, max frames, midinote, loop_start, loopend
519 parse_list_uint32_t(message, sm, 6, 0);
520 int16_t * ram = pcm_load(sm[0], sm[2], AMY_SAMPLE_RATE, 2, sm[3], sm[4], sm[5]);
521 start_receiving_sample(sm[2], sm[1], ram);
522 return 1;
523 }
524 else if (cmd == 'O') {
525 //zO: stop sampling from any bus
526 stop_receiving_sample();
527 return 1;
528 }
529 else if (cmd == 'D') {
530 // zD: Dump data over MIDI sysex.
531 // zD[Z] — dump all active instrument state + global effects.
532 // zD<filename>[Z] — dump file contents from the filesystem.
533 // The filename/payload is "rest of message" — we consume everything
534 // to the end of the C string. A trailing 'Z' (end-of-message marker
535 // some senders append) is stripped, so interior capital-Z characters
536 // in the filename (e.g. "/ZIPFILE.py") are preserved. Limitation:
537 // filenames whose last char is 'Z' are not addressable.
538 char filename[MAX_FILENAME_LEN];
539 uint16_t len = 0;
540 while (message[len] && len < MAX_FILENAME_LEN - 1) {
541 filename[len] = message[len];
542 len++;
543 }
544 filename[len] = '\0';
545 if (len > 0 && filename[len - 1] == 'Z') {
546 filename[--len] = '\0';
547 }
548 if (filename[0] == '\0') {
549 amy_dump_state_to_sysex();
550 } else {
551 amy_dump_file_to_sysex(filename);
552 }
553 // Consume the whole rest of the message so the outer parser exits.
554 {
555 uint16_t total = 0;
556 const char *scan = message - 1; // back to 'D'
557 while (scan[total]) total++;
558 return total;
559 }
560 }
561 else if (cmd == 'A') {
562 // zA: Update sketch.py on disk with current AMY state (calls update_file_hook).
563 // Takes optional filename; defaults to /user/current/sketch.py on AMYboard.
564 // Payload semantics match zD: the filename is "rest of message", a
565 // trailing 'Z' terminator is stripped, and interior capital-Z chars
566 // in the filename are preserved.
567 char filename[MAX_FILENAME_LEN];
568 uint16_t len = 0;
569 while (message[len] && len < MAX_FILENAME_LEN - 1) {
570 filename[len] = message[len];
571 len++;
572 }
573 filename[len] = '\0';
574 if (len > 0 && filename[len - 1] == 'Z') {
575 filename[--len] = '\0';
576 }
577 if (amy_global.config.amy_external_update_file_hook) {
578 if (filename[0]) {
579 amy_global.config.amy_external_update_file_hook(filename);
580 } else {
581 amy_global.config.amy_external_update_file_hook("/user/current/sketch.py");
582 }
583 }
584 {
585 uint16_t total = 0;
586 const char *scan = message - 1;
587 while (scan[total]) total++;
588 return total;
589 }
590 }
591 else if (cmd == 'P') {
592 // zP: Execute Python code on host (e.g. zPimport amyboard; amyboard.restart_sketch()Z).
593 // Payload semantics match zD: the code string is "rest of message",
594 // a trailing 'Z' terminator is stripped, and interior capital-Z chars
595 // in the code are preserved.
596 char code[256];
597 uint16_t len = 0;
598 while (message[len] && len < sizeof(code) - 1) {
599 code[len] = message[len];
600 len++;
601 }
602 code[len] = '\0';
603 if (len > 0 && code[len - 1] == 'Z') {
604 code[--len] = '\0';
605 }
606 if (amy_global.config.amy_external_exec_hook) {
607 amy_global.config.amy_external_exec_hook(code);
608 }
609 {
610 uint16_t total = 0;
611 const char *scan = message - 1;
612 while (scan[total]) total++;
613 return total;
614 }
615 }
616 else if (cmd == 'Y') {
617 // zY: sequencer transport. zY1 starts the sequencer, zY0 stops it. Lets a
618 // host drive playback without MIDI clock sync (see external_midi_sync).
619 if (atoi(message)) sequencer_midi_start();
620 else sequencer_midi_stop();
621 return 1;
622 }
623 else if (cmd == 'C') {
624 // zC: external MIDI clock sync. zC1 makes the sequencer follow incoming
625 // MIDI realtime clock/start/stop (0xF8/0xFA/0xFC); zC0 (the default)
626 // ignores them and uses the internal clock. Same switch as the C API's
627 // amy_external_midi_sync(), reachable over the wire so hosts that only
628 // speak the wire protocol (web builds, sysex) can flip it.
629 amy_external_midi_sync(atoi(message) ? 1 : 0);
630 return 1;
631 }
632 else fprintf(stderr, "Unrecognized transfer-level command '%s'\n", message - 1);
633 return 0;
634}

637int _next_alpha(char *s) {
638 // Return how many chars to skip to get to the next alphabet (command prefix) (or EOS).
639 int p = 0;
640 while (*(s + p)) {
641 char c = *(s + p);
642 if (isalpha(c)) break;
643 ++p;
644 }
645 return p;
646}

657// given a string return a parsed event
658int amy_parse_message(char * message, amy_event *e) {
659 peek_stack("parse_message");
660 int length = strlen(message);
661 char cmd = '\0';
662 uint16_t pos = 0;
663
664 // Check if we're in a transfer block, if so, parse it and leave this loop.
665 // FILE transfers (zT, used to write files over MIDI sysex) arrive async
666 // while a sketch may also be running, so we ONLY route them to the
667 // transfer handler when the data is sysex-originated -- otherwise a
668 // sketch calling amy.send(note=36) mid-transfer would get its wire
669 // command base64-decoded as file data and corrupt the file.
670 //
671 // AUDIO transfers (amy.load_sample / load_sample_bytes) are different:
672 // Python sends every chunk synchronously in a tight loop within the same
673 // call, so no other amy.send() can interleave. They route regardless of
674 // the sysex flag (which they don't carry, since send_raw goes through
675 // the regular wire path).
676 extern bool amy_parsing_from_sysex;
677 if (amy_global.transfer_flag == AMY_TRANSFER_TYPE_AUDIO ||
678 (amy_parsing_from_sysex && amy_global.transfer_flag == AMY_TRANSFER_TYPE_FILE)) {
679 parse_transfer_message(message, length);
680 e->status = EVENT_TRANSFER_DATA;
681 return length;
682 }
683
684 while(pos < length) {
685 cmd = message[pos];
686 char *arg = message + pos + 1;
687 if(isalpha(cmd)) {
688 switch(cmd) {
689 case 'a': parse_coef_message(arg, e->amp_coefs);break;
690 case 'A': {
691 char bp_msg[MAX_PARAM_LEN];
692 copy_param_list_substring(bp_msg, arg);
693 parse_event_breakpoints(bp_msg, e->eg0_times, e->eg0_values);
694 e->bp_is_set[0] = 1;
695 break;
696 }
697 case 'B': {
698 char bp_msg[MAX_PARAM_LEN];
699 copy_param_list_substring(bp_msg, arg);
700 parse_event_breakpoints(bp_msg, e->eg1_times, e->eg1_values);
701 e->bp_is_set[1] = 1;
702 break;
703 }
704 case 'b': e->feedback = atoff(arg); break;
705 case 'c': e->chained_osc = atoi(arg); break;
706 /* C available */
707 case 'd': parse_coef_message(arg, e->duty_coefs);break;
708 case 'D': show_debug(atoi(arg)); break;
709 case 'f': parse_coef_message(arg, e->freq_coefs);break;
710 case 'F': parse_coef_message(arg, e->filter_freq_coefs); break;
711 case 'G': e->filter_type = atoi(arg); break;
712 /* g used for Alles for client # */
713 case 'H': parse_list_uint32_t(arg, e->sequence, 3, 0); break;
714 case 'h': if (AMY_HAS_REVERB) {
715 float reverb_params[4];
716 parse_list_float(arg, reverb_params, 4, AMY_UNSET_VALUE(e->reverb_level));
717 e->reverb_level = reverb_params[0];
718 e->reverb_liveness = reverb_params[1];
719 e->reverb_damping = reverb_params[2];
720 e->reverb_xover_hz = reverb_params[3];
721 }
722 break;
723 /* i is used by alles for sync index -- but only for sync messages -- ok to use here but test */
724 case 'i': pos += amy_parse_synth_layer_message(arg, e); break; // Skip over second cmd letter, if any, or entire MIDI CC code string.
725 case 'I': e->ratio = atoff(arg); break;
726 case 'j': e->tempo = atoff(arg); break;
727 /* J available */
728 // chorus.level
729 case 'k': if(AMY_HAS_CHORUS) {
730 float chorus_params[4];
731 parse_list_float(arg, chorus_params, 4, AMY_UNSET_FLOAT);
732 e->chorus_level = chorus_params[0];
733 e->chorus_max_delay = chorus_params[1];
734 e->chorus_lfo_freq = chorus_params[2];
735 e->chorus_depth = chorus_params[3];
736 }
737 break;
738 case 'K': e->patch_number = atoi(arg); break;
739 case 'l': e->velocity=atoff(arg); break;
740 case 'L': e->mod_source=atoi(arg); break;
741 case 'm': e->portamento_ms=atoi(arg); break;
742 case 'M': if (AMY_HAS_ECHO) {
743 float echo_params[5];
744 parse_list_float(arg, echo_params, 5, AMY_UNSET_FLOAT);
745 e->echo_level = echo_params[0];
746 e->echo_delay_ms = echo_params[1];
747 e->echo_max_delay_ms = echo_params[2];
748 e->echo_feedback = echo_params[3];
749 e->echo_filter_coef = echo_params[4];
750 }
751 break;
752 case 'n': e->midi_note=atoff(arg); break;
753 case 'N': e->latency_ms = atoi(arg); break;
754 case 'o': e->algorithm=atoi(arg); break;
755 case 'O': parse_algo_source(arg, e->algo_source); break;
756 case 'p': e->preset=atoi(arg); break;
757 case 'P': e->trigger_phase=atoff(arg); break;
758 /* q unused */
759 case 'Q': parse_coef_message(arg, e->pan_coefs); break;
760 case 'r': parse_voices(arg, e->voices); break;
761 case 'R': e->resonance=atoff(arg); break;
762 case 's': e->pitch_bend = atoff(arg); break;
763 case 'S':
764 e->reset_osc = atoi(arg);
765 // if we're resetting all of AMY, do it now
766 if (e->reset_osc & (RESET_AMY | RESET_TIMEBASE | RESET_EVENTS | RESET_SYNTHS)) {
767 if(e->reset_osc & RESET_AMY) {
768 amy_stop();
769 amy_start(amy_global.config);
770 }
771 // if we're resetting timebase, do it NOW
772 if(e->reset_osc & RESET_TIMEBASE) {
773 amy_reset_sysclock();
774 }
775 if(e->reset_osc & RESET_EVENTS) {
776 amy_deltas_reset();
777 }
778 if(e->reset_osc & RESET_SYNTHS) {
779 amy_reset_oscs();
780 }
781 AMY_UNSET(e->reset_osc);
782 }
783 break;
784 /* t used for time */
785 case 't': e->time=atol(arg); break;
786 case 'T': e->eg_type[0] = atoi(arg); break;
787 case 'u': patches_store_patch(e, arg); pos = strlen(message) - 1; break; // patches_store_patch processes the patch as all the rest of the message and maybe sets patch.
788 /* U used by Alles for sync */
789 case 'v': e->osc=((atoi(arg)) % (AMY_OSCS+1)); break; // allow osc wraparound
790 case 'V': parse_list_float(arg, e->volume, AMY_NUM_BUSES, AMY_UNSET_VALUE(e->volume[0])); break;
791 case 'w': e->wave=atoi(arg); break;
792 /* W used by Tulip for CV, external_channel */
793 case 'X': e->eg_type[1] = atoi(arg); break;
794 case 'x': {
795 float eq[3] = {AMY_UNSET_VALUE(e->eq_l), AMY_UNSET_VALUE(e->eq_m), AMY_UNSET_VALUE(e->eq_h)};
796 parse_list_float(arg, eq, 3, AMY_UNSET_VALUE(e->eq_l));
797 e->eq_l = eq[0];
798 e->eq_m = eq[1];
799 e->eq_h = eq[2];
800 }
801 break;
802 case 'y': e->bus = atoi(arg); break;
803 /* Y still available */
804 case 'z': {
805 pos += amy_parse_transfer_layer_message(arg);
806 break;
807 }
808 /* Z used for end of message */
809 case 'Z':
810 ++pos;
811 goto end;
812 default:
813 break;
814 }
815 }
816 // Skip over arg, line up for the next cmd.
817 ++pos; // move over the current command.
818 if (pos > length) fprintf(stderr, "parse string overrun %d %d %s\n", pos, length, message);
819 pos += _next_alpha(message + pos); // Skip over any non-alpha argument to the current command.
820 }
821 end:
822 // Return exactly how many characters we used.
823 return pos;
824}
