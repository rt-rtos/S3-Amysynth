12float atoff(const char *s) {
13 // Returns float value corresponding to parseable prefix of s.
14 // Unlike atof(), it does not recognize scientific format ('e' or 'E')
15 // and will stop parsing there. Needed for message strings that contain
16 // 'e' as a command prefix.
17 float frac = 0;
18 // Skip leading spaces.
19 while (*s == ' ') ++s;
20 float whole = (float)atoi(s);
21 int is_negative = (s[0] == '-'); // Can't use (whole < 0) because of "-0.xx".
22 //const char *s_in = s; // for debug message.
23 s += strspn(s, "-0123456789");
24 if (*s == '.') {
25 // Float with a decimal part.
26 // Step over dp
27 ++s;
28 // Extract fractional part.
29 int fraclen = strspn(s, "0123456789");
30 char fracpart[8];
31 // atoi() will overflow for values larger than 2^31, so only decode a prefix.
32 if (fraclen > 6) {
33 for(int i = 0; i < 7; ++i) {
34 fracpart[i] = s[i];
35 }
36 fracpart[7] = '\0';
37 s = fracpart;
38 fraclen = 7;
39 }
40 frac = (float)atoi(s);
41 frac /= powf(10.f, (float)fraclen);
42 if (is_negative) frac = -frac;
43 }
44 //fprintf(stderr, "input was %s output is %f + %f = %f\n", s_in, whole, frac, whole+frac);
45 return whole + frac;
46}

73#define PARSE_LIST(type) \
74 int parse_list_##type(char *message, type *vals, int max_num_vals, type skipped_val) { \
75 uint16_t c = 0, last_c; \
76 uint16_t stop = strspn(message, PARSE_LIST_STRSPN1(skipped_val)); \
77 int num_vals_received = 0; \
78 while(c < stop && num_vals_received < max_num_vals) { \
79 *vals = PARSE_LIST_ATO(skipped_val)(message + c); \
80 while (message[c] == ' ') ++c; \
81 last_c = c; \
82 c += strspn(message + c, PARSE_LIST_STRSPN2(skipped_val)); \
83 if (last_c == c) \
84 *vals = skipped_val; \
85 while (message[c] != ',' && message[c] != 0 && c < MAX_MESSAGE_LEN) c++; \
86 ++c; \
87 ++vals; \
88 ++num_vals_received; \
89 } \
90 if (c < stop) { \
91 fprintf(stderr, "WARNING: parse__list_##type: More than %d values in \"%s\"\n", \
92 max_num_vals, message); \
93 } else { /* pad to end */ \
94 for (int i = num_vals_received; i < max_num_vals; ++i) { \
95 *vals++ = skipped_val; \
96 } \
97 } \
98 return num_vals_received; \
99 }

120char *copy_with_trim(char *dest, size_t dest_len, const char *src, size_t src_len) {
121 // Copy a string while trimming leading and trailing spaces.
122 const char *s = src;
123 char *d = dest;
124 size_t d_writ = 0;
125 size_t s_read = 0;
126 size_t trimmed_src_len = src_len;
127 // scan for spaces at end
128 while (trimmed_src_len > 0 && isspace((unsigned char)src[trimmed_src_len - 1])) {
129 --trimmed_src_len;
130 }
131 // skip over leading spaces
132 while (s_read < trimmed_src_len && isspace((unsigned char)*s)) {
133 ++s;
134 ++s_read;
135 }
136 while(s_read < trimmed_src_len && d_writ < (dest_len - 1)) {
137 *d++ = *s++;
138 ++s_read;
139 ++d_writ;
140 }
141 *d = '\0'; // terminator.
142 return (char*) (src + src_len);
143}

196float int_db_to_float_lin(uint32_t db) {
197 // in interp_partials.h, we store amplitudes as integer dB values in range 0..100.
198 float lin = 0;
199 if (AMY_IS_UNSET(db)) return AMY_UNSET_VALUE(lin);
200 lin = powf(10.0f, ((((float)db) - 100.0f) / 20.0f)) - 0.001f;
201 if (lin < 0) return 0;
202 return lin;
203}

220 float vals[2 * MAX_BREAKPOINTS];

231 } else {
232 AMY_UNSET(times_ms[bp_index]);
233 }
234 } else {
235 values[bp_index] = vals[i];
236 }
237 }
238 return num_vals;
239}

262static void parse_event_breakpoints(char *message, uint32_t *times_ms, float *values) {
263 int num_vals = 0;
264 for (int i = 0; i < MAX_BREAKPOINTS; ++i) {
265 AMY_UNSET(times_ms[i]);
266 AMY_UNSET(values[i]);
267 }
268 if (message[0] == '.' && message[1] == '.') {
269 num_vals = parse_breakpoint_event_core_int_db(message + 2, times_ms, values);
270 } else {
271 num_vals = parse_breakpoint_event_core_float_lin(message, times_ms, values);
272 }
273 for (int i = num_vals; i < 2 * MAX_BREAKPOINTS; ++i) {
274 int bp_index = (i >> 1);
275 if (bp_index < MAX_BREAKPOINTS) {
276 if ((i % 2) == 0) AMY_UNSET(times_ms[bp_index]);
277 else AMY_UNSET(values[bp_index]);
278 }
279 }

346int midi_mapping_from_message(char *message, char cmd, int instr_num, int skip_chars) {
347 // MIDI CC mapping ic<C>,<L>,<N>,<X>,<O>,<CODE>, see https://github.com/shorepine/amy/issues/524
348 // ic255 clears all MIDI CC mappings for this synth (short form, no extra fields needed).
349 size_t pos = 0;
350 size_t mlen = strlen(message);
351 // An empty payload ("ic"/"io" at end of message) would return -1 below,
352 // which exactly cancels the outer parser's advance and wedges it in an
353 // infinite loop on the same 'i'.
354 if (mlen == 0) return 0;
355 while (pos < mlen) {
356 // Break the mapping on ZZs (for K257).
357 size_t sub_mlen, next_pos;
358 char *end = strstr(message + pos, "ZZ");
359 if (end != NULL) {
360 sub_mlen = end - (message + pos);
361 next_pos = pos + sub_mlen + 2; // Step over "ZZ"
362 } else {
363 sub_mlen = mlen - pos;
364 next_pos = mlen;
365 }
366 while (sub_mlen > 0 && message[pos + sub_mlen - 1] == 'Z') {
367 --sub_mlen;
368 }
369 // Parse the fragment.
370 int32_t code, is_log;
371 float min_val = 0, max_val = 0, offset_val = 0;
372 int type = (cmd == 'c') ? MIDI_MAP_TYPE_CC : MIDI_MAP_TYPE_NOTE;
373 AMY_UNSET(code);
374 AMY_UNSET(is_log);
375 skip_chars = parse_midi_mapping_payload(message + pos, &code, &is_log, &min_val, &max_val, &offset_val);
376 if (*(message + pos + skip_chars) != ',') {
377 if (AMY_IS_UNSET(code) || AMY_IS_SET(is_log)) {
378 // Either parsing bailed without even a CC code, or it got past the is_log, meaning it wasn't a bare ic<NUM> command.
379 fprintf(stderr, "synth_layer: midi mapping payload didn't parse for %s.\n", message - 1);
380 return pos + skip_chars; // maybe the rest will parse?
381 }
382 // Else we got an incomplete message with a valid CC code - clear it
383 midi_clear_mapping(instr_num, type, code); // (handles 255 as special case).
384 return pos + skip_chars;
385 }
386 ++skip_chars; // step over the "," before the wire string template.
387 midi_store_mapping(instr_num, type, code, is_log, min_val, max_val, offset_val, message + pos + skip_chars, sub_mlen - skip_chars);
388 pos = next_pos;
389 if (pos < mlen && message[pos] != 'i') break;
390 cmd = message[pos + 1];
391 pos += 2;
392 }
393 return pos - 1;
394}

396int parse_cv_trigger_payload(char *message, int32_t *p_gate_cv, float *p_thresh_high, float *p_thresh_low, int32_t *p_pitch_cv, float *p_pitch_scale, float *p_pitch_offset) {
397 char *m = message;
398 m += parse_val_int32_t(m, p_gate_cv);
399 if (m[0] != ',') goto end; else ++m;
400 m += parse_val_float(m, p_thresh_high);
401 if (m[0] != ',') goto end; else ++m;
402 m += parse_val_float(m, p_thresh_low);
403 if (m[0] != ',') goto end; else ++m;

420int cv_trigger_from_message(char *message, int instr_num, int skip_chars) {
421 // i<synth>ig<gate_cv>,<thresh_high>,<thresh_low>,<pitch_cv>,<pitch_scale>,<pitch_offset>,<wire_template>
422 int32_t gate_cv, pitch_cv;
423 uint8_t pitch_cv_uint8;
424 float thresh_high = 0, thresh_low = 0, pitch_scale = 0, pitch_offset = 0;
425 AMY_UNSET(gate_cv);
426 AMY_UNSET(pitch_cv);
427 AMY_UNSET(thresh_high);
428 skip_chars = parse_cv_trigger_payload(message, &gate_cv, &thresh_high, &thresh_low, &pitch_cv, &pitch_scale, &pitch_offset);
429 if (*(message + skip_chars) != ',') {
430 if (AMY_IS_UNSET(gate_cv) || AMY_IS_SET(thresh_high)) {
431 // Either parsing bailed without even a gate CV, or it got past the thresh, meaning it wasn't a bare ic<NUM> command.
432 fprintf(stderr, "cv_trigger: payload didn't parse for %s.\n", message - 1);
433 return skip_chars; // maybe the rest will parse?
434 }
435 // Else we got an incomplete message with a valid gate_cv - clear all triggers for that CV.
436 cv_trigger_clear_mappings(gate_cv);
437 return skip_chars;
438 }
439 ++skip_chars; // step over the "," before the wire string template.
440 if (AMY_IS_SET(pitch_cv))
441 pitch_cv_uint8 = pitch_cv;
442 else
443 AMY_UNSET(pitch_cv_uint8);
444 cv_trigger_new(gate_cv, thresh_high, thresh_low,
445 pitch_cv_uint8, pitch_scale, pitch_offset, message + skip_chars);
446 // Consume rest of message but leave the trailing 'Z' for the outer parser.
447 int remainder = strlen(message);
448 if (remainder > 0 && message[remainder - 1] == 'Z') remainder--;
449 skip_chars = remainder;
450 return skip_chars;
451}

453// Parser for synth-layer ('i') prefix.
454int amy_parse_synth_layer_message(char *message, amy_event *e) {
455 int skip_chars = 1; // default is to skip one extra char.
456 if (message[0] >= '0' && message[0] <= '9') {
457 // It's just the instrument number.
458 e->synth = atoi(message);
459 return 0; // no extra skip.
460 }
461 char cmd = message[0];
462 message++;
463 if (cmd == 'd') e->synth_delay_ms = atoi(message);
464 else if (cmd == 'f') e->synth_flags = atoi(message);
465 else if (cmd == 'g') skip_chars = cv_trigger_from_message(message, e->synth, skip_chars);
466 else if (cmd == 'm') e->grab_midi_notes = atoi(message);
467 else if (cmd == 'M') e->note_source_channel = atoi(message); // To mark MIDI-in notes.
468 else if (cmd == 'n') e->oscs_per_voice = atoi(message);
469 else if (cmd == 'p') e->pedal = atoi(message);
470 else if (cmd == 't') e->to_synth = atoi(message);
471 else if (cmd == 'v') e->num_voices = atoi(message);
472 else if (cmd == 'V') e->synth_level = atoff(message); // Per-instrument level, default 1.
473 else if (cmd == 'y') e->bus = atoi(message); // 'i1iy1' is the same as 'i1y1'.
474 else if (cmd == 'c' || cmd == 'o') skip_chars = midi_mapping_from_message(message, cmd, e->synth, skip_chars);
475 else fprintf(stderr, "Unrecognized synth-level command '%s'\n", message - 1);
476 return skip_chars;
477}

479// Parser for the 'G' prefix: a digit is filter_type as ever; a letter is a
480// distortion sub-command. GC<v> and GF<v> enable clip and fold (0 turns the
481// stage off), GH<bits>[,<rate>] enables the bitcrusher (GH0 turns it off),
482// GD<coefs> and GM<coefs> carry the drive and wet/dry coef vectors shared by
483// every stage - a single value sets just the constant term, so scalar use
484// reads as before. Stages are independent: enabled stages stack in
485// clip -> fold -> crush order, and each command touches only its own stage.
486// The commands say what to do, not where: an event carrying a 'v' shapes that
487// osc, one without shapes the bus the event addresses ('y', else the synth's
488// bus, else the default). At bus scope only the constant term of GD/GM is
489// used, since a bus sum has no per-note modulation sources to combine.
490int amy_parse_dist_layer_message(char *message, amy_event *e) {
491 if (message[0] >= '0' && message[0] <= '9') {
492 // It's just the filter type.
493 e->filter_type = atoi(message);
494 return 0; // no extra skip.
495 }
496 char cmd = message[0];
497 message++;
498 if (cmd == 'C') e->dist_clip = (atoff(message) != 0);
499 else if (cmd == 'F') e->dist_fold = (atoff(message) != 0);
500 else if (cmd == 'H') {
501 uint16_t vals[2];
502 parse_list_uint16_t(message, vals, 2, AMY_UNSET_VALUE(vals[0]));
503 if (vals[0] == 0) {
504 e->dist_crush = 0;
505 } else {
506 e->dist_crush = 1;
507 if (AMY_IS_SET(vals[0])) e->dist_bits = (uint8_t)MIN(vals[0], 24);
508 if (AMY_IS_SET(vals[1])) e->dist_rate = vals[1];
509 }
510 }
511 else if (cmd == 'D') parse_coef_message(message, e->dist_drive_coefs);
512 else if (cmd == 'M') parse_coef_message(message, e->dist_mix_coefs);
513 else fprintf(stderr, "Unrecognized distortion command '%s'\n", message - 1);
514 return 1; // skip the sub-command letter.
515}

517// Parse a sample-load parameter list ('z'/'zS' messages): comma-separated
518// unsigned integers, except the midinote field which may be fractional (e.g.
519// a sample tuned 4 cents sharp of C4 is "60.04"). parse_list_uint32_t cannot
520// parse these lists: its leading charset scan treats the list as ending at
521// the first '.', which would silently zero every field after a fractional
522// midinote. Absent fields parse as 0.
523static void parse_sample_load_params(char *message, uint32_t *vals, int num_vals,
524 int midinote_field, float *midinote) {
525 *midinote = 0;
526 uint16_t c = 0;
527 for (int f = 0; f < num_vals; ++f) {
528 vals[f] = (uint32_t)strtoul(message + c, NULL, 10);
529 if (f == midinote_field) *midinote = atoff(message + c);
530 while (message[c] != ',' && message[c] != 0 && c < MAX_MESSAGE_LEN) c++;
531 if (message[c] == ',') c++;
532 }

535// Parser for transfer-layer ('z') prefix. Returns how much of a message to skip
536uint16_t amy_parse_transfer_layer_message(char *message) {
537
538 if (message[0] >= '0' && message[0] <= '9') {
539 // z: Signal to start loading sample.
540 // Params: preset number, length(frames), samplerate, midinote, loopstart, loopend.
541 uint32_t sm[6]; // preset, length, SR, midinote, loop_start, loopend
542 float midinote;
543 parse_sample_load_params(message, sm, 6, 3, &midinote);
544 if(sm[1]==0) { // remove preset
545 pcm_unload_preset(sm[0]);
546 } else {
547 amy_execute_deltas();
548 int16_t * ram = pcm_load(sm[0], sm[1], sm[2], 1, midinote, sm[4], sm[5]);
549 start_receiving_transfer(sm[1]*2, (uint8_t*)ram);
550 }
551 return 0;
552 }
553 char cmd = message[0];
554 message++;
555 if (cmd == 'T') {
556 // zT: Signal to start loading file.
557 //Params: Destination name, file size.
558 uint32_t file_size = 0;
559 char filename[MAX_FILENAME_LEN];
560 uint16_t len = parse_list_file_transfer_params(message, filename, sizeof(filename), &file_size);
561 if (filename[0] != '\0') {
562 sequencer_midi_stop(); // Stop sequencer (and sketch loop) during file transfer.
563 start_receiving_file_transfer(file_size, filename);
564 }
565 return len;
566 }
567 else if (cmd == 'F') {
568 // zF: setup PCM preset from WAV filename on disk.
569 // Params: Preset number, filename, midi note
570
571 uint32_t preset = 0;
572 uint32_t midinote = 0;
573 char filename[MAX_FILENAME_LEN];
574 uint16_t len = parse_list_file_params(message, &preset, filename, sizeof(filename),
575 &midinote);
576 if (filename[0] != '\0') {
577 amy_global.transfer_stored_bytes = midinote;
578 strncpy(amy_global.transfer_filename, filename, MAX_FILENAME_LEN);
579 amy_global.transfer_file_handle = preset;
580 // For tulip/amyboard we have to load the PCM file from the MP "task"
581 #if (defined AMYBOARD) || (defined TULIP)
582 mp_sched_schedule(MP_OBJ_FROM_PTR(&tulip_pcm_load_file_obj), mp_const_none);
583 #else
584 pcm_load_file();
585 #endif
586
587 }
588 return len;
589 }
590 else if (cmd == 'S') {
591 // zS: sample from BUS[1] to a memorypcm patch.
592 // Params: Preset number, bus, max length in frames,midinote,loopstart,loopend
593 uint32_t sm[6]; // preset, bus, max frames, midinote, loop_start, loopend
594 float midinote;
595 parse_sample_load_params(message, sm, 6, 3, &midinote);
596 int16_t * ram = pcm_load(sm[0], sm[2], AMY_SAMPLE_RATE, 2, midinote, sm[4], sm[5]);
597 start_receiving_sample(sm[2], sm[1], ram);
598 return 1;
599 }
600 else if (cmd == 'O') {
601 //zO: stop sampling from any bus
602 stop_receiving_sample();
603 return 1;
604 }
605 else if (cmd == 'D') {
606 // zD: Dump data over MIDI sysex.
607 // zD[Z] — dump all active instrument state + global effects.
608 // zD<filename>[Z] — dump file contents from the filesystem.
609 // The filename/payload is "rest of message" — we consume everything
610 // to the end of the C string. A trailing 'Z' (end-of-message marker
611 // some senders append) is stripped, so interior capital-Z characters
612 // in the filename (e.g. "/ZIPFILE.py") are preserved. Limitation:
613 // filenames whose last char is 'Z' are not addressable.
614 char filename[MAX_FILENAME_LEN];
615 uint16_t len = 0;
616 while (message[len] && len < MAX_FILENAME_LEN - 1) {
617 filename[len] = message[len];
618 len++;
619 }
620 filename[len] = '\0';
621 if (len > 0 && filename[len - 1] == 'Z') {
622 filename[--len] = '\0';
623 }
624 if (filename[0] == '\0') {
625 amy_dump_state_to_sysex();
626 } else {
627 amy_dump_file_to_sysex(filename);
628 }
629 // Consume the whole rest of the message so the outer parser exits.
630 {
631 uint16_t total = 0;
632 const char *scan = message - 1; // back to 'D'
633 while (scan[total]) total++;
634 return total;
635 }
636 }
637 else if (cmd == 'P') {
638 // zP: Execute Python code on host (e.g. zPimport amyboard; amyboard.restart_sketch()Z).
639 // Payload semantics match zD: the code string is "rest of message",
640 // a trailing 'Z' terminator is stripped, and interior capital-Z chars
641 // in the code are preserved.
642 char code[256];
643 uint16_t len = 0;
644 while (message[len] && len < sizeof(code) - 1) {
645 code[len] = message[len];
646 len++;
647 }
648 code[len] = '\0';
649 if (len > 0 && code[len - 1] == 'Z') {
650 code[--len] = '\0';
651 }
652 if (amy_global.config.amy_external_exec_hook) {
653 amy_global.config.amy_external_exec_hook(code);
654 }
655 {
656 uint16_t total = 0;
657 const char *scan = message - 1;
658 while (scan[total]) total++;
659 return total;
660 }
661 }
662 else if (cmd == 'Y') {
663 // zY: sequencer transport. zY1 starts the sequencer, zY0 stops it. Lets a
664 // host drive playback without MIDI clock sync (see external_midi_sync).
665 if (atoi(message)) sequencer_midi_start();
666 else sequencer_midi_stop();
667 return 1;
668 }
669 else if (cmd == 'C') {
670 // zC: external MIDI clock sync. zC1 makes the sequencer follow incoming
671 // MIDI realtime clock/start/stop (0xF8/0xFA/0xFC); zC2 makes AMY the
672 // clock master, sending those messages out (0xF8 at 24 PPQ from the
673 // internal tempo, 0xFA/0xFC on transport start/stop); zC0 (the default)
674 // neither follows nor sends and uses the internal clock. Same switch as
675 // the C API's amy_external_midi_sync(), reachable over the wire so hosts
676 // that only speak the wire protocol (web builds, sysex) can flip it.
677 amy_external_midi_sync((uint8_t)atoi(message));
678 return 1;
679 }
680 else fprintf(stderr, "Unrecognized transfer-level command '%s'\n", message - 1);
681 return 0;
682}

685int _next_alpha(char *s) {
686 // Return how many chars to skip to get to the next alphabet (command prefix) (or EOS).
687 int p = 0;

689 char c = *(s + p);
690 if (isalpha(c)) break;
691 ++p;
692 }
693 return p;

707// Called from amy_add_message when the first char is 'H', indicating a ticks message.
708// It claims the rest of the message as its payload -- stored as a raw
709// wire string and only parsed when it comes due -- so a schedule command
710// is only ever honored as the first command of a message.
711void handle_ticks_message(char *message) {
712 assert(message[0] == 'H');
713 uint32_t ticks[3] = {0, 0, 0};
714 int num_vals = parse_list_uint32_t(message + 1, ticks, 3, 0);
715 uint16_t schedule_len = 1 + _next_alpha(message + 1);
716 char *payload = message + schedule_len;
717 uint16_t payload_len = (uint16_t)strlen(payload);
718 char *stripped = (char *)malloc_caps(payload_len + 1, amy_global.config.ram_caps_sequencer);
719 // LOCAL EDIT (S3-Amysynth): sequencer caps + events-pool fallback; see
720 // the matching site in amy_add_event (api.c).
721 if (stripped == NULL)
722 stripped = (char *)malloc_caps(payload_len + 1, amy_global.config.ram_caps_events);
723 if (stripped == NULL) {
724 amy_oom("ticks_message");
725 } else {
726 memcpy(stripped, payload, payload_len + 1);
727 // A tag is only "given" if all 3 values were present; fewer
728 // than that (a 1- or 2-value ticks=) stores anonymously.
729 sequencer_add_wire(ticks[TICKS_TICK], ticks[TICKS_PERIOD], ticks[TICKS_TAG],
730 num_vals >= 3, stripped);

734// given a string return a parsed event
735//
736// Transfer payloads never reach here: amy_add_message() traps them before
737// any parsing is attempted (see the comment there), so this only ever sees
738// real wire commands.
739int amy_parse_message(char * message, amy_event *e) {
740 peek_stack("parse_message");
741 int length = strlen(message);
742 char cmd = '\0';
743 uint16_t pos = 0;
744
745 while(pos < length) {
746 cmd = message[pos];
747 char *arg = message + pos + 1;
748 if(isalpha(cmd)) {
749 switch(cmd) {
750 case 'a': parse_coef_message(arg, e->amp_coefs);break;
751 case 'A': {
752 char bp_msg[MAX_PARAM_LEN];
753 copy_param_list_substring(bp_msg, arg);
754 parse_event_breakpoints(bp_msg, e->eg0_times, e->eg0_values);
755 e->bp_is_set[0] = 1;
756 break;
757 }
758 case 'B': {
759 char bp_msg[MAX_PARAM_LEN];
760 copy_param_list_substring(bp_msg, arg);
761 parse_event_breakpoints(bp_msg, e->eg1_times, e->eg1_values);
762 e->bp_is_set[1] = 1;
763 break;
764 }
765 case 'b': e->feedback = atoff(arg); break;
766 case 'c': e->chained_osc = atoi(arg); break;
767 /* C available */
768 case 'd': parse_coef_message(arg, e->duty_coefs);break;
769 case 'D': show_debug(atoi(arg)); break;
770 case 'f': parse_coef_message(arg, e->freq_coefs);break;
771 case 'F': parse_coef_message(arg, e->filter_freq_coefs); break;
772 case 'G': pos += amy_parse_dist_layer_message(arg, e); break; // Skip over second cmd letter, if any.
773 /* g used for Alles for client # */
774 // 'H' is the ticks= schedule command, it's caught in amy_add_message before this.
775 //case 'H': parse_list_uint32_t(arg, e->ticks, 3, 0); break;
776 case 'h': if (AMY_HAS_REVERB) {
777 float reverb_params[4];
778 parse_list_float(arg, reverb_params, 4, AMY_UNSET_VALUE(e->reverb_level));
779 e->reverb_level = reverb_params[0];
780 e->reverb_liveness = reverb_params[1];
781 e->reverb_damping = reverb_params[2];
782 e->reverb_xover_hz = reverb_params[3];
783 }
784 break;
785 /* i is used by alles for sync index -- but only for sync messages -- ok to use here but test */
786 case 'i': pos += amy_parse_synth_layer_message(arg, e); break; // Skip over second cmd letter, if any, or entire MIDI CC code string.
787 case 'I': e->ratio = atoff(arg); break;
788 case 'j': e->tempo = atoff(arg); break;
789 /* J available */
790 // chorus.level
791 case 'k': if(AMY_HAS_CHORUS) {
792 float chorus_params[4];
793 parse_list_float(arg, chorus_params, 4, AMY_UNSET_FLOAT);
794 e->chorus_level = chorus_params[0];
795 e->chorus_max_delay = chorus_params[1];
796 e->chorus_lfo_freq = chorus_params[2];
797 e->chorus_depth = chorus_params[3];
798 }
799 break;
800 case 'K': e->patch_number = atoi(arg); break;
801 case 'l': e->velocity=atoff(arg); break;
802 case 'L': parse_mod_source(arg, e->mod_source); break;
803 case 'm': e->portamento_ms=atoi(arg); break;
804 case 'M': if (AMY_HAS_ECHO) {
805 float echo_params[5];
806 parse_list_float(arg, echo_params, 5, AMY_UNSET_FLOAT);
807 e->echo_level = echo_params[0];
808 e->echo_delay_ms = echo_params[1];
809 e->echo_max_delay_ms = echo_params[2];
810 e->echo_feedback = echo_params[3];
811 e->echo_filter_coef = echo_params[4];
812 }
813 break;
814 case 'n': e->midi_note=atoff(arg); break;
815 case 'N': e->latency_ms = atoi(arg); break;
816 case 'o': e->algorithm=atoi(arg); break;
817 case 'O': parse_algo_source(arg, e->algo_source); break;
818 case 'p':
819 // 'p' is the preset/sampler layer: a bare number is the preset,
820 // a sub-letter addresses a PCM param. Sampler params live here
821 // rather than at the top level because single letters are nearly
822 // exhausted (44 of 52 allocated) and this corner keeps growing.
823 if (arg[0] == 'o') { // 'po' is PCM sample_offset.
824 e->sample_offset = atoi(arg + 1);
825 ++pos;
826 } else if (arg[0] == 'F') { // 'pF' is PCM fit (ticks).
827 e->fit_ticks = atoff(arg + 1);
828 ++pos;
829 } else if (arg[0] == 'S') { // 'pS' is PCM fit grain search half-width.
830 e->fit_search = atoi(arg + 1);
831 ++pos;
832 } else {
833 e->preset = atoi(arg);
834 }
835 break;
836 case 'P': e->trigger_phase=atoff(arg); break;
837 /* q unused */
838 case 'Q': parse_coef_message(arg, e->pan_coefs); break;
839 //case 'r': parse_voices(arg, e->voices); break; // 'r' deprecated, you basically never control a voice directly from the API. Planning to use it for multi-amyboard.
840 case 'R': e->resonance=atoff(arg); break;
841 case 's': e->pitch_bend = atoff(arg); break;
842 case 'S':
843 e->reset_osc = atoi(arg);
844 // These two can only happen here, on the parse side, because
845 // neither survives being carried IN a delta: RESET_AMY tears
846 // AMY down and restarts it, and RESET_EVENTS empties the very
847 // queue the delta would be sitting in. Every other reset bit
848 // -- RESET_TIMEBASE included -- travels as an ordinary delta,
849 // so it works identically from amy_add_event() and honours
850 // time=/ticks= like the rest of the API.
851 if (e->reset_osc & (RESET_AMY | RESET_EVENTS)) {
852 if(e->reset_osc & RESET_AMY) {
853 amy_stop();
854 amy_start(amy_global.config);
855 }
856 if(e->reset_osc & RESET_EVENTS) {
857 amy_deltas_reset();
858 }
859 // Clear only the bits handled here. Unsetting the whole
860 // field dropped everything it was combined with, so e.g.
861 // RESET_EVENTS|RESET_ALL_OSCS silently skipped the osc
862 // reset. Unset it entirely if nothing is left, since a
863 // reset_osc of 0 means "reset oscillator 0".
864 e->reset_osc &= ~(uint32_t)(RESET_AMY | RESET_EVENTS);
865 if (e->reset_osc == 0) AMY_UNSET(e->reset_osc);
866 }
867 break;
868 /* t no longer used (was time=) */
869 case 'T': e->eg_type[0] = atoi(arg); break;
870 case 'u': patches_store_patch(e, arg); pos = strlen(message) - 1; break; // patches_store_patch processes the patch as all the rest of the message and maybe sets patch.
871 /* U used by Alles for sync */
872 case 'v': e->osc=((atoi(arg)) % (AMY_OSCS+1)); break; // allow osc wraparound
873 case 'V': e->volume = atoff(arg); break;
874 case 'w': if (arg[0] == 'w') { // 'ww' is wave submode.
875 e->mode=atoi(arg + 1);
876 ++pos;
877 } else {
878 e->wave=atoi(arg);
879 }
880 break;
881 /* W used by Tulip for CV, external_channel */
882 case 'X': e->eg_type[1] = atoi(arg); break;
883 case 'x': {
884 float eq[3] = {AMY_UNSET_VALUE(e->eq_l), AMY_UNSET_VALUE(e->eq_m), AMY_UNSET_VALUE(e->eq_h)};
885 parse_list_float(arg, eq, 3, AMY_UNSET_VALUE(e->eq_l));
886 e->eq_l = eq[0];
887 e->eq_m = eq[1];
888 e->eq_h = eq[2];
889 }
890 break;
891 case 'y': e->bus = atoi(arg); break;
892 case 'z': {
893 pos += amy_parse_transfer_layer_message(arg);
894 break;
895 }
896 /* Z used for end of message */
897 case 'Z':
898 ++pos;
899 goto end;
900 default:
901 break;
902 }
903 }
904 // Skip over arg, line up for the next cmd.
905 ++pos; // move over the current command.
906 if (pos > length) fprintf(stderr, "parse string overrun %d %d %s\n", pos, length, message);
907 pos += _next_alpha(message + pos); // Skip over any non-alpha argument to the current command.
908 }
909 end:
910 // Return exactly how many characters we used.
911 return pos;
912}
