34void patches_init(int max_memory_patches) {
35 max_num_memory_patches = max_memory_patches;
36 uint8_t *alloc_base = malloc_caps(
37 max_num_memory_patches * sizeof(struct delta *)
38 + max_num_memory_patches * sizeof(uint16_t)

40 + amy_global.config.max_voices * sizeof(uint16_t)
41 + max_num_memory_patches * sizeof(uint8_t),

43 );
44 memory_patch_deltas = (struct delta **)alloc_base;
45 memory_patch_oscs = (uint16_t *)(memory_patch_deltas + max_num_memory_patches);
46 osc_to_voice = (uint16_t *)(memory_patch_oscs + max_num_memory_patches);
47 voice_to_base_osc = (uint16_t *)(osc_to_voice + AMY_OSCS);
48 memory_patch_auto = (uint8_t *)(voice_to_base_osc + amy_global.config.max_voices);
49 bzero(memory_patch_deltas, max_num_memory_patches * sizeof(struct delta *));
50 bzero(memory_patch_auto, max_num_memory_patches * sizeof(uint8_t));
51 patches_reset();
52}

54void patches_reset_patch(int patch_number) {
55 int patch_index = patch_number - _PATCHES_FIRST_USER_PATCH;
56 if (patch_index < 0 || patch_index >= (int)max_num_memory_patches) {
57 fprintf(stderr, "reset patch number %" PRId32 " is out of range (%" PRId32 " .. %" PRId32 ")\n",
58 (int32_t)(patch_index + _PATCHES_FIRST_USER_PATCH),
59 (int32_t)_PATCHES_FIRST_USER_PATCH,
60 (int32_t)(_PATCHES_FIRST_USER_PATCH + (int32_t)max_num_memory_patches));
61 return;
62 }
63 if (memory_patch_deltas[patch_index] != NULL) delta_release_list(memory_patch_deltas[patch_index]);
64 memory_patch_deltas[patch_index] = NULL;
65 memory_patch_oscs[patch_index] = 0;
66 memory_patch_auto[patch_index] = 0;
67}

82// LOCAL EDIT (S3-Amysynth, 2026-07-28): read accessor for voice_to_base_osc.
83// The table is a plain global with no declaration in any header, so an embedder
84// that needs a voice's oscillators - to display live per-voice engine state, for
85// instance - has no supported way to ask for it. Read-only, and checked for both

88 if (voice_to_base_osc == NULL || voice >= amy_global.config.max_voices)
89 return false;

91 return false;
92 if (base_osc != NULL)
93 *base_osc = voice_to_base_osc[voice];
94 return true;
95}

161void update_num_oscs_for_patch_number(int patch_number) {
162 int patch_index = patch_number - _PATCHES_FIRST_USER_PATCH;
163 if (patch_index < 0 || patch_index >= (int)max_num_memory_patches) {

173 if (d->osc >= num_oscs) num_oscs = d->osc + 1;
174 d = d->next;
175 }
176 memory_patch_oscs[patch_index] = num_oscs;
177}

189void add_deltas_to_queue_with_baseosc(struct delta *d, int base_osc, uint16_t oscs_per_voice, struct delta **queue, uint32_t time) {
190 //fprintf(stderr, "add_deltas_to_queue_with_baseosc: added %d baseosc %d time %d\n", delta_list_len(d), base_osc, time);
191 struct delta d_offset;
192 while(d) {
193 d_offset = *d;
194 // These deltas are voice-relative, so they answer to the same bound as
195 // a live event does (see osc_ref_within_voice): a stored patch that
196 // names more oscs than the voice it is being replayed into has, or
197 // points one of them outside it, must not write into the neighbours.
198 if (!osc_ref_within_voice((int)d_offset.osc, oscs_per_voice, "stored")) {
199 d = d->next;
200 continue;
201 }
202 d_offset.osc += base_osc;
203 if (d_offset.param == CHAINED_OSC || d_offset.param == RESET_OSC
204 || (d_offset.param >= MOD_SOURCE_START && d_offset.param < MOD_SOURCE_END)
205 || (d_offset.param >= ALGO_SOURCE_START && d_offset.param < ALGO_SOURCE_START + MAX_ALGO_OPS))
206 if (!(AMY_IS_UNSET((int16_t)d_offset.data.i) || AMY_IS_UNSET((uint16_t)d_offset.data.i))) { // CHAINED_OSC is uint16_t, but ALGO_SOURCE is int16_t.
207 // RESET_OSC's payload is a mask of RESET_* bits as often as it
208 // is an osc number, so it is not range-checked (as in
209 // EVENT_TO_DELTA_WITH_BASEOSC).
210 if (d_offset.param != RESET_OSC &&
211 !osc_ref_within_voice((int)d_offset.data.i, oscs_per_voice, "stored reference")) {
212 d = d->next;
213 continue;
214 }
215 d_offset.data.i += base_osc;
216 }
217 d_offset.time = time;
218 // assume the d->time is 0 and that's good.
219 add_delta_to_queue(&d_offset, &amy_global.delta_queue);
220 d = d->next;
221 }
222}

224void snprintcodefloat(char *s, size_t max_len, char *code, float val) {
225 // Skip trailing zeros (and decimal point) when printing a float.
226 float testval = val;
227 char fmt[7]; // = "%s%.0f";
228 fmt[0] = '%';
229 fmt[1] = 's';
230 fmt[2] = '%';
231 fmt[3] = '.';
232 fmt[4] = '0';
233 fmt[5] = 'f';
234 fmt[6] = 0;
235 for (int dp = 0; dp < 3; ++dp) {
236 if (testval == roundf(testval)) {
237 snprintf(s, max_len, fmt, code, val);
238 return;
239 }
240 testval *= 10.0f;
241 ++fmt[4]; // increase the DP digit
242 }
243 snprintf(s, max_len, fmt, code, val);
244}

613struct delta *deltas_to_event(struct delta *queue, struct amy_event *event) {
614 // Consume deltas from queue and push into event. Return pointer to first non-consumed delta.
615 if (queue == NULL) return NULL;
616 event->osc = queue->osc;
617 event->time = queue->time;
618 uint32_t breakpoint_times[MAX_BREAKPOINT_SETS][MAX_BREAKPOINTS];
619 float breakpoint_values[MAX_BREAKPOINT_SETS][MAX_BREAKPOINTS];
620 for (int i = 0; i < MAX_BREAKPOINT_SETS; ++i) {
621 for (int j = 0; j < MAX_BREAKPOINTS; ++j) {
622 AMY_UNSET(breakpoint_times[i][j]);
623 AMY_UNSET(breakpoint_values[i][j]);
624 }
625 }
626 int highest_breakpoint[MAX_BREAKPOINT_SETS] = {-1, -1};
627 while(queue != NULL) {
628 if (queue->osc != event->osc || queue->time != event->time) break; // delta doesn't fit this event.
629 switch (queue->param) {
630 _CASE_I(wave, WAVE)
631 _CASE_I(preset, PRESET)
632 _CASE_F(midi_note, MIDI_NOTE)
633 _CASE_F(feedback, FEEDBACK)
634 _CASE_F(trigger_phase, PHASE)
635 _CASE_I(sample_offset, SAMPLE_OFFSET)
636 _CASE_F(fit_ticks, FIT)
637 _CASE_I(fit_search, FIT_SEARCH)
638 _CASE_F(pitch_bend, PITCH_BEND)
639 _CASE_I(latency_ms, LATENCY)
640 _CASE_F(tempo, TEMPO)
641 _CASE_LOG(ratio, RATIO)
642 _CASE_F(resonance, RESONANCE)
643 _CASE_I(portamento_ms, PORTAMENTO)
644 _CASE_I(chained_osc, CHAINED_OSC)
645 _CASE_I(reset_osc, RESET_OSC)
646 _CASE_I(note_source_channel, NOTE_SOURCE_CHANNEL)
647 _CASE_I(filter_type, FILTER_TYPE)
648 _CASE_I(dist_clip, DIST_CLIP_EN)
649 _CASE_I(dist_fold, DIST_FOLD_EN)
650 _CASE_I(dist_crush, DIST_CRUSH_EN)
651 _CASE_I(dist_bits, DIST_BITS)
652 _CASE_I(dist_rate, DIST_RATE)
653 _CASE_I(algorithm, ALGORITHM)
654 _CASE_F(eq_l, EQ_L)
655 _CASE_F(eq_m, EQ_M)
656 _CASE_F(eq_h, EQ_H)
657 _CASE_F(echo_max_delay_ms, ECHO_MAX_DELAY_MS)
658 _CASE_F(echo_level, ECHO_LEVEL)
659 _CASE_F(echo_delay_ms, ECHO_DELAY_MS)
660 _CASE_F(echo_feedback, ECHO_FEEDBACK)
661 _CASE_F(echo_filter_coef, ECHO_FILTER_COEF)
662 _CASE_F(chorus_max_delay, CHORUS_MAX_DELAY)
663 _CASE_F(chorus_level, CHORUS_LEVEL)
664 _CASE_F(chorus_lfo_freq, CHORUS_LFO_FREQ)
665 _CASE_F(chorus_depth, CHORUS_DEPTH)
666 _CASE_F(reverb_level, REVERB_LEVEL)
667 _CASE_F(reverb_liveness, REVERB_LIVENESS)
668 _CASE_F(reverb_damping, REVERB_DAMPING)
669 _CASE_F(reverb_xover_hz, REVERB_XOVER_HZ)
670 // Bus distortion comes back through the same event fields the per-osc
671 // stage uses; the event's own osc says which scope it will be read at
672 // on the way back in, exactly as it does for VOLUME below.
673 _CASE_I(dist_clip, BUS_DIST_CLIP_EN)
674 _CASE_I(dist_fold, BUS_DIST_FOLD_EN)
675 _CASE_I(dist_crush, BUS_DIST_CRUSH_EN)
676 _CASE_I(dist_bits, BUS_DIST_BITS)
677 _CASE_I(dist_rate, BUS_DIST_RATE)
678 case BUS_DIST_DRIVE: event->dist_drive_coefs[COEF_CONST] = queue->data.f; break;
679 case BUS_DIST_MIX: event->dist_mix_coefs[COEF_CONST] = queue->data.f; break;
680 _CASE_I(eg_type[0], EG0_TYPE)
681 _CASE_I(eg_type[1], EG1_TYPE)
682 _CASE_F(velocity, VELOCITY)
683 _CASE_I(bus, BUS)
684 // A VOLUME delta names its bus in osc, like the other bus-directed
685 // params, so it comes back as "this bus, one volume".
686 case VOLUME: event->bus = queue->osc; event->volume = queue->data.f; break;
687 default: // blocks, not handled by case
688 _TEST_COEFS(amp_coefs, AMP)
689 _TEST_FREQ_COEFS(freq_coefs, FREQ)
690 _TEST_FREQ_COEFS(filter_freq_coefs, FILTER_FREQ)
691 _TEST_COEFS(duty_coefs, DUTY)
692 _TEST_COEFS(pan_coefs, PAN)
693 _TEST_DRIVE_COEFS(dist_drive_coefs, DIST_LOGDRIVE)
694 _TEST_COEFS(dist_mix_coefs, DIST_MIX)
695 for (int i = 0; i < MAX_ALGO_OPS; ++i) {
696 if ((int)queue->param == (int)ALGO_SOURCE_START + i)
697 event->algo_source[i] = queue->data.i;
698 }
699 for (int i = 0; i < NUM_MOD_SOURCES; ++i) {
700 if ((int)queue->param == (int)MOD_SOURCE_START + i)
701 event->mod_source[i] = queue->data.i;
702 }
703 for (int i = 0; i < MAX_BREAKPOINT_SETS; ++i) {
704 for (int j = 0; j < MAX_BREAKPOINTS; ++j) {
705 if ((int)queue->param == (int)BP_START + (j * 2) + (i * MAX_BREAKPOINTS * 2)) {
706 //event->bp[i].time[j] = queue->data.i;
707 breakpoint_times[i][j] = queue->data.i;
708 if (j > highest_breakpoint[i]) highest_breakpoint[i] = j;
709 }
710 else if ((int)queue->param == (int)BP_START + (j * 2 + 1) + (i * MAX_BREAKPOINTS * 2)) {
711 //event->bp[i].value[j] = queue->data.f;
712 breakpoint_values[i][j] = queue->data.f;
713 if (j > highest_breakpoint[i]) highest_breakpoint[i] = j;
714 }
715 }
716 }
717 break;
718 }
719 queue = queue->next;
720 }
721 uint32_t *bp_times_ms[MAX_BREAKPOINT_SETS] = {event->eg0_times, event->eg1_times};
722 float *bp_values[MAX_BREAKPOINT_SETS] = {event->eg0_values, event->eg1_values};
723 for (int i = 0; i < MAX_BREAKPOINT_SETS; ++i) {
724 if (highest_breakpoint[i] >= 0) {
725 event->bp_is_set[i] = 1;
726 for (int j = 0; j <= highest_breakpoint[i]; ++j) {
727 if (AMY_IS_SET(breakpoint_times[i][j])) {
728 float t_ms = ((float)breakpoint_times[i][j]) * 1000.0f / (float)AMY_SAMPLE_RATE;
729 int32_t t_rounded = (int32_t)roundf(t_ms);
730 if (t_rounded < 0) t_rounded = 0;
731 if (t_rounded >= SHRT_MAX) t_rounded = SHRT_MAX - 1;
732 bp_times_ms[i][j] = (int16_t)t_rounded;
733 }
734 if (AMY_IS_SET(breakpoint_values[i][j])) {
735 bp_values[i][j] = breakpoint_values[i][j];
736 }
737 }
738 }
739 }
740 return queue;
741}

799void set_event_for_osc(int base_osc, int rel_osc, struct amy_event *event) {
800 // Set fields in the event to configure the osc away from default.
801 // We assume event has already been cleared.
802 // We do not set the osc field of the event.
803 int osc = base_osc + rel_osc;
804 // synth[] entries are NULL until ensure_osc_allocd() allocates them, and a
805 // voice can own oscs that were never touched -- a synth configured with
806 // oscs_per_voice= and no patch reserves the oscs without allocating any of
807 // them. Reading such an osc's config is not an error: it is at its
808 // defaults, so there is nothing to emit. Dereferencing it, which is what
809 // this did, segfaulted amy_get_synth_commands()/amy_dump_state() for every
810 // such synth.
811 if (osc < 0 || osc >= AMY_OSCS || synth[osc] == NULL) return;
812 // Generate the reference "empty synth".
813 struct synthinfo empty_synth;
814 // We need to have space for the breakpoints.
815 uint32_t times[MAX_BREAKPOINT_SETS * MAX_BREAKPOINTS];
816 float values[MAX_BREAKPOINT_SETS * MAX_BREAKPOINTS];
817 for (int i = 0; i < MAX_BREAKPOINT_SETS; ++i) {
818 empty_synth.max_num_breakpoints[i] = synth[osc]->max_num_breakpoints[i];
819 empty_synth.breakpoint_times[i] = times + i * MAX_BREAKPOINTS;
820 empty_synth.breakpoint_values[i] = values + i * MAX_BREAKPOINTS;
821 }
822 reset_osc_by_pointer(&empty_synth, /* msynth */ NULL);
823 // Go through parameter fields picking out the ones that are nondefault.
824 EVENT_FROM_OSC(wave);
825 EVENT_FROM_OSC(preset);
826 // Note and velocity are special for "note on" events, don't reflect them in config.
827 //EVENT_FROM_OSC(midi_note);
828 //EVENT_FROM_OSC(velocity);
829 EVENT_FROM_OSC_ARRAY(amp_coefs, NUM_COMBO_COEFS);
830 EVENT_FROM_OSC_ARRAY_FREQ(logfreq_coefs, freq_coefs, NUM_COMBO_COEFS);
831 EVENT_FROM_OSC_ARRAY_FREQ(filter_logfreq_coefs, filter_freq_coefs, NUM_COMBO_COEFS);
832 EVENT_FROM_OSC_ARRAY(duty_coefs, NUM_COMBO_COEFS);
833 EVENT_FROM_OSC_ARRAY(pan_coefs, NUM_COMBO_COEFS);
834 EVENT_FROM_OSC(feedback);
835 EVENT_FROM_OSC(trigger_phase);
836 EVENT_FROM_OSC(sample_offset);
837 EVENT_FROM_OSC(fit_ticks);
838 EVENT_FROM_OSC(fit_search);
839 EVENT_FROM_OSC_MAPPED(logratio, ratio, exp2f);
840 EVENT_FROM_OSC(resonance);
841 EVENT_FROM_OSC_MAPPED(portamento_alpha, portamento_ms, alpha_to_portamento_ms);
842 EVENT_FROM_OSC_BASEOSC(chained_osc);
843 EVENT_FROM_OSC_ARRAY_BASEOSC(mod_source, NUM_MOD_SOURCES);
844 EVENT_FROM_OSC(algorithm);
845 EVENT_FROM_OSC(filter_type);
846 // Distortion: one enable per stage that is on (a stage that is off is
847 // the default, and saying so would put GC0GF0GH0 on every osc), and the
848 // drive rail back in linear units, the way it was sent.
849 if (synth[osc]->dist_stages & DIST_CLIP) event->dist_clip = 1;
850 if (synth[osc]->dist_stages & DIST_FOLD) event->dist_fold = 1;

921void *yield_synth_events(uint8_t instr_num, struct amy_event *event, bool include_fx, void *state) {
922 // Return a sequence of events defining a synth.
923 // state = NULL on first call and it returns state to be passed on next call. Returns NULL when event sequence is finished.
924 // Find oscs for synth.
925 uint16_t voices[MAX_VOICES_PER_INSTRUMENT];
926 int num_voices = instrument_get_num_voices(instr_num, voices);
927 if (num_voices < 1) {
928 fprintf(stderr, "yield_synth_events: synth %" PRId32" has no voices.\n", (int32_t)instr_num);
929 return NULL; // instrument not allocated.
930 }
931 uint32_t flags = instrument_get_flags(instr_num);
932 uint16_t bus = instrument_get_bus(instr_num);
933 uint16_t voice = voices[0];
934 uint16_t base_osc = voice_to_base_osc[voice];
935 int num_oscs = num_oscs_for_voice(voice);
936 // The "state" indicates which osc within the voice we're going to report for.
937 int state_val = (intptr_t)state;
938 //fprintf(stderr, "yield_synth_events(%d) voice=%d num_oscs=%d state_val=%d\n", instr_num, voice, num_oscs, (int)state_val);
939 amy_clear_event(event);
940 // Always use first output (state val 0) as preamble.
941 int first_osc_state_val = 1;
942 int last_osc_state_val = num_oscs + 1;
943 if (state_val == 0) {
944 // Always emit a preamble with num_voices and oscs_per_voice
945 // We could include the patch but we don't, because we're going to set the values by hand.
946 event->num_voices = instrument_get_num_voices(instr_num, NULL);
947 event->oscs_per_voice = instrument_get_oscs_per_voice(instr_num);
948 if (flags != 0) event->synth_flags = flags;
949 float level = instrument_get_level(instr_num);
950 if (level != 1.0f) event->synth_level = level;
951 // Bus is specified in set_event_for_bus_fx, no need to include here.
952 if (!include_fx && bus != 0) event->bus = bus;
953 } else if (state_val >= first_osc_state_val && state_val < last_osc_state_val) {
954 event->osc = state_val - first_osc_state_val;
955 //fprintf(stderr, "2 base_osc %d, event->osc %d, state_val %d first_osc_state_val %d last_osc_state_val %d\n",
956 // base_osc, event->osc, state_val, first_osc_state_val, last_osc_state_val);
957 set_event_for_osc(base_osc, event->osc, event);
958 } else if (include_fx && (state_val == last_osc_state_val)) {
959 // optional final event contains the global/bus settings (volume, eq, chorus, echo, reverb).
960 // The bus is determined by the bus value on the base_osc, probably OK.
961 set_event_for_bus_fx(event, instrument_get_bus(instr_num), &amy_global);
962 }
963 ++state_val;
964 if (state_val == last_osc_state_val + (include_fx ? 1 : 0)) state_val = 0; // Indicate this is the final event.
965 return (void *)((intptr_t)state_val);
966}

968#define STATE_START_OF_MIDI_TPLT_CMDS 1024
969
970void *yield_synth_commands(uint8_t instr_num, char *s, size_t len, bool include_fx, void *state) {
971 // Generator to return multiple wirecode strings to reconfigure a synth.
972 int state_val = (intptr_t)state;
973 //fprintf(stderr, "yield_synth_commands: synth %d state %d\n", instr_num, state_val);
974 s[0] = '\0'; // By default, return an empty string.
975 if (state_val < STATE_START_OF_MIDI_TPLT_CMDS) {
976 amy_event event = amy_default_event();
977 state_val = (intptr_t)yield_synth_events(instr_num, &event, include_fx, (void *)(intptr_t)state_val);
978 sprint_event(&event, s, len, /* wirecode= */ true);
979 if (state_val == 0) {
980 // Push the state machine on to the MIDI codes
981 state_val = STATE_START_OF_MIDI_TPLT_CMDS;
982 }
983 } else {
984 // MIDI CC part
985 bool found = false;
986 int type = MIDI_MAP_TYPE_CC;
987 int starting_code = state_val - STATE_START_OF_MIDI_TPLT_CMDS;
988 for (int next_code = starting_code; next_code < 256; ++next_code) {
989 int next_midi_code = next_code;
990 if (next_midi_code >= 128) {
991 next_midi_code -= 128;
992 type = MIDI_MAP_TYPE_NOTE;
993 }
994 if (midi_fetch_mapping_command(instr_num, type, next_midi_code, s, len) == true) {
995 state_val = STATE_START_OF_MIDI_TPLT_CMDS + next_code + 1;
996 found = true;
997 break;
998 }
999 }
1000 if (found == false) {
1001 // We hit the bottom of the MIDI CCs
1002 state_val = 0; // Will terminate the yield cycle.
1003 }
1004 }
1005 return (void *)(intptr_t)state_val;
1006}

1026void parse_patch_string_to_queue(char *message, int base_osc, uint16_t oscs_per_voice, struct delta **queue, uint8_t synth, uint32_t time, bool is_first_voice) {
1027 // Work though the patch string and send to voices.
1028 // Now actually initialize the newly-allocated osc blocks with the patch
1029 //fprintf(stderr, "parse_patch_string: message %s base_osc %d synth %d time %d is_first_voice %d\n", message, base_osc, synth, time, is_first_voice);
1030 amy_event e;
1031 size_t pos = 0;
1032 // A patch string can re-shape the voice it is being loaded into, partway
1033 // through itself: the drum kits open with `if3iv1in38Z`, which turns the
1034 // 1-osc voice patch_oscs promised into a 38-osc one before the per-drum
1035 // commands that follow. So the bound the rest of the string is held to is
1036 // whatever the string last said, not what we were told before it ran.
1037 uint16_t voice_oscs = oscs_per_voice;
1038 do {
1039 amy_clear_event(&e);
1040 e.time = time;
1041 if (message[pos] == 'i') {
1042 // It's a synth-layer message, it needs a synth defined.
1043 e.synth = synth;
1044 }
1045 pos = yield_event_from_message(message, &e, pos);
1046 if (pos > 0) {
1047 // This event re-shapes the voice; everything after it answers to
1048 // the new size. (Taking effect from this event on, not the next
1049 // one, is right: the event that carries oscs_per_voice is
1050 // synth-level and names no oscs of its own.)
1051 if (AMY_IS_SET(e.oscs_per_voice)) voice_oscs = e.oscs_per_voice;
1052 // A patch's global FX phrase (a Juno patch's trailing
1053 // "x...k1..." -- 127 of the ROM Juno patches carry one) is
1054 // bus-directed but has no synth attached, so its params fell
1055 // to the default bus 0. Give it the synth being loaded and
1056 // the bus resolves from the synth, exactly as if the synth
1057 // had sent it itself.
1058 if (event_addresses_bus(&e)) e.synth = synth;
1059 if (event_addresses_oscs(&e) || is_first_voice)
1060 amy_event_to_deltas_queue(&e, base_osc, voice_oscs, queue);
1061 }
1062 } while (pos > 0);
1063}

1070void patches_store_patch(amy_event *e, char * patch_string) {
1071 peek_stack("store_patch");
1072 // amy patch string. Either pull patch_number from e, or allocate a new one and write it to e.
1073 // Patch is stored in ram.
1074 //fprintf(stderr, "store_patch: synth %d patch_num %d patch '%s'\n", e->synth, e->patch, patch_string);
1075 bool auto_assigned = !AMY_IS_SET(e->patch_number);
1076 if (auto_assigned) {
1077 // We need a number, and it only has to live long enough to carry this
1078 // patch_string into patches_load_patch, which frees the slot as soon as
1079 // it has loaded it. So: take any free slot (never used, or handed back
1080 // by an earlier load), else refuse — the deliberately out-of-range
1081 // number falls to the range check below, which prints and drops the
1082 // store.
1083 for (uint32_t i = 0; i < max_num_memory_patches; ++i) {
1084 if (memory_patch_deltas[i] == NULL && memory_patch_oscs[i] == 0) {
1085 e->patch_number = i + _PATCHES_FIRST_USER_PATCH;
1086 break;
1087 }
1088 }
1089 if (!AMY_IS_SET(e->patch_number))
1090 e->patch_number = _PATCHES_FIRST_USER_PATCH + max_num_memory_patches;
1091 //fprintf(stderr, "store_patch: auto-assigning patch number %d for '%s'\n", e->patch_number, patch_string);
1092 }
1093 int patch_index = (int)e->patch_number - _PATCHES_FIRST_USER_PATCH;
1094 if (patch_index < 0 || patch_index >= (int)max_num_memory_patches) {
1095 fprintf(stderr, "patch number %" PRId32 " is out of range (%" PRId32 " .. %" PRId32 ")\n",
1096 (int32_t)(patch_index + _PATCHES_FIRST_USER_PATCH),
1097 (int32_t)_PATCHES_FIRST_USER_PATCH,
1098 (int32_t)(_PATCHES_FIRST_USER_PATCH + (int32_t)max_num_memory_patches));
1099 // Don't leave the rejected (possibly auto-assigned) number in the
1100 // event, or the rest of event handling would try to load it.
1101 AMY_UNSET(e->patch_number);
1102 return;
1103 }
1104 if (patch_index >= next_user_patch_index) next_user_patch_index = patch_index + 1;
1105 // A re-store REPLACES the slot's contents. parse_patch_string_to_queue
1106 // appends to whatever list is standing, so without this a second store
1107 // to the same number merged two patches into one slot.
1108 if (memory_patch_deltas[patch_index] != NULL || memory_patch_oscs[patch_index])
1109 patches_reset_patch(e->patch_number);
1110 memory_patch_auto[patch_index] = auto_assigned ? 1 : 0;
1111 // Store the patch as deltas and find out how many oscs this message uses
1112 parse_patch_string_to_queue(patch_string, 0, /* oscs_per_voice= */ 0, &memory_patch_deltas[patch_index], e->synth, e->time, true);
1113 update_num_oscs_for_patch_number(patch_index + _PATCHES_FIRST_USER_PATCH);
1114 //fprintf(stderr, "store_patch: patch %d max_osc %d patch %s #deltas %d (e->num_vx=%d)\n", patch_index, max_osc, patch_string, delta_list_len(memory_patch_deltas[patch_index]), e->num_voices);
1115}

1135uint8_t patches_voices_for_note_onoff_event(amy_event *e, uint16_t voices[], uint32_t synth_flags, bool *pstolen) {
1136 // Identify the specific voice (or voices) for a note on/off event.
1137 // e->velocity is assumed to be set when this function is called.
1138 int num_voices = 0;
1139 if (AMY_IS_UNSET(e->midi_note) && AMY_IS_UNSET(e->preset) && instrument_get_num_voices(e->synth, NULL) != 1) {
1140 // velocity without midi_note is valid for velocity==0 => all-notes-off.
1141 if (e->velocity != 0) {
1142 // Attempted a note-on to all voices, suppress.
1143 fprintf(stderr, "note-on with no note for synth %" PRId32 " - ignored.\n", (int32_t)e->synth);
1144 return 0;
1145 }
1146 // All notes off - find out which voices are actually currently active, so we can turn them off.
1147 num_voices = instrument_all_notes_off(e->synth, voices);
1148 } else {
1149 // It's a note-on or note-off event, so the instrument mechanism chooses which single voice to use.
1150 uint16_t note = 0;

1152 note = (uint8_t)roundf(e->midi_note);
1153 if (AMY_IS_SET(e->preset)) {
1154 // This event includes a note *and* a preset, so it's like a drum sample note on.
1155 // Wrap the preset number into the note, so we don't allocate the same pitch for different drums to the same voice.
1156 note += 128 * e->preset;
1157 }
1158 bool is_note_off = (e->velocity == 0);
1159 voices[0] = instrument_voice_for_note_event(e->synth, note, is_note_off, pstolen);
1160 if (voices[0] == _INSTRUMENT_NO_VOICE) {
1161 // For now, I think this can only happen with a note-off that has no matching note-on.
1162 //fprintf(stderr, "synth %d did not find a voice, dropping message.\n", e->synth);
1163 // No, it also happens with note-offs when pedal is down.
1164 return 0;
1165 }
1166 num_voices = 1;
1167 }
1168 //fprintf(stderr, "instrument %d vel %d note %d voice %d\n", e->synth, (int)roundf(127.f * e->velocity), (int)roundf(e->midi_note), voices[0]);
1169 return num_voices;
1170}

1207uint8_t patches_voices_for_event(amy_event *e, uint16_t voices[]) {
1208 // Convert an event that may specify a synth into a number of specific voices.
1209 uint8_t num_voices = 0;
1210 uint32_t synth_flags = 0;
1211 // The synth may have been CONSUMED earlier in this event's processing:
1212 // releasing an instrument (num_voices=0) unsets e->synth precisely so
1213 // the rest of the event is not forwarded to it
1214 // (patches_voices_for_load_synth). Without this guard the unset
1215 // sentinel (255) reached instrument_get_num_voices below and printed
1216 // "instrument_number 255 is out of range" once per released synth —
1217 // and on a device stderr is a blocking UART line in the audio path.
1218 if (AMY_IS_UNSET(e->synth)) return 0;
1219 // We have an instrument specified - decide which of its voices are actually to be used.
1220 if (AMY_IS_SET(e->pedal)) {
1221 // Pedal events are a special case
1222 bool sustain = (e->pedal != 0);
1223 synth_flags = instrument_get_flags(e->synth);
1224 if (synth_flags & SYNTH_FLAGS_NEGATE_PEDAL) {
1225 sustain = !sustain; // Some MIDI pedals report backwards.
1226 }
1227 // A sustain release can result in note-off events for multiple voices.
1228 num_voices = instrument_sustain(e->synth, sustain, voices);
1229 if (num_voices) {
1230 e->velocity = 0;
1231 }
1232 //fprintf(stderr, "synth %d pedal %d num_voices %d\n", e->synth, e->pedal, num_voices);
1233 } else if (AMY_IS_SET(e->velocity)) {
1234 bool stolen = false;
1235 synth_flags = instrument_get_flags(e->synth);
1236 num_voices = patches_voices_for_note_onoff_event(e, voices, synth_flags, &stolen);
1237 if (stolen) {
1238 // Here, we issue a quick note-off for the stolen note, to support short decay. Kind of an abstraction violation.
1239 struct delta d = {
1240 .time = e->time,
1241 .osc = voice_to_base_osc[voices[0]],
1242 .param = VELOCITY,
1243 .data.i = 0,
1244 .next = NULL,
1245 };
1246 add_delta_to_queue(&d, &amy_global.delta_queue);
1247 //fprintf(stderr, "synth %d note %d: voice %d stolen, osc %d time %d added note-off\n", e->synth, (int)roundf(e->midi_note), voices[0], d.osc, d.time);
1248 }
1249 // Apply noteon_delay_ms to note-on events.
1250 if (instrument_noteon_delay_ms(e->synth)) {
1251 uint32_t playback_time = amy_sysclock();
1252 if(AMY_IS_SET(e->time)) playback_time = e->time;
1253 playback_time += instrument_noteon_delay_ms(e->synth);
1254 // See amy_process_event(): dodge the u32 "unset" sentinel.
1255 if(AMY_IS_UNSET(playback_time)) playback_time++;
1256 e->time = playback_time;
1257 //fprintf(stderr, "synth %d note %d delay %d time %d\n", e->synth, (int)roundf(e->midi_note), instrument_noteon_delay_ms(e->synth), e->time);
1258 }
1259 } else {
1260 // Not note on/off, treat the synth as a shorthand for *all* the voices.
1261 num_voices = instrument_get_num_voices(e->synth, voices);
1262 }
1263 if (AMY_IS_SET(e->velocity) && e->velocity == 0 && (synth_flags & SYNTH_FLAGS_IGNORE_NOTE_OFFS)) {
1264 return 0; // Ignore the note off, as requested.
1265 }
1266 return num_voices;
1267}

1269// This is called when i get an event with synth set.
1270// If the event also has synth and patch_number specified (a "load patch"), those are handled before this is called.
1271// So i know that the patch / voice alloc already exists and the patch has already been set!
1272void patches_event_has_voices(amy_event *e, struct delta **queue) {
1273 peek_stack("has_voices");
1274 //fprintf(stderr, "patches_event_has_voices:\n");
1275 //fprintf_event_stderr(e);
1276
1277 patches_grab_synth_tier(e);
1278 // If this event is only changing bus parameters (FX), those parts have already been handled before we see it.
1279 // A pure pedal event doesn't strictly address oscs, but it can result in note-off events.
1280 if (!event_addresses_oscs(e) && AMY_IS_UNSET(e->pedal))
1281 return; // Early exit.
1282
1283 uint8_t synth = e->synth;
1284 int synth_flags = 0;
1285 if (AMY_IS_SET(e->synth)) {
1286 synth_flags = instrument_get_flags(e->synth);
1287 }
1288 // Should we invoke MIDI note-on cmd rules?
1289 if (synth_flags & SYNTH_FLAGS_NOTES_VIA_MIDI
1290 && AMY_IS_SET(e->midi_note)
1291 && (e->note_source_channel != e->synth)) {
1292 // Route note-on event via MIDI to invoke midi_note_cmds. We pass the synth
1293 // number straight through as the channel rather than packing it into a MIDI
1294 // status byte, so synths above 16 reach their own mappings.
1295 uint8_t data[2];
1296 data[0] = 0x7F & (uint8_t)(e->midi_note);
1297 uint8_t velocity = 255; // fake-note-on magic value.

1489void patches_load_patch(amy_event *e) {
1490 // Given an event with a synth (instrument) set up a synth.
1491 // Common case is to call with a patch to load, but can also have just
1492 // a oscs_per_voice value.
1493 // (also called if instrument & num_voices even if no patch specified, to change #voices).
1494 // This means to set/reset the voices and load the messages (from ROM or memory) and set them.
1495 peek_stack("load_patch");
1496 // A RE-PATCH KEEPS THE INSTRUMENT'S BUS. Loading a patch rebuilds the
1497 // voices' oscs (reset to bus 0) and re-registers the instrument, so a
1498 // patch change that didn't re-state the bus dropped the synth out of
1499 // its own mix and onto the default bus, into everybody else's FX.
1500 // Inheriting the standing bus here makes the rest of the pipeline
1501 // (instrument_add_new below, the per-osc BUS deltas fanned out by
1502 // patches_event_has_voices) re-route exactly as an explicit bus would.
1503 if (AMY_IS_UNSET(e->bus) && instrument_number_exists(e->synth, NULL)) {
1504 int prev_bus = instrument_get_bus(e->synth);
1505 if (prev_bus > 0) e->bus = (uint16_t)prev_bus;
1506 }
1507 uint16_t voices[MAX_VOICES_PER_INSTRUMENT];
1508 uint8_t num_voices = 0;
1509 uint16_t oscs_per_voice = 0;
1510 uint16_t patch_number = e->patch_number; // Need to match type of e->patch_number so AMY_IS_UNSET(patch_number) will work.
1511 //fprintf(stderr, "load_patch synth %d patch_number %d num_voices %d oscs_per_voice %d\n", e->synth, e->patch_number, e->num_voices, e->oscs_per_voice);
1512 // CLONE ON GROW. Changing num_voices on a synth that already has voices
1513 // rebuilds every voice, and the only config to rebuild them from used to be
1514 // the synth's stored patch -- so growing a synth silently threw away every
1515 // edit made since the patch was loaded, and reverted voice 0 to the pristine
1516 // patch. Snapshot the live voice instead, BEFORE the reallocation below
1517 // resets its oscs, and use that as the source for all the new voices. A
1518 // synth that was configured osc-by-osc (or with oscs_per_voice= and no patch
1519 // at all) can now grow too; before, it had no usable patch to reload and the
1520 // whole event was dropped.
1521 struct delta *clone_deltas = NULL;
1522 uint16_t clone_oscs_per_voice = 0;
1523 // num_voices=0 is a deletion: nothing will ever replay the snapshot, so
1524 // don't spend delta-pool churn capturing one.
1525 if (AMY_IS_UNSET(patch_number) && AMY_IS_UNSET(e->oscs_per_voice)
1526 && (AMY_IS_UNSET(e->num_voices) || e->num_voices > 0)
1527 && AMY_IS_SET(e->synth) && instrument_number_exists(e->synth, NULL))
1528 clone_oscs_per_voice = snapshot_voice_config(e->synth, &clone_deltas, e->time);
1529 // An AUTO-ASSIGNED memory patch is a courier, not a patch. Its number was
1530 // invented by patches_store_patch so a patch_string could be handed to this
1531 // load; the caller never saw it and can never ask for it again. So it is
1532 // consumed here: the number is not stored on the instrument (the voices
1533 // themselves are the synth's config now that growth clones them), and the
1534 // slot goes back to the pool on the way out, whether this load worked or
1535 // not.
1536 bool auto_patch = false;
1537 if (AMY_IS_SET(patch_number)) {
1538 int patch_index = (int)patch_number - _PATCHES_FIRST_USER_PATCH;
1539 auto_patch = (patch_index >= 0 && patch_index < (int)max_num_memory_patches
1540 && memory_patch_auto[patch_index]);
1541 }
1542 num_voices = patches_voices_for_load_synth(e, voices);
1543 if (num_voices == 0) {
1544 if (AMY_IS_UNSET(e->num_voices)) {
1545 // Print a warning unless we deliberately set the voices to zero to release the synth.
1546 fprintf(stderr, "synth %" PRId32 ": no voices selected, ignored (e->num_voices %" PRId32 "...)\n",
1547 (int32_t)e->synth, (int32_t)e->num_voices);
1548 }
1549 goto done;
1550 } else {
1551 // Reset every osc belonging to each voice so stale state doesn't persist
1552 // before we reallocate and reinitialize them below.
1553 for (uint8_t v = 0; v < num_voices; v++) {
1554 if (AMY_IS_SET(voice_to_base_osc[voices[v]])) {
1555 uint16_t base = voice_to_base_osc[voices[v]];
1556 for (uint16_t j = 0; j < AMY_OSCS - base; j++) {
1557 if (osc_to_voice[base + j] != voices[v]) break;
1558 schedule_osc_reset(e->time, base + j, NULL);
1559 }
1560 }
1561 }
1562 }
1563
1564 // At this point, we have the voices[] array and num_voices set up to be initialized.
1565 char *message = NULL;
1566 struct delta *deltas = NULL;
1567 // Figure out the #oscs per voice, setup message or deltas if available.
1568 if (AMY_IS_SET(e->oscs_per_voice)) {
1569 oscs_per_voice = e->oscs_per_voice;
1570 if (AMY_IS_SET(patch_number)) {
1571 fprintf(stderr, "WARN: synth %" PRId32 ": oscs_per_voice %" PRIu16 " made me ignore patch number %" PRIu16 "\n",
1572 (int32_t)e->synth, e->oscs_per_voice, patch_number);
1573 }
1574 } else if (clone_oscs_per_voice > 0) {
1575 // Reshaping a live synth: every voice comes from the snapshot of the
1576 // existing voice taken above, not from the stored patch. The synth keeps
1577 // whatever patch number it had (nothing here changes it) -- it is no
1578 // longer what defines the sound.
1579 oscs_per_voice = clone_oscs_per_voice;
1580 deltas = clone_deltas;
1581 patch_number = instrument_get_patch_number(e->synth);
1582 } else {
1583 if (AMY_IS_UNSET(patch_number))
1584 patch_number = instrument_get_patch_number(e->synth);
1585 if(patch_number < _PATCHES_FIRST_USER_PATCH) {
1586 // Built-in patch. The table has reserved (NULL) gaps -- e.g. 259-383
1587 // before the drum kit bank at 384 -- and ends well before the user
1588 // range, so check both before dereferencing.
1589 if (patch_number >= _PATCHES_NUM_BUILTIN || patch_commands[patch_number] == NULL) {
1590 fprintf(stderr, "patch_number %" PRIu16 " is not a defined built-in patch (synth %" PRId32 "), ignored\n",
1591 patch_number, (int32_t)e->synth);
1592 goto done;
1593 }
1594 message = (char*)patch_commands[patch_number];

1608 patch_number, (int32_t)_PATCHES_FIRST_USER_PATCH,
1609 (int32_t)(_PATCHES_FIRST_USER_PATCH + (int32_t)max_num_memory_patches - 1),

1611 goto done;

1619 (int32_t)e->synth, (int32_t)e->num_voices);
1620 goto done;
1621 }
1622 }
1623 }

1628 }
1629
1630 for(uint8_t v=0;v<num_voices;v++) {
1631 // Find the first osc with oscs_per_voice free oscs.
1632 bool found = false;

1635 #ifdef TULIP

1644 // Don't scan if the block would wrap AMY_OSCS.
1645 if (base_osc > AMY_OSCS - oscs_per_voice) continue;
1646 int available_oscs = 0;
1647 // Are there num_voices x oscs_per_voice free oscs after this one?
1648 for(uint16_t osc = base_osc; osc < base_osc + oscs_per_voice; ++osc) {
1649 if (AMY_IS_SET(osc_to_voice[osc])) {
1650 break;
1651 }
1652 ++available_oscs;
1653 }
1654 if(available_oscs == oscs_per_voice) {
1655 //fprintf(stderr, "found %d consecutive oscs starting at %d for voice %d\n", oscs_per_voice, base_osc, voices[v]);
1656 voice_to_base_osc[voices[v]] = base_osc;
1657 for(uint16_t osc = base_osc; osc < base_osc + oscs_per_voice; ++osc) {
1658 //fprintf(stderr, "setting osc %d for voice %d to amy osc %d\n", osc - base_osc, voices[v], osc);
1659 osc_to_voice[osc] = voices[v];
1660 schedule_osc_reset(e->time, osc, NULL);
1661 }
1662 // exit the loop
1663 found = true;
1664 }
1665 }
1666 if(!found) {
1667 fprintf(stderr, "cannot find %" PRIu16 " oscs for patch %" PRIu16 " for voice %" PRIu16 ". not setting this voice\n",
1668 oscs_per_voice, patch_number, voices[v]);

1670 } // end of loop setting up voice_to_base_osc for all voices[v]
1671
1672 // Finally, store as an instrument if instrument number is specified.
1673 if (AMY_IS_SET(e->synth)) {
1674 uint32_t flags = 0;
1675 if (AMY_IS_SET(e->synth_flags)) flags = e->synth_flags;
1676 uint16_t bus = 0;
1677 if (AMY_IS_SET(e->bus)) bus = e->bus;

1679 // records no patch at all rather than a number nobody can use.
1680 uint16_t stored_patch_number = patch_number;
1681 if (auto_patch) AMY_UNSET(stored_patch_number);

1683 }

1685 bool is_first_voice = true; // flag for once-only messages.
1686 for(uint8_t v = 0; v < num_voices; v++) {
1687 if(AMY_IS_SET(voice_to_base_osc[voices[v]])) {
1688 if (deltas) {
1689 add_deltas_to_queue_with_baseosc(deltas, voice_to_base_osc[voices[v]], oscs_per_voice, &amy_global.delta_queue, e->time);
1690 } else if (message) {
1691 parse_patch_string_to_queue(message, voice_to_base_osc[voices[v]], oscs_per_voice, &amy_global.delta_queue, e->synth, e->time, is_first_voice);

1695 }
1696 }
1697done:

1701 // this load failed), and nothing can ask for it by number, so hand it back.
1702 if (auto_patch) patches_reset_patch(e->patch_number);
