26void patches_init(int max_memory_patches) {
27 max_num_memory_patches = max_memory_patches;
28 uint8_t *alloc_base = malloc_caps(
29 max_num_memory_patches * sizeof(struct delta *)
30 + max_num_memory_patches * sizeof(uint16_t)
31 + AMY_OSCS * sizeof(uint8_t)
32 + amy_global.config.max_voices * sizeof(uint16_t),
33 amy_global.config.ram_caps_synth
34 );
35 memory_patch_deltas = (struct delta **)alloc_base;
36 memory_patch_oscs = (uint16_t *)(memory_patch_deltas + max_num_memory_patches);
37 osc_to_voice = (uint8_t *)(memory_patch_oscs + max_num_memory_patches);
38 voice_to_base_osc = (uint16_t *)(osc_to_voice + AMY_OSCS);
39 bzero(memory_patch_deltas, max_num_memory_patches * sizeof(struct delta *));
40 patches_reset();
41}

43void patches_reset_patch(int patch_number) {
44 int patch_index = patch_number - _PATCHES_FIRST_USER_PATCH;
45 if (patch_index < 0 || patch_index >= (int)max_num_memory_patches) {
46 fprintf(stderr, "reset patch number %" PRId32 " is out of range (%" PRId32 " .. %" PRId32 ")\n",
47 (int32_t)(patch_index + _PATCHES_FIRST_USER_PATCH),
48 (int32_t)_PATCHES_FIRST_USER_PATCH,
49 (int32_t)(_PATCHES_FIRST_USER_PATCH + (int32_t)max_num_memory_patches));
50 return;
51 }
52 if (memory_patch_deltas[patch_index] != NULL) delta_release_list(memory_patch_deltas[patch_index]);
53 memory_patch_deltas[patch_index] = NULL;
54 memory_patch_oscs[patch_index] = 0;
55}

423struct delta *deltas_to_event(struct delta *queue, struct amy_event *event) {
424 // Consume deltas from queue and push into event. Return pointer to first non-consumed delta.
425 if (queue == NULL) return NULL;
426 event->osc = queue->osc;
427 event->time = queue->time;
428 uint32_t breakpoint_times[MAX_BREAKPOINT_SETS][MAX_BREAKPOINTS];
429 float breakpoint_values[MAX_BREAKPOINT_SETS][MAX_BREAKPOINTS];
430 for (int i = 0; i < MAX_BREAKPOINT_SETS; ++i) {
431 for (int j = 0; j < MAX_BREAKPOINTS; ++j) {
432 AMY_UNSET(breakpoint_times[i][j]);
433 AMY_UNSET(breakpoint_values[i][j]);
434 }
435 }
436 int highest_breakpoint[MAX_BREAKPOINT_SETS] = {-1, -1};
437 while(queue != NULL) {
438 if (queue->osc != event->osc || queue->time != event->time) break; // delta doesn't fit this event.
439 switch (queue->param) {
440 _CASE_I(wave, WAVE)
441 _CASE_I(preset, PRESET)
442 _CASE_F(midi_note, MIDI_NOTE)
443 _CASE_F(feedback, FEEDBACK)
444 _CASE_F(trigger_phase, PHASE)
445 _CASE_F(pitch_bend, PITCH_BEND)
446 _CASE_I(latency_ms, LATENCY)
447 _CASE_F(tempo, TEMPO)
448 _CASE_LOG(ratio, RATIO)
449 _CASE_F(resonance, RESONANCE)
450 _CASE_I(portamento_ms, PORTAMENTO)
451 _CASE_I(chained_osc, CHAINED_OSC)
452 _CASE_I(reset_osc, RESET_OSC)
453 _CASE_I(mod_source, MOD_SOURCE)
454 _CASE_I(note_source_channel, NOTE_SOURCE_CHANNEL)
455 _CASE_I(filter_type, FILTER_TYPE)
456 _CASE_I(algorithm, ALGORITHM)
457 _CASE_F(eq_l, EQ_L)
458 _CASE_F(eq_m, EQ_M)
459 _CASE_F(eq_h, EQ_H)
460 _CASE_F(echo_max_delay_ms, ECHO_MAX_DELAY_MS)
461 _CASE_F(echo_level, ECHO_LEVEL)
462 _CASE_F(echo_delay_ms, ECHO_DELAY_MS)
463 _CASE_F(echo_feedback, ECHO_FEEDBACK)
464 _CASE_F(echo_filter_coef, ECHO_FILTER_COEF)
465 _CASE_F(chorus_max_delay, CHORUS_MAX_DELAY)
466 _CASE_F(chorus_level, CHORUS_LEVEL)
467 _CASE_F(chorus_lfo_freq, CHORUS_LFO_FREQ)
468 _CASE_F(chorus_depth, CHORUS_DEPTH)
469 _CASE_F(reverb_level, REVERB_LEVEL)
470 _CASE_F(reverb_liveness, REVERB_LIVENESS)
471 _CASE_F(reverb_damping, REVERB_DAMPING)
472 _CASE_F(reverb_xover_hz, REVERB_XOVER_HZ)
473 _CASE_I(eg_type[0], EG0_TYPE)
474 _CASE_I(eg_type[1], EG1_TYPE)
475 _CASE_F(velocity, VELOCITY)
476 default: // blocks, not handled by case
477 _TEST_COEFS(amp_coefs, AMP)
478 _TEST_FREQ_COEFS(freq_coefs, FREQ)
479 _TEST_FREQ_COEFS(filter_freq_coefs, FILTER_FREQ)
480 _TEST_COEFS(duty_coefs, DUTY)
481 _TEST_COEFS(pan_coefs, PAN)
482 for (int bus = 0; bus < AMY_NUM_BUSES; ++bus) {
483 if ((int)queue->param == (int)VOLUME_BASE + bus)
484 event->volume[bus] = queue->data.f;
485 }
486 for (int i = 0; i < MAX_ALGO_OPS; ++i) {
487 if ((int)queue->param == (int)ALGO_SOURCE_START + i)
488 event->algo_source[i] = queue->data.i;
489 }
490 for (int i = 0; i < MAX_BREAKPOINT_SETS; ++i) {
491 for (int j = 0; j < MAX_BREAKPOINTS; ++j) {
492 if ((int)queue->param == (int)BP_START + (j * 2) + (i * MAX_BREAKPOINTS * 2)) {
493 //event->bp[i].time[j] = queue->data.i;
494 breakpoint_times[i][j] = queue->data.i;
495 if (j > highest_breakpoint[i]) highest_breakpoint[i] = j;
496 }
497 else if ((int)queue->param == (int)BP_START + (j * 2 + 1) + (i * MAX_BREAKPOINTS * 2)) {
498 //event->bp[i].value[j] = queue->data.f;
499 breakpoint_values[i][j] = queue->data.f;
500 if (j > highest_breakpoint[i]) highest_breakpoint[i] = j;
501 }
502 }
503 }
504 break;
505 }
506 queue = queue->next;
507 }
508 uint32_t *bp_times_ms[MAX_BREAKPOINT_SETS] = {event->eg0_times, event->eg1_times};
509 float *bp_values[MAX_BREAKPOINT_SETS] = {event->eg0_values, event->eg1_values};
510 for (int i = 0; i < MAX_BREAKPOINT_SETS; ++i) {
511 if (highest_breakpoint[i] >= 0) {
512 event->bp_is_set[i] = 1;
513 for (int j = 0; j <= highest_breakpoint[i]; ++j) {
514 if (AMY_IS_SET(breakpoint_times[i][j])) {
515 float t_ms = ((float)breakpoint_times[i][j]) * 1000.0f / (float)AMY_SAMPLE_RATE;
516 int32_t t_rounded = (int32_t)roundf(t_ms);
517 if (t_rounded < 0) t_rounded = 0;
518 if (t_rounded >= SHRT_MAX) t_rounded = SHRT_MAX - 1;
519 bp_times_ms[i][j] = (int16_t)t_rounded;
520 }
521 if (AMY_IS_SET(breakpoint_values[i][j])) {
522 bp_values[i][j] = breakpoint_values[i][j];
523 }
524 }
525 }
526 }
527 return queue;
528}

665void *yield_synth_events(uint8_t instr_num, struct amy_event *event, bool include_fx, void *state) {
666 // Return a sequence of events defining a synth.
667 // state = NULL on first call and it returns state to be passed on next call. Returns NULL when event sequence is finished.
668 // Find oscs for synth.
669 uint16_t voices[MAX_VOICES_PER_INSTRUMENT];
670 int num_voices = instrument_get_num_voices(instr_num, voices);
671 if (num_voices < 1) {
672 fprintf(stderr, "yield_synth_events: synth %" PRId32" has no voices.\n", (int32_t)instr_num);
673 return NULL; // instrument not allocated.
674 }
675 uint32_t flags = instrument_get_flags(instr_num);
676 uint16_t voice = voices[0];
677 uint16_t base_osc = voice_to_base_osc[voice];
678 int num_oscs = num_oscs_for_voice(voice);
679 // The "state" indicates which osc within the voice we're going to report for.
680 int state_val = (intptr_t)state;
681 //fprintf(stderr, "yield_synth_events(%d) voice=%d num_oscs=%d state_val=%d\n", instr_num, voice, num_oscs, (int)state_val);
682 amy_clear_event(event);
683 int first_osc_state_val = 0;
684 int last_osc_state_val = num_oscs;
685 if (flags != 0) {
686 ++first_osc_state_val;
687 ++last_osc_state_val;
688 if (state_val == 0) {
689 event->synth_flags = flags;
690 }
691 }
692 if (state_val >= first_osc_state_val && state_val < last_osc_state_val) {
693 event->osc = state_val - first_osc_state_val;
694 //fprintf(stderr, "2 base_osc %d, event->osc %d, state_val %d first_osc_state_val %d last_osc_state_val %d\n",
695 // base_osc, event->osc, state_val, first_osc_state_val, last_osc_state_val);
696 set_event_for_osc(base_osc, event->osc, event);
697 } else if (include_fx && (state_val == last_osc_state_val)) {
698 // optional final event contains the global/bus settings (volume, eq, chorus, echo, reverb).
699 // The bus is determined by the bus value on the base_osc, probably OK.
700 set_event_for_bus_fx(event, synth[base_osc]->bus, &amy_global);
701 }
702 ++state_val;
703 if (state_val == last_osc_state_val + (include_fx ? 1 : 0)) state_val = 0; // Indicate this is the final event.
704 return (void *)((intptr_t)state_val);
705}

707#define STATE_START_OF_MIDI 1024
708void *yield_synth_commands(uint8_t instr_num, char *s, size_t len, bool include_fx, void *state) {
709 // Generator to return multiple wirecode strings to reconfigure a synth.
710 int state_val = (intptr_t)state;
711 //fprintf(stderr, "yield_synth_commands: synth %d state %d\n", instr_num, state_val);
712 s[0] = '\0'; // By default, return an empty string.
713 if (state_val < STATE_START_OF_MIDI) {
714 amy_event event = amy_default_event();
715 state_val = (intptr_t)yield_synth_events(instr_num, &event, include_fx, (void *)(intptr_t)state_val);
716 sprint_event(&event, s, len, /* wirecode= */ true);
717 if (state_val == 0) {
718 // Push the state machine on to the MIDI codes
719 state_val = STATE_START_OF_MIDI;
720 }
721 } else {
722 // MIDI CC part
723 bool found = false;
724 int type = MIDI_MAP_TYPE_CC;
725 for (int next_midi_code = state_val - STATE_START_OF_MIDI; next_midi_code < 128; ++next_midi_code) {
726 if (midi_fetch_mapping_command(instr_num, type, next_midi_code, s, len) == true) {
727 state_val = STATE_START_OF_MIDI + next_midi_code + 1;
728 found = true;
729 break;
730 }
731 }
732 if (found == false) {
733 // We hit the bottom of the MIDI CCs
734 state_val = 0; // Will terminate the yield cycle.
735 }
736 }
737 return (void *)(intptr_t)state_val;
738}

809uint8_t patches_voices_for_note_onoff_event(amy_event *e, uint16_t voices[], uint32_t synth_flags, bool *pstolen) {
810 // Identify the specific voice (or voices) for a note on/off event.
811 // e->velocity is assumed to be set when this function is called.
812 int num_voices = 0;
813 if (AMY_IS_UNSET(e->midi_note) && AMY_IS_UNSET(e->preset) && instrument_get_num_voices(e->synth, NULL) != 1) {
814 // velocity without midi_note is valid for velocity==0 => all-notes-off.
815 if (e->velocity != 0) {
816 // Attempted a note-on to all voices, suppress.
817 fprintf(stderr, "note-on with no note for synth %" PRId32 " - ignored.\n", (int32_t)e->synth);
818 return 0;
819 }
820 // All notes off - find out which voices are actually currently active, so we can turn them off.
821 num_voices = instrument_all_notes_off(e->synth, voices);
822 } else {
823 // It's a note-on or note-off event, so the instrument mechanism chooses which single voice to use.
824 uint16_t note = 0;
825 if (AMY_IS_SET(e->midi_note)) // midi note can be unset if preset is set.
826 note = (uint8_t)roundf(e->midi_note);
827 if (AMY_IS_SET(e->preset)) {
828 // This event includes a note *and* a preset, so it's like a drum sample note on.
829 // Wrap the preset number into the note, so we don't allocate the same pitch for different drums to the same voice.
830 note += 128 * e->preset;
831 }
832 bool is_note_off = (e->velocity == 0);
833 voices[0] = instrument_voice_for_note_event(e->synth, note, is_note_off, pstolen);
834 if (voices[0] == _INSTRUMENT_NO_VOICE) {
835 // For now, I think this can only happen with a note-off that has no matching note-on.
836 //fprintf(stderr, "synth %d did not find a voice, dropping message.\n", e->synth);
837 // No, it also happens with note-offs when pedal is down.
838 return 0;
839 }
840 num_voices = 1;
841 }
842 //fprintf(stderr, "instrument %d vel %d note %d voice %d\n", e->synth, (int)roundf(127.f * e->velocity), (int)roundf(e->midi_note), voices[0]);
843 return num_voices;
844}

847uint8_t patches_voices_for_event(amy_event *e, uint16_t voices[]) {
848 // Convert an event that may specify a synth into a number of specific voices.
849 uint8_t num_voices = 0;
850 uint32_t synth_flags = 0;
851 if (!AMY_IS_SET(e->synth)) {
852 // No instrument, just directly naming the voices.
853 num_voices = copy_voices(e->voices, voices);
854 } else { // We have an instrument specified - decide which of its voices are actually to be used.
855 // It's a mistake to specify both synth (instrument) and voices, warn user we're ignoring voices.

1062void patches_load_patch(amy_event *e) {

1064 // Common case is to call with a patch to load, but can also have just
1065 // a oscs_per_voice value.
1066 // (also called if instrument & num_voices even if no patch specified, to change #voices).
1067 // This means to set/reset the voices and load the messages (from ROM or memory) and set them.
1068 peek_stack("load_patch");
1069 uint16_t voices[MAX_VOICES_PER_INSTRUMENT];
1070 uint8_t num_voices = 0;
1071 uint16_t oscs_per_voice = 0;
1072 uint16_t patch_number = e->patch_number; // Need to match type of e->patch_number so AMY_IS_UNSET(patch_number) will work.
1073 //fprintf(stderr, "load_patch synth %d patch_number %d num_voices %d oscs_per_voice %d\n", e->synth, e->patch_number, e->num_voices, e->oscs_per_voice);

1075 num_voices = patches_voices_for_load_synth(e, voices);
1076 } else if (AMY_IS_SET(e->voices[0])) {
1077 num_voices = copy_voices(e->voices, voices);
1078 }
1079 if (num_voices == 0) {
1080 if (AMY_IS_UNSET(e->num_voices)) {
1081 // Print a warning unless we deliberately set the voices to zero to release the synth.
1082 fprintf(stderr, "synth %" PRId32 ": no voices selected, ignored (e->num_voices %" PRId32 " e->voices [0] %" PRIu16 "...)\n",
1083 (int32_t)e->synth, (int32_t)e->num_voices, e->voices[0]);
1084 }

1086 } else {
1087 // Reset every osc belonging to each voice so stale state doesn't persist
1088 // before we reallocate and reinitialize them below.
1089 for (uint8_t v = 0; v < num_voices; v++) {

1092 for (uint16_t j = 0; j < AMY_OSCS - base; j++) {

1118 fprintf(stderr, "patch_number %" PRIu16 " is not a defined built-in patch (synth %" PRId32 "), ignored\n",
1119 patch_number, (int32_t)e->synth);
1120 return;
1121 }
1122 message = (char*)patch_commands[patch_number];
1123 oscs_per_voice = patch_oscs[patch_number];
1124 } else {

1127 oscs_per_voice = memory_patch_oscs[patch_index];
1128 if(oscs_per_voice > 0){

1130 } else {

1134 return;
1135 }
1136 }
1137 }

1140 // Release all the oscs of any voices we're re-using before we start re-allocating oscs.
1141 release_voice_oscs(voices[v]);
1142 }

1163 good = good & (AMY_IS_UNSET(osc_to_voice[osc + j]));
1164 }

1220 }
