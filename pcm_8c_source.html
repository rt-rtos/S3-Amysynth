61// Get either memory preset, file preset or baked in preset for preset number.
62// For ROM presets, fill the caller-provided rom_local and return it.
63memorypcm_preset_t * get_preset_for_preset_number(uint16_t preset_number,
64 memorypcm_preset_t *rom_local) {
65 // Get the memory preset. If we can't find it, it could be a ROM preset. So copy params in from ROM preset
66 memorypcm_ll_t *preset = memorypcm_ll_start;
67 while(preset != NULL) {
68 if(preset->preset_number == preset_number) {
69 if(preset->preset->sample_ram != NULL || preset->preset->file_handle > 0) {
70 return preset->preset;
71 }
72 }
73 preset = preset->next;
74 }
75
76#ifdef GAMMA9001
77 // Gamma9001 drum banks live at GAMMA9001_PRESET_BASE+, read straight out
78 // of the platform-provided blob (memory presets above may still shadow them).
79 if (preset_number >= GAMMA9001_PRESET_BASE &&
80 preset_number < GAMMA9001_PRESET_BASE + GAMMA9001_NUM_SAMPLES &&
81 gamma9001_pcm != NULL && rom_local != NULL) {
82 const pcm_map_t *g = &gamma9001_map[preset_number - GAMMA9001_PRESET_BASE];
83 memset(rom_local, 0, sizeof(*rom_local));
84 rom_local->sample_ram = (int16_t *)gamma9001_pcm + g->offset;
85 rom_local->length = g->length;
86 rom_local->loopstart = g->loopstart;
87 rom_local->loopend = g->loopend;
88 rom_local->midinote = g->midinote;
89 rom_local->samplerate = GAMMA9001_SAMPLE_RATE;
90 rom_local->log2sr = log2f((float)GAMMA9001_SAMPLE_RATE / ZERO_LOGFREQ_IN_HZ);
91 rom_local->type = AMY_PCM_TYPE_GAMMA;
92 rom_local->channels = 1;
93 return rom_local;
94 }
95#endif
96
97 // No memory preset found, so try ROM preset. default to 0 if out of range
98 if (preset_number >= pcm_samples) preset_number = 0;
99 if (rom_local == NULL) {
100 return NULL;
101 }
102 memset(rom_local, 0, sizeof(*rom_local));
103 const pcm_map_t cpreset = pcm_map[preset_number];
104 uint32_t offset = cpreset.offset;
105 uint32_t length = cpreset.length;
106#ifdef PCM_LENGTH
107 if (offset >= PCM_LENGTH) {
108 offset = 0;
109 length = 0;
110 } else if (length > (PCM_LENGTH - offset)) {
111 length = PCM_LENGTH - offset;
112 }
113#endif
114 rom_local->sample_ram = (int16_t*)pcm + offset;
115 rom_local->length = length;
116 rom_local->loopstart = cpreset.loopstart;
117 rom_local->loopend = cpreset.loopend;
118 if (rom_local->loopstart > rom_local->length) {
119 rom_local->loopstart = 0;
120 }
121 if (rom_local->loopend > rom_local->length) {
122 rom_local->loopend = rom_local->length;
123 }
124 rom_local->midinote = cpreset.midinote;
125 rom_local->samplerate = PCM_AMY_SAMPLE_RATE;
126 rom_local->log2sr = PCM_AMY_LOG2_SAMPLE_RATE;
127 rom_local->type = AMY_PCM_TYPE_ROM;
128 rom_local->channels = 1;
129 return rom_local;
130}

145// Hann window for the granular time-stretcher, Q15. At 50% overlap Hann
146// windows sum to exactly 1.0 (COLA), so two overlapping grains reconstruct
147// unity gain. Filled once at pcm_init (float trig at init time only; the
148// render path is all fixed point).
149static int16_t stretch_win[PCM_STRETCH_GRAIN];
150
151void pcm_init() {
152 memorypcm_ll_start = NULL;
153 for (int i = 0; i < PCM_STRETCH_GRAIN; ++i) {
154 float w = 0.5f * (1.0f - cosf(2.0f * (float)M_PI * (float)i / (float)PCM_STRETCH_GRAIN));
155 stretch_win[i] = (int16_t)(w * 32767.0f);
156 }
157}
158void pcm_deinit() {
159 pcm_unload_all_presets();
160}

196// A file-backed preset streams through a small sliding buffer rather than
197// sitting in a table we can index freely, so there is nothing to loop back
198// into: render_pcm refills from the file each block and rewinds phase to the
199// top of the fresh buffer. A PCM_LOOP* mode on such a preset can never do
200// what it says.
201//
202// Rather than accept the command and quietly do something else at note-on,
203// refuse it where it is issued -- when the mode changes, and when the preset
204// number changes -- so the configuration never reaches a state it can't
205// honor, and the user hears about it while the offending command is still in
206// front of them. Called with the *proposed* mode and preset; returns false if
207// the command should be dropped.
208//
209// (Whole-file looping *would* be implementable on top of the fseek+re-parse
210// rewind pcm_note_on already does; what a stream can never honor is the
211// loopstart/loopend marks. That's a bigger change than this one.)
212bool pcm_loop_config_allowed(uint16_t osc, uint16_t mode, uint16_t preset_number,
213 bool mode_is_the_new_part) {
214 // mode means nothing outside PCM, so don't second-guess other waves.
215 if (synth[osc]->wave != PCM) return true;
216 if (!mode_is_looping(mode)) return true;
217 const char *filename = NULL;
218 if (!preset_is_file(preset_number, &filename)) return true;
219 if (mode_is_the_new_part) {
220 fprintf(stderr, "amy: osc %d preset %d streams from %s, which cannot loop; "
221 "ignoring mode=%d. Use load_sample() to loop.\n",
222 osc, preset_number, filename ? filename : "a file", mode);
223 } else {
224 fprintf(stderr, "amy: preset %d streams from %s, which cannot loop, but osc %d "
225 "is in mode=%d; ignoring preset=%d. Set a non-loop mode first, "
226 "or use load_sample().\n",
227 preset_number, filename ? filename : "a file", osc, mode, preset_number);
228 }
229 return false;
230}

233#define PCM_ZERO_THRESH 16
234// How far forward to search for a zero crossing

237// LOCAL EDIT: retrig-behavior gate. 0 (default) = fade-restart: ramp the old
238// tail to zero over a FIXED PCM_RETRIG_FADE_FRAMES span, then splice to the
239// new note - click-free with constant onset latency. 1 = upstream behavior:
240// defer the restart to the next zero crossing, which costs a VARIABLE
241// 0..PCM_MAX_ZERO_SEARCH_LEN-frame latency per hit (audible as an
242// inconsistent transient on steady retrigs into a long low-frequency tail,
243// e.g. a bass-drum four-on-floor).
244#ifndef AMY_PCM_RETRIG_ZERO_CROSS
245#define AMY_PCM_RETRIG_ZERO_CROSS 0
246#endif
247// Old-tail fade length in sample-table frames; 64 pairs with the >>6 in the
248// per-sample gain (render_pcm).
249#define PCM_RETRIG_FADE_FRAMES 64

251int pcm_find_next_zero_crossing(uint16_t osc, uint32_t base_index) {
252 // Find next zero or zero crossing beyond base_index in PCM under osc.
253 int index = -1;
254 if(AMY_IS_SET(synth[osc]->preset)) {
255 memorypcm_preset_t rom_local;
256 memorypcm_preset_t *preset =
257 get_preset_for_preset_number(synth[osc]->preset, &rom_local);
258 uint32_t sample_length = preset->length;
259 const LUTSAMPLE* table = preset->sample_ram;
260 int last_sign = 0;
261 int sign;
262 if (preset->type != AMY_PCM_TYPE_FILE
263 && table != NULL
264 && sample_length != 0) {
265 const LUTSAMPLE* table = preset->sample_ram;
266 //uint32_t start_index = base_index; // for debug only
267 LUTSAMPLE min_val = SAMPLE_MAX;
268 LUTSAMPLE val;
269 for(uint16_t i=0; i < PCM_MAX_ZERO_SEARCH_LEN; i++) {
270 // For non-file samples, we have to check for end of sample/looping.
271 if (base_index >= sample_length) break;
272 if (preset->channels == 2) {
273 if (synth[osc]->wave == PCM_LEFT) {
274 val = table[base_index * 2];
275 } else if (synth[osc]->wave == PCM_RIGHT) {
276 val = table[base_index * 2 + 1];
277 } else { // PCM or PCM_MIX
278 val = (LUTSAMPLE)(((int32_t)table[base_index * 2] + (int32_t)table[base_index * 2 + 1]) / 2);
279 }
280 } else {
281 val = table[base_index];
282 }
283 sign = 1;
284 if (val < 0) {sign = -1; val = -val;}
285 if (val < min_val) {
286 min_val = val;
287 index = base_index;
288 }
289 if ((val <= PCM_ZERO_THRESH) || ((sign * last_sign) == -1))
290 break;
291 last_sign = sign;
292 ++base_index;
293 }
294 //fprintf(stderr, "time %.3f: pcm_find_zero: osc %d start %d base %d len %d min_val %d index %d sign %d last %d\n",
295 // amy_global.time, osc, start_index, base_index, sample_length, min_val, index, sign, last_sign);
296 }
297 }
298 return index;
299}

302// "fit": real-time pitch-invariant time stretch / time-invariant pitch shift.
303//
304// Synchronous granular overlap-add: two Hann-windowed grains, 50% overlap.
305// Every PCM_STRETCH_HOP output samples a new grain spawns at the input
306// timeline position, which advances at (input frames / output samples) set
307// by the fit target -- independent of the per-sample read step *within* a
308// grain, which carries the pitch (note tuning + preset samplerate
309// conversion). Decoupling those two rates is the whole trick:
310// fit=N ticks : timeline rate = remaining_frames / target_samples,
311// grain step = pitch ratio -> duration locked, pitch free
312// fit=0 : timeline rate = preset_sr / AMY_SAMPLE_RATE (original
313// duration), grain step = pitch ratio -> pitch shift at
314// constant length
315// This was chosen over an FFT phase vocoder deliberately: no FFT, no big
316// tables, a handful of multiplies per sample, all s.15/Q16 fixed point --
317// it runs on every AMY platform including ESP32-S3/RP2350 class parts.
318//
319// The first note-on hop starts BOTH grains at the same input position, one
320// at the top of the window and one at the halfway (peak) point. Their
321// windows sum to 1 over the first hop, so the first ~12 ms reproduce the
322// input exactly: drum transients are not softened by a window fade-in.
323
324// Read one frame for the stretcher, honoring PCM_LEFT/PCM_RIGHT/mix.
325static inline LUTSAMPLE pcm_stretch_read(const LUTSAMPLE *table, uint8_t channels, uint16_t wave, uint32_t idx) {
326 if (channels == 2) {
327 if (wave == PCM_LEFT) return table[idx * 2];
328 if (wave == PCM_RIGHT) return table[idx * 2 + 1];
329 return (LUTSAMPLE)(((int32_t)table[idx * 2] + (int32_t)table[idx * 2 + 1]) / 2);
330 }
331 return table[idx];
332}

334// Configure the stretcher at note-on. preset must be in-memory (not FILE).
335static void pcm_stretch_note_on(uint16_t osc, memorypcm_preset_t *preset) {
336 pcm_stretch_t *st = &synth[osc]->stretch;
337 uint32_t start_frame = INT_OF_P(synth[osc]->phase, PCM_INDEX_BITS);
338 if (start_frame >= preset->length) start_frame = 0;
339 uint32_t remaining = preset->length - start_frame;
340 // Target output duration in samples at AMY_SAMPLE_RATE.
341 float target;
342 if (synth[osc]->fit_ticks > 0) {
343 target = synth[osc]->fit_ticks * (float)amy_global.us_per_tick * (float)AMY_SAMPLE_RATE / 1000000.0f;
344 } else {
345 // fit=0: keep the sample's own duration; note pitch changes only.
346 target = (float)remaining * (float)AMY_SAMPLE_RATE / (float)preset->samplerate;
347 }
348 if (target < 1.0f) target = 1.0f;
349 st->hop_advance_q16 = (uint32_t)(((float)remaining * (float)PCM_STRETCH_HOP / target) * 65536.0f);
350 st->in_pos_q16 = ((uint64_t)start_frame << 16) + st->hop_advance_q16;
351 for (int g = 0; g < PCM_STRETCH_GRAINS; ++g) {
352 st->grain[g].active = 1;
353 st->grain[g].start_frame = start_frame;
354 st->grain[g].phase_q16 = 0;
355 // Grain 0 starts at the window peak (as if born a hop ago), grain 1
356 // at the top; see the note above about preserving the transient.
357 st->grain[g].win_pos = (g == 0) ? PCM_STRETCH_HOP : 0;
358 }
359 st->hop_counter = PCM_STRETCH_HOP;
360 // fit=N ticks is tempo-locked, so remember what a tick was worth here;
361 // render_pcm_stretch rescales the rate if that changes mid-note. fit=0
362 // asks for the sample's own duration, which no tempo can move, so it is
363 // flagged (0) as not tempo-locked.
364 st->us_per_tick_ref = (synth[osc]->fit_ticks > 0) ? amy_global.us_per_tick : 0;
365 st->ended = 0;
366 st->active = 1;
367}

369// WSOLA-style spawn alignment: search this many frames either side of the
370// nominal grain start for the best phase match with the still-playing grain.
371// Without it, a pure tone through the stretcher lands on the nearest line of
372// a comb spaced at the hop rate (~86 Hz at 44.1 kHz) instead of the exact
373// target pitch; broadband drums barely care, tonal material does.
374//
375// The half-width is per-osc at runtime (fit_search / 'pS'); PCM_STRETCH_SEARCH
376// is the default when it's unset and PCM_STRETCH_SEARCH_MAX the ceiling any
377// request is clamped to. Setting PCM_STRETCH_SEARCH to 0 compiles the search
378// out entirely for very small parts, and then 'pS' does nothing.
379//
380// How wide it needs to be is set by the lowest pitch present: the search has
381// to be able to reach half a period, so the default 64 covers periods down to
382// about AMY_SAMPLE_RATE/128 -- fine from ~345 Hz up (at 44.1 kHz), useless on
383// a bass line, where the aligner lands on a comb line instead of the note. A
384// 110 Hz tone needs 200 frames of reach, and sounds like 141 Hz without them
385// (experiments/sampler/fit_search_test.py measures it).
386#define PCM_STRETCH_SEARCH 64
387#define PCM_STRETCH_SEARCH_MAX 512
388// Multiplies per candidate offset, so the whole search costs
389// (2*search+1)*TAPS per hop no matter how wide it is: the comparison window
390// grows with the search (it also has to span about a period, or the
391// correlation can't tell one lag from another) but the stride opens up to
392// match, keeping the tap count fixed. Decimating like that aliases, which
393// would matter if the wide settings were for bright material -- they're for
394// bass, where the taps still land many per period.
395#define PCM_STRETCH_SEARCH_TAPS 64
396
397// Resolve the per-osc request ('pS'): unset means the default, anything over
398// the ceiling is clamped rather than refused so a part can ask for "as wide
399// as you can" without knowing the build's budget.
400static inline uint16_t pcm_stretch_search_width(uint16_t osc) {
401 uint16_t s = synth[osc]->fit_search;
402 if (AMY_IS_UNSET(s)) return PCM_STRETCH_SEARCH;
403 if (s > PCM_STRETCH_SEARCH_MAX) return PCM_STRETCH_SEARCH_MAX;
404 return s;
405}

407static int32_t pcm_stretch_best_offset(const LUTSAMPLE *table, uint8_t channels, uint16_t wave,
408 uint32_t length, uint32_t target, uint32_t nominal,
409 uint16_t search) {
410#if PCM_STRETCH_SEARCH > 0
411 if (search == 0) return 0;
412 uint32_t stride = (search > PCM_STRETCH_SEARCH_TAPS) ? (search / PCM_STRETCH_SEARCH_TAPS) : 1;
413 uint32_t win = stride * PCM_STRETCH_SEARCH_TAPS; // == 64 at the default
414 // Bail if either segment would run off the sample.
415 if (target + win >= length) return 0;
416 int32_t lo = -(int32_t)search, hi = (int32_t)search;
417 if ((int32_t)nominal + lo < 0) lo = -(int32_t)nominal;
418 if (nominal + hi + win >= length) {
419 if (nominal + win >= length) return 0;
420 hi = (int32_t)(length - win - 1 - nominal);
421 }
422 if (lo > hi) return 0;
423 int32_t best_d = 0, best_corr = INT32_MIN;
424 for (int32_t d = lo; d <= hi; ++d) {
425 int32_t corr = 0;
426 for (uint32_t i = 0; i < win; i += stride) {
427 int32_t a = pcm_stretch_read(table, channels, wave, target + i) >> 4;
428 int32_t b = pcm_stretch_read(table, channels, wave, nominal + d + i) >> 4;
429 corr += a * b;
430 }
431 if (corr > best_corr) { best_corr = corr; best_d = d; }
432 }
433 return best_d;
434#else
435 (void)table; (void)channels; (void)wave; (void)length; (void)target; (void)nominal;
436 (void)search;
437 return 0;
438#endif
439}

441// Spawn a replacement grain at the current input timeline position.
442static void pcm_stretch_spawn(pcm_stretch_t *st, memorypcm_preset_t *preset, uint16_t wave,
443 bool looping, uint32_t loopstart, uint32_t loopend, uint16_t search) {
444 uint32_t length = preset->length;
445 st->hop_counter = PCM_STRETCH_HOP;
446 if (st->ended) return;
447 uint32_t frame = (uint32_t)(st->in_pos_q16 >> 16);
448 if (looping) {
449 // Wrap the timeline into the loop region.
450 uint32_t le = (loopend > loopstart && loopend <= length) ? loopend : length;
451 uint32_t ls = (loopstart < le) ? loopstart : 0;
452 if (frame >= le) {
453 frame = ls + ((frame - le) % (le - ls));
454 st->in_pos_q16 = ((uint64_t)frame << 16) | (st->in_pos_q16 & 0xffff);
455 }
456 } else if (frame >= length) {
457 // Input exhausted; let the live grains fade out.
458 st->ended = 1;
459 return;
460 }
461 // Reuse the expired grain (it hits win_pos == PCM_STRETCH_GRAIN exactly
462 // when the spawn is due); fall back to the oldest.
463 int gi = 0;
464 for (int g = 0; g < PCM_STRETCH_GRAINS; ++g) {
465 if (!st->grain[g].active) { gi = g; break; }
466 if (st->grain[g].win_pos > st->grain[gi].win_pos) gi = g;
467 }
468 // Phase-align the new grain against the one it will overlap: search near
469 // the nominal start for the segment that best matches what that grain is
470 // about to play. The timeline itself (in_pos) is never adjusted, so the
471 // alignment jitter can't accumulate into a tempo error.
472 int32_t d = 0;
473 int other = 1 - gi; // PCM_STRETCH_GRAINS == 2
474 if (st->grain[other].active) {
475 uint32_t target = st->grain[other].start_frame + (st->grain[other].phase_q16 >> 16);
476 d = pcm_stretch_best_offset(preset->sample_ram, preset->channels, wave, length, target, frame,
477 search);
478 }
479 st->grain[gi].active = 1;
480 st->grain[gi].start_frame = (uint32_t)((int32_t)frame + d);
481 st->grain[gi].phase_q16 = st->in_pos_q16 & 0xffff;
482 st->grain[gi].win_pos = 0;
483 st->in_pos_q16 += st->hop_advance_q16;
484}

486static SAMPLE render_pcm_stretch(SAMPLE *buf, uint16_t osc, memorypcm_preset_t *preset) {
487 pcm_stretch_t *st = &synth[osc]->stretch;
488 // A tempo change has to reach notes that are ALREADY sounding: fit= locks
489 // a note to a number of ticks, and a tick just got longer or shorter, so
490 // a note left alone would run on at the old tempo and land off the grid.
491 //
492 // The timeline rate is (input frames / target output samples) and the
493 // target is proportional to us_per_tick, so following the change is a
494 // rescale of the existing rate, not a recomputation from what is left.
495 // That distinction matters: "input frames remaining" is a finite quantity
496 // only for a one-shot, while a looping preset wraps forever -- the
497 // rescale is correct for both, and stays correct across repeated changes
498 // because the reference is updated each time.
499 if (st->us_per_tick_ref != 0 && st->us_per_tick_ref != amy_global.us_per_tick
500 && amy_global.us_per_tick > 0) {
501 st->hop_advance_q16 = (uint32_t)(((uint64_t)st->hop_advance_q16
502 * st->us_per_tick_ref) / amy_global.us_per_tick);
503 st->us_per_tick_ref = amy_global.us_per_tick;
504 }
505 SAMPLE max_value = 0;
506 float logfreq = msynth[osc]->logfreq;
507 if (AMY_IS_SET(synth[osc]->midi_note)) {
508 logfreq -= logfreq_for_midi_note(preset->midinote);
509 }
510 float playback_freq = freq_of_logfreq(preset->log2sr + logfreq);
511 // Same exact-native-rate correction as render_pcm.
512 if (logfreq == 0 && AMY_IS_UNSET(synth[osc]->midi_note))
513 playback_freq = (float)preset->samplerate;
514 // Per-sample read step within a grain: the pitch, recomputed per block so
515 // envelopes/LFOs on freq keep working.
516 uint32_t pitch_step_q16 = (uint32_t)((playback_freq / (float)AMY_SAMPLE_RATE) * 65536.0f);
517 SAMPLE amp = F2S(msynth[osc]->amp);
518 const LUTSAMPLE *table = preset->sample_ram;
519 uint32_t length = preset->length;
520 // Re-read each block, so 'pS' can be swept while the note is sounding.
521 uint16_t search = pcm_stretch_search_width(osc);
522 bool looping = mode_is_looping(msynth[osc]->state);
523 uint32_t loopstart = msynth[osc]->loopstart;
524 uint32_t loopend = (msynth[osc]->loopend > loopstart && msynth[osc]->loopend <= length) ? msynth[osc]->loopend : length;
525 uint16_t i = 0;
526 if (msynth[osc]->pcm_delay) {
527 // sample_offset: leave the head of the note-on block silent.
528 i = msynth[osc]->pcm_delay;
529 msynth[osc]->pcm_delay = 0;
530 }
531 for (; i < AMY_BLOCK_SIZE; i++) {
532 if (st->hop_counter == 0)
533 pcm_stretch_spawn(st, preset, synth[osc]->wave, looping, loopstart, loopend, search);
534 st->hop_counter--;
535 SAMPLE out = 0;
536 uint8_t any_active = 0;
537 for (int g = 0; g < PCM_STRETCH_GRAINS; ++g) {
538 if (!st->grain[g].active) continue;
539 any_active = 1;
540 uint32_t idx = st->grain[g].start_frame + (st->grain[g].phase_q16 >> 16);
541 if (looping && idx >= loopend)
542 idx = loopstart + ((idx - loopend) % (loopend - loopstart));
543 if (idx < length) {
544 LUTSAMPLE b = pcm_stretch_read(table, preset->channels, synth[osc]->wave, idx);
545 LUTSAMPLE c = (idx + 1 < length) ? pcm_stretch_read(table, preset->channels, synth[osc]->wave, idx + 1) : b;
546 SAMPLE frac = (SAMPLE)(st->grain[g].phase_q16 & 0xffff) << (S_FRAC_BITS - 16);
547 SAMPLE samp = L2S(b) + MUL4_SS(L2S(c - b), frac);
548 out += MUL0_SS(samp, L2S(stretch_win[st->grain[g].win_pos]));
549 }
550 st->grain[g].phase_q16 += pitch_step_q16;
551 if (++st->grain[g].win_pos >= PCM_STRETCH_GRAIN) st->grain[g].active = 0;
552 }
553 if (!any_active) {
554 if (st->ended) {
555 synth[osc]->status = SYNTH_OFF;
556 st->active = 0;
557 }
558 break;
559 }
560 SAMPLE value = buf[i] + MUL4_SS(amp, out);
561 buf[i] = value;
562 if (value < 0) value = -value;
563 if (value > max_value) max_value = value;
564 }
565 return max_value;
566}

568void pcm_note_on(uint16_t osc) {
569 if(AMY_IS_SET(synth[osc]->preset)) {
570 memorypcm_preset_t rom_local;
571 memorypcm_preset_t *preset =
572 get_preset_for_preset_number(synth[osc]->preset, &rom_local);
573 if (preset->type == AMY_PCM_TYPE_FILE) {
574 if (preset->file_handle != 0) {
575 wave_info_t info = {0};
576 uint32_t data_bytes = 0;
577 amy_global.config.amy_external_fseek_hook(preset->file_handle, 0);
578 if (wave_parse_header(preset->file_handle, &info, &data_bytes)) {
579 preset->channels = info.channels;
580 preset->samplerate = info.sample_rate;
581 preset->log2sr = log2f((float)info.sample_rate / ZERO_LOGFREQ_IN_HZ);
582 preset->file_bytes_remaining = data_bytes;
583 } else {
584 amy_global.config.amy_external_fclose_hook(preset->file_handle);
585 }
586 }
587 } else if (preset->type == AMY_PCM_TYPE_ROM) {
588 // baked-in PCM - don't overrun.
589 if(synth[osc]->preset >= pcm_samples) synth[osc]->preset = 0;
590 }
591 PHASOR phase;
592 if (AMY_IS_SET(synth[osc]->trigger_phase)) {
593 // trigger_phase (P) sets the sample start point for this
594 // note-on (start_frame / 2^PCM_INDEX_BITS).
595 phase = F2P(synth[osc]->trigger_phase);
596 } else {
597 phase = 0; // s16.15 index into the table; as if a PHASOR into a 16 bit sample table.
598 }
599 // Does this note-on want the granular fit engine? Only for in-memory
600 // presets: it needs random access, which a streamed file can't give.
601 bool want_stretch = AMY_IS_SET(synth[osc]->fit_ticks)
602 && preset->type != AMY_PCM_TYPE_FILE
603 && preset->sample_ram != NULL && preset->length > 0;
604 bool fresh_start = true;
605 if (synth[osc]->status == SYNTH_AUDIBLE && preset->type != AMY_PCM_TYPE_FILE
606 && !want_stretch && !synth[osc]->stretch.active) {
607 // Restarting a currently-playing (non-file) PCM, splice the reonset
608 // onto the old tail to avoid a click (mechanism per the gate below).
609 // (Not for the fit engine: its grains are windowed, so a restart is click-free by construction.)
610 fresh_start = false;
611 uint32_t base_index = INT_OF_P(synth[osc]->phase, PCM_INDEX_BITS);
612#if AMY_PCM_RETRIG_ZERO_CROSS
613 // Restarting a currently-playing (non-file) PCM, delay reonset to next zero crossing to avoid click.
614 msynth[osc]->loopend = pcm_find_next_zero_crossing(osc, base_index);
615#else
616 // LOCAL EDIT: fade-restart - play PCM_RETRIG_FADE_FRAMES more of
617 // the old tail under a linear ramp to zero (applied in render_pcm),
618 // then splice. Constant latency where the zero-cross defer's is
619 // tail-phase-dependent; see the gate comment above.
620 uint32_t fade_end = base_index + PCM_RETRIG_FADE_FRAMES;
621 if (preset->length && fade_end >= preset->length)
622 fade_end = preset->length - 1;
623 msynth[osc]->loopend = fade_end;
624#endif
625 msynth[osc]->loopstart = INT_OF_P(phase, PCM_INDEX_BITS);
626 msynth[osc]->state = PCM_LOOP_ONCE_INTERNAL;
627 msynth[osc]->next_state = synth[osc]->mode;
628 //fprintf(stderr, "time %.3f osc %d RESTART amp %.3f last_amp %.3f\n", amy_global.time, osc, msynth[osc]->amp, msynth[osc]->last_amp);
629 } else {
630 synth[osc]->phase = phase;
631 msynth[osc]->loopstart = preset->loopstart;
632 msynth[osc]->loopend = preset->loopend;
633 // Copy the looping mode from the wave mode field. Can be updated on note_off.
634 msynth[osc]->state = synth[osc]->mode;
635 }
636 // sample_offset (po): begin this note-on partway into its render
637 // block, so slices of arbitrary length can butt-join sample-
638 // accurately. Only meaningful on a fresh start; a zero-crossing
639 // restart already has fuzzy timing by design.
640 msynth[osc]->pcm_delay = 0;
641 if (fresh_start && AMY_IS_SET(synth[osc]->sample_offset))
642 msynth[osc]->pcm_delay = synth[osc]->sample_offset % AMY_BLOCK_SIZE;
643 if (want_stretch) pcm_stretch_note_on(osc, preset);
644 else synth[osc]->stretch.active = 0;
645 // Make sure PCM waveforms are excluded from auto-termination, so we don't cut-off samples with silent gaps. May be modified by note_off.
646 synth[osc]->terminate_on_silence = 0;
647 }
648}

655void pcm_note_off(uint16_t osc) {
656 if(AMY_IS_SET(synth[osc]->preset)) {
657 if (msynth[osc]->state == PCM_PLAY_STOP
658 || msynth[osc]->state == PCM_LOOP_STOP) {
659 // PCM mode where note off causes immediate stop.
660 //
661 // This used to seek phase past the end of the sample and let
662 // render_pcm notice on the next block. That worked only for
663 // in-memory presets: a streamed one refills from the file and
664 // resets phase to 0 every block, so the seek was thrown away and
665 // the clip played on to end-of-file, ignoring note-off entirely.
666 // PCM_PLAY_STOP is the DEFAULT mode, so that hit every
667 // disk_sample() note-off. Stopping the osc says what we mean and
668 // works for both kinds -- and it no longer needs the preset
669 // lookup that the seek needed just to find the sample length.
670 synth[osc]->status = SYNTH_OFF;
671 } else if (msynth[osc]->state == PCM_LOOP_FOREVER) {
672 // Sending one note-off to a LOOP_FOREVER loop downgrades it to a stoppable loop.
673 msynth[osc]->state = PCM_LOOP;
674 // Allow the engine to terminate it when it goes to silence (e.g. from envelope).
675 synth[osc]->terminate_on_silence = 1;
676 } else if (msynth[osc]->state == PCM_LOOP || msynth[osc]->state == PCM_PLAY) {
677 // Looping was enabled but after stop we just play through to the end.
678 // (sending a second note-off will stop it immediately).
679 msynth[osc]->state = PCM_PLAY_STOP;
680 }
681 }
682}

704SAMPLE render_pcm(SAMPLE* buf, uint16_t osc) {
705 if(AMY_IS_SET(synth[osc]->preset)) {
706 SAMPLE max_value = 0;
707 memorypcm_preset_t rom_local;
708 memorypcm_preset_t *preset =
709 get_preset_for_preset_number(synth[osc]->preset, &rom_local);
710 // fit= notes render through the granular stretch engine instead.
711 if (synth[osc]->stretch.active && preset->type != AMY_PCM_TYPE_FILE
712 && preset->sample_ram != NULL && preset->length > 0) {
713 return render_pcm_stretch(buf, osc, preset);
714 }
715 float logfreq = msynth[osc]->logfreq;
716 // If osc[midi_note] is set, shift the freq by the preset's default base_note.
717 if (AMY_IS_SET(synth[osc]->midi_note)) {
718 logfreq -= logfreq_for_midi_note(preset->midinote);
719 }
720 float playback_freq = freq_of_logfreq(preset->log2sr + logfreq);
721 // Unpitched playback (no note, no freq mods): use the preset's rate
722 // exactly. The log2f/exp2f roundtrip above lands a hair off the
723 // native rate, which makes untransposed samples drift by ~1 sample
724 // every few seconds -- enough to spoil sample-accurate butt-joins of
725 // slices against their source.
726 if (logfreq == 0 && AMY_IS_UNSET(synth[osc]->midi_note))
727 playback_freq = (float)preset->samplerate;
728 uint32_t sample_length = preset->length;
729 if (preset->type == AMY_PCM_TYPE_FILE) {
730 float frames_per_output = playback_freq / (float)AMY_SAMPLE_RATE;
731 uint32_t frames_needed = (uint32_t)ceilf(frames_per_output * AMY_BLOCK_SIZE) + 1;
732 uint32_t max_frames = AMY_BLOCK_SIZE * PCM_FILE_BUFFER_MULT;
733 if (frames_needed > max_frames) {
734 frames_needed = max_frames;
735 }
736 sample_length = fill_sample_from_file(preset, frames_needed);
737 if(sample_length != frames_needed) {
738 // reached end of file
739 synth[osc]->status = SYNTH_OFF;
740 }
741 synth[osc]->phase = 0;
742 }
743 if (preset->sample_ram == NULL || sample_length == 0) {
744 synth[osc]->status = SYNTH_OFF;
745 return 0;
746 }
747
748 SAMPLE amp = F2S(msynth[osc]->amp);
749 PHASOR step = F2P((playback_freq / (float)AMY_SAMPLE_RATE) / (float)(1 << (PCM_INDEX_BITS - PCM_INDEX_STEP_EXTRA_BITS)));
750 const LUTSAMPLE* table = preset->sample_ram;
751 uint32_t base_index_base = INT_OF_P(synth[osc]->phase, PCM_INDEX_BITS);
752 uint32_t base_index = base_index_base;
753 PHASOR phase = (synth[osc]->phase - (base_index_base << PCM_INDEX_FRAC_BITS)) << PCM_INDEX_STEP_EXTRA_BITS;
754 // sample_offset (po): a fresh note-on starts partway into this block,
755 // leaving the head silent, so slices can butt-join sample-accurately.
756 uint16_t start_i = 0;
757 if (msynth[osc]->pcm_delay) {
758 start_i = msynth[osc]->pcm_delay;
759 msynth[osc]->pcm_delay = 0;
760 }
761 for(uint16_t i=start_i; i < AMY_BLOCK_SIZE; i++) {
762 SAMPLE frac = S_FRAC_OF_P(phase, PCM_INDEX_BITS - PCM_INDEX_STEP_EXTRA_BITS);
763 LUTSAMPLE b = 0;
764 LUTSAMPLE c = 0;
765 uint32_t next_index = base_index + 1;
766 // For non-file samples, we have to check for end of sample/looping.
767 if(preset->type != AMY_PCM_TYPE_FILE) {
768 if ((msynth[osc]->state == PCM_LOOP
769 || msynth[osc]->state == PCM_LOOP_ONCE_INTERNAL
770 || msynth[osc]->state == PCM_LOOP_STOP
771 || msynth[osc]->state == PCM_LOOP_FOREVER)
772 && base_index >= msynth[osc]->loopend) { // loopend
773 // still looping. The state may be modified by pcm_note_off.
774 // back to loopstart
775 phase &= ((1L << (PCM_INDEX_FRAC_BITS + PCM_INDEX_STEP_EXTRA_BITS)) - 1);
776 base_index_base = msynth[osc]->loopstart + (base_index - msynth[osc]->loopend);
777 if (msynth[osc]->state == PCM_LOOP_ONCE_INTERNAL) {
778 msynth[osc]->state = msynth[osc]->next_state; // Only loops once.
779 msynth[osc]->loopstart = preset->loopstart;
780 msynth[osc]->loopend = preset->loopend;
781 }
782 //fprintf(stderr, "time %.3f sample %d LOOP: old_index %d new_index %d phase 0x%lx\n", amy_global.time, i, base_index, base_index_base, phase);
783 base_index = base_index_base;
784 } else if(base_index >= sample_length) { // end
785 synth[osc]->status = SYNTH_OFF;// is this right?
786 buf[i] = 0;
787 break;
788 }
789 }
790 if (preset->channels == 2) {
791 uint32_t base_offset = base_index * 2;
792 uint32_t next_offset = next_index * 2;
793 if (synth[osc]->wave == PCM_LEFT) {
794 b = table[base_offset];
795 c = (next_index < sample_length) ? table[next_offset] : b;
796 } else if (synth[osc]->wave == PCM_RIGHT) {
797 b = table[base_offset + 1];
798 c = (next_index < sample_length) ? table[next_offset + 1] : b;
799 } else { // PCM or PCM_MIX
800 LUTSAMPLE bl = table[base_offset];
801 LUTSAMPLE br = table[base_offset + 1];
802 b = (LUTSAMPLE)(((int32_t)bl + (int32_t)br) / 2);
803 if (next_index < sample_length) {
804 LUTSAMPLE cl = table[next_offset];
805 LUTSAMPLE cr = table[next_offset + 1];
806 c = (LUTSAMPLE)(((int32_t)cl + (int32_t)cr) / 2);
807 } else {
808 c = b;
809 }
810 }
811 } else {
812 b = table[base_index];
813 c = (next_index < sample_length) ? table[next_index] : b;
814 }
815 SAMPLE sample = L2S(b) + MUL4_SS(L2S(c - b), frac);
816#if !AMY_PCM_RETRIG_ZERO_CROSS
817 // LOCAL EDIT: retrig fade - gain ramps 1 -> 0 across the fade span
818 // armed in pcm_note_on. After the splice the state has already
819 // advanced to next_state, so new-note samples pass unscaled.
820 if (preset->type != AMY_PCM_TYPE_FILE
821 && msynth[osc]->state == PCM_LOOP_ONCE_INTERNAL) {
822 uint32_t rem = msynth[osc]->loopend - base_index;
823 if (rem > PCM_RETRIG_FADE_FRAMES) rem = PCM_RETRIG_FADE_FRAMES;
824 sample = MUL4_SS(sample, (SAMPLE)(int32_t)rem * SHIFTR(F2S(1.0f), 6));
825 }
826#endif
827 SAMPLE value = buf[i] + MUL4_SS(amp, sample);
828 buf[i] = value;
829 if (value < 0) value = -value;
830 if (value > max_value) max_value = value;
831 phase = P_WRAPPED_SUM(phase, step);
832 base_index = base_index_base + INT_OF_P(phase, PCM_INDEX_BITS - PCM_INDEX_STEP_EXTRA_BITS);
833 }
834 //synth[osc]->phase = phase;
835 synth[osc]->phase = I2P(base_index, PCM_INDEX_BITS) + (S_FRAC_OF_P(phase, PCM_INDEX_BITS - PCM_INDEX_STEP_EXTRA_BITS) >> (S_FRAC_BITS - (PCM_INDEX_FRAC_BITS)) ); // + PCM_INDEX_STEP_EXTRA_BITS
836 //fprintf(stderr, "\rtime %.3f osc %d render_pcm7: preset %d len %d base_ix 0x%lx phase 0x%lx sfracofp 0x%lx step 0x%lx synthphase 0x%lx amp %.3f\n",
837 // amy_global.time, osc, synth[osc]->preset, preset->length, base_index, phase, S_FRAC_OF_P(phase, PCM_INDEX_BITS - PCM_INDEX_STEP_EXTRA_BITS) >> (S_FRAC_BITS - (PCM_INDEX_FRAC_BITS + PCM_INDEX_STEP_EXTRA_BITS)), step, synth[osc]->phase, S2F(msynth[osc]->amp));
838 return max_value;
839 // i don't believe we ever need to detect silence in a sample. it will shut itself off at the end.
840 }
841 return 0;
842}

856int pcm_load_file() {
857 // We pass the inputs to this as aliases in the amy_global structure. This is to not destroy the MP heap for amy->AMYboard
858 uint8_t midinote = amy_global.transfer_stored_bytes;

861
862 pcm_unload_preset(preset_number);
863 if (filename == NULL || filename[0] == '\0') {
864 return 0;
865 }
866 if (amy_global.config.amy_external_fopen_hook == NULL || amy_global.config.amy_external_fclose_hook == NULL) {
867 fprintf(stderr, "fopen hook not enabled on platform\n");
868 return 0;
869 }
870 uint32_t handle = amy_global.config.amy_external_fopen_hook((char *)filename, "rb");
871 if (handle == 0) {
872 fprintf(stderr, "Could not open file %s\n", filename);

874 }

881 }

885 }

887 memorypcm_ll_t *new_preset_pointer = malloc_caps(
888 sizeof(memorypcm_ll_t) + sizeof(memorypcm_preset_t) + buffer_frames * sizeof(int16_t),
889 amy_global.config.ram_caps_sample);
890 if (new_preset_pointer == NULL) {
891 fprintf(stderr, "No RAM left for sample load\n");
892 return 0;
893 }
894 new_preset_pointer->next = memorypcm_ll_start;
895 memorypcm_ll_start = new_preset_pointer;
896 new_preset_pointer->preset_number = preset_number;
897 memorypcm_preset_t *memory_preset =
898 (memorypcm_preset_t *)(((uint8_t *)new_preset_pointer) + sizeof(memorypcm_ll_t));
899 strncpy(memory_preset->filename, filename, MAX_FILENAME_LEN - 1);
900 memory_preset->filename[MAX_FILENAME_LEN - 1] = '\0';
901 memory_preset->channels = info.channels;
902 memory_preset->samplerate = info.sample_rate;
903 memory_preset->log2sr = log2f((float)info.sample_rate / ZERO_LOGFREQ_IN_HZ);
904 memory_preset->midinote = midinote;
905 memory_preset->length = total_frames;
906 memory_preset->type = AMY_PCM_TYPE_FILE;
907 memory_preset->file_bytes_remaining = total_frames * info.channels * 2;
908 memory_preset->file_handle = handle;
909 memory_preset->sample_ram = malloc_caps(buffer_frames * info.channels * sizeof(int16_t),
910 amy_global.config.ram_caps_sample);
911 new_preset_pointer->preset = memory_preset;
912 //fprintf(stderr, "read file %s frames %ld channels %d preset %d handle %ld\n", filename, total_frames, info.channels, preset_number, handle);
913 return 1;
914}

917// load mono samples (let python parse wave files) into preset #
918// set loopstart, loopend, midinote, samplerate (and log2sr)
919// return the allocated sample ram that AMY will fill in.
920int16_t * pcm_load(uint16_t preset_number, uint32_t length, uint32_t samplerate, uint8_t channels, float midinote, uint32_t loopstart, uint32_t loopend) {
921 // if preset was already a memorypcm, we need to unload it
922 pcm_unload_preset(preset_number); // this is a no-op if preset doesn't exist or is a const pcm
923 // now alloc a new LL entry and preset (the old LL entry is removed with pcm_unload_preset)
924 memorypcm_ll_t *new_preset_pointer = malloc_caps(sizeof(memorypcm_ll_t) + sizeof(memorypcm_preset_t) + length * channels * sizeof(int16_t),
925 amy_global.config.ram_caps_sample);
926 if(new_preset_pointer == NULL) {
927 fprintf(stderr, "No RAM left for sample load\n");
928 return NULL; // no ram for sample
929 }
930 new_preset_pointer->next = memorypcm_ll_start;
931 memorypcm_ll_start = new_preset_pointer;
932 new_preset_pointer->preset_number = preset_number;
933 memorypcm_preset_t *memory_preset = (memorypcm_preset_t *)(((uint8_t *)new_preset_pointer) + sizeof(memorypcm_ll_t));
934 memory_preset->samplerate = samplerate;
935 memory_preset->log2sr = log2f((float)samplerate / ZERO_LOGFREQ_IN_HZ);
936 memory_preset->midinote = midinote;
937 memory_preset->loopstart = loopstart;
938 memory_preset->length = length;
939 memory_preset->channels = channels;
940 memory_preset->filename[0] = '\0';
941 memory_preset->file_bytes_remaining = 0;
942 memory_preset->file_handle = 0;
943 memory_preset->type = AMY_PCM_TYPE_MEMORY;
