61// Get either memory preset, file preset or baked in preset for preset number.
62// For ROM presets, fill the caller-provided rom_local and return it.
63memorypcm_preset_t * get_preset_for_preset_number(uint16_t preset_number,
64 memorypcm_preset_t *rom_local) {
65 // Get the memory preset. If we can't find it, it could be a ROM preset. So copy params in from ROM preset
66 memorypcm_ll_t *preset = memorypcm_ll_start;
67 while(preset != NULL) {
68 if(preset->preset_number == preset_number) {
69 if(preset->preset->sample_ram != NULL || preset->preset->file_handle > 0) {
70 return preset->preset;
71 }
72 }
73 preset = preset->next;
74 }
75
76#ifdef GAMMA9001
77 // Gamma9001 drum banks live at GAMMA9001_PRESET_BASE+, read straight out
78 // of the platform-provided blob (memory presets above may still shadow them).
79 if (preset_number >= GAMMA9001_PRESET_BASE &&
80 preset_number < GAMMA9001_PRESET_BASE + GAMMA9001_NUM_SAMPLES &&
81 gamma9001_pcm != NULL && rom_local != NULL) {
82 const pcm_map_t *g = &gamma9001_map[preset_number - GAMMA9001_PRESET_BASE];
83 memset(rom_local, 0, sizeof(*rom_local));
84 rom_local->sample_ram = (int16_t *)gamma9001_pcm + g->offset;
85 rom_local->length = g->length;
86 rom_local->loopstart = g->loopstart;
87 rom_local->loopend = g->loopend;
88 rom_local->midinote = g->midinote;
89 rom_local->samplerate = GAMMA9001_SAMPLE_RATE;
90 rom_local->log2sr = log2f((float)GAMMA9001_SAMPLE_RATE / ZERO_LOGFREQ_IN_HZ);
91 rom_local->type = AMY_PCM_TYPE_GAMMA;
92 rom_local->channels = 1;
93 return rom_local;
94 }
95#endif
96
97 // No memory preset found, so try ROM preset. default to 0 if out of range
98 if (preset_number >= pcm_samples) preset_number = 0;
99 if (rom_local == NULL) {
100 return NULL;
101 }
102 memset(rom_local, 0, sizeof(*rom_local));
103 const pcm_map_t cpreset = pcm_map[preset_number];
104 uint32_t offset = cpreset.offset;
105 uint32_t length = cpreset.length;
106#ifdef PCM_LENGTH
107 if (offset >= PCM_LENGTH) {
108 offset = 0;
109 length = 0;
110 } else if (length > (PCM_LENGTH - offset)) {
111 length = PCM_LENGTH - offset;
112 }
113#endif
114 rom_local->sample_ram = (int16_t*)pcm + offset;
115 rom_local->length = length;
116 rom_local->loopstart = cpreset.loopstart;
117 rom_local->loopend = cpreset.loopend;
118 if (rom_local->loopstart > rom_local->length) {
119 rom_local->loopstart = 0;
120 }
121 if (rom_local->loopend > rom_local->length) {
122 rom_local->loopend = rom_local->length;
123 }
124 rom_local->midinote = cpreset.midinote;
125 rom_local->samplerate = PCM_AMY_SAMPLE_RATE;
126 rom_local->log2sr = PCM_AMY_LOG2_SAMPLE_RATE;
127 rom_local->type = AMY_PCM_TYPE_ROM;
128 rom_local->channels = 1;
129 return rom_local;
130}

169void pcm_note_on(uint16_t osc) {
170 if(AMY_IS_SET(synth[osc]->preset)) {
171 memorypcm_preset_t rom_local;
172 memorypcm_preset_t *preset =
173 get_preset_for_preset_number(synth[osc]->preset, &rom_local);
174 if (preset->type == AMY_PCM_TYPE_FILE) {
175 if (preset->file_handle != 0) {
176 wave_info_t info = {0};
177 uint32_t data_bytes = 0;
178 amy_global.config.amy_external_fseek_hook(preset->file_handle, 0);
179 if (wave_parse_header(preset->file_handle, &info, &data_bytes)) {
180 preset->channels = info.channels;
181 preset->samplerate = info.sample_rate;
182 preset->log2sr = log2f((float)info.sample_rate / ZERO_LOGFREQ_IN_HZ);
183 preset->file_bytes_remaining = data_bytes;
184 } else {
185 amy_global.config.amy_external_fclose_hook(preset->file_handle);
186 }
187 }
188 } else if (preset->type == AMY_PCM_TYPE_ROM) {
189 // baked-in PCM - don't overrun.
190 if(synth[osc]->preset >= pcm_samples) synth[osc]->preset = 0;
191 }
192
193 synth[osc]->phase = 0; // s16.15 index into the table; as if a PHASOR into a 16 bit sample table.
194 // Special case: We use the msynth feedback flag to indicate note-off for looping PCM. As a result, it's explicitly NOT set in amy:hold_and_modify for PCM voices. Set it here.
195 msynth[osc]->feedback = synth[osc]->feedback;
196
197 // Make sure PCM waveforms are excluded from auto-termination, so we don't cut-off samples with silent gaps.
198 synth[osc]->terminate_on_silence = 0;
199 }
200}

207void pcm_note_off(uint16_t osc) {
208 if(AMY_IS_SET(synth[osc]->preset)) {
209 uint32_t length = 0;
210 memorypcm_preset_t rom_local;
211 memorypcm_preset_t *preset =
212 get_preset_for_preset_number(synth[osc]->preset, &rom_local);
213 if(preset != NULL) {
214 length = preset->length;
215 }
216 if(msynth[osc]->feedback == 0) {
217 // Non-looping note: Set phase to the end to cause immediate stop.
218 synth[osc]->phase = F2P(length / (float)(1 << PCM_INDEX_BITS));
219 } else {
220 // Looping is requested, disable future looping, sample will play through to end.
221 // (sending a second note-off will stop it immediately).
222 msynth[osc]->feedback = 0;
223 }
224 }

247SAMPLE render_pcm(SAMPLE* buf, uint16_t osc) {
248 if(AMY_IS_SET(synth[osc]->preset)) {
249 SAMPLE max_value = 0;
250 memorypcm_preset_t rom_local;
251 memorypcm_preset_t *preset =
252 get_preset_for_preset_number(synth[osc]->preset, &rom_local);
253 float logfreq = msynth[osc]->logfreq;
254 // If osc[midi_note] is set, shift the freq by the preset's default base_note.
255 if (AMY_IS_SET(synth[osc]->midi_note)) {
256 logfreq -= logfreq_for_midi_note(preset->midinote);
257 }
258 float playback_freq = freq_of_logfreq(preset->log2sr + logfreq);
259 uint32_t sample_length = preset->length;
260 if (preset->type == AMY_PCM_TYPE_FILE) {
261 float frames_per_output = playback_freq / (float)AMY_SAMPLE_RATE;
262 uint32_t frames_needed = (uint32_t)ceilf(frames_per_output * AMY_BLOCK_SIZE) + 1;
263 uint32_t max_frames = AMY_BLOCK_SIZE * PCM_FILE_BUFFER_MULT;
264 if (frames_needed > max_frames) {
265 frames_needed = max_frames;
266 }
267 sample_length = fill_sample_from_file(preset, frames_needed);
268 if(sample_length != frames_needed) {
269 // reached end of file
270 synth[osc]->status = SYNTH_OFF;
271 }
272 synth[osc]->phase = 0;
273 }
274 if (preset->sample_ram == NULL || sample_length == 0) {
275 synth[osc]->status = SYNTH_OFF;
276 return 0;
277 }
278
279 SAMPLE amp = F2S(msynth[osc]->amp);
280 PHASOR step = F2P((playback_freq / (float)AMY_SAMPLE_RATE) / (float)(1 << PCM_INDEX_BITS));
281 const LUTSAMPLE* table = preset->sample_ram;
282 uint32_t base_index = INT_OF_P(synth[osc]->phase, PCM_INDEX_BITS);
283 for(uint16_t i=0; i < AMY_BLOCK_SIZE; i++) {
284 SAMPLE frac = S_FRAC_OF_P(synth[osc]->phase, PCM_INDEX_BITS);
285 LUTSAMPLE b = 0;
286 LUTSAMPLE c = 0;
287 uint32_t next_index = base_index + 1;
288 if (base_index >= sample_length) {
289 if (preset->type != AMY_PCM_TYPE_FILE) {
290 synth[osc]->status = SYNTH_OFF;
291 }
292 buf[i] = 0;
293 continue;
294 }
295 if (preset->channels == 2) {
296 uint32_t base_offset = base_index * 2;
297 uint32_t next_offset = next_index * 2;
298 if (synth[osc]->wave == PCM_LEFT) {
299 b = table[base_offset];
300 c = (next_index < sample_length) ? table[next_offset] : b;
301 } else if (synth[osc]->wave == PCM_RIGHT) {
302 b = table[base_offset + 1];
303 c = (next_index < sample_length) ? table[next_offset + 1] : b;
304 } else { // PCM or PCM_MIX
305 LUTSAMPLE bl = table[base_offset];
306 LUTSAMPLE br = table[base_offset + 1];
307 b = (LUTSAMPLE)(((int32_t)bl + (int32_t)br) / 2);
308 if (next_index < sample_length) {
309 LUTSAMPLE cl = table[next_offset];
310 LUTSAMPLE cr = table[next_offset + 1];
311 c = (LUTSAMPLE)(((int32_t)cl + (int32_t)cr) / 2);
312 } else {
313 c = b;
314 }
315 }
316 } else {
317 b = table[base_index];
318 c = (next_index < sample_length) ? table[next_index] : b;
319 }
320 SAMPLE sample = L2S(b) + MUL4_SS(L2S(c - b), frac);
321 synth[osc]->phase = P_WRAPPED_SUM(synth[osc]->phase, step);
322 base_index = INT_OF_P(synth[osc]->phase, PCM_INDEX_BITS);
323
324 if(preset->type != AMY_PCM_TYPE_FILE) {
325 // For non-file samples, we have to check for end of sample/looping.
326 if(base_index >= sample_length) { // end
327 synth[osc]->status = SYNTH_OFF;// is this right?
328 sample = 0;
329 } else {
330 if(msynth[osc]->feedback > 0) { // still looping. The feedback flag is cleared by pcm_note_off.
331 if(base_index >= preset->loopend) { // loopend
332 // back to loopstart
333 int32_t loop_len = preset->loopend - preset->loopstart;
334 synth[osc]->phase -= F2P(loop_len / (float)(1 << PCM_INDEX_BITS));
335 base_index -= loop_len;
336 }
337 }
338 }
339 }
340 SAMPLE value = buf[i] + MUL4_SS(amp, sample);
341 buf[i] = value;
342 if (value < 0) value = -value;
343 if (value > max_value) max_value = value;
344 }
345 //printf("render_pcm: osc %d preset %d len %d base_ix %d phase %f step %f tablestep %f amp %f\n",
346 // osc, synth[osc]->preset, preset->length, base_index, P2F(synth[osc]->phase), P2F(step), (1 << PCM_INDEX_BITS) * P2F(step), S2F(msynth[osc]->amp));
347 return max_value;
348 // i don't believe we ever need to detect silence in a sample. it will shut itself off at the end.
349 }
350 return 0;
351}

365int pcm_load_file() {
366 // We pass the inputs to this as aliases in the amy_global structure. This is to not destroy the MP heap for amy->AMYboard
367 uint8_t midinote = amy_global.transfer_stored_bytes;
368 uint16_t preset_number = amy_global.transfer_file_handle;
369 char * filename = amy_global.transfer_filename;
370
371 pcm_unload_preset(preset_number);
372 if (filename == NULL || filename[0] == '\0') {
373 return 0;
374 }
375 if (amy_global.config.amy_external_fopen_hook == NULL || amy_global.config.amy_external_fclose_hook == NULL) {
376 fprintf(stderr, "fopen hook not enabled on platform\n");
377 return 0;
378 }
379 uint32_t handle = amy_global.config.amy_external_fopen_hook((char *)filename, "rb");
380 if (handle == 0) {
381 fprintf(stderr, "Could not open file %s\n", filename);
382 return 0;
383 }
384 wave_info_t info = {0};
385 uint32_t data_bytes = 0;
386 if (!wave_parse_header(handle, &info, &data_bytes)) {
387 fprintf(stderr, "Could not parse WAVE file %s\n", filename);
388 amy_global.config.amy_external_fclose_hook(handle);
389 return 0;
390 }
391 uint32_t total_frames = 0;
392 if (info.channels > 0) {
393 total_frames = data_bytes / (info.channels * 2);
394 }
395 uint32_t buffer_frames = AMY_BLOCK_SIZE * PCM_FILE_BUFFER_MULT;
396 memorypcm_ll_t *new_preset_pointer = malloc_caps(
397 sizeof(memorypcm_ll_t) + sizeof(memorypcm_preset_t) + buffer_frames * sizeof(int16_t),
398 amy_global.config.ram_caps_sample);
399 if (new_preset_pointer == NULL) {
400 fprintf(stderr, "No RAM left for sample load\n");
401 return 0;
402 }
403 new_preset_pointer->next = memorypcm_ll_start;
404 memorypcm_ll_start = new_preset_pointer;
405 new_preset_pointer->preset_number = preset_number;
406 memorypcm_preset_t *memory_preset =
407 (memorypcm_preset_t *)(((uint8_t *)new_preset_pointer) + sizeof(memorypcm_ll_t));
408 strncpy(memory_preset->filename, filename, MAX_FILENAME_LEN - 1);
409 memory_preset->filename[MAX_FILENAME_LEN - 1] = '\0';
410 memory_preset->channels = info.channels;
411 memory_preset->samplerate = info.sample_rate;
412 memory_preset->log2sr = log2f((float)info.sample_rate / ZERO_LOGFREQ_IN_HZ);
413 memory_preset->midinote = midinote;
414 memory_preset->length = total_frames;
415 memory_preset->type = AMY_PCM_TYPE_FILE;
416 memory_preset->file_bytes_remaining = total_frames * info.channels * 2;
417 memory_preset->file_handle = handle;
418 memory_preset->sample_ram = malloc_caps(buffer_frames * info.channels * sizeof(int16_t),
419 amy_global.config.ram_caps_sample);
420 new_preset_pointer->preset = memory_preset;
421 //fprintf(stderr, "read file %s frames %ld channels %d preset %d handle %ld\n", filename, total_frames, info.channels, preset_number, handle);
422 return 1;
423}

429int16_t * pcm_load(uint16_t preset_number, uint32_t length, uint32_t samplerate, uint8_t channels, uint8_t midinote, uint32_t loopstart, uint32_t loopend) {

454 if(loopend == 0) { // loop whole sample

457 memory_preset->loopend = loopend;

463void pcm_unload_preset(uint16_t preset_number) {
464 // run through the LL looking for the preset
465 memorypcm_ll_t **preset_pointer = &memorypcm_ll_start;
466 while(*preset_pointer != NULL) {
467 if((*preset_pointer)->preset_number == preset_number) {
468 memorypcm_ll_t *next = (*preset_pointer)->next;
469 fclose_if_file((*preset_pointer)->preset);
470 // free the memory we allocated
471 free((*preset_pointer));
472 // close up the list
473 *preset_pointer = next;
474 return;
475 } else {
476 preset_pointer = &(*preset_pointer)->next;
477 }
478 }
479 //fprintf(stderr, "pcm_unload_preset: preset %d not found\n", preset_number); // This happens during a routine load_preset.
480}
