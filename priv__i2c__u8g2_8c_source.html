32static uint8_t i2c_u8g2_byte_cb(u8x8_t *u8x8, uint8_t msg, uint8_t arg_int, void *arg_ptr)
33{
34 (void)u8x8;
35 static uint8_t buffer[I2C_U8G2_TX_BUFFER_SIZE];
36 static uint8_t buf_idx;
37
38 switch (msg) {
39 case U8X8_MSG_BYTE_INIT: {
40 i2c_device_config_t dev_config = {
41 .dev_addr_length = I2C_ADDR_BIT_LEN_7,
42 .device_address = s_device_address,
43 .scl_speed_hz = s_scl_speed_hz,
44 .scl_wait_us = 0,
45 .flags.disable_ack_check = false,
46 };
47
48 esp_err_t ret = i2c_master_bus_add_device(s_i2c_bus_handle, &dev_config,
49 &s_display_dev_handle);
50 if (ret != ESP_OK) {
51 ESP_LOGE(TAG, "Failed to add display device to I2C bus (%s)", esp_err_to_name(ret));
52 return 0;
53 }
54
55 i2c_u8g2_handle_t *handle = i2c_u8g2_handle_from_u8x8(u8x8);
56 if (handle != NULL) {
57 handle->display_dev_handle = s_display_dev_handle;
58 }
59
60 ESP_LOGI(TAG, "I2C master driver initialized successfully");
61 break;
62 }
63
64 case U8X8_MSG_BYTE_START_TRANSFER:
65 buf_idx = 0;
66 break;
67
68 case U8X8_MSG_BYTE_SET_DC:
69 break;
70
71 case U8X8_MSG_BYTE_SEND:
72 for (size_t i = 0; i < arg_int; ++i) {
73 if (buf_idx >= sizeof(buffer)) {
74 ESP_LOGE(TAG, "U8G2 transfer buffer overflow (%u >= %u)",
75 (unsigned int)buf_idx,
76 (unsigned int)sizeof(buffer));
77 return 0;
78 }
79 buffer[buf_idx++] = *((uint8_t *)arg_ptr + i);
80 }
81 break;
82
83 case U8X8_MSG_BYTE_END_TRANSFER:
84 if (buf_idx > 0 && s_display_dev_handle != NULL) {
85 esp_err_t ret = i2c_master_transmit(s_display_dev_handle, buffer, buf_idx, s_timeout_ms);
86 if (ret != ESP_OK) {
87 ESP_LOGE(TAG, "I2C transfer failed (%s)", esp_err_to_name(ret));
88 return 0;
89 }
90 }
91 break;
92
93 default:
94 return 0;
95 }
96
97 return 1;
98}

100static uint8_t i2c_u8g2_gpio_and_delay_cb(u8x8_t *u8x8, uint8_t msg,
101 uint8_t arg_int, void *arg_ptr)
102{
103 (void)u8x8;
104 (void)arg_ptr;
105
106 switch (msg) {
107 case U8X8_MSG_GPIO_AND_DELAY_INIT:
108 ESP_LOGI(TAG, "GPIO and delay initialization completed");
109 break;
110
111 case U8X8_MSG_DELAY_MILLI:
112 vTaskDelay(pdMS_TO_TICKS(arg_int));
113 break;
114
115 case U8X8_MSG_DELAY_10MICRO:
116 esp_rom_delay_us(arg_int * 10);
117 break;
118
119 case U8X8_MSG_DELAY_100NANO:
120 __asm__ __volatile__("nop");
121 break;
122
123 case U8X8_MSG_DELAY_I2C:
124 if (arg_int == 0) {
125 return 1;
126 }
127 esp_rom_delay_us(5 / arg_int);
128 break;
129
130 case U8X8_MSG_GPIO_RESET:
131 break;
132
133 default:
134 return 0;
135 }
136
137 return 1;

231 ESP_RETURN_ON_ERROR(i2c_del_master_bus(handle->i2c_bus_handle),

234 s_i2c_bus_handle = NULL;
235 }
236
237 if (s_active_handle == handle) {
238 s_active_handle = NULL;
239 }
