1/* Project snapshot serializer + loader orchestration.
2 *
3 * Field order is the format: every ser_ writer and its de_ or parse_ reader
4 * counterpart below must walk fields in exactly the same sequence, or every
5 * saved project silently corrupts on load. The shared env/filter/lfo
6 * sub-block helpers exist so the LAYR per-track voice_params_t blocks and the
7 * ARP/DRON sections (which embed the same env/filter/lfo shapes) cannot
8 * drift from each other.
9 *
10 * Two-phase load: parse_* validates and stages everything into heap-allocated
11 * scratch (never touching live state); only once every section has parsed
12 * clean does the apply phase run. Any parse/validation failure frees the
13 * scratch and returns false with zero live-state changes.
14 */

62{

81/* Reads all 5 fields unconditionally (keeps the reader position correct),
82 * then resets the whole sub-block to a safe bypass default if filter_type is
83 * out of range instead of trusting a bogus enum downstream. */
84static bool de_filter(tlv_reader_t *r, seq_filter_t *f)
85{
86 uint8_t ft, en;
87 float cutoff, resonance, env_amount;
88 if (!tlv_get_u8(r, &ft)) return false;
89 if (!tlv_get_f32(r, &cutoff)) return false;
90 if (!tlv_get_f32(r, &resonance)) return false;
91 if (!tlv_get_u8(r, &en)) return false;
92 if (!tlv_get_f32(r, &env_amount)) return false;
93
94 if (ft > SEQ_FILTER_LPF24) {
95 *f = (seq_filter_t){0};
96 return true;
97 }
98 f->filter_type = ft;
99 f->cutoff_hz = cutoff;
100 f->resonance = resonance;
101 f->enabled = en != 0;
102 f->filter_env_amount = env_amount;
103 return true;
104}

116/* Same "read everything, then validate" shape as de_filter: an out-of-range
117 * enum resets the whole LFO sub-block rather than propagating a bad value.
118 * The 6th byte changed meaning: section v1 stored a single target index, v2+
119 * stores a target-set bitmask. Migrate index -> bit when reading an old file. */
120static bool de_lfo(tlv_reader_t *r, seq_lfo_t *l, uint8_t ver)
121{
122 uint8_t en, mode, wave, rate, depth, tgt;
123 if (!tlv_get_u8(r, &en)) return false;
124 if (!tlv_get_u8(r, &mode)) return false;
125 if (!tlv_get_u8(r, &wave)) return false;
126 if (!tlv_get_u8(r, &rate)) return false;
127 if (!tlv_get_u8(r, &depth)) return false;
128 if (!tlv_get_u8(r, &tgt)) return false;
129
130 if (mode > LFO_MODE_RETRIG || wave >= LFO_WAVE_COUNT ||
131 rate >= LFO_RATE_COUNT) {
132 *l = (seq_lfo_t){0};
133 return true;
134 }
135 l->enabled = en != 0;
136 l->mode = (lfo_mode_t)mode;
137 l->wave = (lfo_wave_t)wave;
138 l->rate = (lfo_rate_t)rate;
139 l->depth = (depth > 100) ? 100 : depth;
140 l->targets = (ver < 2)
141 ? ((tgt < LFO_TARGET_COUNT) ? LFO_TGT_BIT(tgt) : 0u) /* v1 index */
142 : (uint8_t)(tgt & LFO_TGT_ALL); /* v2 bitmask */
143 return true;
144}

162{
163 voice_params_init_defaults(vp); /* zeroed baseline + unity amp_trim */
164 if (!de_env(r, &vp->env)) return false;
165 if (!de_env(r, &vp->env1)) return false;
166 if (!de_filter(r, &vp->filter)) return false;
167 if (!de_lfo(r, &vp->lfo, ver)) return false;

170 if (!tlv_get_u8(r, &e1a)) return false;
171 if (!tlv_get_u8(r, &fa)) return false;
172 if (!tlv_get_u8(r, &la)) return false;
173 vp->env_authored = ea != 0;
174 vp->env1_authored = e1a != 0;
175 vp->filter_authored = fa != 0;
176 vp->lfo_authored = la != 0;
177 if (!tlv_get_f32(r, &vp->amp_trim)) return false;
178 if (vp->amp_trim < 0.0f) vp->amp_trim = 0.0f;
179 if (vp->amp_trim > 1.0f) vp->amp_trim = 1.0f;
180 return true;
181}

280/* ── LAYR section ─────────────────────────────────────────────────────────
281 * Strict: any truncated read rejects the whole file (unlike GLOB, a partial
282 * LAYR body cannot be safely defaulted field-by-field once step arrays are
283 * involved). Per-value clamps normalize out-of-range data instead of
284 * rejecting, except for the layer-count/type/first-layer-is-drum invariants
285 * enforced by the caller after every LAYR section has parsed. */
286
287static void ser_layer(tlv_writer_t *w, const seq_layer_t *L)
288{
289 size_t h = tlv_begin_section(w, TAG_LAYR, 3); /* v2: LFO target bitmask;
290 * v3: +gate_pct, +portamento_ms */
291 tlv_put_u8(w, (uint8_t)L->type);
292 tlv_put_u8(w, L->num_steps);
293 tlv_put_u16(w, L->patch);
294 tlv_put_u32(w, L->synth_flags);
295 tlv_put_u8(w, L->num_voices);
296 tlv_put_u8(w, L->chord_mode ? 1 : 0);
297 tlv_put_u8(w, L->chord_root);
298 tlv_put_u8(w, (uint8_t)L->chord_type);
299 tlv_put_u8(w, L->swing_pct);
300 for (int t = 0; t < SEQ_TRACKS; t++) {
301 tlv_put_u8(w, L->track_base_note[t]);
302 tlv_put_u16(w, L->track_patch[t]);
303 tlv_put_u16(w, L->track_pcm_preset[t]);
304 tlv_put_u8(w, L->repeat_rate[t]);
305 tlv_put_u8(w, L->mute[t] ? 1 : 0);
306 tlv_put_u8(w, L->solo[t] ? 1 : 0);
307 ser_vp(w, &L->vp[t]);
308 }
309 tlv_put_bytes(w, L->grid, sizeof L->grid);
310 tlv_put_bytes(w, L->step_note, sizeof L->step_note);
311 tlv_put_bytes(w, L->step_prob, sizeof L->step_prob);
312 tlv_put_bytes(w, L->step_ratchet, sizeof L->step_ratchet);
313 tlv_put_bytes(w, L->step_cond_type, sizeof L->step_cond_type);
314 tlv_put_bytes(w, L->step_cond_param, sizeof L->step_cond_param);
315 tlv_put_bytes(w, L->step_transform, sizeof L->step_transform);
316 tlv_put_bytes(w, L->step_quant_bypass, sizeof L->step_quant_bypass);
317 tlv_put_bytes(w, L->step_nudge, sizeof L->step_nudge);
318 tlv_put_bytes(w, L->step_velocity_adj, sizeof L->step_velocity_adj);
319 tlv_put_bytes(w, L->step_ratchet_taper, sizeof L->step_ratchet_taper);
320 /* v3: per-layer melodic NoteFX (gate length + glide). Appended at the tail so
321 * v2 readers stop cleanly before them and v2 files parse under the ver<3 path.*/
322 tlv_put_u8(w, L->gate_pct);
323 tlv_put_u16(w, L->portamento_ms);
324 tlv_end_section(w, h);
325}

334static bool parse_layer(tlv_reader_t *b, seq_layer_t *L, uint8_t ver)
335{
336 memset(L, 0, sizeof *L);
337
338 uint8_t type_raw;
339 if (!tlv_get_u8(b, &type_raw)) return false;
340 L->type = (type_raw > SEQ_LAYER_MELODIC) ? SEQ_LAYER_MELODIC
341 : (seq_layer_type_t)type_raw;
342
343 if (!tlv_get_u8(b, &L->num_steps)) return false;
344 if (L->num_steps != 16 && L->num_steps != 32) L->num_steps = 16;
345
346 if (!tlv_get_u16(b, &L->patch)) return false;
347 L->patch = clamp_patch(L->patch);
348
349 if (!tlv_get_u32(b, &L->synth_flags)) return false;
350 if (!tlv_get_u8(b, &L->num_voices)) return false;
351 /* AMY's instrument_init() aborts outside 1..MAX_VOICES_PER_INSTRUMENT
352 * (32); a CRC-valid but hand-edited file must not crash the load. */
353 if (L->num_voices < 1) L->num_voices = 1;
354 if (L->num_voices > 32) L->num_voices = 32;
355 { uint8_t v; if (!tlv_get_u8(b, &v)) return false; L->chord_mode = v != 0; }
356 if (!tlv_get_u8(b, &L->chord_root)) return false;
357 { uint8_t v; if (!tlv_get_u8(b, &v)) return false;
358 L->chord_type = (v >= CHORD_TYPE_COUNT) ? CHORD_MAJ : (chord_type_t)v; }
359 if (!tlv_get_u8(b, &L->swing_pct)) return false;
360 /* Bulk import bypasses sequencer_core_set_layer_swing(), so the engine
361 * ceiling must be enforced here: the swing-offset math requires the
362 * delay to stay short of one full step. */
363 if (L->swing_pct > SEQ_SWING_MAX) L->swing_pct = SEQ_SWING_MAX;
364
365 for (int t = 0; t < SEQ_TRACKS; t++) {
366 if (!tlv_get_u8(b, &L->track_base_note[t])) return false;
367 if (!tlv_get_u16(b, &L->track_patch[t])) return false;
368 L->track_patch[t] = clamp_patch(L->track_patch[t]);
369 if (!tlv_get_u16(b, &L->track_pcm_preset[t])) return false;
370 { uint8_t v; if (!tlv_get_u8(b, &v)) return false; L->repeat_rate[t] = v; }
371 { uint8_t v; if (!tlv_get_u8(b, &v)) return false; L->mute[t] = v != 0; }
372 { uint8_t v; if (!tlv_get_u8(b, &v)) return false; L->solo[t] = v != 0; }
373 if (!de_vp(b, &L->vp[t], ver)) return false;
374 }
375
376 if (!tlv_get_bytes(b, L->grid, sizeof L->grid)) return false;
377 if (!tlv_get_bytes(b, L->step_note, sizeof L->step_note)) return false;
378 if (!tlv_get_bytes(b, L->step_prob, sizeof L->step_prob)) return false;
379 if (!tlv_get_bytes(b, L->step_ratchet, sizeof L->step_ratchet)) return false;
380 if (!tlv_get_bytes(b, L->step_cond_type, sizeof L->step_cond_type)) return false;
381 if (!tlv_get_bytes(b, L->step_cond_param, sizeof L->step_cond_param)) return false;
382 if (!tlv_get_bytes(b, L->step_transform, sizeof L->step_transform)) return false;
383 if (!tlv_get_bytes(b, L->step_quant_bypass, sizeof L->step_quant_bypass)) return false;
384 if (!tlv_get_bytes(b, L->step_nudge, sizeof L->step_nudge)) return false;
385 if (!tlv_get_bytes(b, L->step_velocity_adj, sizeof L->step_velocity_adj)) return false;
386 if (!tlv_get_bytes(b, L->step_ratchet_taper, sizeof L->step_ratchet_taper)) return false;
387
388 for (int t = 0; t < SEQ_TRACKS; t++) {
389 for (int s = 0; s < SEQ_MAX_STEPS; s++) {
390 L->grid[t][s] = L->grid[t][s] ? 1 : 0;
391 L->step_quant_bypass[t][s] = L->step_quant_bypass[t][s] ? 1 : 0;
392 if (L->step_prob[t][s] > 100) L->step_prob[t][s] = 100;
393 if (L->step_ratchet[t][s] < 1) L->step_ratchet[t][s] = 1;
394 if (L->step_ratchet[t][s] > SEQ_MAX_RATCHET) L->step_ratchet[t][s] = SEQ_MAX_RATCHET;
395 if (L->step_cond_type[t][s] >= SEQ_STEP_COND_COUNT) L->step_cond_type[t][s] = SEQ_STEP_COND_NONE;
396 if (L->step_cond_param[t][s] > 8) L->step_cond_param[t][s] = 8; /* FILL divisor caps at 8 */
397 if (L->step_transform[t][s] >= SEQ_STEP_TRANSFORM_COUNT) L->step_transform[t][s] = SEQ_STEP_TRANSFORM_NONE;
398 }
399 }
400
401 /* v3: per-layer melodic NoteFX. Pre-v3 files predate the fields — default to
402 * the legacy gate (glide off). Clamp to the live control ranges either way. */
403 if (ver >= 3) {
404 if (!tlv_get_u8(b, &L->gate_pct)) return false;
405 if (!tlv_get_u16(b, &L->portamento_ms)) return false;
406 } else {
407 L->gate_pct = SEQ_MELODIC_GATE_DEFAULT_PCT;
408 L->portamento_ms = 0;
409 }
410 if (L->gate_pct < 10) L->gate_pct = 10;
411 if (L->gate_pct > 100) L->gate_pct = 100;
412 if (L->portamento_ms > SEQ_MELODIC_PORTAMENTO_MAX_MS)
413 L->portamento_ms = SEQ_MELODIC_PORTAMENTO_MAX_MS;
414
415 return true;
416}

440{
441 size_t h = tlv_begin_section(w, TAG_ARP, 2); /* v2: LFO target is a bitmask */
442 tlv_put_u8(w, arp_get_enabled() ? 1 : 0);
443 tlv_put_u8(w, (uint8_t)arp_get_source());
444 tlv_put_u16(w, arp_get_wave());

750bool project_snapshot_load(uint8_t slot)
751{
752 uint8_t *payload = NULL;
753 size_t len = 0;
754 char name[PROJECT_NAME_LEN];
755 if (!project_store_read(slot, &payload, &len, name)) return false;
756
757 seq_layer_t *staged_layers =
758 heap_caps_malloc(sizeof(seq_layer_t) * MAX_LAYERS, MALLOC_CAP_SPIRAM);
759 if (!staged_layers) {
760 free(payload);
761 ESP_LOGE(TAG, "load slot %u: SPIRAM allocation failed", slot);
762 return false;
763 }
764
765 staged_glob_t staged_glob; memset(&staged_glob, 0, sizeof staged_glob);
766 staged_arp_t staged_arp; memset(&staged_arp, 0, sizeof staged_arp);
767 staged_drone_t staged_drone; memset(&staged_drone, 0, sizeof staged_drone);
768 staged_prog_t staged_prog; memset(&staged_prog, 0, sizeof staged_prog);
769 uint8_t staged_layer_count = 0;
770 bool got_glob = false, got_arp = false, got_drone = false, got_prog = false;
771
772 tlv_reader_t r;
773 tlv_reader_init(&r, payload, len);
774 bool ok = true;
775 uint32_t tag; uint8_t ver; tlv_reader_t body;
776 while (ok && tlv_next_section(&r, &tag, &ver, &body)) {
777 switch (tag) {
778 case TAG_GLOB:
779 if (got_glob || ver != 1) { ok = false; break; }
780 ok = parse_glob(&body, &staged_glob);
781 got_glob = ok;
782 break;
783 case TAG_LAYR:
784 if (ver < 1 || ver > 3 || staged_layer_count >= MAX_LAYERS) { ok = false; break; }
785 ok = parse_layer(&body, &staged_layers[staged_layer_count], ver);
786 if (ok) staged_layer_count++;
787 break;
788 case TAG_ARP:
789 if (got_arp || ver < 1 || ver > 2) { ok = false; break; }
790 ok = parse_arp(&body, &staged_arp, ver);
791 got_arp = ok;
792 break;
793 case TAG_DRON:
794 if (got_drone || ver != 1) { ok = false; break; }
795 ok = parse_drone(&body, &staged_drone);
796 got_drone = ok;
797 break;
798 case TAG_PROG:
799 if (got_prog || ver != 1) { ok = false; break; }
800 ok = parse_prog(&body, &staged_prog);
801 got_prog = ok;
802 break;
803 default:
804 break; /* unknown section: ignore (forward-compat) */
805 }
806 }
807 if (r.err) ok = false; /* truncated/corrupt outer stream */
808
809 if (ok && (!got_glob || staged_layer_count < 1 ||
810 staged_layers[0].type != SEQ_LAYER_DRUM)) {
811 ok = false;
812 }
813 /* Exactly one drum layer, and only at index 0: add_layer(DRUM) always
814 * binds the fixed SEQ_DRUM_SYNTH_BASE slots, so a second drum layer
815 * would alias the first one's AMY synths. */
816 for (uint8_t i = 1; ok && i < staged_layer_count; i++) {
817 if (staged_layers[i].type == SEQ_LAYER_DRUM) ok = false;
818 }
819
820 if (!ok) {
821 free(staged_layers);
822 free(payload);
823 ESP_LOGW(TAG, "load slot %u: validation failed, no changes made", slot);
824 return false;
825 }
826
827 /* ── Phase 2: apply. Nothing above this point touched live state. ── */
828 sequencer_core_set_playing(false);
829 arp_core_clear_all();
830
831 while (sequencer_core_get_num_layers() > 1) {
832 sequencer_core_delete_layer((uint8_t)(sequencer_core_get_num_layers() - 1));
833 }
834 for (uint8_t i = 1; i < staged_layer_count; i++) {
835 sequencer_core_add_layer(staged_layers[i].type, staged_layers[i].num_steps);
836 }
837 for (uint8_t i = 0; i < staged_layer_count; i++) {
838 sequencer_core_import_layer(i, &staged_layers[i]);
839 }
840
841 apply_glob(&staged_glob);
842 if (got_arp) apply_arp(&staged_arp);
843 if (got_drone) apply_drone(&staged_drone);
844 if (got_prog) apply_prog(&staged_prog);
845
846 synth_ui_reload_mirror_from_core();
847
848 free(staged_layers);
849 free(payload);
850 ESP_LOGI(TAG, "load slot %u ('%s') OK: %u layer(s)", slot, name, staged_layer_count);
851 return true;
852}
