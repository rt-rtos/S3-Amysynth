62static bool de_env(tlv_reader_t *r, seq_env_t *e)
63{
64 if (!tlv_get_u32(r, &e->attack_ms)) return false;
65 if (!tlv_get_u32(r, &e->decay_ms)) return false;

83/* Reads every field first (keeps the reader position correct), then bypasses
84 * the whole sub-block if filter_type is out of range rather than passing a
85 * bogus enum downstream.
86 * `has_feedback`: does this file carry the KS feedback field (LAYR v7+,
87 * ARP v5+)? Caller resolves it, like de_lfo's flags. Older files derive it
88 * from Q via the legacy mapping so they still sound the same. */
89static bool de_filter(tlv_reader_t *r, seq_filter_t *f, bool has_feedback)
90{
91 uint8_t ft, en;
92 float cutoff, resonance, env_amount, feedback = 0.0f;
93 if (!tlv_get_u8(r, &ft)) return false;
94 if (!tlv_get_f32(r, &cutoff)) return false;
95 if (!tlv_get_f32(r, &resonance)) return false;
96 if (!tlv_get_u8(r, &en)) return false;
97 if (!tlv_get_f32(r, &env_amount)) return false;
98 if (has_feedback) {
99 if (!tlv_get_f32(r, &feedback)) return false;
100 feedback = SEQ_CLAMP_F32(feedback, 0.0f, 1.0f);
101 } else {
102 feedback = sequencer_core_ks_feedback_from_q(resonance);
103 }
104
105 if (ft >= SEQ_FILTER_COUNT) {
106 *f = (seq_filter_t){0};
107 return true;
108 }
109 f->filter_type = ft;
110 f->cutoff_hz = cutoff;
111 f->resonance = resonance;
112 f->enabled = en != 0;
113 f->filter_env_amount = env_amount;
114 f->feedback = feedback;
115 return true;
116}

118/* ── Distortion codec (LAYR v12+, ARP v9+) ───────────────────────────────
119 * `present` is the caller's version gate, and it is load-bearing rather than
120 * cosmetic for LAYR: the block lives at the end of each track's vp, mid-body,
121 * so a wrong gate would desync every following track. ARP's copy really is
122 * section-final. `present = false` leaves the caller's initialised default
123 * (type OFF) untouched. Values are re-clamped on read: a truncated or
124 * hand-edited body must not push an out-of-range type into AMY. */
125static void ser_dist(tlv_writer_t *w, const seq_dist_t *d)
126{
127 tlv_put_u8(w, d->type);
128 tlv_put_u8(w, d->drive);
129 tlv_put_u8(w, d->bits);
130 tlv_put_u8(w, d->rate);
131 tlv_put_u8(w, d->mix);
132}

134static bool de_dist(tlv_reader_t *r, seq_dist_t *d, bool present)
135{
136 if (!present) return true; /* keep the caller's default */
137 uint8_t type, drive, bits, rate, mix;
138 if (!tlv_get_u8(r, &type)) return false;
139 if (!tlv_get_u8(r, &drive)) return false;
140 if (!tlv_get_u8(r, &bits)) return false;
141 if (!tlv_get_u8(r, &rate)) return false;
142 if (!tlv_get_u8(r, &mix)) return false;
143 d->type = type; d->drive = drive; d->bits = bits; d->rate = rate; d->mix = mix;
144 voice_dist_clamp(d);
145 return true;
146}

148static void ser_lfo(tlv_writer_t *w, const seq_lfo_t *l)
149{
150 tlv_put_u8(w, l->enabled ? 1 : 0);
151 tlv_put_u8(w, (uint8_t)l->mode);
152 tlv_put_u8(w, (uint8_t)l->wave);
153 tlv_put_u8(w, (uint8_t)l->rate);
154 tlv_put_u8(w, l->depth);
155 tlv_put_u8(w, l->targets); /* section v2+: target-set bitmask (v1: index) */
156 tlv_put_u8(w, l->wob_rate); /* LAYR v5+ / ARP v3+: WOBBLE second-order LFO */
157 tlv_put_u8(w, l->wob_depth);
158 tlv_put_u8(w, l->wob_reach); /* LAYR v6+ / ARP v4+: reach (0 both,
159 * 1 depth, 2 rate). Older firmware reads
160 * 2 as nonzero = depth-only - benign. */
161 tlv_put_u8(w, l->flt_oct_q); /* LAYR v8+ / ARP v7+: FILTER swing in
162 * quarter-octaves; 0 = legacy depth-derived */
163}

165/* Same "read everything, then validate" shape as de_filter. The 6th byte
166 * changed meaning: v1 = single target index, v2+ = target-set bitmask;
167 * migrate index -> bit when reading an old file.
168 * Presence flags are caller-resolved because the version threshold differs
169 * per containing section: `wobble` = two WOBBLE bytes (LAYR v5+, ARP v3+),
170 * `wob_mode` = reach byte (LAYR v6+, ARP v4+; absent = depth+rate; 0 both,
171 * 1 depth-only, 2 rate-only),
172 * `flt_oct` = FILTER octave-swing byte (LAYR v8+, ARP v7+; absent leaves the
173 * 0 sentinel = old depth-derived filter law). */
174static bool de_lfo(tlv_reader_t *r, seq_lfo_t *l, uint8_t ver, bool wobble,
175 bool wob_mode, bool flt_oct)
176{
177 uint8_t en, mode, wave, rate, depth, tgt;
178 uint8_t wrate = 0, wdepth = 0, wdeponly = 0, foct = 0;
179 if (!tlv_get_u8(r, &en)) return false;
180 if (!tlv_get_u8(r, &mode)) return false;
181 if (!tlv_get_u8(r, &wave)) return false;
182 if (!tlv_get_u8(r, &rate)) return false;
183 if (!tlv_get_u8(r, &depth)) return false;
184 if (!tlv_get_u8(r, &tgt)) return false;
185 if (wobble) {
186 if (!tlv_get_u8(r, &wrate)) return false;
187 if (!tlv_get_u8(r, &wdepth)) return false;
188 }
189 if (wob_mode) {
190 if (!tlv_get_u8(r, &wdeponly)) return false;
191 }
192 if (flt_oct) {
193 if (!tlv_get_u8(r, &foct)) return false;
194 }
195
196 if (mode > LFO_MODE_RETRIG || wave >= LFO_WAVE_COUNT ||
197 rate >= LFO_RATE_COUNT) {
198 *l = (seq_lfo_t){0};
199 return true;
200 }
201 l->enabled = en != 0;
202 l->mode = (lfo_mode_t)mode;
203 l->wave = (lfo_wave_t)wave;
204 l->rate = (lfo_rate_t)rate;
205 l->depth = (depth > 100) ? 100 : depth;
206 l->targets = (ver < 2)
207 ? ((tgt < LFO_TARGET_COUNT) ? LFO_TGT_BIT(tgt) : 0u) /* v1 index */
208 : (uint8_t)(tgt & LFO_TGT_ALL); /* v2 bitmask */
209 l->wob_rate = (wrate < LFO_RATE_COUNT) ? wrate : 0;
210 /* Snap to a whole-dB authoring step (voice_config.h): pre-dB files carry
211 * 5 %-grid values, where a stored 5 % reads as OFF while the modulator
212 * still runs. */
213 l->wob_depth = voice_wob_db_to_depth(voice_wob_depth_to_db(wdepth));
214 l->wob_reach = (wdeponly < WOB_REACH_COUNT) ? wdeponly : 0;
215 l->flt_oct_q = (foct > VOICE_LFO_FLT_OCT_Q_MAX)
216 ? (uint8_t)VOICE_LFO_FLT_OCT_Q_MAX : foct;
217 return true;
218}

222static void ser_vp(tlv_writer_t *w, const voice_params_t *vp)
223{
224 ser_env(w, &vp->env);
225 ser_env(w, &vp->env1);
226 ser_filter(w, &vp->filter);
227 ser_lfo(w, &vp->lfo);
228 tlv_put_u8(w, vp->env_authored ? 1 : 0);
229 tlv_put_u8(w, vp->env1_authored ? 1 : 0);
230 tlv_put_u8(w, vp->filter_authored ? 1 : 0);
231 tlv_put_u8(w, vp->lfo_authored ? 1 : 0);
232 tlv_put_f32(w, vp->amp_trim);
233 ser_dist(w, &vp->dist); /* LAYR v12+ */
234 tlv_put_u8(w, vp->dist_authored ? 1 : 0);
235}

237static bool de_vp(tlv_reader_t *r, voice_params_t *vp, uint8_t ver)
238{
239 voice_params_init_defaults(vp); /* zeroed baseline + unity amp_trim */
240 if (!de_env(r, &vp->env)) return false;
241 if (!de_env(r, &vp->env1)) return false;
242 if (!de_filter(r, &vp->filter, ver >= 7)) return false; /* LAYR: feedback v7+ */
243 if (!de_lfo(r, &vp->lfo, ver, ver >= 5, ver >= 6, ver >= 8)) return false; /* LAYR: wobble v5+, reach v6+, flt_oct v8+ */
244 uint8_t ea, e1a, fa, la;
245 if (!tlv_get_u8(r, &ea)) return false;
246 if (!tlv_get_u8(r, &e1a)) return false;
247 if (!tlv_get_u8(r, &fa)) return false;
248 if (!tlv_get_u8(r, &la)) return false;
249 vp->env_authored = ea != 0;
250 vp->env1_authored = e1a != 0;
251 vp->filter_authored = fa != 0;
252 vp->lfo_authored = la != 0;
253 if (!tlv_get_f32(r, &vp->amp_trim)) return false;
254 vp->amp_trim = SEQ_CLAMP_F32(vp->amp_trim, 0.0f, 1.0f);
255 /* LAYR v12+: distortion, appended to each track's vp block - which sits
256 * INSIDE the per-track loop, so this is mid-body, not section-end. The
257 * version gate is what keeps the walk aligned: on a pre-v12 file both
258 * sides skip these bytes and the next track starts where it should.
259 * Unread, the block keeps voice_params_init_defaults()' OFF default. */
260 if (!de_dist(r, &vp->dist, ver >= 12)) return false;
261 if (ver >= 12) {
262 uint8_t da;
263 if (!tlv_get_u8(r, &da)) return false;
264 vp->dist_authored = da != 0;
265 }
266 return true;
267}

317static bool parse_glob(tlv_reader_t *b, staged_glob_t *g)
318{
319 /* Seed from live state so a short/older section leaves the tail as-is. */
320 g->bpm = sequencer_core_get_bpm();
321 g->master_volume = amy_fx_get_master_volume();
322 g->quant_enabled = sequencer_core_get_quantizer_enabled();
323 g->quant_root = sequencer_core_get_quantizer_root_note();
324 g->quant_scale = sequencer_core_get_quantizer_scale();
325 g->drum_engine = sequencer_core_get_drum_engine();
326 g->fx = s_fx;
327
328 uint8_t v;
329 if (!tlv_get_u16(b, &g->bpm)) return true;
330 if (!tlv_get_f32(b, &g->master_volume)) return true;
331 if (!tlv_get_u8(b, &v)) return true;
332 g->quant_enabled = v != 0;
333 if (!tlv_get_u8(b, &g->quant_root)) return true;
334 if (!tlv_get_u8(b, &g->quant_scale)) return true;
335 if (!tlv_get_u8(b, &v)) return true;
336 g->drum_engine = (v == SEQ_DRUM_PCM) ? SEQ_DRUM_PCM : SEQ_DRUM_SYNTH;
337 if (!tlv_get_i8(b, &g->fx.eq_low_db)) return true;
338 if (!tlv_get_i8(b, &g->fx.eq_mid_db)) return true;
339 if (!tlv_get_i8(b, &g->fx.eq_high_db)) return true;
340 if (!tlv_get_u8(b, &g->fx.echo_level)) return true;
341 if (!tlv_get_u8(b, &g->fx.chorus_level)) return true;
342 if (!tlv_get_u8(b, &g->fx.reverb_level)) return true;
343 if (!tlv_get_i16(b, &g->fx.echo_delay_ms)) return true;
344 if (!tlv_get_i16(b, &g->fx.echo_feedback)) return true;
345 if (!tlv_get_i16(b, &g->fx.echo_tone)) return true;
346 if (!tlv_get_i16(b, &g->fx.reverb_liveness))return true;
347 if (!tlv_get_i16(b, &g->fx.reverb_damping)) return true;
348 if (!tlv_get_i16(b, &g->fx.reverb_xover_hz))return true;
349 if (!tlv_get_i16(b, &g->fx.chorus_rate)) return true;
350 if (!tlv_get_i16(b, &g->fx.chorus_depth)) return true;

358 if (!tlv_get_u8(b, &g->fx.bus_dist_mix)) return true;

360}

378/* ── LAYR section ─────────────────────────────────────────────────────────
379 * Strict: any truncated read rejects the whole file - unlike GLOB, a partial
380 * LAYR body cannot be defaulted field-by-field once step arrays are involved.
381 * Out-of-range values clamp rather than reject, except the layer-count/type/
382 * first-layer-is-drum invariants the caller enforces after all LAYR sections
383 * have parsed. */
384
385static void ser_layer(tlv_writer_t *w, const seq_layer_t *L)
386{
387 size_t h = tlv_begin_section(w, TAG_LAYR, 14); /* v2: LFO target bitmask;
388 * v3: +gate_pct, +portamento_ms;
389 * v4: +groove_pct;
390 * v5: LFO +wob_rate/+wob_depth;
391 * v6: LFO +wob_depth_only;
392 * v7: filter +feedback (KS);
393 * v8: LFO +flt_oct_q;
394 * v9: step cond enum+param ->
395 * independent every+prev
396 * (same two array slots);
397 * v10: +track_pcm_mode;
398 * v11: +step_pitch_ofs;
399 * v12: vp +dist/+dist_authored;
400 * v13: +fm_algo_override;
401 * v14: +vp_src[], +vp_layer */
402 tlv_put_u8(w, (uint8_t)L->type);
403 tlv_put_u8(w, L->num_steps);
404 tlv_put_u16(w, L->patch);
405 tlv_put_u32(w, L->synth_flags);
406 tlv_put_u8(w, L->num_voices);
407 tlv_put_u8(w, L->chord_mode ? 1 : 0);
408 tlv_put_u8(w, L->chord_root);
409 tlv_put_u8(w, (uint8_t)L->chord_type);
410 tlv_put_u8(w, L->swing_pct);
411 for (int t = 0; t < SEQ_TRACKS; t++) {
412 tlv_put_u8(w, L->track_base_note[t]);
413 tlv_put_u16(w, L->track_patch[t]);
414 tlv_put_u16(w, L->track_pcm_preset[t]);
415 tlv_put_u8(w, L->track_pcm_mode[t]); /* v10+ */
416 tlv_put_u8(w, L->repeat_rate[t]);
417 tlv_put_u8(w, L->mute[t] ? 1 : 0);
418 tlv_put_u8(w, L->solo[t] ? 1 : 0);
419 ser_vp(w, &L->vp[t]);
420 }
421 tlv_put_bytes(w, L->grid, sizeof L->grid);
422 tlv_put_bytes(w, L->step_note, sizeof L->step_note);
423 tlv_put_bytes(w, L->step_pitch_ofs, sizeof L->step_pitch_ofs); /* v11+ */
424 tlv_put_bytes(w, L->step_prob, sizeof L->step_prob);
425 tlv_put_bytes(w, L->step_ratchet, sizeof L->step_ratchet);
426 tlv_put_bytes(w, L->step_every, sizeof L->step_every);
427 tlv_put_bytes(w, L->step_prev, sizeof L->step_prev);
428 tlv_put_bytes(w, L->step_transform, sizeof L->step_transform);
429 tlv_put_bytes(w, L->step_quant_bypass, sizeof L->step_quant_bypass);
430 tlv_put_bytes(w, L->step_nudge, sizeof L->step_nudge);
431 tlv_put_bytes(w, L->step_velocity_adj, sizeof L->step_velocity_adj);
432 tlv_put_bytes(w, L->step_ratchet_taper, sizeof L->step_ratchet_taper);
433 /* v3: melodic NoteFX (gate length + glide). Tail-appended so v2 readers
434 * stop cleanly before them. */
435 tlv_put_u8(w, L->gate_pct);
436 tlv_put_u16(w, L->portamento_ms);
437 /* v4: NoteFX GROOVE (accent-curve amount), same tail-append. */
438 tlv_put_u8(w, L->groove_pct);
439 /* v13: live FM algorithm override (Shift+Turn), same tail-append. */
440 tlv_put_u8(w, L->fm_algo_override);
441 /* v14: voice-block source selector per row + the layer's shared block. */
442 for (int t = 0; t < SEQ_TRACKS; t++) tlv_put_u8(w, L->vp_src[t]);
443 ser_vp(w, &L->vp_layer);
444 tlv_end_section(w, h);
445}

447static uint16_t clamp_patch(uint16_t patch)
448{
449 if (patch > SEQ_PATCH_FULL_MAX) return 0;
450 if (sequencer_core_patch_compiled_out(patch)) return 0;
451 return patch;
452}

454static bool parse_layer(tlv_reader_t *b, seq_layer_t *L, uint8_t ver)
455{
456 memset(L, 0, sizeof *L);
457
458 uint8_t type_raw;
459 if (!tlv_get_u8(b, &type_raw)) return false;
460 L->type = (type_raw > SEQ_LAYER_MELODIC) ? SEQ_LAYER_MELODIC
461 : (seq_layer_type_t)type_raw;
462
463 if (!tlv_get_u8(b, &L->num_steps)) return false;
464 if (L->num_steps != 16 && L->num_steps != 32) L->num_steps = 16;
465
466 if (!tlv_get_u16(b, &L->patch)) return false;
467 L->patch = clamp_patch(L->patch);
468
469 if (!tlv_get_u32(b, &L->synth_flags)) return false;
470 if (!tlv_get_u8(b, &L->num_voices)) return false;
471 /* AMY's instrument_init() aborts outside 1..MAX_VOICES_PER_INSTRUMENT
472 * (32); a CRC-valid but hand-edited file must not crash the load. */
473 L->num_voices = SEQ_CLAMP_U8(L->num_voices, 1, 32);
474 { uint8_t v; if (!tlv_get_u8(b, &v)) return false; L->chord_mode = v != 0; }
475 if (!tlv_get_u8(b, &L->chord_root)) return false;
476 { uint8_t v; if (!tlv_get_u8(b, &v)) return false;
477 L->chord_type = (v >= CHORD_TYPE_COUNT) ? CHORD_MAJ : (chord_type_t)v; }
478 if (!tlv_get_u8(b, &L->swing_pct)) return false;
479 /* Bulk import bypasses sequencer_core_set_layer_swing(), so enforce the
480 * engine ceiling here: swing-offset math requires the delay to stay
481 * short of one full step. */
482 if (L->swing_pct > SEQ_SWING_MAX) L->swing_pct = SEQ_SWING_MAX;
483
484 for (int t = 0; t < SEQ_TRACKS; t++) {
485 if (!tlv_get_u8(b, &L->track_base_note[t])) return false;
486 if (!tlv_get_u16(b, &L->track_patch[t])) return false;
487 L->track_patch[t] = clamp_patch(L->track_patch[t]);
488 if (!tlv_get_u16(b, &L->track_pcm_preset[t])) return false;
489 if (ver >= 10) {
490 if (!tlv_get_u8(b, &L->track_pcm_mode[t])) return false;
491 } else {
492 L->track_pcm_mode[t] = 0; /* pre-v10: engine default (one-shot) */
493 }
494 { uint8_t v; if (!tlv_get_u8(b, &v)) return false; L->repeat_rate[t] = v; }
495 { uint8_t v; if (!tlv_get_u8(b, &v)) return false; L->mute[t] = v != 0; }
496 { uint8_t v; if (!tlv_get_u8(b, &v)) return false; L->solo[t] = v != 0; }
497 if (!de_vp(b, &L->vp[t], ver)) return false;
498 }
499
500 if (!tlv_get_bytes(b, L->grid, sizeof L->grid)) return false;
501 if (!tlv_get_bytes(b, L->step_note, sizeof L->step_note)) return false;
502 if (ver >= 11) {
503 if (!tlv_get_bytes(b, L->step_pitch_ofs, sizeof L->step_pitch_ofs)) return false;
504 } else {
505 memset(L->step_pitch_ofs, 0, sizeof L->step_pitch_ofs); /* pre-v11: neutral */
506 }
507 if (!tlv_get_bytes(b, L->step_prob, sizeof L->step_prob)) return false;
508 if (!tlv_get_bytes(b, L->step_ratchet, sizeof L->step_ratchet)) return false;
509 /* v9 stores every+prev directly; v<=8 stored a cond enum (0 NONE / 1 FILL
510 * / 2 PREV) and its FILL divisor in the same two byte-array slots - read
511 * into the new fields and remap below. */
512 if (!tlv_get_bytes(b, L->step_every, sizeof L->step_every)) return false;
513 if (!tlv_get_bytes(b, L->step_prev, sizeof L->step_prev)) return false;
514 if (!tlv_get_bytes(b, L->step_transform, sizeof L->step_transform)) return false;
515 if (!tlv_get_bytes(b, L->step_quant_bypass, sizeof L->step_quant_bypass)) return false;
516 if (!tlv_get_bytes(b, L->step_nudge, sizeof L->step_nudge)) return false;
517 if (!tlv_get_bytes(b, L->step_velocity_adj, sizeof L->step_velocity_adj)) return false;
518 if (!tlv_get_bytes(b, L->step_ratchet_taper, sizeof L->step_ratchet_taper)) return false;
519
520 for (int t = 0; t < SEQ_TRACKS; t++) {
521 for (int s = 0; s < SEQ_MAX_STEPS; s++) {
522 L->grid[t][s] = L->grid[t][s] ? 1 : 0;
523 L->step_quant_bypass[t][s] = L->step_quant_bypass[t][s] ? 1 : 0;
524 if (L->step_prob[t][s] > 100) L->step_prob[t][s] = 100;
525 L->step_ratchet[t][s] = SEQ_CLAMP_U8(L->step_ratchet[t][s], 1, SEQ_MAX_RATCHET);
526 if (ver <= 8) {
527 /* Legacy cond enum in step_every's slot, FILL divisor in
528 * step_prev's: FILL(1) -> every=divisor (capped), PREV(2) ->
529 * prev on; anything else neutral. */
530 uint8_t cond = L->step_every[t][s];
531 uint8_t parm = L->step_prev[t][s];
532 L->step_every[t][s] = (cond == 1)
533 ? SEQ_CLAMP_U8(parm, 1, SEQ_STEP_EVERY_MAX) : 1;
534 L->step_prev[t][s] = (cond == 2) ? 1 : 0;
535 } else {
536 L->step_every[t][s] = SEQ_CLAMP_U8(L->step_every[t][s], 1, SEQ_STEP_EVERY_MAX);
537 L->step_prev[t][s] = L->step_prev[t][s] ? 1 : 0;
538 }
539 if (L->step_transform[t][s] >= SEQ_STEP_TRANSFORM_COUNT) L->step_transform[t][s] = SEQ_STEP_TRANSFORM_NONE;
540 }
541 }
542
543 /* Note fields hold a playable pitch or (melodic only) a chord preset
544 * sentinel (seq_chords.h). Anything else - corrupt, or a sentinel family
545 * this build doesn't know - clamps into the playable range so it cannot
546 * leak into pitch math. */
547 for (int t = 0; t < SEQ_TRACKS; t++) {
548 bool mel = L->type == SEQ_LAYER_MELODIC;
549 if (!(mel && SEQ_NOTE_IS_CHORD(L->track_base_note[t]))) {
550 L->track_base_note[t] = SEQ_CLAMP_U8(L->track_base_note[t],
551 SEQ_MEL_NOTE_MIN, SEQ_MEL_NOTE_MAX);
552 }
553 for (int s = 0; s < SEQ_MAX_STEPS; s++) {
554 if (!(mel && SEQ_NOTE_IS_CHORD(L->step_note[t][s]))) {
555 L->step_note[t][s] = SEQ_CLAMP_U8(L->step_note[t][s],
556 SEQ_MEL_NOTE_MIN, SEQ_MEL_NOTE_MAX);
557 }
558 }
559 }
560
561 /* v3: melodic NoteFX. Pre-v3 files get the legacy gate, glide off.
562 * Clamp to the live control ranges either way. */
563 if (ver >= 3) {
564 if (!tlv_get_u8(b, &L->gate_pct)) return false;
565 if (!tlv_get_u16(b, &L->portamento_ms)) return false;
566 } else {
567 L->gate_pct = SEQ_MELODIC_GATE_DEFAULT_PCT;
568 L->portamento_ms = 0;
569 }
570 L->gate_pct = SEQ_CLAMP_U8(L->gate_pct, 10, 100);
571 if (L->portamento_ms > SEQ_MELODIC_PORTAMENTO_MAX_MS)
572 L->portamento_ms = SEQ_MELODIC_PORTAMENTO_MAX_MS;
573
574 /* v4: NoteFX GROOVE. Pre-v4 files get the full legacy accent curve so
575 * they keep their feel. */
576 if (ver >= 4) {
577 if (!tlv_get_u8(b, &L->groove_pct)) return false;
578 } else {
579 L->groove_pct = 100;
580 }
581 if (L->groove_pct > 100) L->groove_pct = 100;
582
583 /* v13: live FM algorithm override. Pre-v13 (and corrupt values): none.
584 * The configure path reasserts it after the load's patch apply; anything
585 * past AMY's algorithm table would be an unchecked OOB index there. */
586 if (ver >= 13) {
587 if (!tlv_get_u8(b, &L->fm_algo_override)) return false;
588 } else {
589 L->fm_algo_override = SEQ_FM_ALGO_NONE;
590 }
591 if (L->fm_algo_override != SEQ_FM_ALGO_NONE &&
592 L->fm_algo_override >= amy_num_algorithms)
593 L->fm_algo_override = SEQ_FM_ALGO_NONE;
594
595 /* v14: per-row voice-block source + the layer's shared block. Pre-v14:
596 * every row reads its own block (the only behaviour those files had). The
597 * layer block is only meaningful on melodic layers; drum rows are forced
598 * to TRACK so the bank seeding never resolves onto the shared block. */
599 voice_params_init_defaults(&L->vp_layer);
600 if (ver >= 14) {
601 for (int t = 0; t < SEQ_TRACKS; t++) {
602 if (!tlv_get_u8(b, &L->vp_src[t])) return false;
603 if (L->vp_src[t] > SEQ_VP_SRC_LAYER || L->type != SEQ_LAYER_MELODIC)
604 L->vp_src[t] = SEQ_VP_SRC_TRACK;
605 }
606 if (!de_vp(b, &L->vp_layer, ver)) return false;
607 } else {
608 memset(L->vp_src, 0, sizeof L->vp_src);
609 if (L->type == SEQ_LAYER_MELODIC) {
610 L->vp_layer.env = seq_default_melodic_env();
611 L->vp_layer.env1 = seq_default_melodic_env1();
612 }
613 }
614
615 return true;
616}

639static void ser_arp(tlv_writer_t *w)
640{
641 size_t h = tlv_begin_section(w, TAG_ARP, 9); /* v2: LFO target is a bitmask;
642 * v3: LFO +wob_rate/+wob_depth;
643 * v4: LFO +wob_depth_only;
644 * v5: filter +feedback (KS);
645 * v6: +follow_quant (appended);
646 * v7: LFO +flt_oct_q;
647 * v8: -source/-wave (patch
648 * covers the wave range);
649 * v9: +dist (appended) */
650 tlv_put_u8(w, arp_get_enabled() ? 1 : 0);
651 tlv_put_u16(w, arp_get_patch());
652 tlv_put_u8(w, (uint8_t)arp_get_direction());
653 tlv_put_u8(w, arp_get_octaves());
654 tlv_put_u8(w, (uint8_t)arp_get_rate());
655 tlv_put_u8(w, arp_get_gate_pct());
656 tlv_put_u8(w, arp_get_scale());
657 tlv_put_u8(w, arp_get_root_note());
658 tlv_put_u16(w, arp_get_portamento_ms());
659 tlv_put_f32(w, arp_get_amp_scale());
660 for (int i = 0; i < ARP_MAX_SLOTS; i++) tlv_put_i16(w, arp_get_slot((uint8_t)i));
661 seq_env_t e; arp_get_envelope(&e); ser_env(w, &e);
662 seq_env_t e2; arp_get_envelope2(&e2); ser_env(w, &e2);
663 seq_filter_t f; arp_get_filter(&f); ser_filter(w, &f);
664 seq_lfo_t l; arp_get_lfo(&l); ser_lfo(w, &l);
665 tlv_put_u8(w, arp_get_follow_quant() ? 1 : 0); /* v6 */
666 seq_dist_t d; arp_get_dist(&d); ser_dist(w, &d); /* v9 */
667 tlv_end_section(w, h);
668}

670static bool parse_arp(tlv_reader_t *b, staged_arp_t *a, uint8_t ver)
671{
672 uint8_t v;
673 /* The caller memsets the staging block, and an all-zero distortion is not
674 * the OFF default - it is OFF with every secondary parameter below its
675 * legal floor. Seed before parsing so a pre-v9 file restores the same
676 * block a fresh boot would give it. */
677 a->dist = (seq_dist_t){ .type = 0, .drive = 2, .bits = 8, .rate = 8, .mix = 100 };
678 if (!tlv_get_u8(b, &v)) return false;
679 a->enabled = v != 0;
680 /* Pre-v8: a WAVE/PATCH source toggle (u8) plus a raw AMY waveform (u16)
681 * preceded the patch number. The toggle is gone (the patch range covers
682 * the raw waves); consume the fields to keep the walk aligned, discard
683 * the values - a legacy WAVE-mode save falls back to its stored patch. */
684 if (ver < 8) {
685 uint8_t legacy_src;
686 uint16_t legacy_wave;
687 if (!tlv_get_u8(b, &legacy_src)) return false;
688 if (!tlv_get_u16(b, &legacy_wave)) return false;
689 }
690 if (!tlv_get_u16(b, &a->patch)) return false;
691 a->patch = clamp_patch(a->patch);
692 if (!tlv_get_u8(b, &v)) return false;
693 a->dir = (v >= ARP_DIR_COUNT) ? ARP_UP : (arp_dir_t)v;
694 if (!tlv_get_u8(b, &a->octaves)) return false;
695 a->octaves = SEQ_CLAMP_U8(a->octaves, 1, ARP_OCT_MAX);
696 if (!tlv_get_u8(b, &v)) return false;
697 a->rate = (v >= ARP_RATE_COUNT) ? ARP_RATE_1_4 : (arp_rate_t)v;
698 if (!tlv_get_u8(b, &a->gate_pct)) return false;
699 a->gate_pct = SEQ_CLAMP_U8(a->gate_pct, 10, 100);
700 if (!tlv_get_u8(b, &a->scale)) return false;
701 if (a->scale >= quantizer_scale_count()) a->scale = 0;
702 if (!tlv_get_u8(b, &a->root)) return false;
703 if (!tlv_get_u16(b, &a->portamento_ms)) return false;
704 if (a->portamento_ms > ARP_PORTAMENTO_MAX_MS) a->portamento_ms = ARP_PORTAMENTO_MAX_MS;
705 if (!tlv_get_f32(b, &a->amp_scale)) return false;
706 a->amp_scale = SEQ_CLAMP_F32(a->amp_scale, 0.0f, 1.0f);
707 for (int i = 0; i < ARP_MAX_SLOTS; i++) {
708 if (!tlv_get_i16(b, &a->slots[i])) return false;
709 if (a->slots[i] < ARP_REST || a->slots[i] > 127) a->slots[i] = -1;
710 }
711 if (!de_env(b, &a->env)) return false;
712 if (!de_env(b, &a->env2)) return false;
713 if (!de_filter(b, &a->filter, ver >= 5)) return false; /* ARP: feedback v5+ */
714 if (!de_lfo(b, &a->lfo, ver, ver >= 3, ver >= 4, ver >= 7)) return false; /* ARP: wobble v3+, reach v4+, flt_oct v7+ */
715 /* v6: follow the global scale quantizer. Pre-v6 files default OFF
716 * (the arp's own scale). */
717 if (ver >= 6) {
718 if (!tlv_get_u8(b, &v)) return false;
719 a->follow_quant = v != 0;
720 } else {
721 a->follow_quant = false;
722 }
723 /* v9: distortion, appended last. Pre-v9 keeps the staged default. */
724 if (!de_dist(b, &a->dist, ver >= 9)) return false;
725 return true;
726}

890 uint8_t count = sequencer_core_progression_get_count();

895 tlv_put_u8(w, root);

897 tlv_put_u8(w, bars);

902static bool parse_prog(tlv_reader_t *b, staged_prog_t *p)
903{
904 uint8_t v, raw_count;
905 if (!tlv_get_u8(b, &v)) return false;
906 p->enabled = v != 0;
907 if (!tlv_get_u8(b, &raw_count)) return false;
908
909 uint8_t max_entries = sequencer_core_progression_get_max();
910 if (max_entries > PROJECT_PROG_MAX_ENTRIES) max_entries = PROJECT_PROG_MAX_ENTRIES;
911 p->count = (raw_count > max_entries) ? max_entries : raw_count;
912
913 for (uint8_t i = 0; i < raw_count; i++) {
914 uint8_t root, ctraw, bars;
915 if (!tlv_get_u8(b, &root)) return false;
916 if (!tlv_get_u8(b, &ctraw)) return false;
917 if (!tlv_get_u8(b, &bars)) return false;
918 if (i < p->count) {
919 p->root[i] = root;
920 p->chord_type[i] = (ctraw >= CHORD_TYPE_COUNT) ? CHORD_MAJ : (chord_type_t)ctraw;
921 p->duration_bars[i] = bars;
922 }
923 }
924 return true;

940/* ── CHRD section (chord presets, seq_chords.h) ──────────────────────────
941 * v1: u8 slot count, then per slot u8 tone count + SEQ_CHORD_MAX_NOTES note
942 * bytes (fixed width; a wider voicing bumps the version). Old firmware skips
943 * the unknown section: chords vanish there and any chord sentinel in that
944 * file's LAYR notes range-clamps to a top-of-range note, no crash. */
945
946static void ser_chrd(tlv_writer_t *w)
947{
948 size_t h = tlv_begin_section(w, TAG_CHRD, 1);
949 tlv_put_u8(w, SEQ_CHORD_SLOTS);
950 for (uint8_t s = 0; s < SEQ_CHORD_SLOTS; s++) {
951 seq_chord_t c;
952 if (!seq_chords_get(s, &c)) memset(&c, 0, sizeof c);
953 tlv_put_u8(w, c.count);
954 for (uint8_t i = 0; i < SEQ_CHORD_MAX_NOTES; i++) {
955 tlv_put_u8(w, c.notes[i]);
956 }
957 }
958 tlv_end_section(w, h);
959}
960
961static bool parse_chrd(tlv_reader_t *b, seq_chord_t slots[SEQ_CHORD_SLOTS])
962{
963 uint8_t nslots;
964 if (!tlv_get_u8(b, &nslots)) return false;
965 for (uint8_t s = 0; s < nslots; s++) {
966 uint8_t count;
967 uint8_t notes[SEQ_CHORD_MAX_NOTES];
968 if (!tlv_get_u8(b, &count)) return false;
969 for (uint8_t i = 0; i < SEQ_CHORD_MAX_NOTES; i++) {
970 if (!tlv_get_u8(b, &notes[i])) return false;
971 }
972 if (s < SEQ_CHORD_SLOTS) {
973 /* count is advisory: seq_chords_import re-derives it from the
974 * non-zero tones, which also normalizes hand-edited files. */
975 memcpy(slots[s].notes, notes, sizeof notes);
976 slots[s].count = (count > SEQ_CHORD_MAX_NOTES)
977 ? SEQ_CHORD_MAX_NOTES : count;
978 }
979 }
980 return true;
981}

1019bool project_snapshot_load(uint8_t slot)
1020{
1021 uint8_t *payload = NULL;
1022 size_t len = 0;
1023 char name[PROJECT_NAME_LEN];
1024 if (!project_store_read(slot, &payload, &len, name)) return false;
1025
1026 seq_layer_t *staged_layers =
1027 heap_caps_malloc(sizeof(seq_layer_t) * MAX_LAYERS, MALLOC_CAP_SPIRAM);
1028 if (!staged_layers) {
1029 free(payload);
1030 ESP_LOGE(TAG, "load slot %u: SPIRAM allocation failed", slot);
1031 return false;
1032 }
1033
1034 staged_glob_t staged_glob; memset(&staged_glob, 0, sizeof staged_glob);
1035 staged_arp_t staged_arp; memset(&staged_arp, 0, sizeof staged_arp);
1036 staged_drone_t staged_drone; memset(&staged_drone, 0, sizeof staged_drone);
1037 staged_prog_t staged_prog; memset(&staged_prog, 0, sizeof staged_prog);
1038 seq_chord_t staged_chords[SEQ_CHORD_SLOTS];
1039 memset(staged_chords, 0, sizeof staged_chords);
1040 uint8_t staged_layer_count = 0;
1041 bool got_glob = false, got_arp = false, got_drone = false, got_prog = false;
1042 bool got_chrd = false;
1043
1044 tlv_reader_t r;
1045 tlv_reader_init(&r, payload, len);
1046 bool ok = true;
1047 uint32_t tag; uint8_t ver; tlv_reader_t body;
1048 while (ok && tlv_next_section(&r, &tag, &ver, &body)) {
1049 switch (tag) {
1050 case TAG_GLOB:
1051 if (got_glob || ver != 1) { ok = false; break; }
1052 ok = parse_glob(&body, &staged_glob);
1053 got_glob = ok;
1054 break;
1055 case TAG_LAYR:
1056 /* Ceiling must track ser_layer()'s version or the firmware
1057 * rejects its own files. */
1058 if (ver < 1 || ver > 13 || staged_layer_count >= MAX_LAYERS) { ok = false; break; }
1059 ok = parse_layer(&body, &staged_layers[staged_layer_count], ver);
1060 if (ok) staged_layer_count++;
1061 break;
1062 case TAG_ARP:
1063 /* Ceiling must track ser_arp()'s version (see TAG_LAYR). */
1064 if (got_arp || ver < 1 || ver > 9) { ok = false; break; }
1065 ok = parse_arp(&body, &staged_arp, ver);
1066 got_arp = ok;
1067 break;
1068 case TAG_DRON:
1069 if (got_drone || ver != 1) { ok = false; break; }
1070 ok = parse_drone(&body, &staged_drone);
1071 got_drone = ok;
1072 break;
1073 case TAG_PROG:
1074 if (got_prog || ver != 1) { ok = false; break; }
1075 ok = parse_prog(&body, &staged_prog);
1076 got_prog = ok;
1077 break;
1078 case TAG_CHRD:
1079 if (got_chrd || ver != 1) { ok = false; break; }
1080 ok = parse_chrd(&body, staged_chords);
1081 got_chrd = ok;
1082 break;
1083 default:
1084 break; /* unknown section: ignore (forward-compat) */
1085 }
1086 }
1087 if (r.err) ok = false; /* truncated/corrupt outer stream */
1088
1089 if (ok && (!got_glob || staged_layer_count < 1 ||
1090 staged_layers[0].type != SEQ_LAYER_DRUM)) {
1091 ok = false;
1092 }
1093 /* Exactly one drum layer, and only at index 0: add_layer(DRUM) always
1094 * binds the fixed SEQ_DRUM_SYNTH_BASE slots, so a second drum layer
1095 * would alias the first one's AMY synths. */
1096 for (uint8_t i = 1; ok && i < staged_layer_count; i++) {
1097 if (staged_layers[i].type == SEQ_LAYER_DRUM) ok = false;
1098 }
1099
1100 if (!ok) {
1101 free(staged_layers);
1102 free(payload);
1103 ESP_LOGW(TAG, "load slot %u: validation failed, no changes made", slot);
1104 return false;
1105 }
1106
1107 /* ── Phase 2: apply. Nothing above this point touched live state. ── */
1108 sequencer_core_set_playing(false);
1109 arp_core_clear_all();
1110
1111 /* Chord table BEFORE the layer imports: import sizes each row's voice
1112 * count from the chord a loaded sentinel references (seq_track_num_voices),
1113 * so the table must already hold the file's voicings. A file without a
1114 * CHRD section clears the table - the project is the whole persisted
1115 * state, chords included. */
1116 seq_chords_import(got_chrd ? staged_chords : NULL);
1117
1118 while (sequencer_core_get_num_layers() > 1) {
1119 sequencer_core_delete_layer((uint8_t)(sequencer_core_get_num_layers() - 1));
1120 }
1121 for (uint8_t i = 1; i < staged_layer_count; i++) {
1122 sequencer_core_add_layer(staged_layers[i].type, staged_layers[i].num_steps);
1123 }
1124 for (uint8_t i = 0; i < staged_layer_count; i++) {
1125 sequencer_core_import_layer(i, &staged_layers[i]);
1126 }
1127
1128 apply_glob(&staged_glob);
1129 if (got_arp) apply_arp(&staged_arp);
1130 if (got_drone) apply_drone(&staged_drone);
1131 if (got_prog) apply_prog(&staged_prog);
1132
1133 /* The layer import writes solo[] wholesale rather than through the setter,
1134 * so nothing has applied the loaded solo state yet. Do it after the arp and
1135 * drone applies, or their duck would be undone by the enable that follows. */
1136 sequencer_core_notify_solo_changed();
1137
1138 synth_ui_reload_mirror_from_core();
1139
1140 free(staged_layers);
1141 free(payload);
1142 ESP_LOGI(TAG, "load slot %u ('%s') OK: %u layer(s)", slot, name, staged_layer_count);
1143 return true;
1144}
