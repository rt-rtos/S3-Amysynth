26static void put_hdr(uint8_t hdr[PROJECT_HDR_LEN], const char *name,
27 uint32_t payload_len, uint32_t crc)
28{
29 memset(hdr, 0, PROJECT_HDR_LEN);

32 tlv_put_u32(&w, PROJECT_MAGIC);
33 tlv_put_u16(&w, PROJECT_FMT_VERSION);
34 tlv_put_u16(&w, 0); /* reserved */

36 if (name) strncpy(nm, name, PROJECT_NAME_LEN - 1);

40}

42bool project_store_write(uint8_t slot, const char *name,
43 const uint8_t *payload, size_t payload_len)
44{
45 /* Mirror the read-side payload bounds: writing a size read() would later
46 * refuse must fail HERE, not turn the slot silently unreadable. */
47 if (!project_fs_ok() || slot >= CONFIG_SYNTH_PROJECT_MAX_SLOTS || !payload
48 || payload_len == 0 || payload_len > (1u << 20))
49 return false;
50
51 uint8_t hdr[PROJECT_HDR_LEN];
52 put_hdr(hdr, name, (uint32_t)payload_len,
53 project_crc32(payload, payload_len));
54
55 char tmp_path[SLOT_MAX_PATH], final_path[SLOT_MAX_PATH];
56 slot_path(slot, true, tmp_path);
57 slot_path(slot, false, final_path);
58
59 FILE *f = fopen(tmp_path, "wb");
60 if (!f) { ESP_LOGE(TAG, "open %s failed", tmp_path); return false; }
61 bool ok = fwrite(hdr, 1, PROJECT_HDR_LEN, f) == PROJECT_HDR_LEN
62 && fwrite(payload, 1, payload_len, f) == payload_len
63 && fflush(f) == 0
64 && fsync(fileno(f)) == 0;
65 ok = (fclose(f) == 0) && ok;
66 if (!ok) { unlink(tmp_path); ESP_LOGE(TAG, "write slot %u failed", slot); return false; }
67
68 /* LittleFS rename won't overwrite an existing target, so the old file is
69 * unlinked first. Power loss inside the unlink..rename window leaves the
70 * slot empty (new data still in .tmp, reaped at next mount) - a narrow,
71 * accepted tradeoff; the old data was about to be replaced anyway. */
72 unlink(final_path);
73 if (rename(tmp_path, final_path) != 0) {
74 unlink(tmp_path);
75 ESP_LOGE(TAG, "rename slot %u failed", slot);
76 return false;
77 }
78 return true;
79}

81bool project_store_read(uint8_t slot, uint8_t **out, size_t *out_len,
82 char name_out[PROJECT_NAME_LEN])
83{
84 if (!project_fs_ok() || slot >= CONFIG_SYNTH_PROJECT_MAX_SLOTS || !out)
85 return false;
86
87 char path[SLOT_MAX_PATH];
88 slot_path(slot, false, path);
89 FILE *f = fopen(path, "rb");
90 if (!f) return false;
91
92 uint8_t hdr[PROJECT_HDR_LEN];
93 bool ok = fread(hdr, 1, PROJECT_HDR_LEN, f) == PROJECT_HDR_LEN;
94
95 uint32_t magic = 0, payload_len = 0, crc = 0;
96 uint16_t ver = 0, reserved = 0;
97 char nm[PROJECT_NAME_LEN] = {0};
98 if (ok) {
99 tlv_reader_t r;
100 tlv_reader_init(&r, hdr, PROJECT_HDR_LEN);
101 ok = tlv_get_u32(&r, &magic) && tlv_get_u16(&r, &ver)
102 && tlv_get_u16(&r, &reserved)
103 && tlv_get_bytes(&r, nm, PROJECT_NAME_LEN)
104 && tlv_get_u32(&r, &payload_len) && tlv_get_u32(&r, &crc);
105 }
106 ok = ok && magic == PROJECT_MAGIC && ver <= PROJECT_FMT_VERSION
107 && payload_len > 0 && payload_len <= (1u << 20);
108
109 uint8_t *buf = NULL;
110 if (ok) {
111 buf = heap_caps_malloc(payload_len, MALLOC_CAP_SPIRAM);
112 ok = buf && fread(buf, 1, payload_len, f) == payload_len
113 && project_crc32(buf, payload_len) == crc;
114 }
115 fclose(f);
116
117 if (!ok) {
118 free(buf);
119 ESP_LOGW(TAG, "slot %u invalid/corrupt - refused", slot);
120 return false;
121 }
122 *out = buf;
123 if (out_len) *out_len = payload_len;
124 if (name_out) memcpy(name_out, nm, PROJECT_NAME_LEN);
125 return true;
126}

202void project_store_cleanup_tmp(void)
203{
204 if (!project_fs_ok())
205 return;
206
207 /* Enumerate the slots rather than scanning the directory: the only temp
208 * files this module can create are slot_path(slot, tmp) for slot <
209 * MAX_SLOTS, so the name comes from the same function that writes it.
210 * Nothing to bound, no path to assemble, and it cannot unlink a .tmp that
211 * is not ours. Absent files return ENOENT, which is the normal case. */
212 for (uint8_t slot = 0; slot < CONFIG_SYNTH_PROJECT_MAX_SLOTS; slot++) {
213 char path[SLOT_MAX_PATH];
214 slot_path(slot, true, path);
215 unlink(path);
216 }
217}
