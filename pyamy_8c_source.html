178 amy_start(amy_config); // initializes amy

238static PyObject * inject_midi_bytes_wrapper(PyObject *self, PyObject *args) {
239 // Feed a raw MIDI byte stream through the real byte-stream parser
240 // (convert_midi_bytes_to_messages) -- exercises running status, real-time
241 // interleaving, etc. Unlike inject_midi, which hands a pre-formed 3-byte
242 // message straight to amy_event_midi_message_received, this is the only way
243 // to test the parser from Python.
244 PyObject *seq;
245 int usb = 0;
246 if (!PyArg_ParseTuple(args, "O|i", &seq, &usb))
247 return NULL;
248 PyObject *fast = PySequence_Fast(seq, "inject_midi_bytes expects a list/tuple/bytes of ints");
249 if (fast == NULL) return NULL;
250 Py_ssize_t n = PySequence_Fast_GET_SIZE(fast);
251 uint8_t *buf = (uint8_t *)malloc(n > 0 ? (size_t)n : 1);
252 for (Py_ssize_t i = 0; i < n; ++i)
253 buf[i] = (uint8_t)(PyLong_AsLong(PySequence_Fast_GET_ITEM(fast, i)) & 0xFF);
254 Py_DECREF(fast);
255 convert_midi_bytes_to_messages(buf, (size_t)n, (uint8_t)usb);
256 free(buf);
257 return Py_None;
258}
