151static PyObject * live_wrapper(PyObject *self, PyObject *args, PyObject *kwargs) {
152 // Parse and validate every kwarg into a local config BEFORE tearing down the
153 // running AMY: a rejected kwarg then leaves audio playing instead of
154 // silently killing it (and leaving AMY stopped for the next live() call).
155 amy_config_t amy_config = amy_default_config();
156 Py_ssize_t pos = 0;
157 PyObject *key_obj = NULL;
158 PyObject *value_obj = NULL;
159
160 if (PyTuple_Size(args) != 0) {
161 PyErr_SetString(PyExc_TypeError, "live() no longer accepts positional args; use keyword args like live(playback_device_id=..., capture_device_id=...)");
162 return NULL;
163 }
164
165 while (kwargs != NULL && PyDict_Next(kwargs, &pos, &key_obj, &value_obj)) {
166 const char *key = PyUnicode_AsUTF8(key_obj);
167 if (key == NULL) return NULL;
168 if (parse_live_kwarg(&amy_config, key, value_obj) != 0) {
169 return NULL;
170 }
171 }
172
173 amy_config.audio = AMY_AUDIO_IS_MINIAUDIO;
174 amy_stop();
175 amy_start(amy_config); // initializes amy
176 Py_RETURN_NONE;
177}

220static PyObject * inject_midi_bytes_wrapper(PyObject *self, PyObject *args) {
221 // Feed a raw MIDI byte stream through the real byte-stream parser
222 // (convert_midi_bytes_to_messages), so running status and interleaved
223 // real-time bytes behave exactly as they do for actual MIDI input. This is
224 // the only MIDI entry point from Python: there is no scheduling parameter
225 // because live MIDI has no time of its own -- it plays when it arrives.
226 PyObject *seq;
227 int usb = 0;
228 if (!PyArg_ParseTuple(args, "O|i", &seq, &usb))
229 return NULL;
230 PyObject *fast = PySequence_Fast(seq, "inject_midi_bytes expects a list/tuple/bytes of ints");
231 if (fast == NULL) return NULL;
232 Py_ssize_t n = PySequence_Fast_GET_SIZE(fast);
233 uint8_t *buf = (uint8_t *)malloc(n > 0 ? (size_t)n : 1);
234 for (Py_ssize_t i = 0; i < n; ++i)
235 buf[i] = (uint8_t)(PyLong_AsLong(PySequence_Fast_GET_ITEM(fast, i)) & 0xFF);
236 Py_DECREF(fast);
237 convert_midi_bytes_to_messages(buf, (size_t)n, (uint8_t)usb);
238 free(buf);
239 Py_RETURN_NONE;
240}
