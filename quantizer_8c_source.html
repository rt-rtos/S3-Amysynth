16/* Floored integer division (rounds toward negative infinity), unlike C's
17 * built-in division which truncates toward zero. Needed so negative note/octave
18 * offsets map to the correct octave (e.g. -1/12 should be -1, not 0). */
19static int32_t quantizer_floor_div(int32_t numerator, int32_t denominator)
20{
21 int32_t quotient = numerator / denominator;
22 int32_t remainder = numerator % denominator;
23 /* If there's a remainder and the signs differ, C truncated up; correct it. */
24 if (remainder != 0 && ((remainder < 0) != (denominator < 0))) {
25 quotient -= 1;
26 }
27 return quotient;
28}

54/* Convert a scale degree (which step of the scale, can be negative or beyond
55 * one octave) into a MIDI note. The degree is split into an octave count and an
56 * in-scale index; the index picks a semitone offset from the scale's interval
57 * table, and the octave adds/subtracts 12 semitones each. */
58uint8_t quantizer_degree_to_midi(int32_t scale_degree, uint8_t root_note,
59 const musical_scale_t *scale)
60{
61 if (scale == NULL || scale->size == 0) {
62 return 60;
63 }
64
65 /* floor-div/mod so negative degrees wrap into a valid [0, size) index. */
66 int32_t octave = quantizer_floor_div(scale_degree, scale->size);
67 int32_t scale_index = scale_degree - (octave * scale->size);
68 if (scale_index < 0) {
69 scale_index += scale->size;
70 }
71
72 int32_t midi_note = (int32_t)root_note + (octave * 12) + scale->intervals[scale_index];
73 return quantizer_clamp_midi(midi_note);
74}

76/* Snap an arbitrary MIDI note to the nearest note in the given scale/key.
77 * Strategy: generate every scale note across the octave the input falls in plus
78 * the neighbours (to handle notes near an octave boundary), and keep the
79 * closest one. Ties break toward the lower note for deterministic results. */
80uint8_t quantizer_snap_midi_note(uint8_t midi_note, uint8_t root_note,
81 const musical_scale_t *scale)
82{
83 if (scale == NULL || scale->size == 0) {
84 return midi_note;
85 }
86
87 int32_t best_note = (int32_t)midi_note;
88 int32_t best_distance = 9999;
89 /* Locate which octave (relative to root) the input note sits in. */
90 int32_t relative_note = (int32_t)midi_note - (int32_t)root_note;
91 int32_t base_octave = quantizer_floor_div(relative_note, 12);
92
93 /* Search the surrounding octaves so candidates just across an octave line
94 * are still considered as the nearest match. */
95 for (int32_t octave = base_octave - 1; octave <= base_octave + 1; octave++) {
96 for (uint8_t i = 0; i < scale->size; i++) {
97 int32_t candidate = (int32_t)root_note + (octave * 12) + scale->intervals[i];
98 if (candidate < 0 || candidate > 127) {
99 continue; /* out of MIDI range */
100 }
101
102 int32_t distance = candidate - (int32_t)midi_note;
103 if (distance < 0) {
104 distance = -distance; /* absolute distance in semitones */
105 }
106
107 if (distance < best_distance ||
108 (distance == best_distance && candidate < best_note)) {
109 best_distance = distance;
110 best_note = candidate;
111 }
112 }
113 }
114
115 return (uint8_t)best_note;
116}

143uint8_t quantizer_snap_to_chord(uint8_t midi_note, uint8_t root,
144 chord_type_t chord_type)
145{
146 if ((unsigned)chord_type >= CHORD_TYPE_COUNT) return midi_note;
147 const int8_t *intervals = s_chord_intervals[chord_type];
148 uint8_t root_pc = root % 12;
149
150 int32_t best_note = (int32_t)midi_note;
151 int32_t best_distance = 9999;
152
153 /* Search chord tones across all reachable MIDI octaves. */
154 for (int32_t oct = -1; oct <= 10; oct++) {
155 for (int i = 0; i < 6; i++) {
156 if (intervals[i] < 0) break;
157 int32_t candidate = oct * 12 + (int32_t)root_pc + intervals[i];
158 if (candidate < 0 || candidate > 127) continue;
159 int32_t dist = candidate - (int32_t)midi_note;
160 if (dist < 0) dist = -dist;
161 if (dist < best_distance ||
162 (dist == best_distance && candidate < best_note)) {
163 best_distance = dist;
