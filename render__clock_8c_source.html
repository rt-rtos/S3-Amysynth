20// Requested counter resolution. The GPTimer prescaler is an INTEGER divider of
21// the source clock (see gptimer_select_periph_clock(): prescale = src / res,
22// truncated), so only resolutions that divide the source exactly are granted as
23// asked. On the S3 the default source is APB @ 80 MHz and the minimum prescale
24// is 2, so 40 MHz is both the highest achievable resolution and an exact one
25// (80 / 2). It is deliberately NOT 3 MHz: 80/3 truncates to a prescale of 26,
26// silently granting 3,076,923 Hz - 2.56% fast - which paced every render block
27// (and therefore the whole sequencer clock) 2.56% early.
28//
29// A perfectly exact block period is impossible here regardless: it would need a
30// resolution that is an integer multiple of 187.5 Hz, and no integer divisor of
31// 80 MHz yields one. At 40 MHz the rounded period is off by ~1.6 ppm, far below
32// the S3-vs-host crystal tolerance that the USB ring already has to cope with.
33#define RENDER_CLOCK_RESOLUTION_HZ (40 * 1000 * 1000)
34

36static TaskHandle_t s_render_task = NULL;

38// GPTimer alarm ISR. Runs on the core that called gptimer_enable() (i.e. the
39// render task's core), so the notify + wake stay core-local with no cross-core
40// latency. Kept in IRAM and minimal: just a counting task notification.
41static bool IRAM_ATTR render_clock_on_alarm(gptimer_handle_t timer,
42 const gptimer_alarm_event_data_t *edata,
43 void *user_ctx)
44{
45 (void)timer;
46 (void)edata;
47 (void)user_ctx;
48 BaseType_t higher_prio_woken = pdFALSE;
49 // Counting give: each missed/queued tick increments the notification value,
50 // so render_clock_wait() can report a backlog (>1) as an overrun signal.
51 vTaskNotifyGiveFromISR(s_render_task, &higher_prio_woken);
52 return higher_prio_woken == pdTRUE; // request context switch if needed
53}

55esp_err_t render_clock_start(uint32_t block_frames, uint32_t sample_rate_hz)
56{
57 if (s_timer != NULL) {
58 return ESP_OK; // already started
59 }
60
61 if (block_frames == 0 || sample_rate_hz == 0) {
62 return ESP_ERR_INVALID_ARG;
63 }
64
65 s_render_task = xTaskGetCurrentTaskHandle();
66
67 const gptimer_config_t timer_config = {
68 .clk_src = GPTIMER_CLK_SRC_DEFAULT,
69 .direction = GPTIMER_COUNT_UP,
70 .resolution_hz = RENDER_CLOCK_RESOLUTION_HZ,
71 .intr_priority = 0, // auto-select
72 };
73 esp_err_t err = gptimer_new_timer(&timer_config, &s_timer);
74 if (err != ESP_OK) {
75 ESP_LOGE(TAG, "gptimer_new_timer failed: %s", esp_err_to_name(err));
76 s_timer = NULL;
77 return err;
78 }
79
80 // Derive the alarm period from the resolution the driver actually GRANTED,
81 // never from the one requested above. They match on the S3 today, but a
82 // different clock source, a different SoC (source frequency and minimum
83 // divider both change), or DFS on a PM-enabled build can each make them
84 // diverge, and the driver reports that only as a log warning.
85 uint32_t granted_resolution_hz = 0;
86 err = gptimer_get_resolution(s_timer, &granted_resolution_hz);
87 if (err != ESP_OK) {
88 ESP_LOGE(TAG, "gptimer_get_resolution failed: %s", esp_err_to_name(err));
89 goto fail;
90 }
91 if (granted_resolution_hz != (uint32_t)RENDER_CLOCK_RESOLUTION_HZ) {
92 ESP_LOGW(TAG, "requested %d Hz, granted %" PRIu32 " Hz - deriving the block "
93 "period from the granted rate",
94 (int)RENDER_CLOCK_RESOLUTION_HZ, granted_resolution_hz);
95 }
96
97 // Round rather than truncate: a half-tick bias is free to avoid, and at low
98 // granted resolutions truncation is the difference between a few ppm and a
99 // few hundred.
100 const uint32_t period_ticks =
101 (uint32_t)((((uint64_t)block_frames * granted_resolution_hz) + (sample_rate_hz / 2))
102 / (uint64_t)sample_rate_hz);
103 // alarm_count 0 with auto-reload would re-arm instantly and livelock the
104 // core in the alarm ISR. Only reachable via an absurd resolution/rate ratio,
105 // but period_ticks is derived from queried state now, so it gets a guard.
106 if (period_ticks == 0) {
107 ESP_LOGE(TAG, "period_ticks == 0 (resolution %" PRIu32 " Hz too low for %" PRIu32
108 " frames @ %" PRIu32 " Hz)", granted_resolution_hz, block_frames, sample_rate_hz);
109 err = ESP_ERR_INVALID_STATE;
110 goto fail;
111 }
112
113 const gptimer_event_callbacks_t cbs = {
114 .on_alarm = render_clock_on_alarm,
115 };
116 err = gptimer_register_event_callbacks(s_timer, &cbs, NULL);
117 if (err != ESP_OK) {
118 ESP_LOGE(TAG, "gptimer_register_event_callbacks failed: %s", esp_err_to_name(err));
119 goto fail;
120 }
121
122 // Enabling from the render task's context registers the ISR on this core.
123 err = gptimer_enable(s_timer);
124 if (err != ESP_OK) {
125 ESP_LOGE(TAG, "gptimer_enable failed: %s", esp_err_to_name(err));
126 goto fail;
127 }
128
129 const gptimer_alarm_config_t alarm_config = {
130 .alarm_count = period_ticks, // counter runs at granted_resolution_hz
131 .reload_count = 0,
132 .flags.auto_reload_on_alarm = true,
133 };
134 err = gptimer_set_alarm_action(s_timer, &alarm_config);
135 if (err != ESP_OK) {
136 ESP_LOGE(TAG, "gptimer_set_alarm_action failed: %s", esp_err_to_name(err));
137 gptimer_disable(s_timer);
138 goto fail;
139 }
140
141 err = gptimer_start(s_timer);
142 if (err != ESP_OK) {
143 ESP_LOGE(TAG, "gptimer_start failed: %s", esp_err_to_name(err));
144 gptimer_disable(s_timer);
145 goto fail;
146 }
147
148 ESP_LOGI(TAG, "render master clock started: %" PRIu32 " ticks @ %" PRIu32 " Hz "
149 "(%" PRIu32 " ns block period) on core %d",
150 period_ticks, granted_resolution_hz,
151 (uint32_t)(((uint64_t)period_ticks * 1000000000ULL) / granted_resolution_hz),
152 xPortGetCoreID());
153 return ESP_OK;
154
155fail:
156 gptimer_del_timer(s_timer);
157 s_timer = NULL;
158 return err;
159}

161uint32_t render_clock_wait(void)
162{
163 // Block until >=1 tick, return the accumulated count and clear it to 0.
164 // The caller renders exactly ONE block regardless of the returned value.
165 return ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
166}
