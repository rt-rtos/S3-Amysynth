19// Requested counter resolution. The GPTimer prescaler is a TRUNCATED integer
20// divider of the source clock, so only exact divisors are granted as asked. On
21// the S3 (APB @ 80 MHz, minimum prescale 2) 40 MHz is both the highest and an
22// exact one. Deliberately NOT 3 MHz: that truncates to 3,076,923 Hz - 2.56%
23// fast - pacing every render block, and thus the sequencer, 2.56% early.
24// No divisor of 80 MHz gives an exact block period; 40 MHz rounds to ~1.6 ppm,
25// far below the crystal tolerance the USB ring already absorbs.
26#define RENDER_CLOCK_RESOLUTION_HZ (40 * 1000 * 1000)
27

44}

46esp_err_t render_clock_start(uint32_t block_frames, uint32_t sample_rate_hz)
47{
48 if (s_timer != NULL) {
49 return ESP_OK; // already started
50 }
51
52 if (block_frames == 0 || sample_rate_hz == 0) {
53 return ESP_ERR_INVALID_ARG;
54 }
55
56 s_render_task = xTaskGetCurrentTaskHandle();
57
58 const gptimer_config_t timer_config = {
59 .clk_src = GPTIMER_CLK_SRC_DEFAULT,
60 .direction = GPTIMER_COUNT_UP,
61 .resolution_hz = RENDER_CLOCK_RESOLUTION_HZ,
62 .intr_priority = 0, // auto-select
63 };
64 esp_err_t err = gptimer_new_timer(&timer_config, &s_timer);
65 if (err != ESP_OK) {
66 ESP_LOGE(TAG, "gptimer_new_timer failed: %s", esp_err_to_name(err));
67 s_timer = NULL;
68 return err;
69 }
70
71 // Derive the alarm period from the GRANTED resolution, never the requested
72 // one. A different clock source, SoC, or DFS on a PM build can make them
73 // diverge, and the driver only reports that as a log warning.
74 uint32_t granted_resolution_hz = 0;
75 err = gptimer_get_resolution(s_timer, &granted_resolution_hz);
76 if (err != ESP_OK) {
77 ESP_LOGE(TAG, "gptimer_get_resolution failed: %s", esp_err_to_name(err));
78 goto fail;
79 }
80 if (granted_resolution_hz != (uint32_t)RENDER_CLOCK_RESOLUTION_HZ) {
81 ESP_LOGW(TAG, "requested %d Hz, granted %" PRIu32 " Hz - deriving the block "
82 "period from the granted rate",
83 (int)RENDER_CLOCK_RESOLUTION_HZ, granted_resolution_hz);
84 }
85
86 // Round, not truncate: at low granted resolutions truncation is the
87 // difference between a few ppm and a few hundred.
88 const uint32_t period_ticks =
89 (uint32_t)((((uint64_t)block_frames * granted_resolution_hz) + (sample_rate_hz / 2))
90 / (uint64_t)sample_rate_hz);
91 // alarm_count 0 with auto-reload re-arms instantly and livelocks the core
92 // in the ISR; period_ticks comes from queried state, so guard it.
93 if (period_ticks == 0) {
94 ESP_LOGE(TAG, "period_ticks == 0 (resolution %" PRIu32 " Hz too low for %" PRIu32
95 " frames @ %" PRIu32 " Hz)", granted_resolution_hz, block_frames, sample_rate_hz);
96 err = ESP_ERR_INVALID_STATE;
97 goto fail;
98 }
99
100 const gptimer_event_callbacks_t cbs = {
101 .on_alarm = render_clock_on_alarm,
102 };
103 err = gptimer_register_event_callbacks(s_timer, &cbs, NULL);
104 if (err != ESP_OK) {
105 ESP_LOGE(TAG, "gptimer_register_event_callbacks failed: %s", esp_err_to_name(err));
106 goto fail;
107 }
108
109 // Enabling from the render task's context registers the ISR on this core.
110 err = gptimer_enable(s_timer);
111 if (err != ESP_OK) {
112 ESP_LOGE(TAG, "gptimer_enable failed: %s", esp_err_to_name(err));
113 goto fail;
114 }
115
116 const gptimer_alarm_config_t alarm_config = {
117 .alarm_count = period_ticks, // counter runs at granted_resolution_hz
118 .reload_count = 0,
119 .flags.auto_reload_on_alarm = true,
120 };
121 err = gptimer_set_alarm_action(s_timer, &alarm_config);
122 if (err != ESP_OK) {
123 ESP_LOGE(TAG, "gptimer_set_alarm_action failed: %s", esp_err_to_name(err));
124 gptimer_disable(s_timer);
125 goto fail;
126 }
127
128 err = gptimer_start(s_timer);
129 if (err != ESP_OK) {
130 ESP_LOGE(TAG, "gptimer_start failed: %s", esp_err_to_name(err));
131 gptimer_disable(s_timer);
132 goto fail;
133 }
134
135 ESP_LOGI(TAG, "render master clock started: %" PRIu32 " ticks @ %" PRIu32 " Hz "
136 "(%" PRIu32 " ns block period) on core %d",
137 period_ticks, granted_resolution_hz,
138 (uint32_t)(((uint64_t)period_ticks * 1000000000ULL) / granted_resolution_hz),
139 xPortGetCoreID());
140 return ESP_OK;
141
142fail:
143 gptimer_del_timer(s_timer);
144 s_timer = NULL;
145 return err;
146}
