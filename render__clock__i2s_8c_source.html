20// Stereo 16-bit frames, matching AMY_NCHANS / the USB output format. Nothing
21// is wired to the I2S output (no GPIO assigned below) - its only job is to
22// drain its TX DMA queue on schedule, so the buffer stays silent.

24#define RENDER_CLOCK_I2S_BYTES_PER_SAMPLE 2

30esp_err_t render_clock_start(uint32_t block_frames, uint32_t sample_rate_hz)
31{
32 if (s_tx_handle != NULL) {
33 return ESP_OK; // already started
34 }
35
36 s_silence_block_bytes =
37 (size_t)block_frames * RENDER_CLOCK_I2S_CHANNELS * RENDER_CLOCK_I2S_BYTES_PER_SAMPLE;
38 // One-time startup alloc. DMA-capable internal RAM is required by the
39 // GDMA-backed I2S TX descriptors; ~1 KiB at the default 256-frame block.
40 s_silence_block = heap_caps_calloc(1, s_silence_block_bytes,
41 MALLOC_CAP_DMA | MALLOC_CAP_INTERNAL);
42 if (s_silence_block == NULL) {
43 ESP_LOGE(TAG, "silence buffer alloc failed (%u bytes)",
44 (unsigned)s_silence_block_bytes);
45 return ESP_ERR_NO_MEM;
46 }
47
48 i2s_chan_config_t chan_cfg = I2S_CHANNEL_DEFAULT_CONFIG(I2S_NUM_AUTO, I2S_ROLE_MASTER);
49 // Buffer depth = render look-ahead: with the default 2, the render task
50 // can run 1 block period ahead of the drain before i2s_channel_write()
51 // blocks it, vs. zero slack on the GPTimer path.
52 chan_cfg.dma_desc_num = CONFIG_RENDER_CLOCK_I2S_DMA_DESC_NUM;
53 chan_cfg.dma_frame_num = block_frames;
54
55 esp_err_t err = i2s_new_channel(&chan_cfg, &s_tx_handle, NULL);
56 if (err != ESP_OK) {
57 ESP_LOGE(TAG, "i2s_new_channel failed: %s", esp_err_to_name(err));
58 goto fail_no_channel;
59 }
60
61 // Every GPIO unassigned: no pin mux change, no physical signal. In master
62 // role the internal clock divider free-runs and paces the DMA queue
63 // regardless of routing - the only property this backend depends on.
64 const i2s_std_config_t std_cfg = {
65 .clk_cfg = I2S_STD_CLK_DEFAULT_CONFIG(sample_rate_hz),
66 .slot_cfg = I2S_STD_MSB_SLOT_DEFAULT_CONFIG(I2S_DATA_BIT_WIDTH_16BIT,
67 I2S_SLOT_MODE_STEREO),
68 .gpio_cfg = {
69 .mclk = I2S_GPIO_UNUSED,
70 .bclk = I2S_GPIO_UNUSED,
71 .ws = I2S_GPIO_UNUSED,
72 .dout = I2S_GPIO_UNUSED,
73 .din = I2S_GPIO_UNUSED,
74 .invert_flags = {
75 .mclk_inv = false,
76 .bclk_inv = false,
77 .ws_inv = false,
78 },
79 },
80 };
81 err = i2s_channel_init_std_mode(s_tx_handle, &std_cfg);
82 if (err != ESP_OK) {
83 ESP_LOGE(TAG, "i2s_channel_init_std_mode failed: %s", esp_err_to_name(err));
84 goto fail_channel;
85 }
86
87 err = i2s_channel_enable(s_tx_handle);
88 if (err != ESP_OK) {
89 ESP_LOGE(TAG, "i2s_channel_enable failed: %s", esp_err_to_name(err));
90 goto fail_channel;
91 }
92
93 ESP_LOGI(TAG,
94 "render master clock started: I2S TX DMA pacing, %u frames/block @ %u Hz, "
95 "%u DMA buffers, no GPIO assigned, on core %d",
96 (unsigned)block_frames, (unsigned)sample_rate_hz,
97 (unsigned)CONFIG_RENDER_CLOCK_I2S_DMA_DESC_NUM, xPortGetCoreID());
98 return ESP_OK;
99
100fail_channel:
101 i2s_del_channel(s_tx_handle);
102 s_tx_handle = NULL;
103fail_no_channel:
104 heap_caps_free(s_silence_block);
105 s_silence_block = NULL;
106 s_silence_block_bytes = 0;
107 return err;
108}

110uint32_t render_clock_wait(void)
111{
112 // The blocking write is the clock: it returns once the TX DMA queue has
113 // room, i.e. a queued block finished transmitting. No accumulated-tick
114 // count exists here, so this backend always returns 1 (see render_clock.h).
115 size_t written = 0;
116 esp_err_t err = i2s_channel_write(s_tx_handle, s_silence_block, s_silence_block_bytes,
117 &written, portMAX_DELAY);
118 if (err != ESP_OK) {
119 ESP_LOGE(TAG, "i2s_channel_write failed: %s", esp_err_to_name(err));
120 }
121 return 1;
122}
