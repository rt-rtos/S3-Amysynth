1// Per-block render-time statistics (cycle-count histogram + worst-block
2// snapshot). The render budget is one block period: 256 samples @ 48 kHz =
3// 5.333 ms; everything here is denominated in CPU cycles of that budget.
4//
5// Concurrency: the window struct is written only by the render task (Core 1)
6// and read only by the idle loop (Core 0), published through a seqlock: the
7// writer bumps `version` to odd before mutating and back to even after; the
8// reader retries until it sees the same even version on both sides of its
9// copy. Window resets are requested by the reader (s_reset_req) but executed
10// by the writer at the next block boundary, so only one core ever writes the
11// window. No locks, no allocation, nothing blocking on the render side.
12#include "render_stats.h"

25// 240 MHz * 256 / 48000 = 1,280,000 cycles per block.
26#define BUDGET_CYCLES ((uint32_t)((uint64_t)CONFIG_ESP_DEFAULT_CPU_FREQ_MHZ \
27 * 1000000ULL * AMY_BLOCK_SIZE / AMY_SAMPLE_RATE))
28#define HIST_BUCKETS 16u
29// Linear buckets of budget/16 = 6.25% of budget each; the last bucket also
30// absorbs everything at or over budget.
31#define BUCKET_CYCLES (BUDGET_CYCLES / HIST_BUCKETS)
32#define SPIKE_CYCLES ((uint32_t)((uint64_t)BUDGET_CYCLES \
33 * CONFIG_AMYSYNTH_RENDER_STATS_SPIKE_PCT / 100u))
34
35// Compiler barrier: keeps the window mutations from being reordered across
36// the seqlock version bumps. The S3's cores don't reorder stores, so no
37// hardware fence is needed.
38#define BARRIER() __asm__ __volatile__("" ::: "memory")
39
40typedef struct {
41 volatile uint32_t version; // seqlock: odd while the writer is mid-update
42 uint32_t count;
43 uint32_t min_cycles;
44 uint64_t sum_cycles;
45 uint32_t spikes; // blocks at/above the spike threshold
46 uint32_t over_budget; // blocks that exceeded the full block period
47 uint32_t hist[HIST_BUCKETS];
48 // Worst block since the last reset (doubles as the window max):
49 uint32_t worst_cycles;
50 uint32_t worst_block; // amy_global.total_blocks at capture
51 uint16_t worst_qsize; // delta-queue depth at capture
52 uint8_t worst_tick; // a sequencer tick ran inside that block
53} stats_window_t;

55static stats_window_t s_w = { .min_cycles = UINT32_MAX };
56
57static volatile uint32_t s_reset_req = 0; // reader increments to request
58static uint32_t s_reset_ack = 0; // writer-private
59static uint32_t s_block_t0; // writer-private

61void render_stats_block_begin(void)
62{
63 if (s_reset_ack != s_reset_req) {
64 s_reset_ack = s_reset_req;
65 uint32_t v = s_w.version + 1; // -> odd
66 s_w.version = v;
67 BARRIER();
68 memset((char *)&s_w + sizeof(s_w.version), 0,
69 sizeof(s_w) - sizeof(s_w.version));
70 s_w.min_cycles = UINT32_MAX;
71 BARRIER();
72 s_w.version = v + 1; // -> even
73 }
74 s_block_t0 = esp_cpu_get_cycle_count();
75}

77void render_stats_block_end(bool seq_tick_coincident)
78{
79 // Unsigned subtraction stays correct across the ~17.9 s ccount wrap.
80 uint32_t cycles = esp_cpu_get_cycle_count() - s_block_t0;
81
82 s_w.version++; // -> odd
83 BARRIER();
84 s_w.count++;
85 s_w.sum_cycles += cycles;
86 if (cycles < s_w.min_cycles) s_w.min_cycles = cycles;
87 uint32_t b = cycles / BUCKET_CYCLES;
88 s_w.hist[b >= HIST_BUCKETS ? HIST_BUCKETS - 1u : b]++;
89 if (cycles >= SPIKE_CYCLES) s_w.spikes++;
90 if (cycles > BUDGET_CYCLES) s_w.over_budget++;
91 if (cycles > s_w.worst_cycles) {
92 s_w.worst_cycles = cycles;
93 s_w.worst_block = amy_global.total_blocks;
94 s_w.worst_qsize = amy_global.delta_qsize;
95 s_w.worst_tick = seq_tick_coincident ? 1 : 0;
96 }
97 BARRIER();
98 s_w.version++; // -> even
99}

101void render_stats_report(void)
102{
103 stats_window_t copy = { 0 };
104 for (int tries = 0; tries < 8; tries++) {
105 uint32_t v1 = s_w.version;
106 if (v1 & 1u) continue;
107 BARRIER();
108 memcpy(&copy, (const void *)&s_w, sizeof(copy));
109 BARRIER();
110 if (s_w.version == v1) break; // stable copy; else retry
111 }
112 if (copy.count == 0) {
113 ESP_LOGI(TAG, "no blocks in window");
114 return;
115 }
116
117 const uint32_t mhz = CONFIG_ESP_DEFAULT_CPU_FREQ_MHZ;
118 uint32_t mean = (uint32_t)(copy.sum_cycles / copy.count);
119
120 // p99 from the histogram: first bucket where the cumulative count crosses
121 // 99%, reported as that bucket's upper edge.
122 uint32_t p99_target = copy.count - copy.count / 100u;
123 uint32_t cum = 0, p99_bucket = HIST_BUCKETS - 1u;
124 for (uint32_t i = 0; i < HIST_BUCKETS; i++) {
125 cum += copy.hist[i];
126 if (cum >= p99_target) { p99_bucket = i; break; }
127 }
128 uint32_t p99_cycles = (p99_bucket + 1u) * BUCKET_CYCLES;
129
130 ESP_LOGI(TAG,
131 "blk us min/mean/p99/max=%" PRIu32 "/%" PRIu32 "/<=%" PRIu32 "/%" PRIu32
132 " budget-peak=%" PRIu32 "%% spikes>=%d%%=%" PRIu32
133 " over=%" PRIu32 " n=%" PRIu32,
134 copy.min_cycles / mhz, mean / mhz, p99_cycles / mhz,
135 copy.worst_cycles / mhz,
136 (uint32_t)((uint64_t)copy.worst_cycles * 100u / BUDGET_CYCLES),
137 CONFIG_AMYSYNTH_RENDER_STATS_SPIKE_PCT, copy.spikes,
138 copy.over_budget, copy.count);
139 ESP_LOGI(TAG,
140 "worst: blk=%" PRIu32 " qsize=%u tick=%u",
141 copy.worst_block, copy.worst_qsize, copy.worst_tick);
142
143 // One histogram line, 6.25%-of-budget buckets, low to high.
144 char hist[HIST_BUCKETS * 11 + 1];
145 size_t off = 0;
146 for (uint32_t i = 0; i < HIST_BUCKETS; i++) {
147 off += (size_t)snprintf(hist + off, sizeof(hist) - off, "%s%" PRIu32,
148 i ? " " : "", copy.hist[i]);
149 if (off >= sizeof(hist) - 1) break;
150 }
151 ESP_LOGI(TAG, "hist(6.25%%/bucket)=[%s]", hist);
152
153 s_reset_req++; // writer zeroes the window at its next block boundary
154}
