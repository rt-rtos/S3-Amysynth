20void rtos_stats_report(void)
21{
22 UBaseType_t num_tasks = uxTaskGetNumberOfTasks();
23 TaskStatus_t *tasks = malloc(num_tasks * sizeof(TaskStatus_t));
24 if (tasks == NULL) {
25 ESP_LOGW(TAG, "rtos stats: malloc(%u tasks) failed", (unsigned)num_tasks);
26 return;
27 }
28
29 uint32_t total_runtime = 0;
30 // On-demand profiling dump (AMYSYNTH_RTOS_STATS, default n). The IRQ-mask
31 // window is acceptable ONLY because this never runs in release builds and
32 // profiling sessions accept stream artifacts; never wire this into a
33 // default-on periodic path: the TCB walk masks IRQs for milliseconds and
34 // beats against the USB frame clock.
35 // ast-grep-ignore
36 num_tasks = uxTaskGetSystemState(tasks, num_tasks, &total_runtime);
37 if (num_tasks == 0 || total_runtime == 0) {
38 ESP_LOGW(TAG, "rtos stats: snapshot unavailable");
39 free(tasks);
40 return;
41 }
42
43 // Per-task table; also collect per-core IDLE counters for the busy % below.
44 uint64_t idle_now[portNUM_PROCESSORS] = {0};
45 ESP_LOGI(TAG, "RTOS tasks: name core prio stack_hwm cpu%%(cumulative)");
46 for (UBaseType_t i = 0; i < num_tasks; i++) {
47 const TaskStatus_t *t = &tasks[i];
48 uint32_t cpu_pct = (uint32_t)(((uint64_t)t->ulRunTimeCounter * 100ULL) / total_runtime);
49 int core = (int)t->xCoreID; // tskNO_AFFINITY shows as a large value
50 ESP_LOGI(TAG, " %-16s %4d %4u %9u %3u%%",
51 t->pcTaskName,
52 core,
53 (unsigned)t->uxCurrentPriority,
54 (unsigned)t->usStackHighWaterMark,
55 (unsigned)cpu_pct);
56 // The IDLE tasks are named "IDLE0"/"IDLE1" on the IDF kernel.
57 if (strncmp(t->pcTaskName, "IDLE", 4) == 0) {
58 int c = t->pcTaskName[4] - '0';
59 if (c >= 0 && c < portNUM_PROCESSORS) {
60 idle_now[c] = (uint64_t)t->ulRunTimeCounter;
61 }
62 }
63 }
64
65 // Busy % = IDLE-counter delta vs esp_timer wall time (same us time base
66 // with RUN_TIME_STATS_USING_ESP_TIMER=y).
67 static uint64_t s_prev_wall_us = 0;
68 static uint64_t s_prev_idle_us[portNUM_PROCESSORS] = {0};
69 uint64_t now_us = (uint64_t)esp_timer_get_time();
70 uint64_t wall_delta = now_us - s_prev_wall_us;
71 for (int core = 0; core < portNUM_PROCESSORS; core++) {
72 uint64_t idle_delta = idle_now[core] - s_prev_idle_us[core];
73 if (s_prev_wall_us != 0 && wall_delta > 0) {
74 uint32_t idle_pct = (uint32_t)((idle_delta * 100ULL) / wall_delta);
75 if (idle_pct > 100) idle_pct = 100; // clamp sampling skew
76 ESP_LOGI(TAG, "core %d load: busy=%u%% idle=%u%% (interval %u ms)",
77 core, (unsigned)(100 - idle_pct), (unsigned)idle_pct,
78 (unsigned)(wall_delta / 1000ULL));
79 } else {
80 ESP_LOGI(TAG, "core %d load: (baseline captured)", core);
81 }
82 s_prev_idle_us[core] = idle_now[core];
83 }
84 s_prev_wall_us = now_us;
85
86 free(tasks);
87
88 // Heap snapshot: internal vs PSRAM free + largest free block.
89 ESP_LOGI(TAG,
90 "heap: free=%u min_free=%u | internal free=%u largest=%u | psram free=%u largest=%u",
91 (unsigned)esp_get_free_heap_size(),
92 (unsigned)esp_get_minimum_free_heap_size(),
93 (unsigned)heap_caps_get_free_size(MALLOC_CAP_INTERNAL),
94 (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_INTERNAL),
95 (unsigned)heap_caps_get_free_size(MALLOC_CAP_SPIRAM),
96 (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_SPIRAM));
97}
