155// template type for doing sample conversion
156template<typename ToFmt, typename FromFmt>
157struct converting_copy {
158 static void copy(typename ToFmt::sample_t *dest, const typename FromFmt::sample_t *src, uint sample_count);
159};

161// Efficient copies of same sample type
162
163template<class Fmt, uint ChannelCount>
164struct converting_copy<MultiChannelFmt<Fmt, ChannelCount>, MultiChannelFmt<Fmt, ChannelCount>> {
165 static void copy(typename MultiChannelFmt<Fmt, ChannelCount>::sample_t *dest,
166 const typename MultiChannelFmt<Fmt, ChannelCount>::sample_t *src,
167 uint sample_count) {
168 memcpy((void *) dest, (const void *) src, sample_count * MultiChannelFmt<Fmt, ChannelCount>::frame_stride);
169 }
170};
171
172// N channel to N channel
173template<typename ToFmt, typename FromFmt, uint NumChannels>
174struct converting_copy<MultiChannelFmt<ToFmt, NumChannels>, MultiChannelFmt<FromFmt, NumChannels>> {
175 static void copy(typename ToFmt::sample_t *dest, const typename FromFmt::sample_t *src, uint sample_count) {
176 for (uint i = 0; i < sample_count * NumChannels; i++) {
177 *dest++ = sample_converter<ToFmt, FromFmt>::convert_sample(*src++);
178 }
179 }
180};

183// mono->stereo conversion
184template<typename ToFmt, typename FromFmt>
185struct converting_copy<Stereo<ToFmt>, Mono<FromFmt>> {
186 static void copy(typename ToFmt::sample_t *dest, const typename FromFmt::sample_t *src, uint sample_count) {
187 for (; sample_count; sample_count--) {
188 typename ToFmt::sample_t mono_sample = sample_converter<ToFmt, FromFmt>::convert_sample(*src++);
189 *dest++ = mono_sample;
190 *dest++ = mono_sample;
191 }
192 }
193};
194
195// stereo->mono conversion
196template<typename ToFmt, typename FromFmt>
197struct converting_copy<Mono<ToFmt>, Stereo<FromFmt>> {
198 static void copy(typename ToFmt::sample_t *dest, const typename FromFmt::sample_t *src, uint sample_count) {
199 for (; sample_count; sample_count--) {
200 // average first in case precision is better in source
201 typename FromFmt::sample_t averaged_sample = (src[0] + src[1]) / 2;
202 src += 2;
203 *dest++ = sample_converter<ToFmt, FromFmt>::convert_sample(averaged_sample);
204 }
205 }
206};

208template<typename ToFmt, typename FromFmt>
209audio_buffer_t *consumer_pool_take(audio_connection_t *connection, bool block) {
210 struct buffer_copying_on_consumer_take_connection *cc = (struct buffer_copying_on_consumer_take_connection *) connection;
211 // for now we block until we have all the data in consumer buffers
212 audio_buffer_t *buffer = get_free_audio_buffer(cc->core.consumer_pool, block);
213 if (!buffer) return NULL;
214 assert(buffer->format->sample_stride == ToFmt::frame_stride);
215
216 uint32_t pos = 0;
217 while (pos < buffer->max_sample_count) {
218 if (!cc->current_producer_buffer) {
219 cc->current_producer_buffer = get_full_audio_buffer(cc->core.producer_pool, block);
220 if (!cc->current_producer_buffer) {
221 assert(!block);
222 if (!pos) {
223 queue_free_audio_buffer(cc->core.consumer_pool, buffer);
224 return NULL;
225 }
226 break;
227 }
228 assert(cc->current_producer_buffer->format->format->channel_count == FromFmt::channel_count);
229 assert(cc->current_producer_buffer->format->sample_stride == FromFmt::frame_stride);
230 cc->current_producer_buffer_pos = 0;
231 }
232 uint sample_count = std::min(buffer->max_sample_count - pos,
233 cc->current_producer_buffer->sample_count - cc->current_producer_buffer_pos);
234 converting_copy<ToFmt, FromFmt>::copy(
235 ((typename ToFmt::sample_t *) buffer->buffer->bytes) + pos * ToFmt::channel_count,
236 ((typename FromFmt::sample_t *) cc->current_producer_buffer->buffer->bytes) +
237 cc->current_producer_buffer_pos * FromFmt::channel_count,
238 sample_count);
239 pos += sample_count;
240 cc->current_producer_buffer_pos += sample_count;
241 if (cc->current_producer_buffer_pos == cc->current_producer_buffer->sample_count) {
242 queue_free_audio_buffer(cc->core.producer_pool, cc->current_producer_buffer);
243 cc->current_producer_buffer = NULL;
244 }
245 }
246 buffer->sample_count = pos;
247 return buffer;
248}

250template<typename ToFmt, typename FromFmt>
251void producer_pool_blocking_give(audio_connection_t *connection, audio_buffer_t *buffer) {
252 struct producer_pool_blocking_give_connection *pbc = (struct producer_pool_blocking_give_connection *) connection;
253 // for now we block until we have all the data in consumer buffers
254 uint32_t pos = 0;
255 while (pos < buffer->sample_count) {
256 if (!pbc->current_consumer_buffer) {
257 pbc->current_consumer_buffer = get_free_audio_buffer(pbc->core.consumer_pool, true);
258 pbc->current_consumer_buffer_pos = 0;
259 }
260 uint sample_count = std::min(buffer->sample_count - pos,
261 pbc->current_consumer_buffer->max_sample_count - pbc->current_consumer_buffer_pos);
262 assert(buffer->format->sample_stride == FromFmt::frame_stride);
263 assert(buffer->format->format->channel_count == FromFmt::channel_count);
264 converting_copy<ToFmt, FromFmt>::copy(
265 ((typename ToFmt::sample_t *) pbc->current_consumer_buffer->buffer->bytes) +
266 pbc->current_consumer_buffer_pos * ToFmt::channel_count,
267 ((typename FromFmt::sample_t *) buffer->buffer->bytes) + pos * FromFmt::channel_count, sample_count);
268 pos += sample_count;
269 pbc->current_consumer_buffer_pos += sample_count;
270 if (pbc->current_consumer_buffer_pos == pbc->current_consumer_buffer->max_sample_count) {
271 pbc->current_consumer_buffer->sample_count = pbc->current_consumer_buffer->max_sample_count;
272 queue_full_audio_buffer(pbc->core.consumer_pool, pbc->current_consumer_buffer);
273 pbc->current_consumer_buffer = NULL;
274 }
275 }
276 // todo this should be a connection configuration (or a seaparate connection type)
277#ifdef BLOCKING_GIVE_SYNCHRONIZE_BUFFERS
278 if (pbc->current_consumer_buffer) {
279 pbc->current_consumer_buffer->sample_count = pbc->current_consumer_buffer_pos;
280 queue_full_audio_buffer(pbc->core.consumer_pool, pbc->current_consumer_buffer);
281 pbc->current_consumer_buffer = NULL;
282 }
283#endif
284 assert(pos == buffer->sample_count);
285 queue_free_audio_buffer(pbc->core.producer_pool, buffer);
286}
