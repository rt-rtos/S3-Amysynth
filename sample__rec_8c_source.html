35bool sample_rec_arm(uint8_t layer_idx, uint8_t track)
36{
37 if (atomic_load_explicit(&s_state, memory_order_acquire) != SAMPLE_REC_IDLE) {
38 return false;
39 }
40 if (track >= SEQ_TRACKS) {

44 /* pcm_load() mutates pcm.c's unlocked memory-preset linked list, which
45 * render_pcm() walks from inside the render body. Take the same lock
46 * add_delta_to_queue() uses so the insert can't race a concurrent
47 * render block (see amy.h's LOCAL EDIT note on amy_grab_lock). */
48 amy_grab_lock();
49 int16_t *buf = pcm_load(SAMPLE_REC_PRESET_NUMBER, SAMPLE_REC_LENGTH,
50 AMY_SAMPLE_RATE, 1, SAMPLE_REC_MIDINOTE, 0, 0);
51 if (buf != NULL) {
52 memset(buf, 0, SAMPLE_REC_LENGTH * sizeof(int16_t));
53 }
54 amy_release_lock();
55
56 if (buf == NULL) {
57 ESP_LOGW(TAG, "arm failed: no RAM for %lu-sample slot",
58 (unsigned long)SAMPLE_REC_LENGTH);
59 return false;
60 }
61
62 s_buf = buf;
63 s_target_layer = layer_idx;
64 s_target_track = track;
65 atomic_store_explicit(&s_write_index, 0, memory_order_relaxed);
66 atomic_store_explicit(&s_cancel_requested, false, memory_order_relaxed);
67 atomic_store_explicit(&s_state, SAMPLE_REC_ARMED, memory_order_release);
68 ESP_LOGI(TAG, "armed: L%u T%u, %lu samples", layer_idx + 1u, track + 1u,
69 (unsigned long)SAMPLE_REC_LENGTH);
70 return true;
71}

101void sample_rec_cancel(void)
102{
103 if (atomic_load_explicit(&s_state, memory_order_acquire) == SAMPLE_REC_IDLE) {
104 return;
105 }
106 /* Deferred: the actual pcm_unload_preset() + state reset happens on the
107 * render task's own thread (sample_rec_render_tick), never here. Freeing
108 * the buffer from this task while the render task might be mid-write
109 * (RECORDING) would be a cross-core use-after-free; letting the render
110 * task tear down its own capture at a block boundary makes it single-
111 * writer instead. */
112 atomic_store_explicit(&s_cancel_requested, true, memory_order_release);
113}

129void sample_rec_render_tick(const int16_t *interleaved_stereo, uint16_t frames)
130{
131 sample_rec_state_t st = atomic_load_explicit(&s_state, memory_order_acquire);
132 if (st == SAMPLE_REC_IDLE) {
133 return;
134 }
135
136 if (atomic_load_explicit(&s_cancel_requested, memory_order_acquire)) {
137 amy_grab_lock();
138 pcm_unload_preset(SAMPLE_REC_PRESET_NUMBER);
139 amy_release_lock();
140 s_buf = NULL;
141 atomic_store_explicit(&s_write_index, 0, memory_order_relaxed);
142 atomic_store_explicit(&s_cancel_requested, false, memory_order_relaxed);
143 atomic_store_explicit(&s_state, SAMPLE_REC_IDLE, memory_order_release);
144 ESP_LOGI(TAG, "cancelled");
145 return;
146 }
147
148 if (st != SAMPLE_REC_RECORDING) {
149 return; /* ARMED or READY: nothing to capture this block */
150 }
151
152 uint32_t idx = atomic_load_explicit(&s_write_index, memory_order_relaxed);
153 uint32_t remaining = SAMPLE_REC_LENGTH - idx;
154 uint32_t n = ((uint32_t)frames < remaining) ? (uint32_t)frames : remaining;
155
156 int16_t *dst = s_buf + idx;
157 for (uint32_t i = 0; i < n; i++) {
158 int32_t l = interleaved_stereo[i * 2];
159 int32_t r = interleaved_stereo[i * 2 + 1];
160 dst[i] = (int16_t)((l + r) / 2);
161 }
162
163 idx += n;
164 atomic_store_explicit(&s_write_index, idx, memory_order_release);
165
166 if (idx >= SAMPLE_REC_LENGTH) {
167 atomic_store_explicit(&s_state, SAMPLE_REC_READY, memory_order_release);
168 ESP_LOGI(TAG, "capture complete (%lu samples)", (unsigned long)SAMPLE_REC_LENGTH);
169 }
170}
