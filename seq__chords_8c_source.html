19/* Normalize a raw voicing: keep non-zero tones clamped to the melodic range,
20 * sort ascending, zero-fill the tail; count derives from what survived. */
21static seq_chord_t chord_normalize(const seq_chord_t *chord)
22{
23 seq_chord_t c = { .notes = {0}, .count = 0 };
24 for (uint8_t i = 0; i < SEQ_CHORD_MAX_NOTES; i++) {
25 if (chord->notes[i] == 0) continue;
26 c.notes[c.count++] = sequencer_core_clamp_melodic_note(chord->notes[i]);
27 }
28 for (uint8_t i = 1; i < c.count; i++) { /* insertion sort, n<=4 */
29 uint8_t v = c.notes[i];
30 int8_t j = (int8_t)(i - 1);
31 while (j >= 0 && c.notes[j] > v) { c.notes[j + 1] = c.notes[j]; j--; }
32 c.notes[j + 1] = v;
33 }
34 return c;
35}

92uint8_t seq_chords_resolve(uint8_t idx, int transpose,
93 uint8_t out[SEQ_CHORD_MAX_NOTES])
94{
95 if (idx >= SEQ_CHORD_SLOTS) return 0;

98 /* Clamp, never drop: extreme progressions compress the voicing at the
99 * range edges instead of thinning it. */
100 out[i] = sequencer_core_clamp_melodic_note((int32_t)c->notes[i] + transpose);

105uint8_t seq_chords_selector_step(uint8_t cur, int dir)
106{
107 int step = (dir >= 0) ? 1 : -1;
108
109 if (SEQ_NOTE_IS_CHORD(cur)) {
110 /* Walk the chord zone, skipping undefined slots; off either end we
111 * re-enter the plain range at the matching edge. */
112 int i = (int)SEQ_CHORD_INDEX(cur);
113 for (i += step; i >= 0 && i < SEQ_CHORD_SLOTS; i += step) {
114 if (seq_chords_is_defined((uint8_t)i)) return SEQ_CHORD_NOTE(i);
115 }
116 return (step > 0) ? SEQ_MEL_NOTE_MIN : SEQ_MEL_NOTE_MAX;
117 }
118
119 /* Plain range. Normalize a stray out-of-range value first. */
120 if (cur < SEQ_MEL_NOTE_MIN) cur = SEQ_MEL_NOTE_MIN;
121 if (cur > SEQ_MEL_NOTE_MAX) cur = SEQ_MEL_NOTE_MAX;
122
123 int next = (int)cur + step;
124 if (next >= SEQ_MEL_NOTE_MIN && next <= SEQ_MEL_NOTE_MAX) return (uint8_t)next;
125
126 /* Off the top: first defined chord, else wrap to the bottom.
127 * Off the bottom: last defined chord, else wrap to the top. */
128 if (step > 0) {
129 for (uint8_t i = 0; i < SEQ_CHORD_SLOTS; i++) {
130 if (seq_chords_is_defined(i)) return SEQ_CHORD_NOTE(i);
131 }
132 return SEQ_MEL_NOTE_MIN;
133 }
134 for (int i = SEQ_CHORD_SLOTS - 1; i >= 0; i--) {
135 if (seq_chords_is_defined((uint8_t)i)) return SEQ_CHORD_NOTE(i);
136 }
137 return SEQ_MEL_NOTE_MAX;
138}
