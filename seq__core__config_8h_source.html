42/* ── Timing ──────────────────────────────────────────────────────────── */
43#define SEQ_TICKS_PER_STEP (AMY_SEQUENCER_PPQ / 4)
44/* Max swing: percent of one step by which odd 16ths are delayed. Mirrors
45 * DRONE_SWING_MAX and must stay < 100 so a swung note-on never crosses into the
46 * next step's slot. The Digitakt 50-80% scalar maps to 0-60 here. */
47#define SEQ_SWING_MAX 66
48/* A musical bar = 16 steps. Fixed regardless of layer length so the bar
49 * counter and repeat-rate are independent of which layers are active. */
50#define SEQ_TICKS_PER_BAR (16u * SEQ_TICKS_PER_STEP)
51/* Per-step micro-timing range in sequencer ticks, folded in only at the
52 * step->absolute-tick conversion in sequencer_emit_step(). +-6 is half a step -
53 * a strong but musical push/drag, safely below the smallest loop period. A
54 * chosen range, not a spec; the step_nudge setter clamps to +-this. */
55#define SEQ_STEP_NUDGE_MAX 6
56/* Tempo-synced LFO frequency ceilings. NATIVE keeps AMY-carrier LFOs
57 * sub-audible - fast rates at high BPM would otherwise cross 20 Hz into
58 * AM/sideband territory. SW also protects the 20 Hz software stepper: under ~4
59 * steps per LFO cycle the stepped waveform degenerates (a sine at exactly 2
60 * steps/cycle aliases to DC). Fast rates cap here rather than being hidden from
61 * the pickers, so the selectable range stays uniform across surfaces. */
62#define SEQ_LFO_NATIVE_MAX_HZ 20.0f
63#define SEQ_LFO_SW_MAX_HZ 5.0f
64/* Drum gate: fraction of a step the note is held before its note-off, i.e.
65 * choke vs ring; a patch's own release tail still plays out afterwards.
66 * Kconfig-tunable, default 1/2 step. NOTE: the default drum engine is PCM
67 * (gamma banks) rather than tonal patches. */
68#define SEQ_GATE_DRUM ((SEQ_TICKS_PER_STEP * CONFIG_SEQ_DRUM_GATE_NUMERATOR) / CONFIG_SEQ_DRUM_GATE_DENOMINATOR)
69/* Default per-layer melodic gate (the NoteFX GATE control) as a % of the step.
70 * Derived from the same num/den fraction as SEQ_GATE_MELODIC, and defined from
71 * the fraction rather than SEQ_TICKS_PER_STEP so it stays PPQ-independent and
72 * usable in TUs that do not include amy.h (project_snapshot.c). Control spans
73 * 10..100% (100% = legato). */
74#if CONFIG_SEQ_MELODIC_EXPRESSIVE_DEFAULTS
75#define SEQ_GATE_MELODIC ((SEQ_TICKS_PER_STEP * CONFIG_SEQ_MELODIC_GATE_NUMERATOR) / CONFIG_SEQ_MELODIC_GATE_DENOMINATOR)

136/* ── Arpeggiator tag space ───────────────────────────────────────────────
137 * Arp tags sit just above the sequencer's tag space:
138 * step on/off : 0 .. MAX_LAYERS*SEQ_TRACKS*SEQ_MAX_STEPS*2 - 1 (0..1023)
139 * previews : 1024 .. 1024 + MAX_LAYERS*SEQ_TRACKS*2 - 1 (..1055)
140 * so the arp starts at 1056 and needs ARP_MAX_SLOTS*ARP_OCT_MAX*2 = 64 tags.
141 *
142 * AMY's sequencer_add_wire rejects `tag >= max_sequences`, so sequences[]
143 * needs at least (highest_tag + 1) user-addressable entries. Keep main.c's
144 * amy_cfg.max_sequencer_tags in sync. */
145#define SEQ_ARP_TAG_BASE 1056u
146#define SEQ_ARP_TAG_COUNT (ARP_MAX_SLOTS * ARP_OCT_MAX * 2) /* = 64 */
147#define SEQ_ARP_TAG_MAX (SEQ_ARP_TAG_BASE + SEQ_ARP_TAG_COUNT - 1) /* 1119 */
148
149/* ── Ratchet tag space ────────────────────────────────────────────────────
150 * A decorated step never uses the plain per-step ON/OFF pair: instead
151 * sequencer_core_service_tick() one-shot schedules up to SEQ_MAX_RATCHET
152 * note-on/off pairs per firing, each on its own statically assigned tag so
153 * sub-hits never overwrite each other. Sits just above the arp's tag space.
154 * Same off-by-one rule: main.c's amy_cfg.max_sequencer_tags must stay
155 * >= SEQ_RATCHET_TAG_MAX + 2. */
156#define SEQ_RATCHET_TAG_BASE (SEQ_ARP_TAG_MAX + 1u) /* 1120 */
157#define SEQ_RATCHET_TAG_COUNT (MAX_LAYERS * SEQ_TRACKS * SEQ_MAX_RATCHET * 2) /* 128 */
158#define SEQ_RATCHET_TAG_MAX (SEQ_RATCHET_TAG_BASE + SEQ_RATCHET_TAG_COUNT - 1) /* 1247 */
159
160/* ── Chord tag space ──────────────────────────────────────────────────────
161 * Chord steps always take the decorated one-shot path (a chord sentinel in
162 * step_note makes the step decorated, see seq_core_trig.c), so chord tones need
163 * one-shot tags only, never a periodic per-step block. Tone 0 reuses the
164 * ratchet pair; higher tones get a statically assigned pair per (layer, track,
165 * ratchet-slot, tone), plus a second small block for the edit-preview path.
166 * Each tag costs ~20 B of internal DRAM in AMY, which is why the one-shot
167 * scheme (640 tags) beat a periodic per-step chord block (~3.5 K tags). Same
168 * off-by-one rule: main.c must stay >= SEQ_CHORD_PREVIEW_TAG_MAX + 2 - it
169 * derives max_sequencer_tags from that expression rather than a literal,
170 * because this ceiling moves with SEQ_CHORD_MAX_NOTES. */
171#define SEQ_CHORD_TAG_BASE (SEQ_RATCHET_TAG_MAX + 1u) /* 1248 */
172#define SEQ_CHORD_TAG_COUNT (MAX_LAYERS * SEQ_TRACKS * SEQ_MAX_RATCHET \
173 * (SEQ_CHORD_MAX_NOTES - 1) * 2) /* 512 */
174#define SEQ_CHORD_TAG_MAX (SEQ_CHORD_TAG_BASE + SEQ_CHORD_TAG_COUNT - 1) /* 1759 */
175#define SEQ_CHORD_PREVIEW_TAG_BASE (SEQ_CHORD_TAG_MAX + 1u) /* 1760 */
176#define SEQ_CHORD_PREVIEW_TAG_COUNT (MAX_LAYERS * SEQ_TRACKS \
177 * (SEQ_CHORD_MAX_NOTES - 1) * 2) /* 128 */
178#define SEQ_CHORD_PREVIEW_TAG_MAX (SEQ_CHORD_PREVIEW_TAG_BASE + \
179 SEQ_CHORD_PREVIEW_TAG_COUNT - 1) /* 1887 */
