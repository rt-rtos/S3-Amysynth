38/* ── Timing ──────────────────────────────────────────────────────────── */
39#define SEQ_TICKS_PER_STEP (AMY_SEQUENCER_PPQ / 4)
40/* Max swing (shuffle): percent of one step by which odd 16ths are delayed.
41 * Mirrors the drone's DRONE_SWING_MAX. Stays < 100 so a swung note-on never
42 * crosses into the next step's slot. Musician mapping: the Digitakt 50-80%
43 * scalar corresponds to 0-60 here (50% = straight = 0). 66 is the ceiling. */
44#define SEQ_SWING_MAX 66
45/* A musical bar = 16 steps. Fixed regardless of layer length so the bar
46 * counter and repeat-rate are independent of which layers are active. */
47#define SEQ_TICKS_PER_BAR (16u * SEQ_TICKS_PER_STEP)
48/* Per-step micro-timing (swing/humanize) range in sequencer ticks, folded in
49 * only at step->absolute-tick conversion in sequencer_emit_step(). +-6 ticks
50 * is half a step (SEQ_TICKS_PER_STEP=12) — a strong but musical push/drag, and
51 * safely below the smallest loop period (16*12=192). Chosen range, NOT a
52 * hardware-confirmed spec; the future step_nudge setter clamps to +-this. */
53#define SEQ_STEP_NUDGE_MAX 6
54/* Drum gate: fraction of a step the note is held before its note-off. Now that
55 * drums are real Juno patches (note-offs honored), this controls choke vs. ring;
56 * the patch's own release tail still plays out after note-off. Tunable via
57 * Kconfig (default 1/2 step).
58 * Drums are no longer Synths with the addition of Gamma banks
59 */
60#define SEQ_GATE_DRUM ((SEQ_TICKS_PER_STEP * CONFIG_SEQ_DRUM_GATE_NUMERATOR) / CONFIG_SEQ_DRUM_GATE_DENOMINATOR)
61/* Default per-layer melodic gate (the NoteFX GATE control), as a % of the step.
62 * Derived from the same fraction as the legacy fixed SEQ_GATE_MELODIC (num/den)
63 * so a fresh layer holds notes exactly as before — seq_step_gate() rounds
64 * pct->ticks, making this round-trip lossless. Defined from the fraction, NOT
65 * SEQ_TICKS_PER_STEP, so it stays PPQ-independent and usable in TUs that don't
66 * include amy.h (e.g. project_snapshot.c). Control spans 10..100% (100% = legato).*/
67#if CONFIG_SEQ_MELODIC_EXPRESSIVE_DEFAULTS
68#define SEQ_GATE_MELODIC ((SEQ_TICKS_PER_STEP * CONFIG_SEQ_MELODIC_GATE_NUMERATOR) / CONFIG_SEQ_MELODIC_GATE_DENOMINATOR)
69#define SEQ_MELODIC_GATE_DEFAULT_PCT \
70 ((100u * CONFIG_SEQ_MELODIC_GATE_NUMERATOR + CONFIG_SEQ_MELODIC_GATE_DENOMINATOR / 2u) \
71 / CONFIG_SEQ_MELODIC_GATE_DENOMINATOR)
72#else
73#define SEQ_GATE_MELODIC ((SEQ_TICKS_PER_STEP * 2) / 3)
74#define SEQ_MELODIC_GATE_DEFAULT_PCT 67u /* round(100 * 2/3) */

126/* ── Arpeggiator tag space ───────────────────────────────────────────────
127 * Arp tags sit just above the sequencer's tag space. The sequencer uses:
128 * step on/off : 0 .. MAX_LAYERS*SEQ_TRACKS*SEQ_MAX_STEPS*2 - 1 (0..1023)
129 * previews : 1024 .. 1024 + MAX_LAYERS*SEQ_TRACKS*2 - 1 (..1055)
130 * so the highest sequencer tag is 1055. We base the arp at 1056 and it needs
131 * ARP_MAX_SLOTS*ARP_OCT_MAX*2 = 64 tags (1056..1119).
132 *
133 * IMPORTANT: AMY's sequencer_add_event guards with `tag > max_sequences`
134 * (NOT >=), so `sequences[]` must have at least (highest_tag + 2) entries to
135 * stay clear of that off-by-one. main.c sets amy_cfg.max_sequencer_tags
136 * accordingly — keep these in sync. */
137#define SEQ_ARP_TAG_BASE 1056u
138#define SEQ_ARP_TAG_COUNT (ARP_MAX_SLOTS * ARP_OCT_MAX * 2) /* = 64 */
139#define SEQ_ARP_TAG_MAX (SEQ_ARP_TAG_BASE + SEQ_ARP_TAG_COUNT - 1) /* 1119 */
140
141/* ── Ratchet tag space ────────────────────────────────────────────────────
142 * A "decorated" step (probability<100, ratchet>1, or a conditional trig) is
143 * never scheduled on the plain per-step ON/OFF tag pair — instead
144 * sequencer_core_service_tick() one-shot schedules up to SEQ_MAX_RATCHET
145 * note-on/off pairs per firing, each on its own dedicated, statically
146 * assigned tag so ratchet sub-hits never overwrite each other's schedule.
147 * Sits just above the arp's tag space; needs
148 * MAX_LAYERS*SEQ_TRACKS*SEQ_MAX_RATCHET*2 tags (4*4*4*2 = 128, 1120..1247).
149 * Same off-by-one rule as the arp block above applies — main.c's
150 * amy_cfg.max_sequencer_tags must stay >= SEQ_RATCHET_TAG_MAX + 2. */
151#define SEQ_RATCHET_TAG_BASE (SEQ_ARP_TAG_MAX + 1u) /* 1120 */
152#define SEQ_RATCHET_TAG_COUNT (MAX_LAYERS * SEQ_TRACKS * SEQ_MAX_RATCHET * 2) /* 128 */
153#define SEQ_RATCHET_TAG_MAX (SEQ_RATCHET_TAG_BASE + SEQ_RATCHET_TAG_COUNT - 1) /* 1247 */
