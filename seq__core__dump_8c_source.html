1/* ── Sequencer state dump (DEV menu one-shot) ──
2 * Prints every layer's track config and step exceptions to the console so
3 * runtime-tuned values (drum pitches/presets above all) can be read back in
4 * one block instead of scraped from per-edit log lines. Deliberately scoped
5 * to sequencer layer/track/step state - global FX, arp and drone have their
6 * own screens and are not this tool's job.
7 *
8 * Output discipline: plain printf (no ESP_LOG prefixes to strip when copying
9 * from the monitor), BEGIN/END markers carrying the firmware version and a
10 * printed-line count so a truncated copy is detectable. Track lines lead
11 * with note= and the drum preset so the tuning harvest reads column-wise;
12 * step state prints lazily - a pattern string per track, plus one line per
13 * step only where something differs from the plain-step neutral values
14 * (prob=100, ratchet=1, every<=1, note==track base, all else 0). Voice
15 * params print lazily too: eg0/eg1/flt lines only when the row's authored
16 * flag is set (unauthored = the patch owns it, nothing to harvest), trim
17 * only when off unity. */
18

44void sequencer_core_dump_state(void)
45{
46 unsigned n_lines = 0;
47
48 DP("==== SEQDUMP BEGIN fw=%s layers=%u engine=%s ====",
49 esp_app_get_description()->version, (unsigned)s_num_layers,
50 sequencer_core_get_drum_engine() == SEQ_DRUM_PCM ? "PCM" : "SYNTH");
51
52 for (uint8_t li = 0; li < s_num_layers; li++) {
53 const seq_layer_t *L = &s_layers[li];
54 bool drum = (L->type == SEQ_LAYER_DRUM);
55
56 if (drum) {
57 DP("L%u DRUM steps=%u swing=%u groove=%u",
58 li + 1u, L->num_steps, L->swing_pct, L->groove_pct);
59 } else {
60 DP("L%u MELODIC steps=%u patch=%u voices=%u swing=%u gate=%u"
61 " porta=%u groove=%u chord=%s root=%u type=%u",
62 li + 1u, L->num_steps, L->patch, L->num_voices, L->swing_pct,
63 L->gate_pct, L->portamento_ms, L->groove_pct,
64 L->chord_mode ? "on" : "off", L->chord_root,
65 (unsigned)L->chord_type);
66 }
67
68 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
69 /* Track line: pitch and timbre first (the harvest columns). */
70 if (drum) {
71 /* PCM preset/mode via the core getters: the seq_layer_t
72 * fields of the same name are UI mirrors refreshed on the
73 * UI's copy only - on this side they hold stale zeros until
74 * a project load writes them. */
75 uint16_t pcm = sequencer_core_get_drum_pcm_preset(li, t);
76 uint8_t pcm_mode = sequencer_core_get_drum_pcm_mode(li, t);
77 DP(" T%u note=%u patch=%u pcm=%u mode=%u(%s) rep=%u"
78 " mute=%u solo=%u",
79 t + 1u, L->track_base_note[t], L->track_patch[t],
80 pcm, pcm_mode, pcm_mode_name(pcm_mode),
81 L->repeat_rate[t],
82 (unsigned)L->mute[t], (unsigned)L->solo[t]);
83 } else {
84 DP(" T%u note=%u rep=%u mute=%u solo=%u",
85 t + 1u, L->track_base_note[t], L->repeat_rate[t],
86 (unsigned)L->mute[t], (unsigned)L->solo[t]);
87 }
88
89 /* The row's OWN block, whichever one it currently reads; the
90 * shared layer block follows the track lines. */
91 const voice_params_t *vp = &L->vp[t];
92 if (L->vp_src[t] == SEQ_VP_SRC_LAYER) {
93 DP(" T%u src=LAYER", t + 1u);
94 }
95 if (vp->env_authored) {
96 DP(" T%u eg0 a=%lu d=%lu s=%u r=%lu type=%u",
97 t + 1u, (unsigned long)vp->env.attack_ms,
98 (unsigned long)vp->env.decay_ms, vp->env.sustain_pct,
99 (unsigned long)vp->env.release_ms, vp->env.eg_type);
100 }
101 if (vp->env1_authored) {
102 DP(" T%u eg1 a=%lu d=%lu s=%u r=%lu type=%u",
103 t + 1u, (unsigned long)vp->env1.attack_ms,
104 (unsigned long)vp->env1.decay_ms, vp->env1.sustain_pct,
105 (unsigned long)vp->env1.release_ms, vp->env1.eg_type);
106 }
107 if (vp->filter_authored) {
108 DP(" T%u flt type=%u en=%u cut=%.0f res=%.2f envamt=%.2f fb=%.2f",
109 t + 1u, vp->filter.filter_type,
110 (unsigned)vp->filter.enabled, (double)vp->filter.cutoff_hz,
111 (double)vp->filter.resonance,
112 (double)vp->filter.filter_env_amount,
113 (double)vp->filter.feedback);
114 }
115 if (vp->amp_trim != 1.0f) {
116 DP(" T%u trim=%.2f", t + 1u, (double)vp->amp_trim);
117 }
118
119 char grid[SEQ_MAX_STEPS + 1];
120 for (uint8_t s = 0; s < L->num_steps; s++)
121 grid[s] = L->grid[t][s] ? 'X' : '.';
122 grid[L->num_steps] = '\0';
123 DP(" T%u |%s|", t + 1u, grid);
124
125 /* Step exceptions: only active steps, only non-neutral fields. */
126 for (uint8_t s = 0; s < L->num_steps; s++) {
127 if (!L->grid[t][s]) continue;
128 bool off_note = (L->step_note[t][s] != L->track_base_note[t]);
129 bool dec = off_note ||
130 L->step_pitch_ofs[t][s] ||
131 L->step_prob[t][s] != 100 ||
132 L->step_ratchet[t][s] != 1 ||
133 L->step_every[t][s] > 1 ||
134 L->step_prev[t][s] ||
135 L->step_transform[t][s] ||
136 L->step_quant_bypass[t][s] ||
137 L->step_nudge[t][s] ||
138 L->step_velocity_adj[t][s] ||
139 L->step_ratchet_taper[t][s];
140 if (!dec) continue;
141
142 char line[128];
143 int n = snprintf(line, sizeof line, " T%u s%u",
144 t + 1u, s + 1u);
145 #define AP(...) \
146 n += snprintf(line + n, sizeof line - (size_t)n, __VA_ARGS__)
147 if (off_note) AP(" note=%u", L->step_note[t][s]);
148 if (L->step_pitch_ofs[t][s]) AP(" pofs=%+d", L->step_pitch_ofs[t][s]);
149 if (L->step_prob[t][s] != 100) AP(" prob=%u", L->step_prob[t][s]);
150 if (L->step_ratchet[t][s] != 1) AP(" rat=%u", L->step_ratchet[t][s]);
151 if (L->step_ratchet_taper[t][s]) AP(" taper=%d", L->step_ratchet_taper[t][s]);
152 if (L->step_every[t][s] > 1) AP(" every=%u", L->step_every[t][s]);
153 if (L->step_prev[t][s]) AP(" prev");
154 if (L->step_transform[t][s]) AP(" xform=%s", xform_name(L->step_transform[t][s]));
155 if (L->step_quant_bypass[t][s]) AP(" qbyp");
156 if (L->step_nudge[t][s]) AP(" nudge=%d", L->step_nudge[t][s]);
157 if (L->step_velocity_adj[t][s]) AP(" vel=%d", L->step_velocity_adj[t][s]);
158 #undef AP
159 DP("%s", line);
160 }
161 }
162
163 /* The layer's shared voice block: which groups it has authority over
164 * (rows with src=LAYER read these; unauthored groups fall through to
165 * the patch). */
166 if (!drum) {
167 const voice_params_t *lv = &L->vp_layer;
168 DP(" LAYER-BLK eg0=%u eg1=%u flt=%u lfo=%u dist=%u",
169 (unsigned)lv->env_authored, (unsigned)lv->env1_authored,
170 (unsigned)lv->filter_authored, (unsigned)lv->lfo_authored,
171 (unsigned)lv->dist_authored);
172 }
173 }
174
175 DP("==== SEQDUMP END layers=%u lines=%u ====",
176 (unsigned)s_num_layers, n_lines + 1u);
177 fflush(stdout);
178}
