52/* Restore the resting coefficient for every target the LFO was driving, so
53 * nothing stays modulated after it is switched off. FILTER restores the
54 * track's authored cutoff; the rest push a neutral constant. */
55static void lfo_restore_target_neutrals(uint8_t layer_idx, uint8_t track,
56 const seq_lfo_t *lfo)
57{
58 const seq_layer_t *layer = &s_layers[layer_idx];
59 const voice_params_t *vp = seq_track_vp(layer_idx, track);
60 for (int t = 0; t < LFO_TARGET_COUNT; t++) {
61 if (!(lfo->targets & LFO_TGT_BIT(t))) continue;
62 if (t == LFO_TARGET_FILTER)
63 sequencer_core_push_filter(layer->synth_id[track], &vp->filter,
64 layer->patch == SEQ_PATCH_KS);
65 else if (t == LFO_TARGET_DIST_DRIVE || t == LFO_TARGET_DIST_MIX) {
66 /* Neutral = the committed dist block; only this caller has it,
67 * so lfo_push_target_neutral leaves DIST alone (like FILTER).
68 * The two dist bits share one restore event - re-pushing the same
69 * block for the second bit would just cost a redundant event. */
70 if (t == LFO_TARGET_DIST_DRIVE ||
71 !LFO_HAS_TGT(lfo, LFO_TARGET_DIST_DRIVE))
72 voice_apply_dist(layer->synth_id[track], &vp->dist);
73 }
74 else
75 lfo_push_target_neutral(layer->synth_id[track], (lfo_target_t)t);
76 }
77}

79/* ── Shared-block fan-out ────────────────────────────────────────────────────
80 * A row's voice block is either its own or the layer's (seq_track_vp). Writing
81 * a block must reach every row that reads it, so the committing setters push
82 * through here instead of to `track` alone: the peer set is {track} for an
83 * own-block row and every LAYER-source row otherwise. */
84uint8_t sequencer_core_melodic_vp_peers(uint8_t layer_idx, uint8_t track,
85 uint8_t peers[SEQ_TRACKS])
86{
87 if (!peers || layer_idx >= s_num_layers || track >= SEQ_TRACKS) return 0;
88 const seq_layer_t *layer = &s_layers[layer_idx];
89 if (layer->vp_src[track] != SEQ_VP_SRC_LAYER) {
90 peers[0] = track;
91 return 1;
92 }
93 uint8_t n = 0;
94 for (uint8_t t = 0; t < SEQ_TRACKS; ++t)
95 if (layer->vp_src[t] == SEQ_VP_SRC_LAYER) peers[n++] = t;
96 return n;
97}

241/* Apply one filter config to a row's AMY synth. Shared by the stored-state
242 * configure below and the live-preview push: the caller decides whether `f` is
243 * the row's committed filter or an editor's scratch copy. */
244static void melodic_filter_apply(uint8_t layer_idx, uint8_t track,
245 const seq_filter_t *f)
246{
247 const seq_layer_t *layer = &s_layers[layer_idx];
248 amy_event *e = amy_helpers_event_begin();
249 e->synth = layer->synth_id[track];
250 if (f->enabled) {
251 e->filter_type = f->filter_type;
252 e->filter_freq_coefs[COEF_CONST] = f->cutoff_hz;
253 e->resonance = f->resonance;
254 /* EG1 -> cutoff depth in octaves, wired only for a non-zero amount so a
255 * plain melodic filter is unchanged. Nothing routes COEF_EG1 to
256 * amplitude on melodic tracks, so there is no double-use to arbitrate
257 * and the amount==0 gate is the only guard needed (as in arp_core.c). */
258 if (f->filter_env_amount != 0.0f) {
259 e->filter_freq_coefs[COEF_EG1] = f->filter_env_amount;
260 }
261 } else {
262 e->filter_type = FILTER_NONE;
263 }
264 /* KS string decay: the authored feedback, pushed directly. 0 = never
265 * authored, so leave AMY's build-time 0.9 default in place. */
266 if (layer->patch == SEQ_PATCH_KS && f->feedback > 0.0f) {
267 e->feedback = SEQ_CLAMP_F32(f->feedback, 0.0f, 1.0f);
268 }
269 amy_helpers_event_send(e);
270
271 /* Guarantee valid EG1 breakpoints whenever the filter env is live, so
272 * filter_freq_coefs[COEF_EG1] modulates a real ramp: AMY treats a
273 * never-configured breakpoint set as a permanent 1.0. Uses the row's stored
274 * EG1 (authored shape or the zeroed default). */
275 if (f->enabled && f->filter_env_amount != 0.0f) {
276 sequencer_core_push_envelope_eg1(layer->synth_id[track], 0,
277 seq_layer_env1(layer_idx, track));
278 }
279}

388/* Active editor-preview slot; contract at the "Live-preview pushes" section
389 * below. WRITTEN by the editor handlers (encoder/button tasks), READ by
390 * sequencer_core_lfo_service() on synth_ui_task - same core, same priority
391 * tier, so a round-robin boundary can land mid-write. seq_core is lock-free
392 * by discipline, so consistency is a generation counter (seqlock): writers
393 * bump to odd, mutate, bump to even; the reader snapshots once per service
394 * tick and falls back to the store if it caught a writer. An aligned 32-bit
395 * counter is single-copy atomic on Xtensa. */
396typedef struct {
397 bool active;
398 uint8_t li, tr;
399 bool filter_valid;
400 seq_filter_t filter;
401 bool lfo_valid;
402 seq_lfo_t lfo;
403} melodic_preview_t;
404static melodic_preview_t s_preview;
405static volatile uint32_t s_preview_gen; /* odd = writer mid-update */
406
407/* Re-point the slot at (li, tr), dropping stale scratch from another track.
408 * Callers hold the write side (odd s_preview_gen). */
409static void preview_slot_touch(uint8_t li, uint8_t tr)
410{
411 if (!s_preview.active || s_preview.li != li || s_preview.tr != tr) {
412 s_preview = (melodic_preview_t){0};
413 s_preview.active = true;
414 s_preview.li = li;
415 s_preview.tr = tr;
416 }
417}

419/* Consistent snapshot for the service loop; returns false when no preview is
420 * active or a writer was mid-update (caller then reads the store as usual -
421 * one 50 ms tick of committed values, self-healing). */
422static bool preview_snapshot(melodic_preview_t *out)
423{
424 uint32_t g0 = s_preview_gen;
425 if (g0 & 1u) return false;
426 *out = s_preview;
427 if (s_preview_gen != g0) return false;
428 return out->active;
429}

431/* Push an EXPLICIT lfo struct to one track's engine state (native topology or
432 * software-service arming) without touching the store. Shared by the
433 * committing setter (which passes the just-stored struct), the live preview
434 * (editor scratch) and cancel-restore (stored struct). */
435static void melodic_lfo_apply_runtime(uint8_t layer_idx, uint8_t track,
436 const seq_lfo_t *lfo)
437{
438 const seq_layer_t *layer = &s_layers[layer_idx];
439#if CONFIG_SEQ_MELODIC_AMY_NATIVE_LFO
440 /* Native topology is melodic-only: it writes voice-relative oscs 1 and 2,
441 * and AMY applies the base_osc offset without a bounds check, so on a
442 * 1-osc-per-voice synth (a PCM drum slot) those events would land on the
443 * NEXT synth's oscillators. A drum layer's `patch` is only the stored
444 * SYNTH-mode selection, so it must never enable the native path. */
445 bool is_native = layer->type == SEQ_LAYER_MELODIC &&
446 sequencer_core_lfo_native_layout(layer->patch, NULL, NULL);
447 if (is_native) {
448 melodic_native_lfo_apply(layer, track, lfo);
449 /* Restore the static target value when disabled: native clears COEF_MOD
450 * but does not push the neutral coef. */
451 if (!lfo->enabled || !is_native_lfo_track(lfo)) {
452 lfo_restore_target_neutrals(layer_idx, track, lfo);
453 }
454 /* s_lfo_hz = 0 makes the software service loop skip native tracks.
455 * Distortion drive/mix now have COEF_MOD rails (dist_*_coefs), driven
456 * by the carrier in melodic_native_lfo_apply, so a native track no
457 * longer arms the stepper for the DIST bits either - every target
458 * rides the carrier. */
459 s_lfo_hz[layer_idx][track] = 0.0f;
460 return;
461 }
462#endif
463 /* Software path: patches without a carrier pair, or native LFO compiled out. */
464 if (!lfo->enabled) {
465 lfo_restore_target_neutrals(layer_idx, track, lfo);
466 s_lfo_hz[layer_idx][track] = 0.0f;
467 } else {
468 s_lfo_hz[layer_idx][track] = seq_lfo_sw_hz(lfo->rate, s_bpm);
469 }
470}

524void __attribute__((optimize("O3", "unroll-loops", "fast-math"))) sequencer_core_lfo_service(void)
525{
526 const float DT = 0.05f; /* 20 Hz */
527 /* One consistent preview snapshot per service tick: an open editor's
528 * scratch overrides the store for its one track, so relational edits are
529 * audible live instead of being stomped back to committed values. */
530 melodic_preview_t prev;
531 bool prev_ok = preview_snapshot(&prev);
532 for (int li = 0; li < s_num_layers; li++) {
533 for (int tr = 0; tr < SEQ_TRACKS; tr++) {
534 bool prev_here = prev_ok &&
535 prev.li == (uint8_t)li &&
536 prev.tr == (uint8_t)tr;
537 bool lfo_previewed = prev_here && prev.lfo_valid;
538 const voice_params_t *vp = seq_track_vp((uint8_t)li, (uint8_t)tr);
539 if (!lfo_previewed && !vp->lfo_authored) continue;
540 const seq_lfo_t *lfo = lfo_previewed ? &prev.lfo : &vp->lfo;
541 if (!lfo->enabled) continue;
542 float hz = s_lfo_hz[li][tr];
543 if (hz <= 0.0f) continue;
544
545 float ph = s_lfo_phase[li][tr] + hz * DT;
546 if (ph >= 1.0f) {
547 ph -= 1.0f;
548 if (lfo->wave == LFO_WAVE_RANDOM)
549 s_lfo_rnd[li][tr] = lfo_next_rand();
550 }
551 s_lfo_phase[li][tr] = ph;
552
553 float val;
554 switch (lfo->wave) {
555 case LFO_WAVE_SINE:
556 val = sinf(2.0f * 3.14159265f * ph); break;
557 case LFO_WAVE_TRIANGLE:
558 val = (ph < 0.5f) ? (4.0f*ph - 1.0f)
559 : (3.0f - 4.0f*ph); break;
560 case LFO_WAVE_SAW_UP: val = 2.0f*ph - 1.0f; break;
561 case LFO_WAVE_SAW_DOWN: val = 1.0f - 2.0f*ph; break;
562 case LFO_WAVE_SQUARE: val = (ph < 0.5f) ? 1.0f : -1.0f; break;
563 case LFO_WAVE_RANDOM: val = s_lfo_rnd[li][tr]; break;
564 default: val = 0.0f; break;
565 }
566
567 float d = (float)lfo->depth / 100.0f;
568 uint8_t syn = s_layers[li].synth_id[tr];
569
570#if CONFIG_SEQ_MELODIC_AMY_NATIVE_LFO
571 /* Native-carrier tracks disarm the stepper wholesale (s_lfo_hz = 0
572 * in melodic_lfo_apply_runtime), so in practice they never reach
573 * this body - every rail, distortion included, rides the carrier's
574 * COEF_MOD. This guard stays as a belt-and-braces: were a native
575 * track ever armed, stepping its rails here would double-modulate. */
576 bool native_track =
577 s_layers[li].type == SEQ_LAYER_MELODIC &&
578 sequencer_core_lfo_native_layout(s_layers[li].patch, NULL, NULL);
579#else
580 const bool native_track = false;
581#endif
582 if (!native_track) {
583 amy_event *e = amy_helpers_event_begin();
584 e->synth = syn;
585 /* Multi-target: each checked target modulates its own COEF_CONST
586 * from the same LFO value. SCAN has no software analog - it needs a
587 * wavetable voice, which always takes the native path. DIST is
588 * handled after this block: native on carrier patches, stepped here
589 * only on PATCH-mode tracks with no free carrier osc. */
590 if (LFO_HAS_TGT(lfo, LFO_TARGET_FILTER)) {
591 const seq_filter_t *fb = (prev_here && prev.filter_valid)
592 ? &prev.filter
593 : &vp->filter;
594 float base = (fb->enabled && fb->cutoff_hz > 0.0f)
595 ? fb->cutoff_hz : 1000.0f;
596 e->filter_freq_coefs[COEF_CONST] =
597 base * powf(2.0f, voice_lfo_filter_octaves(lfo) * val);
598 }
599 if (LFO_HAS_TGT(lfo, LFO_TARGET_AMP))
600 e->amp_coefs[COEF_CONST] = 1.0f - d*(0.5f - 0.5f*val);
601 if (LFO_HAS_TGT(lfo, LFO_TARGET_PAN))
602 e->pan_coefs[COEF_CONST] = 0.5f + d*0.5f*val;
603 amy_helpers_event_send(e);
604 } /* !native_track */
605
606 /* DIST target on a PATCH-mode track (no carrier for a native rail):
607 * step drive/mix around the committed dist block, same law as the
608 * native rail (voice_push_dist_lfo). Native tracks drive it via
609 * COEF_MOD instead, so they are excluded like every other target.
610 * Inert while the shaper is OFF. */
611 if (!native_track && (lfo->targets & LFO_TGT_DIST_MASK))
612 voice_push_dist_lfo(syn, &vp->dist, lfo, val);
613
614 if (!native_track && LFO_HAS_TGT(lfo, LFO_TARGET_PITCH)) {
615 /* freq COEF_CONST is an ABSOLUTE frequency in Hz - AMY maps it
616 * through logfreq_of_freq(x) = log2(x/440). Anchoring the swing
617 * at SEQ_LFO_PITCH_BASE_HZ makes the constant term exactly
618 * d*val octaves, matching the note-neutral reset default of 0.
619 * (A bare ratio here lands ~-8.8 octaves down and mutes the
620 * track - sub-audible playback rate on PCM oscs.)
621 *
622 * Pitch is pushed to osc 0 ONLY: a synth-wide event fans out to
623 * every osc of the voice, rewriting patch-internal modulator
624 * oscs' freq CONST - their RATE - and wrecking patch LFOs
625 * (chorus/PWM) beyond repair short of a patch reload. Stopgap:
626 * multi-carrier patches get vibrato on their first osc only;
627 * revisit with a per-voice offset or a reserved-carrier topo. */
628 amy_event *pe = amy_helpers_event_begin();
629 pe->synth = syn;
630 pe->osc = 0;
631 pe->freq_coefs[COEF_CONST] =
632 SEQ_LFO_PITCH_BASE_HZ * powf(2.0f, d * VOICE_LFO_DEPTH_PITCH * val);
633 amy_helpers_event_send(pe);
634 }
635 }
636 }
637}
