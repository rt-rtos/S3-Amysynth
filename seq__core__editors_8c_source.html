37/* Restore the resting (neutral) coefficient for every target the LFO was
38 * driving — called when the LFO is switched off so nothing stays modulated.
39 * FILTER restores the track's own authored cutoff; the rest push a neutral
40 * constant. Iterates the full target set (multi-target LFOs). */
41static void lfo_restore_target_neutrals(const seq_layer_t *layer, uint8_t track,
42 const seq_lfo_t *lfo)
43{
44 for (int t = 0; t < LFO_TARGET_COUNT; t++) {
45 if (!(lfo->targets & LFO_TGT_BIT(t))) continue;
46 if (t == LFO_TARGET_FILTER)
47 sequencer_core_push_filter(layer->synth_id[track], &layer->vp[track].filter,
48 layer->patch == SEQ_PATCH_KS);
49 else
50 lfo_push_target_neutral(layer->synth_id[track], (lfo_target_t)t);
51 }
52}

167/* Map a Q value (same [0.51, 8.0] range enforced by sequencer_core_set_melodic_filter)
168 * linearly onto AMY's KS oscillator feedback range [0.0, 1.0]. Q=8.0 -> feedback=1.0
169 * is the verified-safe ceiling (lossless two-tap averaging filter, the classic
170 * "infinite sustain" Karplus-Strong case); above 1.0 the KS buffer would diverge. */
171float sequencer_core_ks_feedback_from_q(float q)
172{
173 float n = (q - 0.51f) / (8.0f - 0.51f);
174 return SEQ_CLAMP_F32(n, 0.0f, 1.0f);
175}

178/* Apply one filter config to a row's AMY synth. Shared by the stored-state
179 * configure below and the live-preview push: the caller decides whether `f`
180 * is the row's committed filter or an editor's scratch copy. */
181static void melodic_filter_apply(uint8_t layer_idx, uint8_t track,
182 const seq_filter_t *f)
183{
184 const seq_layer_t *layer = &s_layers[layer_idx];
185 amy_event *e = amy_helpers_event_begin();
186 e->synth = layer->synth_id[track];
187 if (f->enabled) {
188 e->filter_type = f->filter_type;
189 e->filter_freq_coefs[COEF_CONST] = f->cutoff_hz;
190 e->resonance = f->resonance;
191 /* EG1 -> cutoff depth (octaves). Only wired when the user has dialed a
192 * non-zero amount, so a plain melodic filter is byte-for-byte unchanged.
193 * COEF_EG1 is the second (aux) envelope generator; on melodic tracks
194 * nothing routes it to amplitude (no amp_coefs[COEF_EG1] anywhere), so
195 * there is no amp/filter double-use to arbitrate — the amount==0 gate is
196 * the only guard needed. Same convention as arp_core.c:253. */
197 if (f->filter_env_amount != 0.0f) {
198 e->filter_freq_coefs[COEF_EG1] = f->filter_env_amount;
199 }
200 } else {
201 e->filter_type = FILTER_NONE;
202 }
203 if (layer->patch == SEQ_PATCH_KS) {
204 e->feedback = sequencer_core_ks_feedback_from_q(f->resonance);
205 }
206 amy_helpers_event_send(e);
207
208 /* Guarantee valid EG1 breakpoints on this synth whenever the filter env is
209 * live, so filter_freq_coefs[COEF_EG1] modulates a real ramp instead of a
210 * stuck unity gate (AMY treats a never-configured breakpoint set as a
211 * permanent 1.0 — see arp_core.c:247-253). Uses the row's stored EG1
212 * (authored shape, or the zeroed default). Skipped entirely when inert. */
213 if (f->enabled && f->filter_env_amount != 0.0f) {
214 sequencer_core_push_envelope_eg1(layer->synth_id[track], 0,
215 seq_layer_env1(layer_idx, track));
216 }
217}

272 const seq_lfo_t *lfo)
273{
274 if (!lfo || layer_idx >= s_num_layers || track >= SEQ_TRACKS) return;
275 seq_layer_t *layer = &s_layers[layer_idx];
276
277 layer->vp[track].lfo = *lfo;
278 if (layer->vp[track].lfo.depth > 100) layer->vp[track].lfo.depth = 100;
279 layer->vp[track].lfo_authored = true;
280
281#if CONFIG_SEQ_MELODIC_AMY_NATIVE_LFO
282 bool is_wave = sequencer_core_is_wave_patch(layer->patch);
283 if (is_wave) {
284 melodic_configure_native_lfo_track(layer, track);
285 /* Restore static target value when disabled or on software-fallback path
286 * (PAN/RANDOM): native clears COEF_MOD but doesn't push the neutral coef. */
287 if (!lfo->enabled || !is_native_lfo_track(&layer->vp[track].lfo)) {
288 lfo_restore_target_neutrals(layer, track, lfo);
289 }
290 /* s_lfo_hz=0 for native tracks so the service loop skips them;
291 * non-zero for PAN/RANDOM fallback so the service loop picks them up. */
292 s_lfo_hz[layer_idx][track] = (lfo->enabled && is_native_lfo_track(&layer->vp[track].lfo))
293 ? 0.0f
294 : (lfo->enabled ? lfo_rate_to_hz(lfo->rate, s_bpm) : 0.0f);
295 ESP_LOGI(TAG, "LFO L%u T%u %s %.2f Hz d=%u tgt=0x%02x [native]",
296 layer_idx + 1u, track + 1u, lfo->enabled ? "ON" : "OFF",
297 (double)s_lfo_hz[layer_idx][track], lfo->depth, lfo->targets);
298 return;
299 }
300#endif
301
302
303
304
305 /* Software path: non-wave patches, or native LFO disabled at compile time */
306 if (!lfo->enabled) {
307 lfo_restore_target_neutrals(layer, track, lfo);
308 s_lfo_hz[layer_idx][track] = 0.0f;
309 } else {
310 s_lfo_hz[layer_idx][track] = lfo_rate_to_hz(lfo->rate, s_bpm);
311 }
312 ESP_LOGI(TAG, "LFO L%u T%u %s %.2f Hz d=%u tgt=0x%02x",
313 layer_idx + 1u, track + 1u, lfo->enabled ? "ON" : "OFF",
314 (double)s_lfo_hz[layer_idx][track], lfo->depth, lfo->targets);
315}

325void __attribute__((optimize("O3", "unroll-loops", "fast-math"))) sequencer_core_lfo_service(void)
326{
327 const float DT = 0.05f; /* 20 Hz */
328 for (int li = 0; li < s_num_layers; li++) {
329 for (int tr = 0; tr < SEQ_TRACKS; tr++) {
330 if (!s_layers[li].vp[tr].lfo_authored) continue;
331 const seq_lfo_t *lfo = &s_layers[li].vp[tr].lfo;
332 if (!lfo->enabled) continue;
333 float hz = s_lfo_hz[li][tr];
334 if (hz <= 0.0f) continue;
335
336 float ph = s_lfo_phase[li][tr] + hz * DT;
337 if (ph >= 1.0f) {
338 ph -= 1.0f;
339 if (lfo->wave == LFO_WAVE_RANDOM)
340 s_lfo_rnd[li][tr] = lfo_next_rand();
341 }
342 s_lfo_phase[li][tr] = ph;
343
344 float val;
345 switch (lfo->wave) {
346 case LFO_WAVE_SINE:
347 val = sinf(2.0f * 3.14159265f * ph); break;
348 case LFO_WAVE_TRIANGLE:
349 val = (ph < 0.5f) ? (4.0f*ph - 1.0f)
350 : (3.0f - 4.0f*ph); break;
351 case LFO_WAVE_SAW_UP: val = 2.0f*ph - 1.0f; break;
352 case LFO_WAVE_SAW_DOWN: val = 1.0f - 2.0f*ph; break;
353 case LFO_WAVE_SQUARE: val = (ph < 0.5f) ? 1.0f : -1.0f; break;
354 case LFO_WAVE_RANDOM: val = s_lfo_rnd[li][tr]; break;
355 default: val = 0.0f; break;
356 }
357
358 float d = (float)lfo->depth / 100.0f;
359 uint8_t syn = s_layers[li].synth_id[tr];
360
361 amy_event *e = amy_helpers_event_begin();
362 e->synth = syn;
363 /* Multi-target: each checked target modulates its own independent
364 * COEF_CONST from the same LFO value. (SCAN has no software analog —
365 * it needs a wavetable voice, which always takes the native path.) */
366 if (LFO_HAS_TGT(lfo, LFO_TARGET_FILTER)) {
367 float base = (s_layers[li].vp[tr].filter.enabled &&
368 s_layers[li].vp[tr].filter.cutoff_hz > 0.0f)
369 ? s_layers[li].vp[tr].filter.cutoff_hz : 1000.0f;
370 e->filter_freq_coefs[COEF_CONST] = base * powf(2.0f, d * 3.0f * val);
371 }
372 if (LFO_HAS_TGT(lfo, LFO_TARGET_AMP))
373 e->amp_coefs[COEF_CONST] = 1.0f - d*(0.5f - 0.5f*val);
374 if (LFO_HAS_TGT(lfo, LFO_TARGET_PITCH))
375 e->freq_coefs[COEF_CONST] = powf(2.0f, d * val);
376 if (LFO_HAS_TGT(lfo, LFO_TARGET_PAN))
377 e->pan_coefs[COEF_CONST] = 0.5f + d*0.5f*val;
378 amy_helpers_event_send(e);
379 }
380 }
381}

434/* ── Per-track amplitude trim (graph editor amp mode) ────────────────────────
435 * amp_scale is a per-track multiplier applied to note velocity at emit time.
436 * Default 1.0 (unity). Must be initialised to 1.0 in add_layer since
437 * memset zeroes the struct.
438 *
439 * The setter re-emits all steps for the affected track immediately after
440 * storing the value (same pattern as sequencer_core_set_track_repeat_rate).
441 * This is necessary because steps are scheduled ahead-of-time with a period;
442 * the sequencer does not re-emit each step on every tick during steady
443 * playback, so a store-only change would be silent until an unrelated
444 * re-emit event (patch change, play-stop, etc.). */
445
446float sequencer_core_get_melodic_amp_scale(uint8_t layer_idx, uint8_t track)
447{
448 if (layer_idx >= s_num_layers || track >= SEQ_TRACKS) return 1.0f;
449 return s_layers[layer_idx].vp[track].amp_trim;
450}

452void sequencer_core_set_melodic_amp_scale(uint8_t layer_idx, uint8_t track,
453 float v)
454{
455 if (layer_idx >= s_num_layers || track >= SEQ_TRACKS) return;
456 if (v < 0.0f) v = 0.0f;
457 if (v > 1.0f) v = 1.0f;
458 s_layers[layer_idx].vp[track].amp_trim = v;
459 /* Re-emit all steps so the new amplitude takes effect immediately. */
460 seq_layer_t *layer = &s_layers[layer_idx];
461 for (uint8_t s = 0; s < layer->num_steps; s++)
462 sequencer_emit_step(layer_idx, track, s);
463}

465/* ── Live-preview pushes (AMY only; the store is untouched) ──────────────────
466 * The editors audition scratch values against the running engine while the
467 * committed store stays the single source of truth. Cancel restores by
468 * re-pushing the stored state (or reloading the layer's patch when the row was
469 * never authored); confirm goes through the normal setters. Authored flags are
470 * never modified by a preview. */
471
472void sequencer_core_preview_melodic_envelope(uint8_t layer_idx, uint8_t track,
473 const seq_env_t *env)
474{
