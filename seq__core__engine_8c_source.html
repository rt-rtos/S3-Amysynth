8/* ── Bar counter ─────────────────────────────────────────────────────────
9 * sequencer_ticks() is monotonic (never resets on play/stop in normal use).
10 * Capture a baseline at play-start; compute bars elapsed from the delta. */
11uint32_t sequencer_bars_elapsed(void)
12{
13 uint32_t t = sequencer_ticks();
14 if (t < s_bar_baseline) return 0;
15 return (t - s_bar_baseline) / SEQ_TICKS_PER_BAR;
16}

18/* ── Tag helpers ─────────────────────────────────────────────────────── */
19/*
20 * Tag layout (uint32_t — tag space is effectively unlimited):
21 *
22 * ON tag = layer * (SEQ_TRACKS * SEQ_MAX_STEPS * 2)
23 * + track * SEQ_MAX_STEPS + step
24 *
25 * OFF tag = ON tag + (SEQ_TRACKS * SEQ_MAX_STEPS)
26 *
27 * Preview = MAX_LAYERS * (SEQ_TRACKS * SEQ_MAX_STEPS * 2)
28 * + layer * SEQ_TRACKS + track
29 *
30 * Per layer: 4*32*2 = 256 slots; all 4 layers occupy tags 0..1023.
31 * Preview tags start at 1024.
32 */
33static inline uint32_t seq_tag_on(uint8_t layer, uint8_t track, uint8_t step)
34{
35 return (uint32_t)layer * (SEQ_TRACKS * SEQ_MAX_STEPS * 2)
36 + (uint32_t)track * SEQ_MAX_STEPS
37 + step;
38}

40static inline uint32_t seq_tag_off(uint8_t layer, uint8_t track, uint8_t step)
41{
42 return seq_tag_on(layer, track, step)
43 + (uint32_t)(SEQ_TRACKS * SEQ_MAX_STEPS);
44}

59/* Swing / shuffle: delay ODD 16th-steps by swing_pct% of one step so the
60 * off-beats land late, giving a shuffled groove. Even steps (the on-beats) are
61 * untouched. Pure function of the step index + the layer's swing_pct, so the
62 * emitted schedule stays beat-locked and tempo-independent (it is expressed in
63 * ticks, exactly like the drone stutter-grid swing in drone_core.c). Integer
64 * math only — this runs on the Core-0 emit path, never in render/ISR.
65 * swing_pct==0 (the memset-zero default) returns 0 => bit-identical to the
66 * pre-swing schedule. swing_pct is clamped to SEQ_SWING_MAX (<100) so the
67 * result is always a fraction of one step and never crosses the next step. */
68static inline uint32_t sequencer_step_swing_offset(const seq_layer_t *layer,
69 uint8_t step)
70{
71 if ((step & 1u) == 0u || layer->swing_pct == 0) return 0;
72 return ((uint32_t)SEQ_TICKS_PER_STEP * (uint32_t)layer->swing_pct) / 100u;
73}

75float sequencer_step_velocity(const seq_layer_t *layer,
76 uint8_t track, uint8_t step)
77{
78 /* Drums now share the melodic accent+jitter curve for a less "machine-gun"
79 * groove (the old fixed 1.0 made every hit identical). Falls through to the
80 * same expressive path as melodic below. */
81 (void)layer;
82
83#if !CONFIG_SEQ_MELODIC_EXPRESSIVE_DEFAULTS
84 (void)track;
85 (void)step;
86 return 1.0f;
87#else
88
89 /* With the EG0 envelope now shaping onset/tail, we can use a wider dynamic
90 * range without the notes sounding dull — the accent pattern provides the
91 * groove that breaks up the old "machine-gun" monotony. Base level sits
92 * mid-range so accents have room to push up and ghost notes can drop down.
93 * Tracks are spread slightly so stacked voices don't all hit identically. */
94 float velocity = 0.62f + (0.02f * (float)track);
95
96 /* Metric accents: strong downbeat, lighter backbeat, weak off-beats. */
97 if ((step % 4) == 0) {
98 velocity += 0.30f; /* downbeat of each quarter-note */
99 } else if ((step % 4) == 2) {
100 velocity += 0.16f; /* backbeat emphasis */
101 } else {
102 velocity -= 0.04f; /* the in-between 8ths sit back as ghost notes */
103 }
104
105 /* Deterministic per-step jitter so repeated bars are not bit-identical
106 * (light "humanization"). Cycles every 4 steps with a small +/- swing. */
107 static const float jitter[4] = { 0.015f, -0.02f, 0.01f, -0.015f };
108 velocity += jitter[step & 3];
109
110 velocity = SEQ_CLAMP_F32(velocity, 0.45f, 1.0f);
111 return velocity;
112#endif
113}

115/* True when any track in the layer has solo engaged (scans all SEQ_TRACKS,
116 * not just num_tracks, so a stale solo flag on an unused track slot can never
117 * silently affect audibility of the tracks actually in use). */
118static bool sequencer_layer_has_solo(const seq_layer_t *layer)
119{
120 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
121 if (layer->solo[t]) return true;
122 }
123 return false;

181/* Schedule (or cancel) one grid step as a pair of repeating AMY events: a
182 * note-on at the step's position in the bar, and a note-off `gate` ticks later.
183 * Both repeat every `bar_ticks` so the pattern loops automatically. AMY keys
184 * each event by its tag, so re-emitting with the same tag updates in place. */
185void sequencer_emit_step(uint8_t layer_idx, uint8_t track, uint8_t step)
186{
187 seq_layer_t *layer = &s_layers[layer_idx];
188 /* Total ticks in one loop of this layer's pattern. */
189 uint32_t bar_ticks = (uint32_t)layer->num_steps * SEQ_TICKS_PER_STEP;
190 /* Repeat rate: fire every N bars. period scales bar_ticks accordingly.
191 * note-off must wrap against the same period so it lands in the correct
192 * half of the extended window (not just within the first bar). */
193 uint32_t rr = (layer->repeat_rate[track] >= SEQ_REPEAT_2)
194 ? (uint32_t)layer->repeat_rate[track] : 1u;
195 uint32_t period = bar_ticks * rr;
196 /* Plain-trig note-hold incl. the melodic off-beat groove shortening;
197 * shared with the ratchet n==1 path (seq_core_trig.c) via seq_step_gate. */
198 uint8_t gate = (uint8_t)seq_step_gate(layer, step);
199 uint32_t tag_on = seq_tag_on(layer_idx, track, step);
200 uint32_t tag_off = seq_tag_off(layer_idx, track, step);
201 /* +1 so tick 0 stays reserved (AMY treats tick 0 specially as "clear").
202 * Two per-step timing offsets fold into the note-on tick: swing shifts odd
203 * steps later, and micro-timing (patch-06) adds the signed per-step nudge on
204 * top. Both are added into tick_on only; tick_off is derived from it below,
205 * so gate length is preserved. A negative nudge on an early step can land
206 * before the bar origin, so the combined value is wrapped into the tail of
207 * the loop window (a late "drag"). Defaults (swing 0, nudge 0) leave the
208 * on-grid tick byte-identical. */
209 int32_t tick_on_s = (int32_t)(1 + step * SEQ_TICKS_PER_STEP)
210 + (int32_t)sequencer_step_swing_offset(layer, step)
211 + (int32_t)layer->step_nudge[track][step];
212 while (tick_on_s < 1) tick_on_s += (int32_t)period;
213 uint32_t tick_on = (uint32_t)tick_on_s % period;
214 if (tick_on == 0) tick_on = 1;
215 /* Note-off wraps within the full period (not just bar_ticks) so a note at
216 * repeat_rate=2 whose gate spills past bar_ticks still fires correctly. */
217 uint32_t tick_off = (tick_on + gate) % period;
218 float note_velocity = sequencer_step_velocity(layer, track, step);
219 /* Apply per-track amplitude trim (default 1.0; set by graph editor amp mode). */
220 note_velocity *= layer->vp[track].amp_trim;
221 /* Per-step velocity offset (patch-06): signed percentage points, default 0. */
222 note_velocity += (float)layer->step_velocity_adj[track][step] * 0.01f;
223 if (note_velocity < 0.0f) note_velocity = 0.0f;
224 if (note_velocity > 1.0f) note_velocity = 1.0f;
225 if (tick_off == 0) tick_off = 1; /* avoid the reserved tick 0 */
226
227 /* If stopped, this step is off, the track is muted/soloed-out, or the
228 * step carries a probability/ratchet/conditional decoration, cancel the
229 * plain periodic tag pair instead of emitting a note-on. Decorated steps
230 * are one-shot scheduled per loop-iteration by
231 * sequencer_core_service_tick() (seq_core_trig.c) instead, since AMY's
232 * own period-repeat has no hook to gate an individual repetition. */
233 if (!s_playing || !layer->grid[track][step] ||
234 !sequencer_track_audible(layer, track) ||
235 sequencer_core_step_is_decorated(layer, track, step)) {
236 sequencer_emit_clear_tag(tag_on);
237 sequencer_emit_clear_tag(tag_off);
238 return;
239 }
240
241 /* Both drum and melodic layers now have one synth slot per track. */
242 uint8_t synth = layer->synth_id[track];
243
244 amy_helpers_note_send(synth, layer->step_note[track][step], note_velocity,
245 tag_on, tick_on, period);
246 amy_helpers_note_send(synth, layer->step_note[track][step], 0.0f,
247 tag_off, tick_off, period);
248}

273/* Re-resolve a track's note (clamp + optional scale quantization), update every
274 * step on that track to the new note, and re-emit them. When `preview` is set
275 * (interactive editing) also fire a short one-shot so the user hears the note
276 * immediately, even if quantization left the resolved note unchanged. */
277static void sequencer_refresh_track_note(uint8_t layer_idx, uint8_t track,
278 bool preview)
279{
280 if (layer_idx >= s_num_layers) return;
281 seq_layer_t *layer = &s_layers[layer_idx];
282 if (track >= layer->num_tracks) return;
283
284 uint8_t source_note = s_track_source_note[layer_idx][track];
285 uint8_t resolved_note = sequencer_resolve_track_note(layer, source_note);
286
287 /* No change: skip the grid rewrite, but still preview so scrolling within
288 * one scale degree remains audible. */
289 if (layer->track_base_note[track] == resolved_note) {
290 if (preview) {
291 /* Keep the preview path active even when the snapped note does not change. */
292 } else {
293 return;
294 }
295 }
296
297 /* Apply the resolved note to the whole track (all steps play one pitch). */
298 layer->track_base_note[track] = resolved_note;
299 for (uint8_t s = 0; s < layer->num_steps; s++) {
300 layer->step_note[track][s] = resolved_note;
301 }
302
303 for (uint8_t s = 0; s < layer->num_steps; s++) {
304 sequencer_emit_step(layer_idx, track, s);
305 }
306
307 if (!preview) {
308 return;
309 }
310
311 /* One-shot preview: fires a few ticks from now using the same tag slot.
312 * Rapid scrolling overwrites the slot so only the last change is heard.
313 * Velocity matches what a real downbeat step on this track would play
314 * (groove velocity x amp trim) so the preview is honest about level. */
315 float preview_vel = sequencer_step_velocity(layer, track, 0)
316 * layer->vp[track].amp_trim;
317 if (preview_vel > 1.0f) preview_vel = 1.0f;
318 uint32_t fire_tick = sequencer_ticks() + SEQ_PREVIEW_DELAY_TICKS;
319 amy_helpers_note_send(layer->synth_id[track], resolved_note, preview_vel,
320 seq_preview_tag(layer_idx, track), fire_tick, 0);
321 amy_helpers_note_send(layer->synth_id[track], resolved_note, 0.0f,
322 seq_preview_off_tag(layer_idx, track),
323 fire_tick + seq_step_gate(layer, 0), 0);
324
325 ESP_LOGI(TAG, "L%d T%d note -> %d (preview @ tick %lu)",
326 layer_idx + 1, track + 1, resolved_note, (unsigned long)fire_tick);
327}

365/* Start/stop playback. On start, every step is re-emitted so AMY repopulates
366 * its schedule; on stop, each layer's playhead is captured (for a frozen UI)
367 * and all scheduled events are cancelled so nothing keeps triggering. */
368void sequencer_core_set_playing(bool p)
369{
370 if (s_playing == p) return;
371 s_playing = p;
372 if (s_playing) {
373 /* Anchor the bar counter so bars_elapsed is relative to this play-start. */
374 s_bar_baseline = sequencer_ticks();
375 s_prog.entry_start_bar = 0;
376 s_prog.current = 0;
377 for (uint8_t i = 0; i < s_num_layers; i++) {
378 sequencer_core_trig_reset(i);
379 sequencer_resync_layer(i);
380 }
381 } else {
382 /* Bug 1.1: clear arp scheduled events FIRST so repeating arp tags
383 * don't keep firing while the sequencer is paused. */
384 arp_core_clear_all();
385
386 /* Freeze display positions before clearing scheduled events. */
387 for (uint8_t i = 0; i < s_num_layers; i++) {
388 seq_layer_t *layer = &s_layers[i];
389 s_cached_step[i] = seq_playhead_step(layer, sequencer_ticks());
390 sequencer_clear_layer_tags(i);
391 /* Bug-1.1-style fix, same rationale as arp_core_clear_all() above:
392 * decorated steps' one-shot ratchet tags are not touched by
393 * sequencer_clear_layer_tags() (different tag space), so clear
394 * them explicitly or a pending sub-hit could still fire after
395 * the user hits stop. */

441void sequencer_core_arp_emit_note(uint32_t tag_base, uint8_t midi_note,
442 float velocity, uint32_t tick_on,
443 uint32_t gate_ticks, uint32_t period)
444{
445 /* Defensive: never let an out-of-range tag reach AMY. Its sequences[] table
446 * is sized to max_sequencer_tags and add_event has a `tag > max` off-by-one,
447 * so a stray tag would smash the heap. Cap to the reserved arp window. */
448 if (tag_base + 1 > SEQ_ARP_TAG_MAX) {
449 ESP_LOGE(TAG, "arp tag %u out of range (max %u) - dropped",
450 (unsigned)tag_base, (unsigned)SEQ_ARP_TAG_MAX);
451 return;
452 }
453
454 uint32_t tick_off = (period > 0) ? ((tick_on + gate_ticks) % period)
455 : (tick_on + gate_ticks);
456 if (tick_off == 0) tick_off = 1; /* tick 0 is reserved (clear) */
457 if (tick_on == 0) tick_on = 1;
458
459 amy_helpers_note_send(SEQ_ARP_SYNTH, midi_note, velocity,
460 tag_base, tick_on, period);
461 amy_helpers_note_send(SEQ_ARP_SYNTH, midi_note, 0.0f,
462 tag_base + 1, tick_off, period);
463}

520/* ── Per-layer melodic NoteFX: gate length + glide (portamento) ────────────
521 * Both are per-layer scalars edited from the NoteFX menu page. Gate is applied
522 * at emit time (seq_step_gate), so a change must re-emit the layer's steps so
523 * the new note-off ticks take effect immediately, exactly like swing. Glide is
524 * an AMY per-osc setting, pushed straight to the row synths. Both are no-ops on
525 * drum layers (gate uses the fixed SEQ_GATE_DRUM; drums don't glide). */
526void sequencer_core_set_melodic_gate_pct(uint8_t layer_idx, uint8_t gate_pct)
527{
528 if (layer_idx >= s_num_layers) return;
529 seq_layer_t *layer = &s_layers[layer_idx];
530 if (layer->type != SEQ_LAYER_MELODIC) return;
531 uint8_t clamped = (uint8_t)SEQ_CLAMP_U8((int)gate_pct, 10, 100);
532 if (layer->gate_pct == clamped) return;
533 layer->gate_pct = clamped;
534 sequencer_resync_layer(layer_idx); /* re-emit: gate changes note-off ticks */
535}

560void sequencer_core_set_track_mute(uint8_t layer_idx, uint8_t track, bool mute)
561{
562 if (layer_idx >= s_num_layers || track >= SEQ_TRACKS) return;
563 seq_layer_t *layer = &s_layers[layer_idx];
564 if (layer->mute[track] == mute) return;
565 layer->mute[track] = mute;
566 /* Mute only ever changes this one track's own audibility. */
567 for (uint8_t s = 0; s < layer->num_steps; s++) {
568 sequencer_emit_step(layer_idx, track, s);
569 }
570 if (!sequencer_track_audible(layer, track)) {
571 sequencer_kill_synth_voices(layer->synth_id[track]);
572 }
573}

581void sequencer_core_set_track_solo(uint8_t layer_idx, uint8_t track, bool solo)
582{
583 if (layer_idx >= s_num_layers || track >= SEQ_TRACKS) return;
584 seq_layer_t *layer = &s_layers[layer_idx];
585 if (layer->solo[track] == solo) return;
586 layer->solo[track] = solo;
587 /* Solo changes every track's audibility in this layer, not just this
588 * one's, so re-emit the whole layer and hard-kill whichever tracks just
589 * became inaudible (a note already sounding would otherwise ring out
590 * until its scheduled note-off, which re-emit alone does not force). */
591 sequencer_resync_layer(layer_idx);
592 for (uint8_t t = 0; t < layer->num_tracks; t++) {
593 if (!sequencer_track_audible(layer, t)) {
594 sequencer_kill_synth_voices(layer->synth_id[t]);
595 }
596 }
597}
