95 * voicing (progression entry 0); each advance moves them by the delta between
96 * the live chord root and entry 0's root - a rigid transpose, never per-tone

101 if (!s_prog.enabled || s_prog.count == 0) return 0;
102 if (!layer->chord_mode) return 0;
103 return (int)layer->chord_root - (int)s_prog.entries[0].root;
104}

117/* Swing: delay ODD 16th-steps by swing_pct% of one step so off-beats land late.
118 * Pure function of step index + swing_pct, expressed in ticks, so the schedule
119 * stays beat-locked and tempo-independent (same as the drone stutter-grid swing
120 * in drone_core.c). Integer math only - this is the Core-0 emit path.
121 * swing_pct is clamped to SEQ_SWING_MAX (<100), so the offset is always a
122 * fraction of one step and never crosses into the next. */
123static inline uint32_t sequencer_step_swing_offset(const seq_layer_t *layer,
124 uint8_t step)
125{
126 if ((step & 1u) == 0u || layer->swing_pct == 0) return 0;
127 return ((uint32_t)SEQ_TICKS_PER_STEP * (uint32_t)layer->swing_pct) / 100u;
128}

130float sequencer_step_velocity(const seq_layer_t *layer,
131 uint8_t track, uint8_t step)
132{
133 /* Drums share the melodic accent+jitter curve rather than a flat 1.0. */
134 (void)layer;
135
136#if !CONFIG_SEQ_MELODIC_EXPRESSIVE_DEFAULTS
137 (void)track;
138 (void)step;
139 return 1.0f;
140#else
141
142 /* Base level sits mid-range so accents have room to push up and ghost notes
143 * to drop down; tracks are spread slightly so stacked voices don't hit
144 * identically. */
145 float velocity = 0.62f + (0.02f * (float)track);
146
147 /* Metric accents: strong downbeat, lighter backbeat, weak off-beats. */
148 if ((step % 4) == 0) {
149 velocity += 0.30f; /* downbeat of each quarter-note */
150 } else if ((step % 4) == 2) {
151 velocity += 0.16f; /* backbeat emphasis */
152 } else {
153 velocity -= 0.04f; /* in-between 8ths sit back as ghost notes */
154 }
155
156 /* Deterministic per-step jitter (light humanization) so repeated bars are
157 * not bit-identical. Cycles every 4 steps. */
158 static const float jitter[4] = { 0.015f, -0.02f, 0.01f, -0.015f };
159 velocity += jitter[step & 3];
160
161 velocity = SEQ_CLAMP_F32(velocity, 0.45f, 1.0f);
162
163 /* NoteFX GROOVE: blend the accent curve (100%) against flat 1.0 (0%).
164 * After the clamp, so 0% is exactly 1.0. */
165 velocity = 1.0f - ((float)layer->groove_pct * 0.01f) * (1.0f - velocity);
166 return velocity;
167#endif
168}

251/* Schedule (or cancel) one grid step as a pair of repeating AMY events: note-on
252 * at the step's position in the bar, note-off `gate` ticks later, both
253 * repeating every `bar_ticks`. AMY keys events by tag, so re-emitting with the
254 * same tag updates in place. */
255void sequencer_emit_step(uint8_t layer_idx, uint8_t track, uint8_t step)
256{
257 seq_layer_t *layer = &s_layers[layer_idx];
258 /* Total ticks in one loop of this layer's pattern. */
259 uint32_t bar_ticks = (uint32_t)layer->num_steps * SEQ_TICKS_PER_STEP;
260 /* Repeat rate: fire every N bars, so period scales bar_ticks. The note-off
261 * must wrap against the same period to land in the correct half of the
262 * extended window. */
263 uint32_t rr = (layer->repeat_rate[track] >= SEQ_REPEAT_2)
264 ? (uint32_t)layer->repeat_rate[track] : 1u;
265 uint32_t period = bar_ticks * rr;
266 /* Plain-trig note-hold incl. the off-beat groove shortening; shared with
267 * the ratchet n==1 path (seq_core_trig.c) via seq_step_gate. */
268 uint8_t gate = (uint8_t)seq_step_gate(layer, step);
269 uint32_t tag_on = seq_tag_on(layer_idx, track, step);
270 uint32_t tag_off = seq_tag_off(layer_idx, track, step);
271 /* +1 so tick 0 stays reserved (AMY treats tick 0 as "clear"). Two per-step
272 * offsets fold into the note-on tick: swing (odd steps later) and the
273 * signed micro-timing nudge. Both apply to tick_on only - tick_off derives
274 * from it below, preserving gate length. A negative nudge on an early step
275 * can land before the bar origin, so the result is wrapped into the tail of
276 * the loop window (a late "drag"). */
277 int32_t tick_on_s = (int32_t)(1 + step * SEQ_TICKS_PER_STEP)
278 + (int32_t)sequencer_step_swing_offset(layer, step)
279 + (int32_t)layer->step_nudge[track][step];
280 while (tick_on_s < 1) tick_on_s += (int32_t)period;
281 uint32_t tick_on = (uint32_t)tick_on_s % period;
282 if (tick_on == 0) tick_on = 1;
283 /* Wrap within the full period, not bar_ticks, so a repeat_rate=2 note whose
284 * gate spills past bar_ticks still fires correctly. */
285 uint32_t tick_off = (tick_on + gate) % period;
286 float note_velocity = sequencer_step_velocity(layer, track, step);
287 /* Per-track amplitude trim (default 1.0, set by graph editor amp mode). */
288 note_velocity *= layer->vp[track].amp_trim;
289 /* Per-step velocity offset in signed percentage points. */
290 note_velocity += (float)layer->step_velocity_adj[track][step] * 0.01f;
291 note_velocity = SEQ_CLAMP_F32(note_velocity, 0.0f, 1.0f);
292 if (tick_off == 0) tick_off = 1; /* avoid the reserved tick 0 */
293
294 /* Stopped, step off, track inaudible, or the step carries a
295 * probability/ratchet/conditional decoration: cancel the plain periodic tag
296 * pair instead of emitting. Decorated steps are one-shot scheduled per
297 * loop-iteration by sequencer_core_service_tick() (seq_core_trig.c),
298 * because AMY's period-repeat has no hook to gate one repetition. */
299 if (!s_playing || !layer->grid[track][step] ||
300 !sequencer_track_audible(layer, track) ||
301 sequencer_core_step_is_decorated(layer, track, step)) {
302 sequencer_emit_clear_tag(tag_on);
303 sequencer_emit_clear_tag(tag_off);
304 return;
305 }
306
307 /* Drum and melodic layers alike have one synth slot per track. */
308 uint8_t synth = layer->synth_id[track];
309
310 /* Per-step pitch offset, chromatic on top of the resolved step pitch
311 * (TODO: revisit quantizer interplay - bypassed by design for now). The
312 * plain path never carries a chord sentinel (chords force the decorated
313 * path), so plain arithmetic is safe. On and off use the same value:
314 * offs match by note number since AMY v1.2.121. */
315 uint8_t note = layer->step_note[track][step];
316 int8_t pofs = layer->step_pitch_ofs[track][step];
317 if (pofs != 0)
318 note = (uint8_t)SEQ_CLAMP_INT((int)note + (int)pofs, 0, 127);
319
320 amy_helpers_note_send(synth, note, note_velocity,
321 tag_on, tick_on, period);
322 /* PCM drums get no scheduled note-off: pcm_note_off() is a hard phase-jump
323 * to the sample end, so the gate would truncate even the natural tail. The
324 * hit rings to the sample's own end under EG0 (sustain 0 - the decay the
325 * user authors is what shapes the audible length), and non-looping PCM
326 * self-terminates, so nothing is left held. Suppressing only the emit
327 * keeps every velocity-0 kill path (stop/mute/solo) fully effective -
328 * unlike SYNTH_FLAGS_IGNORE_NOTE_OFFS, which swallows those too. SYNTH
329 * mode keeps its note-offs: patch releases shape those tails on purpose. */
330 if (layer->type == SEQ_LAYER_DRUM && s_drum_engine == SEQ_DRUM_PCM) {
331 sequencer_emit_clear_tag(tag_off);
332 } else {
333 amy_helpers_note_send(synth, note, 0.0f,
334 tag_off, tick_off, period);
335 }
336}

361/* Re-resolve a track's note (clamp + optional quantization), write it to every
362 * step, and re-emit. With `preview` set, also fire a short one-shot so the note
363 * is audible even when quantization left it unchanged. */
364static void sequencer_refresh_track_note(uint8_t layer_idx, uint8_t track,
365 bool preview)
366{
367 if (layer_idx >= s_num_layers) return;
368 seq_layer_t *layer = &s_layers[layer_idx];
369 if (track >= layer->num_tracks) return;
370
371 uint8_t source_note = s_track_source_note[layer_idx][track];
372 uint8_t resolved_note = sequencer_resolve_track_note(layer, source_note);
373
374 /* No change: skip the grid rewrite, but still preview so scrolling within
375 * one scale degree remains audible. */
376 if (layer->track_base_note[track] == resolved_note) {
377 if (preview) {
378 /* fall through to the preview below */
379 } else {
380 return;
381 }
382 } else if (SEQ_NOTE_IS_CHORD(layer->track_base_note[track])) {
383 /* Leaving or switching a chord: kill this track's pending chord-tone
384 * one-shots and preview pairs so no tone from the old voicing survives
385 * the transition (clear -> rebuild, every time). */
386 sequencer_core_trig_clear_track_chord(layer_idx, track);
387 seq_chord_preview_clear_from(layer_idx, track, 1);
388 }
389
390 /* All steps on a track play one pitch. */
391 layer->track_base_note[track] = resolved_note;
392 for (uint8_t s = 0; s < layer->num_steps; s++) {
393 layer->step_note[track][s] = resolved_note;
394 }
395
396 for (uint8_t s = 0; s < layer->num_steps; s++) {
397 sequencer_emit_step(layer_idx, track, s);
398 }
399
400 /* Kill anything this track is CURRENTLY sounding at the old pitch. Since
401 * AMY v1.2.121 note-offs match by note number (instrument.c
402 * _instrument_voice_for_note): the re-emits above and the preview
403 * reschedule below rewrite the pending off tags with the NEW pitch, so a
404 * note whose on already fired would never receive a matching off and ring
405 * indefinitely (nearly every preview during a pitch scroll, since preview
406 * spacing < gate). A velocity-0 event with NO midi_note is AMY's
407 * synth-scoped all-notes-off (patches.c) - matched by voice, silent no-op
408 * on an idle instrument. Melodic-only: the 1-voice drum-PCM instruments
409 * take the single-voice path where a no-note event means note 0, and PCM
410 * drums schedule no offs anyway. */
411 if (layer->type == SEQ_LAYER_MELODIC) {
412 amy_event *kill = amy_helpers_event_begin();
413 kill->synth = layer->synth_id[track];
414 kill->velocity = 0.0f;
415 amy_helpers_event_send(kill);
416 }
417
418 if (!preview) {
419 return;
420 }
421
422 /* One-shot preview a few ticks out, reusing the same tag slot so rapid
423 * scrolling only sounds the last change. Velocity matches a real downbeat
424 * step (groove velocity x amp trim) so the level is honest. A chord
425 * sentinel previews the full voicing: tone 0 on the plain preview pair,
426 * extra tones on the chord pairs, pairs past the tone count cleared so a
427 * shrink cannot leave a stale higher tone pending. */
428 uint8_t tones[SEQ_CHORD_MAX_NOTES];
429 uint8_t ntones = seq_track_fire_notes(layer, resolved_note, tones);
430 if (ntones == 0) return; /* undefined chord slot: nothing to audition */
431
432 float preview_vel = sequencer_step_velocity(layer, track, 0)
433 * layer->vp[track].amp_trim;
434 if (preview_vel > 1.0f) preview_vel = 1.0f;
435 uint32_t fire_tick = sequencer_ticks() + SEQ_PREVIEW_DELAY_TICKS;
436 uint32_t off_tick = fire_tick + seq_step_gate(layer, 0);
437 /* PCM drums get no note-off, same rule as sequencer_emit_step(): a preview
438 * while tuning a drum's pitch should sound like the real hit. */
439 bool send_offs = !(layer->type == SEQ_LAYER_DRUM &&
440 s_drum_engine == SEQ_DRUM_PCM);
441 amy_helpers_note_send(layer->synth_id[track], tones[0], preview_vel,
442 seq_preview_tag(layer_idx, track), fire_tick, 0);
443 if (send_offs)
444 amy_helpers_note_send(layer->synth_id[track], tones[0], 0.0f,
445 seq_preview_off_tag(layer_idx, track), off_tick, 0);
446 for (uint8_t i = 1; i < ntones; i++) {
447 amy_helpers_note_send(layer->synth_id[track], tones[i], preview_vel,
448 seq_chord_preview_on_tag(layer_idx, track, i),
449 fire_tick, 0);
450 if (send_offs)
451 amy_helpers_note_send(layer->synth_id[track], tones[i], 0.0f,
452 seq_chord_preview_off_tag(layer_idx, track, i),
453 off_tick, 0);
454 }
455 if (SEQ_NOTE_IS_CHORD(resolved_note)) {
456 seq_chord_preview_clear_from(layer_idx, track, ntones);
457 }
458
459 ESP_LOGI(TAG, "L%d T%d note -> %d (preview @ tick %lu)",
460 layer_idx + 1, track + 1, resolved_note, (unsigned long)fire_tick);
461}

498/* Start/stop playback. Start re-emits every step so AMY repopulates its
499 * schedule; stop captures each layer's playhead (frozen UI) and cancels all
500 * scheduled events so nothing keeps triggering. */
501#if CONFIG_SEQ_OOM_RESYNC
502/* Self-heal after an AMY OOM burst. While internal heap is exhausted (e.g. an
503 * oversized patch load), scheduled events and tag cancels are dropped
504 * silently - only the FIRST failure prints (AMY's print-once policy), so
505 * tracks stay mute until something happens to re-emit them. Poll the OOM
506 * counter from the UI task; when a burst STOPS growing (one quiet service
507 * tick, 50 ms), re-emit everything once. Waiting for quiet avoids resyncing
508 * into the middle of the pressure that caused the drops. */
509void sequencer_core_oom_service(void)
510{
511 static uint32_t s_last_oom = 0;

530void sequencer_core_set_playing(bool p)
531{
532 if (s_playing == p) return;
533 s_playing = p;
534 if (s_playing) {
535 /* Anchor the bar counter to the NEXT absolute bar boundary, not the
536 * play-press tick: AMY fires periodic events on tick % period, so
537 * pattern loops are phase-locked to the absolute tick grid, and
538 * rounding up makes progression bar lines coincide with step 0. The
539 * partial pre-boundary stretch counts as bar 0 (bars_elapsed clamps),
540 * so the first chord keeps its full duration. */
541 uint32_t t = sequencer_ticks();
542 s_bar_baseline = ((t + SEQ_TICKS_PER_BAR - 1) / SEQ_TICKS_PER_BAR)
543 * SEQ_TICKS_PER_BAR;
544 s_prog.entry_start_bar = 0;
545 s_prog.current = 0;
546 /* The progression restarts from entry 0, but the layers/arp may still
547 * hold the chord that was live at stop time - request a re-apply so
548 * what is shown active is what sounds. Drained by the progression
549 * service on synth_ui_task (single-applier). Immediate, because a
550 * correction must bypass the BAR launch-quantize hold. */
551 if (s_prog.enabled) {
552 s_prog_apply_immediate = true;
553 s_prog_apply_pending = true;
554 }
555 for (uint8_t i = 0; i < s_num_layers; i++) {
556 sequencer_core_trig_reset(i);
557 sequencer_resync_layer(i);
558 }
559 /* The pause path cleared the arp schedule and emission is
560 * s_playing-gated, so nothing re-armed it while stopped - request a
561 * coalesced re-emit (drained by arp_core_service() on the UI task). */
562 arp_core_mark_dirty();
563 } else {
564 /* Clear arp scheduled events FIRST so repeating arp tags don't keep
565 * firing while the sequencer is paused. */
566 arp_core_clear_all();
567
568 /* Freeze display positions before clearing scheduled events. */
569 for (uint8_t i = 0; i < s_num_layers; i++) {
570 seq_layer_t *layer = &s_layers[i];
571 s_cached_step[i] = seq_playhead_step(layer, sequencer_ticks());
572 sequencer_clear_layer_tags(i);
573 /* Decorated steps' one-shot ratchet tags live in a different tag
574 * space and survive sequencer_clear_layer_tags(), so clear them
575 * explicitly or a pending sub-hit fires after stop. */
576 sequencer_core_trig_clear_all(i);
577 }
578
579 /* Silence only the sequencer's own synth slots (per-layer melodic/drum
580 * plus the arp). The drone slots are deliberately spared: they manage
581 * their own lifecycle and never auto-resume, so a global notes-off
582 * would silence the drone permanently. */
583 for (uint8_t i = 0; i < s_num_layers; i++) {
584 seq_layer_t *layer = &s_layers[i];
585 for (uint8_t t = 0; t < layer->num_tracks; t++) {
586 sequencer_kill_synth_voices(layer->synth_id[t]);
587 }
588 }
589 sequencer_kill_synth_voices(SEQ_ARP_SYNTH);
590 }
591}

634/* Chord table edit sweep (called by seq_chords_set/clear on the UI task).
635 * Voicing edits need no re-emit - fires read the table live - but pending
636 * one-shots from the old voicing are cleared, a now-undefined slot drops
637 * referencing tracks back to their last plain note, and a changed tone count
638 * reconfigures the layer's voice budget. */

640{

643 bool defined = seq_chords_is_defined(idx);

645 for (uint8_t li = 0; li < s_num_layers; li++) {

656 uint8_t fb = s_track_prev_plain[li][t];
657 if (fb == 0) fb = 60; /* C4 when no plain note was ever set */
658 s_track_source_note[li][t] =
659 sequencer_clamp_layer_note(layer, fb);
660 }
661 sequencer_refresh_track_note(li, t, false);
662 }
663 if (touched && sequencer_layer_voices_stale(li)) {
664 sequencer_reconfigure_layer_paused(li);
665 }

669/* Chord-editor audition: one-shot the (possibly unsaved) voicing through a
670 * melodic track's synth via the preview tags. Tones play exactly as authored -
671 * no progression transpose, the editor edits the "I" voicing. Caveat: a voicing
672 * wider than the track's current voice count voice-steals until assignment
673 * widens it; timbre, the point of the audition, is still accurate. */
674void sequencer_core_audition_chord(uint8_t layer_idx, uint8_t track,
675 const seq_chord_t *chord)
676{
677 if (layer_idx >= s_num_layers || track >= SEQ_TRACKS || chord == NULL) return;
678 seq_layer_t *layer = &s_layers[layer_idx];
679 if (layer->type != SEQ_LAYER_MELODIC) return;
680
681 uint8_t tones[SEQ_CHORD_MAX_NOTES];
682 uint8_t ntones = 0;
683 for (uint8_t i = 0; i < SEQ_CHORD_MAX_NOTES; i++) {
684 if (chord->notes[i] == 0) continue;
685 tones[ntones++] = sequencer_core_clamp_melodic_note(chord->notes[i]);
686 }
687 seq_chord_preview_clear_from(layer_idx, track, ntones ? ntones : 1);
688 if (ntones == 0) return;
689
690 float preview_vel = sequencer_step_velocity(layer, track, 0)
691 * layer->vp[track].amp_trim;
692 if (preview_vel > 1.0f) preview_vel = 1.0f;
693 uint32_t fire_tick = sequencer_ticks() + SEQ_PREVIEW_DELAY_TICKS;
694 uint32_t off_tick = fire_tick + seq_step_gate(layer, 0);
695 amy_helpers_note_send(layer->synth_id[track], tones[0], preview_vel,
696 seq_preview_tag(layer_idx, track), fire_tick, 0);
697 amy_helpers_note_send(layer->synth_id[track], tones[0], 0.0f,
698 seq_preview_off_tag(layer_idx, track), off_tick, 0);
699 for (uint8_t i = 1; i < ntones; i++) {
700 amy_helpers_note_send(layer->synth_id[track], tones[i], preview_vel,
701 seq_chord_preview_on_tag(layer_idx, track, i),
702 fire_tick, 0);
703 amy_helpers_note_send(layer->synth_id[track], tones[i], 0.0f,
704 seq_chord_preview_off_tag(layer_idx, track, i),
705 off_tick, 0);
706 }

709void sequencer_core_arp_emit_note(uint32_t tag_base, uint8_t midi_note,
710 float velocity, uint32_t tick_on,
711 uint32_t gate_ticks, uint32_t period)
712{
713 /* Slaved to the transport like sequencer_emit_step(): nothing may re-arm
714 * the arp's repeating schedule while paused, or an arp refresh during a
715 * pause (param edit, progression apply, project load) leaves the arp
716 * playing alone. sequencer_core_set_playing() marks the arp dirty on
717 * resume; periodic events are phase-locked to the absolute tick grid, so
718 * the rebuild lands back in sync. */
719 if (!s_playing) return;
720
721 /* Never let an out-of-range tag reach AMY: sequences[] is sized to
722 * max_sequencer_tags and add_event has a `tag > max` off-by-one, so a stray
723 * tag smashes the heap. Cap to the reserved arp window. */
724 if (tag_base + 1 > SEQ_ARP_TAG_MAX) {
725 ESP_LOGE(TAG, "arp tag %u out of range (max %u) - dropped",
726 (unsigned)tag_base, (unsigned)SEQ_ARP_TAG_MAX);
727 return;
728 }
729
730 uint32_t tick_off = (period > 0) ? ((tick_on + gate_ticks) % period)
731 : (tick_on + gate_ticks);
732 if (tick_off == 0) tick_off = 1; /* tick 0 is reserved (clear) */
733 if (tick_on == 0) tick_on = 1;
734
735 amy_helpers_note_send(SEQ_ARP_SYNTH, midi_note, velocity,
736 tag_base, tick_on, period);
737 amy_helpers_note_send(SEQ_ARP_SYNTH, midi_note, 0.0f,
738 tag_base + 1, tick_off, period);
739}
