48/* ── Shared step math (single owners; see architecture/sequencer-core.md §3.1,§3.3) ── */

50/* Which step the playhead sits on for `layer` at `ticks`: position within the
51 * bar divided by ticks-per-step. Returns 0 for an empty (num_steps == 0) layer;
52 * callers that must SKIP such a layer keep their own num_steps guard. */
53static inline uint8_t seq_playhead_step(const seq_layer_t *layer, uint32_t ticks)
54{
55 uint32_t bar_ticks = (uint32_t)layer->num_steps * SEQ_TICKS_PER_STEP;
56 if (bar_ticks == 0) return 0;
57 return (uint8_t)((ticks % bar_ticks) / SEQ_TICKS_PER_STEP);
58}
59
60/* Note-hold in ticks for the plain (non-subdivided) trig of `step`: drums are
61 * short/percussive, melodic longer, with off-beat 8ths shortened a touch so
62 * accented downbeats feel legato while in-between notes detach. Only ever
63 * shortens (never past SEQ_GATE_MELODIC) so the note-off always lands before the
64 * next step's note-on. Shared by sequencer_emit_step() and the ratchet n==1
65 * path so the two can never drift. */
66static inline uint16_t seq_step_gate(const seq_layer_t *layer, uint8_t step)
67{
68 if (layer->type == SEQ_LAYER_DRUM) return SEQ_GATE_DRUM;
69
70 /* Melodic note-hold is now a per-layer % of the step (the NoteFX GATE
71 * control). Round pct->ticks so the default (SEQ_MELODIC_GATE_DEFAULT_PCT)
72 * reproduces the legacy fixed gate exactly; 100% yields a full-step
73 * (legato) hold. Off-beat 8ths are still shortened a touch for groove. */
74 uint16_t gate = (uint16_t)(((uint32_t)SEQ_TICKS_PER_STEP * layer->gate_pct
75 + 50u) / 100u);
76 if ((step % 2) == 1 && gate > 2) {
77 gate -= 2;
78 }
79 if (gate < 1) gate = 1; /* never zero — the note must sound */
80 return gate;
81}
