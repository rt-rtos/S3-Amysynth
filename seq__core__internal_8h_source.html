34/* ── Shared step math (single owners; see architecture/sequencer-core.md §3.1,§3.3) ── */
35
36/* Which step the playhead sits on for `layer` at `ticks`: position within the
37 * bar divided by ticks-per-step. Returns 0 for an empty (num_steps == 0) layer;
38 * callers that must SKIP such a layer keep their own num_steps guard. */
39static inline uint8_t seq_playhead_step(const seq_layer_t *layer, uint32_t ticks)
40{
41 uint32_t bar_ticks = (uint32_t)layer->num_steps * SEQ_TICKS_PER_STEP;
42 if (bar_ticks == 0) return 0;
43 return (uint8_t)((ticks % bar_ticks) / SEQ_TICKS_PER_STEP);
44}
45
46/* Note-hold in ticks for the plain (non-subdivided) trig of `step`. Off-beat
47 * 8ths are shortened a touch so accented downbeats feel legato while
48 * in-between notes detach. Only ever shortens, so the note-off always lands
49 * before the next step's note-on. Shared by sequencer_emit_step() and the
50 * ratchet n==1 path so the two cannot drift. */
51static inline uint16_t seq_step_gate(const seq_layer_t *layer, uint8_t step)
52{
53 if (layer->type == SEQ_LAYER_DRUM) return SEQ_GATE_DRUM;
54
55 /* Melodic note-hold is a per-layer % of the step (the NoteFX GATE control),
56 * rounded pct->ticks; 100% is a full-step legato hold. */
57 uint16_t gate = (uint16_t)(((uint32_t)SEQ_TICKS_PER_STEP * layer->gate_pct
58 + 50u) / 100u);
59 if ((step % 2) == 1 && gate > 2) {
60 gate -= 2;
61 }
62 if (gate < 1) gate = 1; /* never zero — the note must sound */
63 return gate;
64}
