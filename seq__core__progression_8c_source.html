14/* Set by input-task entry points (encoder_task / button callback) that change
15 * chord state; consumed once per tick by sequencer_core_progression_service()
16 * which runs in synth_ui_task. This makes synth_ui_task the SINGLE task that
17 * calls chord_progression_apply_current() -> sequencer_refresh_melodic_layers()
18 * -> AMY emit, so progression/manual-chord edits never race the periodic
19 * advance or each other across tasks. */
20volatile bool s_prog_apply_pending = false;

24/* Map a chord type to the closest diatonic scale for arp snap quality. */
25static uint8_t chord_type_to_scale_index(chord_type_t ct)
26{
27 /* Scale indices mirror s_scales[] in quantizer.c:
28 * 0=Chromatic, 1=Major, 2=Natural Minor, 3=Dorian, 4=Phrygian,
29 * 5=Lydian, 6=Mixolydian, 7=Minor Pent, 8=Major Pent */
30 switch (ct) {
31 case CHORD_MAJ: return 1;
32 case CHORD_MIN: return 2;
33 case CHORD_MAJ7: return 1;
34 case CHORD_MIN7: return 3; /* Dorian has the natural 6 common in min7 contexts */
35 case CHORD_DOM7: return 6; /* Mixolydian */
36 case CHORD_SUS2: return 8; /* Major Pentatonic — open, no 3rd */
37 case CHORD_SUS4: return 8;
38 case CHORD_DIM: return 4; /* Phrygian — dark/diminished flavour */
39 case CHORD_AUG: return 5; /* Lydian — raised 4th matches augmented feel */
40 case CHORD_MIN9: return 3; /* Dorian — same family as min7 */
41 case CHORD_MAJ9: return 1; /* Major */
42 default: return 1;
43 }
44}

46void chord_progression_apply_current(void)
47{
48 if (s_prog.count == 0) return;
49 const chord_prog_entry_t *e = &s_prog.entries[s_prog.current];
50
51 /* Update every melodic layer's chord state and re-resolve all tracks. */
52 for (uint8_t li = 0; li < s_num_layers; li++) {
53 seq_layer_t *layer = &s_layers[li];
54 if (layer->type != SEQ_LAYER_MELODIC) continue;
55 layer->chord_mode = true;
56 layer->chord_root = e->root;
57 layer->chord_type = e->chord_type;
58 }
59 sequencer_refresh_melodic_layers(false);
60
61 /* Drive arp root + scale to match the new chord. */
62 arp_set_root_note((uint8_t)(e->root + 60)); /* pitch class → MIDI octave 4 */
63 arp_set_scale(chord_type_to_scale_index(e->chord_type));
64}

66/* Called from synth_ui_task at 20 Hz. This is the SINGLE task that emits chord
67 * changes to AMY: it both (a) drains s_prog_apply_pending, which input-task entry
68 * points (encoder_task / button callback) set after mutating chord state, and
69 * (b) advances the progression when the current entry expires. Funnelling every
70 * chord_progression_apply_current() / sequencer_refresh_melodic_layers() emit
71 * through this one task means edits never race the periodic advance or each other
72 * across tasks — no lock needed because there is only one writer of the emit. */
73void sequencer_core_progression_service(void)
74{
75 /* Drain deferred chord applies first, regardless of playing/enabled state. */
76 if (s_prog_apply_pending) {
77 s_prog_apply_pending = false;
78 if (s_prog.enabled && s_prog.count > 0) {
79 chord_progression_apply_current();
80 } else {
81 /* Disabled (or empty): layers already had chord_mode cleared by the
82 * caller, or a manual per-layer chord was set/cleared; re-resolve all
83 * melodic layers against their own current chord/scale state. */
84 sequencer_refresh_melodic_layers(false);
85 }
86 }
87
88 if (!s_prog.enabled || s_prog.count == 0 || !s_playing) return;
89
90 uint32_t bars = sequencer_bars_elapsed();
91 const chord_prog_entry_t *e = &s_prog.entries[s_prog.current];
92
93 if (bars - s_prog.entry_start_bar >= e->duration_bars) {
94 uint8_t next = (uint8_t)((s_prog.current + 1) % s_prog.count);
95 s_prog.current = next;
96 s_prog.entry_start_bar = bars;
97 chord_progression_apply_current();
98 ESP_LOGI(TAG, "progression -> entry %u (root=%u type=%u)",
99 next, s_prog.entries[next].root, (unsigned)s_prog.entries[next].chord_type);
100 }
101}

105void sequencer_core_progression_set_enabled(bool en)
106{
107 s_prog.enabled = en;
108 if (en && s_prog.count > 0) {
109 /* Anchor entry_start_bar to now so the first entry gets its full
110 * duration regardless of when in a play session this is toggled. */
111 s_prog.current = 0;
112 s_prog.entry_start_bar = sequencer_bars_elapsed();
113 /* Defer the AMY emit to the service tick (single-applier). */
114 s_prog_apply_pending = true;
115 } else if (!en) {
116 /* Disable chord mode on all melodic layers so they return to scale
117 * quantizer; defer the re-resolve emit to the service tick. */
118 for (uint8_t li = 0; li < s_num_layers; li++) {
119 s_layers[li].chord_mode = false;
120 }
121 s_prog_apply_pending = true;
122 }
123}

127void sequencer_core_progression_set_entry(uint8_t idx, uint8_t root,
128 chord_type_t chord_type,
129 uint8_t duration_bars)
130{
131 if (idx >= CHORD_PROG_MAX_ENTRIES) return;
132 s_prog.entries[idx].root = root % 12;
133 s_prog.entries[idx].chord_type = (chord_type < CHORD_TYPE_COUNT)
134 ? chord_type : CHORD_MAJ;
135 s_prog.entries[idx].duration_bars = (duration_bars > 0) ? duration_bars : 4;
136 if (idx >= s_prog.count) s_prog.count = (uint8_t)(idx + 1);
137 /* If we edited the live (currently playing) entry, defer a re-apply so the
138 * audible chord tracks the edit. */
139 if (s_prog.enabled && idx == s_prog.current) s_prog_apply_pending = true;
140}

178/* Append a default entry (Cmaj) if room remains. Returns true on success.
179 * Convenience: the Nth chord defaults to an N-bar duration (1st=1b, 2nd=2b, …),
180 * so a fresh progression staircases 1/2/3/4 bars without manual editing. Beyond
181 * the 4th entry the duration clamps to 4 bars — 5/6/7 aren't in the selectable
182 * duration set {1,2,3,4,8,16}, and the user still edits any of these by hand. */
183bool sequencer_core_progression_add_entry(void)
184{
185 if (s_prog.count >= CHORD_PROG_MAX_ENTRIES) return false;
186 uint8_t idx = s_prog.count;
187 uint8_t entry_num = (uint8_t)(idx + 1); /* 1-based position */
188 s_prog.entries[idx].root = 0; /* C */
189 s_prog.entries[idx].chord_type = CHORD_MAJ;
190 s_prog.entries[idx].duration_bars = entry_num <= 4 ? entry_num : 4;
191 s_prog.count = entry_num;
192 return true;
193}

195/* Delete entry idx, shifting later entries down. Keeps at least one entry. */
196void sequencer_core_progression_delete_entry(uint8_t idx)
197{
198 if (idx >= s_prog.count || s_prog.count <= 1) return;
199 for (uint8_t i = idx; i + 1 < s_prog.count; i++) {
200 s_prog.entries[i] = s_prog.entries[i + 1];
201 }
202 s_prog.count--;
203 /* Fix up the active index/window if it was at or past the deletion point. */
204 bool active_changed = false;
205 if (s_prog.current == idx) {
206 if (s_prog.current >= s_prog.count) s_prog.current = 0;
207 active_changed = true;
208 } else if (s_prog.current > idx) {
209 s_prog.current--; /* same entry, new slot — no audible change */
210 }
211 if (active_changed) {
212 s_prog.entry_start_bar = sequencer_bars_elapsed();
213 if (s_prog.enabled) s_prog_apply_pending = true;
214 }
215}
