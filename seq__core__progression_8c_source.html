4chord_progression_t s_prog = {
5 .entries = {
6 { .root = 0, .chord_type = CHORD_MAJ7, .duration_bars = 1 }, /* 1st chord -> 1b */

14/* Set by input-task entry points that change chord state, consumed once per
15 * tick by sequencer_core_progression_service() on synth_ui_task. That makes
16 * synth_ui_task the SINGLE task calling chord_progression_apply_current(), so
17 * chord edits never race the periodic advance or each other. */
18volatile bool s_prog_apply_pending = false;
19
20/* Set alongside s_prog_apply_pending when the apply is a correction (play-start
21 * re-sync) rather than a musical edit, so it bypasses the BAR
22 * launch-quantize hold and lands on the next service tick. */
23volatile bool s_prog_apply_immediate = false;
24
25/* BAR launch mode: bar index at which a held pending apply was armed;
26 * UINT32_MAX = not armed. Drains once bars_elapsed moves past this value. */
27static uint32_t s_apply_armed_bar = UINT32_MAX;

31/* Map a chord type to the closest diatonic scale for arp snap quality. */
32static uint8_t chord_type_to_scale_index(chord_type_t ct)
33{
34 /* Scale indices mirror s_scales[] in quantizer.c:
35 * 0=Chromatic, 1=Major, 2=Natural Minor, 3=Dorian, 4=Phrygian,
36 * 5=Lydian, 6=Mixolydian, 7=Minor Pent, 8=Major Pent,
37 * 9=Harmonic Minor, 10=Locrian, 11=Whole Tone */
38 switch (ct) {
39 case CHORD_MAJ: return 1;
40 case CHORD_MIN: return 2;
41 case CHORD_MAJ7: return 1;
42 case CHORD_MIN7: return 3; /* Dorian has the natural 6 common in min7 contexts */
43 case CHORD_DOM7: return 6; /* Mixolydian */
44 case CHORD_SUS2: return 8; /* Major Pentatonic - open, no 3rd */
45 case CHORD_SUS4: return 8;
46 case CHORD_DIM: return 10; /* Locrian - its b5 IS the chord tone; a
47 natural 5 clashes a semitone against it */
48 case CHORD_AUG: return 11; /* Whole Tone - contains the #5, same
49 natural-5 clash otherwise */
50 case CHORD_MIN9: return 3; /* Dorian - same family as min7 */
51 case CHORD_MAJ9: return 1; /* Major */
52 default: return 1;
53 }
54}

56void chord_progression_apply_current(void)
57{
58 if (s_prog.count == 0) return;
59 const chord_prog_entry_t *e = &s_prog.entries[s_prog.current];
60
61 /* Update every melodic layer's chord state and re-resolve all tracks. */
62 for (uint8_t li = 0; li < s_num_layers; li++) {
63 seq_layer_t *layer = &s_layers[li];
64 if (layer->type != SEQ_LAYER_MELODIC) continue;
65 layer->chord_mode = true;
66 layer->chord_root = e->root;
67 layer->chord_type = e->chord_type;
68 }
69 sequencer_refresh_melodic_layers(false);
70
71 /* Drive arp root + scale to the new chord, capturing the user's own values
72 * the first time the progression takes over; disable restores them. */
73 if (!s_prog.arp_saved) {
74 s_prog.saved_arp_root = arp_get_root_note();
75 s_prog.saved_arp_scale = arp_get_scale();
76 s_prog.arp_saved = true;
77 }
78 arp_set_root_note((uint8_t)(e->root + 60)); /* pitch class → MIDI octave 4 */
79 arp_set_scale(chord_type_to_scale_index(e->chord_type));
80}

82/* Called from synth_ui_task at 20 Hz - the SINGLE task that emits chord changes
83 * to AMY. It drains s_prog_apply_pending (set by input tasks after mutating
84 * chord state) and advances the progression when the current entry expires.
85 * Funnelling every emit through one task removes the need for a lock: there is
86 * only one writer. */
87void sequencer_core_progression_service(void)
88{
89 /* Drain deferred applies first, regardless of playing/enabled state. In BAR
90 * launch mode a musical edit holds until the next bar line while playing;
91 * corrections and drains while stopped bypass the hold. */
92 if (s_prog_apply_pending) {
93 bool drain_now = true;
94 if (s_prog.apply_at_bar && s_playing && !s_prog_apply_immediate) {
95 uint32_t bars = sequencer_bars_elapsed();
96 if (s_apply_armed_bar == UINT32_MAX) s_apply_armed_bar = bars; /* arm */
97 drain_now = (bars != s_apply_armed_bar); /* crossed a bar line */
98 }
99 if (drain_now) {
100 s_apply_armed_bar = UINT32_MAX;
101 s_prog_apply_pending = false;
102 s_prog_apply_immediate = false;
103 if (s_prog.enabled && s_prog.count > 0) {
104 chord_progression_apply_current();
105 } else {
106 /* Disabled or empty: the caller already cleared chord_mode, or
107 * a manual per-layer chord changed. Re-resolve every melodic
108 * layer against its own chord/scale state. */
109 sequencer_refresh_melodic_layers(false);
110 }
111 }
112 } else {
113 s_apply_armed_bar = UINT32_MAX;
114 }
115
116 if (!s_prog.enabled || s_prog.count == 0 || !s_playing) return;
117
118 uint32_t bars = sequencer_bars_elapsed();
119 /* Mid-session anchors point at the NEXT bar line (entry_start_bar = bars+1),
120 * and the unsigned subtraction below would wrap until it passes. */
121 if (bars < s_prog.entry_start_bar) return;
122 const chord_prog_entry_t *e = &s_prog.entries[s_prog.current];
123
124 if (bars - s_prog.entry_start_bar >= e->duration_bars) {
125 uint8_t next = (uint8_t)((s_prog.current + 1) % s_prog.count);
126 s_prog.current = next;
127 /* Advance by the expiring entry's duration, never "= bars": a service
128 * stall longer than a bar then catches up over successive ticks instead
129 * of permanently shifting the form. */
130 s_prog.entry_start_bar += e->duration_bars;
131 chord_progression_apply_current();
132 ESP_LOGI(TAG, "progression -> entry %u (root=%u type=%u)",
133 next, s_prog.entries[next].root, (unsigned)s_prog.entries[next].chord_type);
134 }
135}

139void sequencer_core_progression_set_enabled(bool en)
140{
141 s_prog.enabled = en;
142 if (en && s_prog.count > 0) {
143 /* The chord applies now (or at the next bar line in BAR mode), but its
144 * duration window counts from the NEXT bar line, so a mid-bar enable
145 * still gives the first entry its full duration - the partial bar is a
146 * free lead-in, which the service's bars < entry_start_bar guard
147 * covers. */
148 s_prog.current = 0;
149 s_prog.entry_start_bar = sequencer_bars_elapsed() + 1;
150 /* Defer the AMY emit to the service tick (single-applier). */
151 s_prog_apply_pending = true;
152 } else if (!en) {
153 /* Drop chord mode everywhere so layers return to the scale quantizer;
154 * the re-resolve emit is deferred to the service tick. */
155 for (uint8_t li = 0; li < s_num_layers; li++) {
156 s_layers[li].chord_mode = false;
157 }
158 /* Return the arp to the user's pre-progression root/scale. These are
159 * state-only setters - the arp marks itself dirty and arp_core_service
160 * re-emits on the UI task - so the single-applier rule holds. */
161 if (s_prog.arp_saved) {
162 arp_set_root_note(s_prog.saved_arp_root);
163 arp_set_scale(s_prog.saved_arp_scale);
164 s_prog.arp_saved = false;
165 }
166 s_prog_apply_pending = true;
167 }
168}

198void sequencer_core_progression_set_entry(uint8_t idx, uint8_t root,
199 chord_type_t chord_type,
200 uint8_t duration_bars)
201{
202 if (idx >= CHORD_PROG_MAX_ENTRIES) return;
203 /* Refuse holes: idx == count appends, anything past that legalizes the
204 * zero-filled entries in between, whose duration_bars 0 would make the
205 * service advance every tick - a 20 Hz chord strobe. */
206 if (idx > s_prog.count) return;
207 s_prog.entries[idx].root = root % 12;
208 s_prog.entries[idx].chord_type = (chord_type < CHORD_TYPE_COUNT)
209 ? chord_type : CHORD_MAJ;
210 s_prog.entries[idx].duration_bars = (duration_bars > 0) ? duration_bars : 4;
211 if (idx >= s_prog.count) s_prog.count = (uint8_t)(idx + 1);
212 /* Editing the live entry defers a re-apply so the audible chord follows. */
213 if (s_prog.enabled && idx == s_prog.current) s_prog_apply_pending = true;
214}

248/* Bars elapsed within the current entry (0-based), for the UI status bar. */
249uint8_t sequencer_core_progression_bars_in_current(void)
250{
251 /* Ticks are monotonic through stop, so without this gate the counter would
252 * keep climbing (and uint8-wrap) while paused. */
253 if (!s_playing) return 0;
254 if (!s_prog.enabled || s_prog.count == 0) return 0;
255 uint32_t bars = sequencer_bars_elapsed();
256 if (bars < s_prog.entry_start_bar) return 0; /* baseline just moved */
257 return (uint8_t)(bars - s_prog.entry_start_bar);
258}

260/* Append a default entry (Cmaj) if room remains. The Nth chord defaults to an
261 * N-bar duration so a fresh progression staircases 1/2/3/4 bars; past the 4th
262 * it clamps to 4, since 5/6/7 are not in the selectable duration set
263 * {1,2,3,4,8,16}. */
264bool sequencer_core_progression_add_entry(void)
265{
266 if (s_prog.count >= CHORD_PROG_MAX_ENTRIES) return false;
267 uint8_t idx = s_prog.count;
268 uint8_t entry_num = (uint8_t)(idx + 1); /* 1-based position */
269 s_prog.entries[idx].root = 0; /* C */
270 s_prog.entries[idx].chord_type = CHORD_MAJ;
271 s_prog.entries[idx].duration_bars = entry_num <= 4 ? entry_num : 4;
272 s_prog.count = entry_num;
273 return true;
274}

276/* Delete entry idx, shifting later entries down. Keeps at least one entry. */
277void sequencer_core_progression_delete_entry(uint8_t idx)
278{
279 if (idx >= s_prog.count || s_prog.count <= 1) return;
280 for (uint8_t i = idx; i + 1 < s_prog.count; i++) {
281 s_prog.entries[i] = s_prog.entries[i + 1];
282 }
283 s_prog.count--;
284 /* Fix the active index/window if it was at or past the deletion point. */
285 bool active_changed = false;
286 if (s_prog.current == idx) {
287 if (s_prog.current >= s_prog.count) s_prog.current = 0;
288 active_changed = true;
289 } else if (s_prog.current > idx) {
290 s_prog.current--; /* same entry, new slot - no audible change */
291 }
292 if (active_changed) {
293 s_prog.entry_start_bar = sequencer_bars_elapsed() + 1;
294 if (s_prog.enabled) s_prog_apply_pending = true;
295 }
