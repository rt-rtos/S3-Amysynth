1/* Bulk layer snapshot support for project save/load.
2 *
3 * The per-field setters in sequencer_core.h each re-emit their own scheduled
4 * steps on every call, and three per-step arrays (step_nudge/step_velocity_adj/
5 * step_ratchet_taper) have no public setters at all. A project loader needs to
6 * overwrite an entire layer's persisted state and settle the AMY side effects
7 * exactly once, so this file copies the layer struct wholesale and then re-runs
8 * the same "configure synth + resync" sequence the layer-management calls use.
9 */
10

16bool sequencer_core_export_layer(uint8_t layer_idx, seq_layer_t *out)
17{
18 if (layer_idx >= s_num_layers || !out) return false;
19 *out = s_layers[layer_idx]; /* RAM-to-RAM copy is fine; only flash
20 serialization must be field-by-field */
21 return true;
22}

24bool sequencer_core_import_layer(uint8_t layer_idx, const seq_layer_t *src)
25{
26 if (layer_idx >= s_num_layers || !src) return false;
27 seq_layer_t *dst = &s_layers[layer_idx];
28 if (dst->type != src->type) return false; /* topology fixed by caller */
29
30 /* Preserve runtime-owned fields. */
31 uint8_t synth_id[SEQ_TRACKS];
32 memcpy(synth_id, dst->synth_id, sizeof synth_id);
33 uint8_t num_tracks = dst->num_tracks;
34
35 *dst = *src;
36 memcpy(dst->synth_id, synth_id, sizeof synth_id);
37 dst->num_tracks = num_tracks;
38 dst->step_page = 0;
39
40 /* Re-push the full sound config to the layer's synth slots. The public
41 * patch setters dedup against the layer's stored patch number - which the
42 * struct copy above has already overwritten - so they would silently
43 * no-op here. Call the configure path directly instead: it reads patch,
44 * num_voices, and synth_flags from the just-copied struct (looping every
45 * drum track's track_patch[] in drum-SYNTH mode) and pushes the authored
46 * env/env1/filter/LFO for melodic layers itself. */
47 sequencer_configure_synth(layer_idx);
48 if (dst->type == SEQ_LAYER_DRUM) {
49 /* Restore each track's PCM preset selection: live-reloads the osc when
50 * the PCM engine is active, otherwise stores the override for the next
51 * engine toggle. */
52 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
53 sequencer_core_set_drum_pcm_preset(layer_idx, t, dst->track_pcm_preset[t]);
54 }
55 }
56
57 sequencer_core_trig_reset_all();
58 sequencer_resync_layer(layer_idx);
59 return true;
60}
