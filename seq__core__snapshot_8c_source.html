1/* Bulk layer snapshot support for project save/load.
2 *
3 * The per-field setters in sequencer_core.h each re-emit their scheduled steps
4 * on every call, and three per-step arrays (step_nudge/step_velocity_adj/
5 * step_ratchet_taper) have no public setters at all. A project loader must
6 * overwrite a whole layer and settle the AMY side effects exactly once, so this
7 * copies the struct wholesale and re-runs the same "configure synth + resync"
8 * sequence the layer-management calls use.
9 */
10

16bool sequencer_core_export_layer(uint8_t layer_idx, seq_layer_t *out)
17{
18 if (layer_idx >= s_num_layers || !out) return false;
19 *out = s_layers[layer_idx]; /* RAM-to-RAM copy is fine; only flash
20 serialization must be field-by-field */
21 /* track_pcm_preset/track_pcm_mode in the core-side struct are UI mirrors
22 * that only the UI's copy ever refreshes - here they hold stale zeros.
23 * Serialize core truth instead, or a saved project resets every drum kit
24 * to preset 0 on load (import_layer replays these through the setters). */
25 if (out->type == SEQ_LAYER_DRUM) {
26 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
27 out->track_pcm_preset[t] = sequencer_core_get_drum_pcm_preset(layer_idx, t);
28 out->track_pcm_mode[t] = sequencer_core_get_drum_pcm_mode(layer_idx, t);
29 }
30 }
31 return true;
32}

34bool sequencer_core_import_layer(uint8_t layer_idx, const seq_layer_t *src)
35{
36 if (layer_idx >= s_num_layers || !src) return false;
37 seq_layer_t *dst = &s_layers[layer_idx];
38 if (dst->type != src->type) return false; /* topology fixed by caller */
39
40 /* Preserve runtime-owned fields. */
41 uint8_t synth_id[SEQ_TRACKS];
42 memcpy(synth_id, dst->synth_id, sizeof synth_id);
43 uint8_t num_tracks = dst->num_tracks;
44
45 *dst = *src;
46 memcpy(dst->synth_id, synth_id, sizeof synth_id);
47 dst->num_tracks = num_tracks;
48 dst->step_page = 0;
49
50 /* Rebuild the per-track source notes from the loaded resolved notes.
51 * s_track_source_note is not persisted, and keeping the pre-load values
52 * makes the first re-resolve after load snap every track back to stale
53 * pitches - and strips a loaded chord sentinel entirely. The resolved note
54 * re-resolves to itself (idempotent snap), so it is a faithful source.
55 * Chord sentinels double as their own source; plain notes also seed the
56 * chord-delete fallback. */
57 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
58 uint8_t n = dst->track_base_note[t];
59 s_track_source_note[layer_idx][t] = n;
60 if (!SEQ_NOTE_IS_CHORD(n)) {
61 s_track_prev_plain[layer_idx][t] = n;
62 }
63 }
64
65 /* Re-push the full sound config. The public patch setters dedup against the
66 * layer's stored patch number, which the struct copy above already
67 * overwrote, so they would silently no-op; the configure path instead reads
68 * patch/num_voices/synth_flags from the copied struct and pushes the
69 * authored env/env1/filter/LFO itself. */
70 sequencer_configure_synth(layer_idx);
71 if (dst->type == SEQ_LAYER_DRUM) {
72 /* Restore each track's PCM mode then preset: mode first so the
73 * preset's live-reload (when PCM is active) applies both in one
74 * pass; otherwise both just store for the next engine toggle. */
75 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
76 sequencer_core_set_drum_pcm_mode(layer_idx, t, dst->track_pcm_mode[t]);
77 sequencer_core_set_drum_pcm_preset(layer_idx, t, dst->track_pcm_preset[t]);
78 }
79 }
80
81 sequencer_core_trig_reset_all();
82 sequencer_resync_layer(layer_idx);
83 return true;
84}
