87uint8_t sequencer_core_add_layer(seq_layer_type_t type, uint8_t num_steps)
88{
89 seq_assert_layers_applier();
90 if (s_num_layers >= MAX_LAYERS) {
91 ESP_LOGW(TAG, "sequencer_core_add_layer: max layers (%d) reached", MAX_LAYERS);
92 return 0xFF;
93 }
94 /* Claim the slot index but do NOT expose it via s_num_layers yet.
95 * The tick path iterates 0..s_num_layers-1; incrementing here would let
96 * the tick see a half-initialised layer. s_num_layers++ is deferred to
97 * after sequencer_configure_synth() completes below. */
98 uint8_t idx = s_num_layers;
99 seq_layer_t *layer = &s_layers[idx];
100 memset(layer, 0, sizeof(seq_layer_t));
101
102 /* Per-row voice params: zeroed/unauthored with amp_trim at unity — the
103 * single defaults source, so no field here needs post-memset repair. */
104 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
105 voice_params_init_defaults(&layer->vp[t]);
106 }
107
108 layer->type = type;
109 layer->num_steps = (num_steps == SEQ_MAX_STEPS) ? SEQ_MAX_STEPS : SEQ_STEPS;
110 layer->num_tracks = SEQ_TRACKS;
111 layer->step_page = 0;
112 /* Melodic NoteFX defaults: gate reproduces the legacy fixed hold, glide off.
113 * memset above zeroed gate_pct — a 0% gate would silence every note. */
114 layer->gate_pct = SEQ_MELODIC_GATE_DEFAULT_PCT;
115 layer->portamento_ms = 0;
116
117 if (type == SEQ_LAYER_DRUM) {
118 /* Drums are now a per-track Juno-patch layer: each track gets its own
119 * fixed synth slot (block 6..9) and its own patch from the curated list,
120 * with note-offs honored (synth_flags = 0) so the patch's own release
121 * envelope shapes the tail. */
122 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
123 layer->synth_id[t] = (uint8_t)(SEQ_DRUM_SYNTH_BASE + t);
124 /* Role-based default patch per track (kick/snare/hat/perc). */
125 layer->track_patch[t] = SEQ_DRUM_DEFAULT_PATCH[t];
126 }
127 layer->patch = layer->track_patch[0]; /* display fallback */
128 layer->synth_flags = 0;
129 layer->num_voices = SEQ_DRUM_VOICES;
130 /* Role-based pitches: pitch IS timbre for these tuned patches AND tunes
131 * the 808 samples in PCM mode (render_pcm shifts by midi_note). Low kick
132 * body, mid snare, high hat tick, mid-high perc; stays user-editable. */
133 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
134 s_track_source_note[idx][t] = SEQ_DRUM_DEFAULT_NOTE[t];
135 layer->track_base_note[t] = SEQ_DRUM_DEFAULT_NOTE[t];
136 for (uint8_t s = 0; s < SEQ_MAX_STEPS; s++) {
137 layer->step_note[t][s] = SEQ_DRUM_DEFAULT_NOTE[t];
138 }
139 }
140 } else {
141 /* Melodic: claim a contiguous block of SEQ_TRACKS synth slots from the
142 * running allocator, one synth per row, so identical pitches on
143 * different rows land in distinct instruments (no voice collapse).
144 * Guard against exceeding AMY's synth ceiling; if we would, reuse the
145 * last valid block (degrades to shared-synth rather than corruption). */
146 uint8_t base = s_next_melodic_synth;
147 if (base + SEQ_TRACKS - 1 > SEQ_MAX_SYNTH) {
148 base = (uint8_t)(SEQ_MAX_SYNTH - (SEQ_TRACKS - 1));
149 ESP_LOGW(TAG, "add_layer[L%d]: melodic synth ceiling reached, "
150 "reusing slots %u..%u", idx + 1, base, base + SEQ_TRACKS - 1);
151 } else {
152 s_next_melodic_synth = (uint8_t)(base + SEQ_TRACKS);
153 }
154 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
155 layer->synth_id[t] = (uint8_t)(base + t);
156 }
157 layer->patch = s_melodic_patch;
158 layer->synth_flags = 0;
159 layer->num_voices = SEQ_MEL_VOICES;
160 /* Default: Cmaj7 voicing — C4 E4 G4 B4 */
161 static const uint8_t mel_notes[SEQ_TRACKS] = {60, 64, 67, 71};
162 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
163 s_track_source_note[idx][t] = mel_notes[t];
164 layer->track_base_note[t] = mel_notes[t];
165 for (uint8_t s = 0; s < SEQ_MAX_STEPS; s++) {
166 layer->step_note[t][s] = mel_notes[t];
167 }
168 layer->vp[t].env = seq_default_melodic_env();
169 layer->vp[t].env1 = seq_default_melodic_env1();
170 }
171 }
172
173 /* step_prob (0% would silence every step) and step_ratchet (0 sub-hits
174 * would fire nothing) need explicit non-zero defaults; step_cond_type/
175 * param and the per-step transform/quant-bypass arrays default correctly
176 * to their zeroed no-op (SEQ_STEP_COND_NONE / SEQ_STEP_TRANSFORM_NONE /
177 * no bypass) and need no explicit init. The per-row amp trim got its
178 * unity default in voice_params_init_defaults() above. */
179 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
180 for (uint8_t s = 0; s < SEQ_MAX_STEPS; s++) {
181 layer->step_prob[t][s] = 100;
182 layer->step_ratchet[t][s] = 1;
183 }
184 }
185
186 CORE_HEAP_CHECK("add_layer: before configure_synth");
187 sequencer_configure_synth(idx);
188 CORE_HEAP_CHECK("add_layer: after configure_synth");
189 /* Layer is fully initialised: now expose it to the tick path. */
190 s_num_layers++;
191 /* Construction-time check of the permanent-drum-layer invariant: slot 0
192 * is the drum layer (created first at boot) and every later add lands
193 * above it. Verified once here rather than trusted throughout. */
194 assert(s_layers[SEQ_DRUM_LAYER_IDX].type == SEQ_LAYER_DRUM);
195 /* Runtime trig bookkeeping (loop counters / last-played / last-step-seen)
196 * is indexed by layer slot, not layer identity; a wholesale reset here
197 * (rather than trying to shift it in lockstep with the new layer) is
198 * cheap and harmless — it only affects FILL/PREV continuity, never the
199 * persisted per-step prob/ratchet/cond data carried in seq_layer_t itself. */
200 sequencer_core_trig_reset_all();
201 ESP_LOGI(TAG, "add_layer[L%d]: type=%d synth0=%d patch=%d steps=%d",
202 idx + 1, type, layer->synth_id[0], layer->patch, layer->num_steps);
203 return idx;
204}

206bool sequencer_core_delete_layer(uint8_t layer_idx)
207{
208 seq_assert_layers_applier();
209 if (s_num_layers <= 1) return false; /* must keep at least 1 */
210 if (layer_idx == SEQ_DRUM_LAYER_IDX) return false; /* drum layer is permanent */
211 if (layer_idx >= s_num_layers) return false;
212 if (s_layers[layer_idx].type == SEQ_LAYER_DRUM) return false;
213
214 /* Clear ALL layers' tags before shifting — indices above layer_idx become
215 * stale after compaction and would fire as ghost notes. Ratchet one-shot
216 * tags live in a separate tag space (seq_core_trig.c) and need the same
217 * treatment. */
218 for (uint8_t i = 0; i < s_num_layers; i++) {
219 sequencer_clear_layer_tags(i);
220 sequencer_core_trig_clear_all(i);
221 }
222
223 /* Release AMY oscillator slots for the deleted layer. */
224 const seq_layer_t *dead = &s_layers[layer_idx];
225 for (uint8_t t = 0; t < dead->num_tracks; t++) {
226 amy_event *e = amy_helpers_event_begin();
227 e->synth = dead->synth_id[t];
228 e->num_voices = 0;
229 amy_helpers_event_send(e);
230 }
231
232 /* Raise the mutation guard so the Core-0 sequencer tick early-returns
233 * instead of reading s_layers[] mid-memmove. The tick tolerates skipped
234 * ticks (trig state is reset wholesale below), so this is inaudible. */
235 s_layers_mutating = true;
236 /* Compact all parallel arrays by shifting survivors down by one slot. */
237 uint8_t tail = (uint8_t)(s_num_layers - layer_idx - 1);
238 if (tail > 0) {
239 memmove(&s_layers[layer_idx],
240 &s_layers[layer_idx + 1],
241 tail * sizeof(s_layers[0]));
242 memmove(&s_cached_step[layer_idx],
243 &s_cached_step[layer_idx + 1],
244 tail * sizeof(s_cached_step[0]));
245 memmove(&s_track_source_note[layer_idx],
246 &s_track_source_note[layer_idx + 1],
247 tail * sizeof(s_track_source_note[0]));
248 memmove(&s_lfo_phase[layer_idx],
249 &s_lfo_phase[layer_idx + 1],
250 tail * sizeof(s_lfo_phase[0]));
251 memmove(&s_lfo_hz[layer_idx],
252 &s_lfo_hz[layer_idx + 1],
253 tail * sizeof(s_lfo_hz[0]));
254 memmove(&s_lfo_rnd[layer_idx],
255 &s_lfo_rnd[layer_idx + 1],
256 tail * sizeof(s_lfo_rnd[0]));
257 }
258 s_num_layers--;
259 /* Table is fully compacted and the count updated; drop the guard. */
260 s_layers_mutating = false;
261 sequencer_core_trig_reset_all(); /* see rationale in sequencer_core_add_layer() */
262
263 /* Resync all surviving layers so their note tags re-register correctly. */
264 if (s_playing) {
265 for (uint8_t i = 0; i < s_num_layers; i++) {
266 sequencer_resync_layer(i);
267 }
268 }
269
270 ESP_LOGI(TAG, "delete_layer[L%u]: %u layers remain", layer_idx + 1u, s_num_layers);
271 return true;
272}
