105uint8_t sequencer_core_add_layer(seq_layer_type_t type, uint8_t num_steps)
106{
107 seq_assert_layers_applier();
108 if (s_num_layers >= MAX_LAYERS) {
109 ESP_LOGW(TAG, "sequencer_core_add_layer: max layers (%d) reached", MAX_LAYERS);
110 return 0xFF;
111 }
112 /* Claim the slot but do NOT expose it via s_num_layers yet: the tick
113 * iterates 0..s_num_layers-1 and would see a half-initialised layer.
114 * s_num_layers++ waits until sequencer_configure_synth() completes. */
115 uint8_t idx = s_num_layers;
116 seq_layer_t *layer = &s_layers[idx];
117 memset(layer, 0, sizeof(seq_layer_t));
118
119 /* Per-row voice params: unauthored with amp_trim at unity. The single
120 * defaults source, so no field here needs post-memset repair. */
121 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
122 voice_params_init_defaults(&layer->vp[t]);
123 }
124 voice_params_init_defaults(&layer->vp_layer);
125 /* vp_src[] is TRACK (0) from the memset: every row reads its own block. */
126
127 layer->type = type;
128 layer->num_steps = (num_steps == SEQ_MAX_STEPS) ? SEQ_MAX_STEPS : SEQ_STEPS;
129 layer->num_tracks = SEQ_TRACKS;
130 layer->step_page = 0;
131 /* NoteFX defaults. Required after the memset: a 0% gate would silence
132 * every note and a 0% groove would flatten dynamics. */
133 layer->gate_pct = SEQ_MELODIC_GATE_DEFAULT_PCT;
134 layer->portamento_ms = 0;
135 layer->groove_pct = 100; /* full accent curve */
136 /* Required after the memset too: 0 is a real FM algorithm. */
137 layer->fm_algo_override = SEQ_FM_ALGO_NONE;
138
139 if (type == SEQ_LAYER_DRUM) {
140 /* Per-track drum layer: each track gets a fixed synth slot from the
141 * reserved block and its own patch from the curated list, with
142 * note-offs honored (synth_flags = 0) so the patch's release shapes
143 * the tail. */
144 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
145 layer->synth_id[t] = (uint8_t)(SEQ_DRUM_SYNTH_BASE + t);
146 layer->track_patch[t] = SEQ_DRUM_DEFAULT_PATCH[t];
147 }
148 layer->patch = layer->track_patch[0]; /* display fallback */
149 layer->synth_flags = 0;
150 layer->num_voices = SEQ_DRUM_VOICES;
151 /* Role-based pitches: pitch IS timbre for these tuned patches, and also
152 * tunes the samples in PCM mode (render_pcm shifts by midi_note).
153 * Seeded from the boot bank's ear-tuned notes[], the same source the
154 * bank selector re-seeds from. */
155 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
156 uint8_t n = sequencer_drum_default_note(t);
157 s_track_source_note[idx][t] = n;
158 s_track_prev_plain[idx][t] = n;
159 layer->track_base_note[t] = n;
160 for (uint8_t s = 0; s < SEQ_MAX_STEPS; s++) {
161 layer->step_note[t][s] = n;
162 }
163 }
164 } else {
165 /* Melodic: claim a contiguous block of SEQ_TRACKS slots from the running
166 * allocator, one synth per row, so identical pitches on different rows
167 * land in distinct instruments instead of collapsing into one voice.
168 * Past AMY's synth ceiling, reuse the last valid block - shared synths
169 * degrade the sound but never corrupt state. */
170 uint8_t base = s_next_melodic_synth;
171 if (base + SEQ_TRACKS - 1 > SEQ_MAX_SYNTH) {
172 base = (uint8_t)(SEQ_MAX_SYNTH - (SEQ_TRACKS - 1));
173 ESP_LOGW(TAG, "add_layer[L%d]: melodic synth ceiling reached, "
174 "reusing slots %u..%u", idx + 1, base, base + SEQ_TRACKS - 1);
175 } else {
176 s_next_melodic_synth = (uint8_t)(base + SEQ_TRACKS);
177 }
178 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
179 layer->synth_id[t] = (uint8_t)(base + t);
180 }
181 layer->patch = s_melodic_patch;
182 layer->synth_flags = 0;
183 layer->num_voices = SEQ_MEL_VOICES;
184 /* Default: Cmaj7 voicing - C4 E4 G4 B4 */
185 static const uint8_t mel_notes[SEQ_TRACKS] = {60, 64, 67, 71};
186 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
187 s_track_source_note[idx][t] = mel_notes[t];
188 s_track_prev_plain[idx][t] = mel_notes[t];
189 layer->track_base_note[t] = mel_notes[t];
190 for (uint8_t s = 0; s < SEQ_MAX_STEPS; s++) {
191 layer->step_note[t][s] = mel_notes[t];
192 }
193 layer->vp[t].env = seq_default_melodic_env();
194 layer->vp[t].env1 = seq_default_melodic_env1();
195 }
196 layer->vp_layer.env = seq_default_melodic_env();
197 layer->vp_layer.env1 = seq_default_melodic_env1();
198 }
199
200 /* step_prob (0% silences every step), step_ratchet (0 sub-hits fire
201 * nothing) and step_every (canonical neutral is 1, though the trig engine
202 * treats 0 the same defensively) need explicit non-zero defaults. The
203 * prev and transform arrays are correct at their zeroed no-op. */
204 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
205 for (uint8_t s = 0; s < SEQ_MAX_STEPS; s++) {
206 layer->step_prob[t][s] = 100;
207 layer->step_ratchet[t][s] = 1;
208 layer->step_every[t][s] = 1;
209 }
210 }
211
212 DIAG_HEAP_CHECK("add_layer: before configure_synth");
213 sequencer_configure_synth(idx);
214 DIAG_HEAP_CHECK("add_layer: after configure_synth");
215 /* Fully initialised: now expose it to the tick path. */
216 s_num_layers++;
217 /* Check the permanent-drum-layer invariant here once rather than trusting
218 * it throughout: slot 0 is the drum layer and every later add is above it. */
219 assert(s_layers[SEQ_DRUM_LAYER_IDX].type == SEQ_LAYER_DRUM);
220 /* Runtime trig bookkeeping is indexed by layer SLOT, not identity, so a
221 * wholesale reset is simpler than shifting it in lockstep and costs only
222 * FILL/PREV continuity - never the persisted per-step data in
223 * seq_layer_t. */
224 sequencer_core_trig_reset_all();
225 ESP_LOGI(TAG, "add_layer[L%d]: type=%d synth0=%d patch=%d steps=%d",
226 idx + 1, type, layer->synth_id[0], layer->patch, layer->num_steps);
227 return idx;
228}

230bool sequencer_core_delete_layer(uint8_t layer_idx)
231{
232 seq_assert_layers_applier();
233 if (s_num_layers <= 1) return false; /* must keep at least 1 */
234 if (layer_idx == SEQ_DRUM_LAYER_IDX) return false; /* drum layer is permanent */
235 if (layer_idx >= s_num_layers) return false;
236 if (s_layers[layer_idx].type == SEQ_LAYER_DRUM) return false;
237
238 /* Clear ALL layers' tags before shifting: indices above layer_idx go stale
239 * on compaction and would fire as ghost notes. Ratchet one-shot tags live
240 * in a separate tag space and need the same treatment. */
241 for (uint8_t i = 0; i < s_num_layers; i++) {
242 sequencer_clear_layer_tags(i);
243 sequencer_core_trig_clear_all(i);
244 }
245
246 /* Release AMY oscillator slots for the deleted layer. */
247 const seq_layer_t *dead = &s_layers[layer_idx];
248 for (uint8_t t = 0; t < dead->num_tracks; t++) {
249 amy_event *e = amy_helpers_event_begin();
250 e->synth = dead->synth_id[t];
251 e->num_voices = 0;
252 amy_helpers_event_send(e);
253 }
254
255 /* Raise the mutation guard so the sequencer tick early-returns instead of
256 * reading s_layers[] mid-memmove. Skipped ticks are inaudible - trig state
257 * is reset wholesale below. */
258 s_layers_mutating = true;
259 /* Compact all parallel arrays by shifting survivors down by one slot. */
260 uint8_t tail = (uint8_t)(s_num_layers - layer_idx - 1);
261 if (tail > 0) {
262 memmove(&s_layers[layer_idx],
263 &s_layers[layer_idx + 1],
264 tail * sizeof(s_layers[0]));
265 memmove(&s_cached_step[layer_idx],
266 &s_cached_step[layer_idx + 1],
267 tail * sizeof(s_cached_step[0]));
268 memmove(&s_track_source_note[layer_idx],
269 &s_track_source_note[layer_idx + 1],
270 tail * sizeof(s_track_source_note[0]));
271 memmove(&s_track_prev_plain[layer_idx],
272 &s_track_prev_plain[layer_idx + 1],
273 tail * sizeof(s_track_prev_plain[0]));
274 memmove(&s_voices_applied[layer_idx],
275 &s_voices_applied[layer_idx + 1],
276 tail * sizeof(s_voices_applied[0]));
277 memmove(&s_lfo_phase[layer_idx],
278 &s_lfo_phase[layer_idx + 1],
279 tail * sizeof(s_lfo_phase[0]));
280 memmove(&s_lfo_hz[layer_idx],
281 &s_lfo_hz[layer_idx + 1],
282 tail * sizeof(s_lfo_hz[0]));
283 memmove(&s_lfo_rnd[layer_idx],
284 &s_lfo_rnd[layer_idx + 1],
285 tail * sizeof(s_lfo_rnd[0]));
286 }
287 s_num_layers--;
288 /* Only the permanent drum layer left: every melodic slot is free again
289 * (AMY released each deleted layer's voices above), so rewind the bump
290 * allocator. Without this, project loads - which delete down to the drum
291 * layer and re-add - creep the counter 4 slots per melodic layer per load
292 * until the arena ceiling's shared-slot degrade kicks in. */
293 if (s_num_layers == 1) s_next_melodic_synth = SEQ_MEL_SYNTH_BASE;
294 /* Table is fully compacted and the count updated; drop the guard. */
295 s_layers_mutating = false;
296 sequencer_core_trig_reset_all(); /* see rationale in sequencer_core_add_layer() */
297
298 /* Resync all surviving layers so their note tags re-register correctly. */
299 if (s_playing) {
300 for (uint8_t i = 0; i < s_num_layers; i++) {
301 sequencer_resync_layer(i);
302 }
303 }
304
305 ESP_LOGI(TAG, "delete_layer[L%u]: %u layers remain", layer_idx + 1u, s_num_layers);
306 return true;
307}
