129/* Apply per-track envelope shape and hat HPF after PCM wave/preset are set. */
130static void sequencer_configure_drum_pcm_voice_params(uint8_t layer_idx)
131{
132 const seq_layer_t *layer = &s_layers[layer_idx];
133 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
134 amy_event *e = amy_helpers_event_begin();
135 e->synth = layer->synth_id[t];
136 e->bp_is_set[0] = 1;
137 e->eg_type[0] = ENVELOPE_LINEAR;
138 e->eg0_times[0] = DRUM_PCM_ATK_MS[t];
139 e->eg0_values[0] = 1.0f;
140 e->eg0_times[1] = DRUM_PCM_DEC_MS[t];
141 e->eg0_values[1] = 0.0f; /* one-shot: no sustain, sample shapes the body */
142 e->eg0_times[2] = DRUM_PCM_REL_MS[t];
143 e->eg0_values[2] = 0.0f;
144 amy_helpers_event_send(e);
145
146 if (t == 2) { /* hat: HPF to strip low-end rumble, add crispness */
147 e = amy_helpers_event_begin();
148 e->synth = layer->synth_id[t];
149 e->filter_type = FILTER_HPF;
150 e->filter_freq_coefs[COEF_CONST] = 3000.0f;
151 e->resonance = 0.5f;
152 amy_helpers_event_send(e);
153 }
154 }

165/* Configure a single melodic synth slot as a bare AMY oscillator.
166 * Mirrors arp_configure_wave_synth() but for melodic tracks (SEQ_MEL_VOICES
167 * voices per synth slot). Envelope, filter, and LFO are applied by the caller.
168 *
169 * In native LFO mode oscs_per_voice=2 is always allocated (even before an LFO
170 * is authored) so that toggling the LFO later never forces a pool resize.
171 * Osc 1 starts dormant (amp=0); sequencer_configure_melodic_lfo() activates it
172 * when an LFO is authored. */
173static void sequencer_configure_melodic_wave_track(uint8_t synth_id,
174 uint16_t patch,
175 uint16_t num_voices,
176 bool filter_authored,
177 float filter_q)
178{
179 static const uint16_t s_wave_for_patch[] = {
180 SINE, SAW_DOWN, SAW_UP, PULSE, TRIANGLE, NOISE, KS,
181 };
182#if CONFIG_AMY_WAVETABLE
183 bool is_wavetable = (patch >= SEQ_PATCH_WAVETABLE_BASE && patch <= SEQ_PATCH_WAVETABLE_MAX);
184 uint16_t wave = WAVETABLE;
185 uint16_t wt_preset = pcm_wavetable_base + (uint16_t)(patch - SEQ_PATCH_WAVETABLE_BASE);
186 if (!is_wavetable) {
187 uint16_t widx = (uint16_t)(patch - SEQ_PATCH_WAVE_BASE);
188 if (widx >= (uint16_t)(sizeof(s_wave_for_patch) / sizeof(s_wave_for_patch[0])))
189 widx = 0;
190 wave = s_wave_for_patch[widx];
191 }
192#else
193 uint16_t widx = (uint16_t)(patch - SEQ_PATCH_WAVE_BASE);
194 if (widx >= (uint16_t)(sizeof(s_wave_for_patch) / sizeof(s_wave_for_patch[0])))
195 widx = 0;
196 uint16_t wave = s_wave_for_patch[widx];
197#endif
198
199 voice_wave_cfg_t cfg = {
200 .synth = synth_id,
201 .num_voices = (uint8_t)num_voices,
202#if CONFIG_SEQ_MELODIC_AMY_NATIVE_LFO
203 .oscs_per_voice = 2, /* osc 1 reserved as native LFO carrier */
204#else
205 .oscs_per_voice = 1,
206#endif
207 .wave = wave,
208 .osc0_amp_const = 1.0f,
209 .osc0_amp_vel = 1.0f,
210 .ks_feedback_authored = filter_authored,
211 .ks_feedback_q = filter_q,
212#if CONFIG_AMY_WAVETABLE
213 .wt_preset = is_wavetable ? (int16_t)wt_preset : -1,
214#else
215 .wt_preset = -1,
216#endif
217 };
218 voice_build_wave(&cfg);
219
220#if CONFIG_SEQ_MELODIC_AMY_NATIVE_LFO
221 /* osc 1: LFO carrier slot — starts dormant until an LFO is authored */
222 amy_event *e = amy_helpers_event_begin();
223 e->synth = synth_id;
224 e->osc = 1;
225 e->amp_coefs[COEF_CONST] = 0.0f; /* dormant */
226 amy_helpers_event_send(e);
227#endif
228}

230/* Push each AUTHORED row's stored envelope to its own per-row synth.
231 * Deferred authority: a row's envelope only overrides the patch's own envelope
232 * once the user has committed it in the graph editor (env_authored[t]==true).
233 * Unauthored rows are left alone so the freshly-loaded patch envelope plays. */
234static void sequencer_configure_melodic_envelope(uint8_t layer_idx)
235{
236 const seq_layer_t *layer = &s_layers[layer_idx];
237 /* Raw-wave primitives (SINE..KS, wavetables) carry no patch envelope of
238 * their own. With no envelope pushed, the carrier's COEF_EG0 stays a
239 * constant 1.0: AMY reads an empty breakpoint set as a permanently open
240 * gate (envelope.c) and a velocity-0 note-off never zeroes it (amy.c), so
241 * the oscillator rings forever — surviving both patch changes and pause,
242 * which both silence voices via that same velocity-0 note-off. Force the
243 * default melodic envelope onto every such unauthored row so a note-off
244 * actually releases it. (KS/NOISE additionally get their sustain floor
245 * applied downstream in sequencer_configure_melodic_envelope_track.) */
246 bool force_wave = sequencer_core_is_wave_patch(layer->patch);
247 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
248 if (layer->vp[t].env_authored || force_wave) {
249 sequencer_configure_melodic_envelope_track(layer_idx, t);
250 }
251 }
252}

254/* Push each AUTHORED row's stored second envelope (EG1) to its own synth.
255 * No "force" case (unlike EG0/KS above): EG1 has no raw-wave fallback role,
256 * it only matters once a row has actually been authored. */
257static void sequencer_configure_melodic_envelope1(uint8_t layer_idx)

262 sequencer_configure_melodic_envelope1_track(layer_idx, t);
263 }
264 }
265}

267/* Push the filter for every authored row in a layer (called after patch reload).
268 *
269 * Unauthored rows normally keep whatever filter the just-loaded patch string
270 * baked in (built-in Juno/DX7 patches carry a G/F/R filter block, so a fresh
271 * voice inherits that patch's LPF). With CONFIG_SEQ_MELODIC_DISABLE_DEFAULT_LPF
272 * set, those unauthored rows instead get a FILTER_NONE event so the raw patch
273 * tone is heard; authored rows always take their stored filter. */
274static void sequencer_configure_melodic_filter(uint8_t layer_idx)
275{
276 const seq_layer_t *layer = &s_layers[layer_idx];
277 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
278 if (layer->vp[t].filter_authored) {
279 sequencer_configure_melodic_filter_track(layer_idx, t);

285 e->filter_type = FILTER_NONE;
286 amy_helpers_event_send(e);
287#endif
288 }
289 }
290}

318/* Apply one routable patch (0..SEQ_PATCH_FULL_MAX) to a synth slot,
319 * dispatching per kind — the SINGLE dispatch for every consumer (melodic
320 * tracks and the arp; the drone has its own excitation model):
321 * raw wave / wavetable → direct oscillator config (no patch string)
322 * bass preset (264-266) → bass_preset_configure_track (oscs_per_voice=2)
323 * FM/ALGO (272-276) → fm preset / live-editable custom voice (7 oscs)
324 * additive (277-279) → additive preset / custom voice (N+1 oscs)
325 * everything else → amy_send_patch() string loader (Juno/DX7/piano)
326 * Returns true when a patch STRING was loaded: those carry global EQ/chorus
327 * commands, so the caller must synth_ui_fx_reassert_global() afterwards
328 * (once, even when applying to several slots). */
329static bool sequencer_apply_patch_kind(uint8_t synth_id, uint16_t patch,
330 uint16_t num_voices, uint32_t synth_flags,
331 bool filter_authored, float filter_q)
332{
333 /* Virtual patch number whose feature is not in this build (browse skips
334 * these, but stored/programmatic values can still arrive): snap to raw
335 * SINE rather than letting a virtual number reach the string loader. */
336 if (sequencer_core_patch_compiled_out(patch)) {
337 sequencer_configure_melodic_wave_track(synth_id, SEQ_PATCH_SINE,
338 num_voices, filter_authored,
339 filter_q);
340 return false;
341 }
342 if (sequencer_core_is_wave_patch(patch)) {
343 sequencer_configure_melodic_wave_track(synth_id, patch, num_voices,
344 filter_authored, filter_q);
345 return false;
346 }
347 if (patch >= SEQ_PATCH_BASS_BASE && patch <= SEQ_PATCH_BASS_MAX) {
348 bass_preset_configure_track(synth_id, patch, num_voices);
349 return false;
350 }
351#if CONFIG_SYNTH_CUSTOM_FM
352 if (patch >= SEQ_PATCH_FM_BASE && patch <= SEQ_PATCH_FM_MAX) {
353 if (patch == SEQ_PATCH_FM_CUSTOM) {
354 fm_voice_configure_track(synth_id, num_voices, &s_fm_voice);
355 } else {
356 fm_preset_configure_track(synth_id, patch, num_voices);
357 }
358 return false;
359 }
360#endif
361#if CONFIG_SYNTH_ADDITIVE
362 if (patch >= SEQ_PATCH_ADDITIVE_BASE && patch <= SEQ_PATCH_ADDITIVE_MAX) {
363 if (patch == SEQ_PATCH_ADDITIVE_CUSTOM) {
364 additive_voice_configure_track(synth_id, num_voices, &s_additive_voice);
365 } else {
366 additive_preset_configure_track(synth_id, patch, num_voices);
367 }
368 return false;
369 }
370#endif
371 amy_send_patch(synth_id, patch, num_voices, synth_flags);
372 return true;
373}

387/* Reassert global FX iff any patch applied since the last flush was a patch
388 * STRING (Juno/DX7/piano): those carry global EQ/chorus commands that
389 * overwrite the user's FX state on load. Raw-wave, bass, and FM patches carry
390 * none and owe nothing. Idempotent; safe to call with owed == false. Every

397/* (Re)configure the AMY synth(s) for layer_idx.
398 * Drums use a single synth (synth_id[0]); melodic layers configure one synth
399 * per row, all sharing the same patch/flags/voice-count but on distinct slots. */
400void sequencer_configure_synth(uint8_t layer_idx)
401{
402 seq_layer_t *layer = &s_layers[layer_idx];
403
404 if (layer->type == SEQ_LAYER_DRUM) {
405 if (s_drum_engine == SEQ_DRUM_PCM) {
406 /* PCM mode: each track's synth slot becomes a 1-osc PCM player loaded
407 * with a built-in 808 sample. We allocate the voice with
408 * oscs_per_voice=1 (no patch string), then set wave=PCM + preset on
409 * osc 0. Note-on/off + velocity + pitch flow through the SAME emit
410 * path as synth mode, so hits get accent/jitter dynamics and the
411 * sample is tuned by midi_note (render_pcm) — not the old clinical
412 * fixed-velocity drumkit path. PCM carries no global EQ/chorus, so no
413 * reassert needed here. */
414 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
415 /* Allocate/realloc the slot as a 1-osc voice (clears old patch). */
416 amy_event *e = amy_helpers_event_begin();
417 e->num_voices = layer->num_voices;
418 e->oscs_per_voice = 1;
419 e->synth = layer->synth_id[t];
420 e->synth_flags = layer->synth_flags; /* 0 */
421 amy_helpers_event_send(e);
422
423 /* Configure osc 0 of this synth as the chosen 808 PCM sample. */
424 e = amy_helpers_event_begin();
425 e->synth = layer->synth_id[t];
426 e->osc = 0;
427 e->wave = PCM;
428 e->preset = drum_pcm_preset_for(layer_idx, t);
429 amy_helpers_event_send(e);
430 }
431 sequencer_configure_drum_pcm_voice_params(layer_idx);
432 return;
433 }
434
435 /* SYNTH mode — per-track: each drum row loads its OWN patch onto its OWN
436 * synth slot, note-offs honored (flags = 0). Mirrors the melodic loop. */
437 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
438 sequencer_kill_synth_voices(layer->synth_id[t]);
439 amy_send_patch(layer->synth_id[t], layer->track_patch[t],
440 layer->num_voices, layer->synth_flags);
441 }
442 /* Every drum SYNTH slot loads a patch string, so a flush is always
443 * owed here (once, after the loop). */
444 seq_flush_patch_fx(true);
445 return;
446 }
447
448 /* Melodic: push the shared patch/flags/voices to each row's own synth.
449 * Kind dispatch (raw wave / bass / FM / patch string) lives in
450 * sequencer_apply_patch_kind() above, shared with the arp. */
451 bool string_patch = false;
452 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
453 sequencer_kill_synth_voices(layer->synth_id[t]);
454 string_patch |= seq_apply_patch(layer->synth_id[t],
455 layer->patch,
456 layer->num_voices,
457 layer->synth_flags,
458 layer->vp[t].filter_authored,
459 layer->vp[t].filter.resonance);
460 }
461 seq_flush_patch_fx(string_patch);
462 sequencer_configure_melodic_envelope(layer_idx);
463 sequencer_configure_melodic_envelope1(layer_idx);
464 sequencer_configure_melodic_filter(layer_idx);
465 sequencer_configure_melodic_lfo(layer_idx);
466 /* Glide is an AMY per-osc setting that a voice rebuild clears, so reassert
467 * it here after the pool/patch has been (re)built. */
468 sequencer_core_push_melodic_portamento(layer_idx);
469}

473void sequencer_core_set_melodic_patch(uint16_t patch_number)
474{
475 /* 0..127: Juno, 128..255: DX7, 256: built-in piano.
476 * 257..263: raw-waveform virtual patches (SEQ_PATCH_SINE..SEQ_PATCH_WAVE_MAX).
477 * 264..266: multi-osc bass presets (SEQ_PATCH_BASS_BASE..SEQ_PATCH_BASS_MAX).

694uint16_t sequencer_core_cycle_drum_pcm_preset(uint8_t layer_idx, uint8_t track,
695 int dir)
696{
697 if (layer_idx >= s_num_layers || track >= SEQ_TRACKS) return 0;
698 if (s_layers[layer_idx].type != SEQ_LAYER_DRUM) return 0;
699
700 /* Combined drum sample domain: the ROM bank (0 .. pcm_wavetable_base-1;
701 * pcm_wavetable_base is the first non-drum map entry — gamma808: 19,
702 * pcm_tiny: 11) followed by the gamma9001 banks (256..391) when their
703 * blob is mounted. The two ranges are walked as one wrapping list via a
704 * linear index. Memory presets (sample_rec overrides, outside both
705 * ranges) step back into the domain at the near end. */
706 uint16_t bound = pcm_wavetable_base;
707 if (bound == 0) return 0;
708 uint16_t total = bound + (drum_gamma_available() ? SEQ_GAMMA_PCM_COUNT : 0);
709
710 uint16_t cur = drum_pcm_preset_for(layer_idx, track);
711 int li_cur;
712 if (cur < bound) {
713 li_cur = (int)cur;
714 } else if (cur >= SEQ_GAMMA_PCM_FIRST &&
715 cur < SEQ_GAMMA_PCM_FIRST + SEQ_GAMMA_PCM_COUNT &&
716 total > bound) {
717 li_cur = (int)bound + (int)(cur - SEQ_GAMMA_PCM_FIRST);
718 } else {
719 /* Memory preset (or unreachable gamma preset): enter at the near end. */
720 li_cur = (dir > 0) ? -1 : 0;
721 }
722
723 int li_next = (li_cur + (dir > 0 ? 1 : -1) + (int)total) % (int)total;
724 uint16_t next = (li_next < (int)bound)
725 ? (uint16_t)li_next
726 : (uint16_t)(SEQ_GAMMA_PCM_FIRST + (uint16_t)(li_next - (int)bound));
727 sequencer_core_set_drum_pcm_preset(layer_idx, track, next);
728 return next;
729}
