44/* ── Drum sample banks (menu "Drum Bank" selector) ──
45 * One row per selectable PCM bank: the compiled-in 808 ROM bank plus the
46 * gamma9001 banks streamed from the 'drums' flash partition. first/count are
47 * PCM preset ranges; roles[] and notes[] seed the four tracks on selection -
48 * a bank pick is a full kit change, sounds AND pitches. Preset numbers
49 * mirror the vendored amy/src/pcm_gamma9001.h map (256=909BD .. 391=Narrow) -
50 * re-verify after any AMY re-vendor. Banks are contiguous in preset space, so
51 * per-track cycling after a seed stays inside the bank until the user walks
52 * out (cycled presets keep the role's note).
53 * notes[] values are per-bank ear-tuned defaults; until a bank has had its
54 * tuning pass they carry the legacy role defaults (39/45/53/82, the
55 * pre-tuning 808 pitches). Editing a kit = touching one row: swap a
56 * role's preset number, note, and/or voice-param blocks here.
57 * eg0/eg1/flt are OPTIONAL per-role voice-param defaults (point rows at
58 * named static const blocks; NULL = the transparent defaults: sample-
59 * natural envelope, no EG1, no filter). Rows the user has NOT authored
60 * get their editable vp model SEEDED from these on every (re)configure,
61 * then the vp is pushed through the same appliers the graph editors use -
62 * so the editors open on the curve that is actually sounding, and a first
63 * touch tweaks from reality instead of a zeroed model. Authored vp wins
64 * and is never re-seeded, same deferred-authority rule as melodic (where
65 * the unauthored model stays zeroed because a patch string's baked
66 * envelope is unreadable; drum defaults are our own data).
67 * The bank is derived from the track's CURRENT preset (not the display
68 * selector), so kit defaults follow cycling within a bank and survive
69 * project reload; presets outside every bank range (e.g. the runtime-
70 * recorded sample slot) fall back to legacy defaults. */

210/* Apply one track's voice params after PCM wave/preset are set.
211 *
212 * Seed-then-push: unauthored blocks of the row's editable vp model are first
213 * seeded from the bank's tuning blocks (or the transparent defaults when the
214 * bank authors none), then the whole model is pushed through the same
215 * appliers the graph editors use. One store sounds AND displays; bank blocks
216 * get full apply fidelity (COEF_EG1 filter-env routing plus the EG1
217 * breakpoint guarantee in melodic_filter_apply). Authored blocks are never
218 * re-seeded, so user edits survive kit changes - the melodic deferred-
219 * authority rule, minus the zeroed unauthored model melodic is stuck with. */
220static void sequencer_configure_drum_pcm_track_params(uint8_t layer_idx,
221 uint8_t track)
222{
223 seq_layer_t *layer = &s_layers[layer_idx];
224 voice_params_t *vp = &layer->vp[track];
225 const seq_drum_bank_t *bank =
226 drum_bank_for_preset(drum_pcm_preset_for(layer_idx, track));
227
228 if (!vp->env_authored) {
229 vp->env = (bank && bank->eg0[track]) ? *bank->eg0[track]
230 : s_drum_env_transparent;
231 }
232 if (!vp->env1_authored) {
233 vp->env1 = (bank && bank->eg1[track]) ? *bank->eg1[track]
234 : (seq_env_t){0};
235 }
236 if (!vp->filter_authored) {
237 vp->filter = (bank && bank->flt[track]) ? *bank->flt[track]
238 : (seq_filter_t){0}; /* bypass */
239 }
240
241 sequencer_configure_melodic_envelope_track(layer_idx, track);
242 /* EG1 is pushed only when a shape exists (authored or bank-seeded): with
243 * neither, the preset reload's osc reset leaves it cleared, and
244 * melodic_filter_apply supplies the breakpoint guarantee if an authored
245 * filter env later needs one. */
246 if (vp->env1_authored || (bank && bank->eg1[track])) {
247 sequencer_configure_melodic_envelope1_track(layer_idx, track);
248 }
249 sequencer_configure_melodic_filter_track(layer_idx, track);
250 /* Only when authored: an unauthored row has nothing to restore, and
251 * pushing its OFF default would add an AMY event per track per kit reload
252 * for a setting nobody touched. */
253 if (vp->dist_authored) sequencer_configure_melodic_dist_track(layer_idx, track);
254}

272/* ── Chord-aware per-track voice budget ───────────────────────────────────
273 * The voice count each melodic row's synth was last configured with, so a
274 * chord assign/remove or slot resize can detect a no-longer-fitting count
275 * without re-deriving AMY-side state. Non-static so delete_layer's compaction
276 * (seq_core_state.c) can shift it with the other per-layer parallel arrays. */
277uint8_t s_voices_applied[MAX_LAYERS][SEQ_TRACKS];

279uint8_t seq_track_num_voices(const seq_layer_t *layer, uint8_t track)
280{
281 /* KS is mono per row, whatever the layer asks for. Each sounding KS voice
282 * needs its own ring in AMY's ks_buffer pool (amy_cfg.ks_oscs = 4, one per
283 * row): voices sharing a ring damp each other and a note-on re-excites the
284 * ring under the tail still reading it. Measured on a 4x2 layer, that cost
285 * 4x the sustained level and let one inaudible note-on lift the rest of the
286 * layer 386%. Clamping here rather than at the stored num_voices keeps the
287 * layer's setting intact for when the patch changes back, and covers chord
288 * rows too - a chord row on a KS layer sounds one note, the accepted
289 * trade for staying inside the ring budget. */
290 if (layer->type == SEQ_LAYER_MELODIC && layer->patch == SEQ_PATCH_KS) return 1;
291 uint8_t v = layer->num_voices;
292 if (layer->type == SEQ_LAYER_MELODIC &&
293 SEQ_NOTE_IS_CHORD(layer->track_base_note[track])) {
294 uint8_t c = seq_chords_count(SEQ_CHORD_INDEX(layer->track_base_note[track]));
295 if (c > v) v = c;
296 }
297 return v;
298}

320/* Effective copy count for one track's build: the authored count, gated to
321 * native-LFO wave patches (KS excluded - every sounding voice needs its own
322 * ks_buffer ring, see seq_track_num_voices) and clamped so
323 * voices * (copies + 2) fits the per-track osc budget. THE single clamp -
324 * the build, the live push and the LFO layout all derive from it, so the
325 * built pool and the carrier index cannot disagree. */
326static uint8_t unison_copies_for(uint16_t patch, uint8_t count, uint8_t voices)
327{
328#if CONFIG_SEQ_MELODIC_AMY_NATIVE_LFO
329 if (count <= 1u) return 1u;
330 if (!sequencer_core_is_wave_patch(patch)) return 1u;
331 if (patch == SEQ_PATCH_KS) return 1u;
332 if (count > VOICE_UNISON_MAX_COPIES) count = VOICE_UNISON_MAX_COPIES;
333 if (voices == 0u) voices = 1u;
334 uint16_t per_voice = SEQ_TRACK_OSC_BUDGET / voices;
335 uint8_t max_copies = (per_voice > 3u) ? (uint8_t)(per_voice - 2u) : 1u;
336 return (count > max_copies) ? max_copies : count;
337#else
338 /* Without the native layout the software pitch stepper owns osc0's freq
339 * CONST and would stomp the detune fan - single copy only. */
340 (void)patch; (void)count; (void)voices;
341 return 1u;
342#endif
343}

359bool seq_track_lfo_native_layout(uint8_t layer_idx, uint8_t track,
360 uint8_t *carrier_osc, uint8_t *coupled_mask)
361{
362 uint8_t c = 0, m = 0;
363 if (layer_idx >= s_num_layers) return false;
364 if (!sequencer_core_lfo_native_layout(s_layers[layer_idx].patch, &c, &m))
365 return false;
366 uint8_t n = seq_track_unison_copies(layer_idx, track);
367 if (n > 1u) { /* wave layout only - unison_copies_for gates on it */
368 c = n; /* pair sits above the copy fan */
369 m = (uint8_t)((1u << n) - 1u); /* native LFO reaches every copy */
370 }
371 if (carrier_osc) *carrier_osc = c;
372 if (coupled_mask) *coupled_mask = m;
373 return true;
374}

386{

394 s_unison[layer_idx] = v;
395 if (layer_idx >= s_num_layers) return; /* applies when it exists */
396 seq_layer_t *layer = &s_layers[layer_idx];
397 if (layer->type != SEQ_LAYER_MELODIC) return;

494 * Deferred authority: a row's envelope only overrides the patch's own envelope
495 * once the user has committed it in the graph editor (env_authored[t]==true).

587/* Apply one routable patch (0..SEQ_PATCH_FULL_MAX) to a synth slot. The SINGLE
588 * kind dispatch for every consumer (melodic tracks and the arp; the drone has
589 * its own excitation model):
590 * raw wave / wavetable -> direct oscillator config (no patch string)
591 * bass preset (264-266) -> bass_preset_configure_track (oscs_per_voice=4)
592 * FM/ALGO (272-276) -> fm preset / live-editable custom voice (7 oscs)
593 * additive (277-279) -> additive preset / custom voice (N+1 oscs)
594 * everything else -> amy_send_patch() string loader
595 * Returns true when a patch STRING was loaded: those carry global EQ/chorus
596 * commands, so the caller owes one synth_ui_fx_reassert_global() afterwards,
597 * even when applying to several slots. */
598/* Clamp a string-patch load's polyphony so oscs_per_voice x voices fits the
599 * per-track osc budget. Built-in piano is 25 oscs/voice: a layer-wide load at
600 * 4 voices/track attempts 400 oscs against the 250-osc pool AND ~76 KB of
601 * internal heap, exhausting ram_caps_events partway (every osc's synthinfo
602 * lives there) - the 2026-08-07 incident. Degrading polyphony keeps the patch
603 * usable and the heap intact. amy_patch_oscs_per_voice() returns 0 for
604 * unknown/virtual numbers - no clamp, the loader rejects those itself. */
605static uint16_t seq_clamp_patch_voices(uint16_t patch, uint16_t num_voices)
606{
607 uint16_t opv = amy_patch_oscs_per_voice(patch);
608 if (opv == 0) return num_voices;
609 uint16_t max_voices = SEQ_TRACK_OSC_BUDGET / opv;
610 if (max_voices < 1) max_voices = 1;
611 if (num_voices > max_voices) {
612 ESP_LOGW(TAG, "patch %u: %u oscs/voice, clamping %u -> %u voices "
613 "(budget %u oscs/track)",
614 (unsigned)patch, (unsigned)opv, (unsigned)num_voices,
615 (unsigned)max_voices, (unsigned)SEQ_TRACK_OSC_BUDGET);
616 return max_voices;
617 }
618 return num_voices;
619}

621static bool sequencer_apply_patch_kind(uint8_t synth_id, uint16_t patch,
622 uint16_t num_voices, uint32_t synth_flags,
623 bool filter_authored, float ks_feedback,
624 const voice_unison_t *uni)
625{
626 /* Virtual patch whose feature is compiled out (browse skips these, but
627 * stored/programmatic values still arrive): snap to raw SINE rather than
628 * letting a virtual number reach the string loader. */
629 if (sequencer_core_patch_compiled_out(patch)) {
630 sequencer_configure_melodic_wave_track(synth_id, SEQ_PATCH_SINE,
631 num_voices, filter_authored,
632 ks_feedback, uni);
633 return false;
634 }
635 if (sequencer_core_is_wave_patch(patch)) {
636 sequencer_configure_melodic_wave_track(synth_id, patch, num_voices,
637 filter_authored, ks_feedback,
638 uni);
639 return false;
640 }
641 /* Bass presets participate in the reserved-LFO-pair contract (they register
642 * their own pool shape); every OTHER branch below configures a foreign osc
643 * topology, voiding the lazy-LFO shape proof (see voice_config.h). */
644 if (patch >= SEQ_PATCH_BASS_BASE && patch <= SEQ_PATCH_BASS_MAX) {
645 bass_preset_configure_track(synth_id, patch, num_voices);
646 return false;
647 }
648#if CONFIG_SYNTH_CUSTOM_FM
649 if (patch >= SEQ_PATCH_FM_BASE && patch <= SEQ_PATCH_FM_MAX) {
650 voice_lfo_mark_foreign(synth_id);
651 if (patch == SEQ_PATCH_FM_CUSTOM) {
652 fm_voice_configure_track(synth_id, num_voices, &s_fm_voice);
653 } else {
654 fm_preset_configure_track(synth_id, patch, num_voices);
655 }
656 return false;
657 }
658#endif
659#if CONFIG_SYNTH_ADDITIVE
660 if (patch >= SEQ_PATCH_ADDITIVE_BASE && patch <= SEQ_PATCH_ADDITIVE_MAX) {
661 voice_lfo_mark_foreign(synth_id);
662 if (patch == SEQ_PATCH_ADDITIVE_CUSTOM) {
663 additive_voice_configure_track(synth_id, num_voices, &s_additive_voice);
664 } else {
665 additive_preset_configure_track(synth_id, patch, num_voices);
666 }
667 return false;
668 }
669#endif
670 voice_lfo_mark_foreign(synth_id);
671 amy_send_patch(synth_id, patch,
672 seq_clamp_patch_voices(patch, num_voices), synth_flags);
673 return true;
674}

697/* (Re)configure the AMY synth(s) for layer_idx.
698 * Drums use a single synth (synth_id[0]); melodic layers configure one synth
699 * per row, all sharing the same patch/flags/voice-count but on distinct slots. */
700void sequencer_configure_synth(uint8_t layer_idx)
701{
702 seq_layer_t *layer = &s_layers[layer_idx];
703
704 if (layer->type == SEQ_LAYER_DRUM) {
705 if (s_drum_engine == SEQ_DRUM_PCM) {
706 /* PCM mode: each track's slot becomes a 1-osc PCM player - allocate
707 * with oscs_per_voice=1 (no patch string), then set wave=PCM +
708 * preset on osc 0. Note-on/off, velocity and pitch flow through the
709 * SAME emit path as synth mode, so hits keep accent/jitter dynamics
710 * and midi_note tunes the sample (render_pcm). PCM carries no
711 * global EQ/chorus, so no reassert is owed. */
712 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
713 /* Allocate/realloc the slot as a 1-osc voice (clears old patch). */
714 amy_event *e = amy_helpers_event_begin();
715 e->num_voices = layer->num_voices;
716 e->oscs_per_voice = 1;
717 e->synth = layer->synth_id[t];
718 e->synth_flags = layer->synth_flags; /* 0 */
719 amy_helpers_event_send(e);
720
721 /* Configure osc 0 of this synth as the chosen 808 PCM sample. */
722 e = amy_helpers_event_begin();
723 e->synth = layer->synth_id[t];
724 e->osc = 0;
725 e->wave = PCM;
726 e->preset = drum_pcm_preset_for(layer_idx, t);
727 if (s_drum_pcm_mode[layer_idx][t])
728 e->mode = s_drum_pcm_mode[layer_idx][t];
729 amy_helpers_event_send(e);
730 }
731 sequencer_configure_drum_pcm_voice_params(layer_idx);
732 return;
733 }
734
735 /* SYNTH mode: each drum row loads its OWN patch onto its OWN slot,
736 * note-offs honored (flags = 0). Mirrors the melodic loop. */
737 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
738 sequencer_kill_synth_voices(layer->synth_id[t]);
739 amy_send_patch(layer->synth_id[t], layer->track_patch[t],
740 layer->num_voices, layer->synth_flags);
741 }
742 /* Every drum SYNTH slot loads a patch string, so a flush is always
743 * owed - once, after the loop. */
744 seq_flush_patch_fx(true);
745 return;
746 }
747
748 /* Melodic: push the shared patch/flags to each row's own synth. Voice count
749 * is per-track - layer->num_voices, widened to the chord tone count on rows
750 * carrying a chord preset, so voices are spent only where chords play. */
751 bool string_patch = false;
752 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
753 uint8_t voices = seq_track_num_voices(layer, t);
754 sequencer_kill_synth_voices(layer->synth_id[t]);
755 const voice_params_t *vp = seq_track_vp(layer_idx, t);
756 voice_unison_t uni = sequencer_core_get_unison(layer_idx);
757 uni.count = unison_copies_for(layer->patch, uni.count, voices);
758 string_patch |= seq_apply_patch(layer->synth_id[t],
759 layer->patch,
760 voices,
761 layer->synth_flags,
762 vp->filter_authored,
763 vp->filter.feedback,
764 &uni);
765 s_voices_applied[layer_idx][t] = voices;
766 }
767 seq_flush_patch_fx(string_patch);
768 sequencer_configure_melodic_envelope(layer_idx);
769 sequencer_configure_melodic_envelope1(layer_idx);
770 sequencer_configure_melodic_filter(layer_idx);
771 sequencer_configure_melodic_dist(layer_idx);
772 sequencer_configure_melodic_lfo(layer_idx);
773 /* Glide is a per-osc AMY setting that a voice rebuild clears - reassert. */
774 sequencer_core_push_melodic_portamento(layer_idx);
775 /* The patch load just rewrote osc 0's baked algorithm - reassert the
776 * live override (no-op without one). */
777 seq_reassert_layer_fm_algo(layer);
778}

782/* Reconfigure a layer's synths with its scheduled events paused.
783 *
784 * Grid steps are repeating events inside AMY's sequencer, fired autonomously.
785 * Leaving them live while a patch load rebuilds the voice/osc tables lets a
786 * note-on resolve against half-updated mappings and strand an osc AUDIBLE
787 * outside any voice, where no later kill can reach it (kills resolve through
788 * the CURRENT voice map) - the intermittent "ringing that survives patch
789 * changes". Same discipline as arp_rebuild(): clear the schedule, rebuild
790 * (configure MUST kill sounding voices per track, their note-offs were just
791 * cleared too), re-emit. sequencer_emit_step() is s_playing-gated, so the
792 * resync no-ops while paused. Cost: the layer goes quiet for the events cleared
793 * mid-flight; the re-emit restores absolute tick positions, keeping groove
794 * phase. Also the reconfigure path for per-track voice-count changes (chord
795 * assign/remove, slot resize) - see seq_core_internal.h. */
796void sequencer_reconfigure_layer_paused(uint8_t layer_idx)
797{
798 sequencer_clear_layer_tags(layer_idx);
799 /* Drop the trig engine's pending one-shots (ratchet + chord tone pairs)
800 * too: a sub-hit scheduled moments ago must not resolve against the
801 * half-rebuilt pool - same stranded-osc mechanism as above. */
802 sequencer_core_trig_clear_all(layer_idx);
803 sequencer_configure_synth(layer_idx);
804 sequencer_resync_layer(layer_idx);
805}

1166uint16_t sequencer_core_cycle_drum_pcm_preset(uint8_t layer_idx, uint8_t track,
1167 int dir)
1168{
1169 if (layer_idx >= s_num_layers || track >= SEQ_TRACKS) return 0;
1170 if (s_layers[layer_idx].type != SEQ_LAYER_DRUM) return 0;
1171
1172 /* Combined drum sample domain: the ROM bank (0 .. pcm_wavetable_base-1,
1173 * that base being the first non-drum map entry) then the gamma9001 banks
1174 * (256..391) when their blob is mounted, walked as one wrapping list via a
1175 * linear index. Memory presets (sample_rec overrides, outside both ranges)
1176 * step back into the domain at the near end. */
1177 uint16_t bound = pcm_wavetable_base;
1178 if (bound == 0) return 0;
1179 uint16_t total = bound + (drum_gamma_available() ? SEQ_GAMMA_PCM_COUNT : 0);
1180
1181 uint16_t cur = drum_pcm_preset_for(layer_idx, track);
1182 int li_cur;
1183 if (cur < bound) {
1184 li_cur = (int)cur;
1185 } else if (cur >= SEQ_GAMMA_PCM_FIRST &&
1186 cur < SEQ_GAMMA_PCM_FIRST + SEQ_GAMMA_PCM_COUNT &&
1187 total > bound) {
1188 li_cur = (int)bound + (int)(cur - SEQ_GAMMA_PCM_FIRST);
1189 } else {
1190 /* Memory preset (or unreachable gamma preset): enter at the near end. */
1191 li_cur = (dir > 0) ? -1 : 0;
1192 }
1193
1194 int li_next = (li_cur + (dir > 0 ? 1 : -1) + (int)total) % (int)total;
1195 uint16_t next = (li_next < (int)bound)
1196 ? (uint16_t)li_next
1197 : (uint16_t)(SEQ_GAMMA_PCM_FIRST + (uint16_t)(li_next - (int)bound));
1198 sequencer_core_set_drum_pcm_preset(layer_idx, track, next);
1199 return next;
1200}
