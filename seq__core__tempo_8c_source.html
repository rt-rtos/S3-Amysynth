28float lfo_rate_to_hz(lfo_rate_t rate, uint16_t bpm)
29{
30 float b = (float)bpm;
31 switch (rate) {
32 case LFO_RATE_1_8: return b / 30.0f; /* 1/8 note */
33 case LFO_RATE_1_4: return b / 60.0f; /* 1/4 note */
34 case LFO_RATE_1_2: return b / 120.0f; /* 1/2 note */
35 case LFO_RATE_1BAR: return b / 240.0f; /* 1 bar (4/4) */
36 case LFO_RATE_2BAR: return b / 480.0f;
37 case LFO_RATE_4BAR: return b / 960.0f;
38 default: return b / 240.0f;
39 }
40}
