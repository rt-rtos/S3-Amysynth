34float lfo_rate_to_hz(lfo_rate_t rate, uint16_t bpm)
35{
36 float b = (float)bpm;
37 float hz;
38 switch (rate) {
39 case LFO_RATE_1_8: hz = b / 30.0f; break; /* 1/8 note */
40 case LFO_RATE_1_4: hz = b / 60.0f; break; /* 1/4 note */
41 case LFO_RATE_1_2: hz = b / 120.0f; break; /* 1/2 note */
42 case LFO_RATE_1BAR: hz = b / 240.0f; break; /* 1 bar (4/4) */
43 case LFO_RATE_2BAR: hz = b / 480.0f; break;
44 case LFO_RATE_4BAR: hz = b / 960.0f; break;
45 case LFO_RATE_1_16: hz = b / 15.0f; break;
46 case LFO_RATE_1_32: hz = b / 7.5f; break;
47 case LFO_RATE_1_4T: hz = b / 40.0f; break; /* triplet = 1.5x straight */
48 case LFO_RATE_1_8T: hz = b / 20.0f; break;
49 case LFO_RATE_1_16T: hz = b / 10.0f; break;
50 case LFO_RATE_1_32T: hz = b / 5.0f; break;
51 default: hz = b / 240.0f; break;
52 }
53 /* Sub-audible ceiling: at high BPM the fastest divisions cross into
54 * audio-rate AM. Cap rather than hide rates from the pickers. */
55 if (hz > SEQ_LFO_NATIVE_MAX_HZ) hz = SEQ_LFO_NATIVE_MAX_HZ;
56 return hz;
57}

59float lfo_next_rand(void)
60{
61 s_lfo_rng_state ^= s_lfo_rng_state << 13;
62 s_lfo_rng_state ^= s_lfo_rng_state >> 17;
63 s_lfo_rng_state ^= s_lfo_rng_state << 5;
64 return (float)(s_lfo_rng_state >> 17) / 32767.0f * 2.0f - 1.0f;
65}

67void lfo_push_target_neutral(uint8_t synth_id, lfo_target_t target)
68{
69 amy_event *e = amy_helpers_event_begin();
70 e->synth = synth_id;
71 switch (target) {
72 case LFO_TARGET_FILTER: e->filter_freq_coefs[COEF_CONST] = 8000.0f; break;
73 case LFO_TARGET_AMP: e->amp_coefs[COEF_CONST] = 1.0f; break;
74 /* Absolute Hz, not a ratio: 440 = SEQ_LFO_PITCH_BASE_HZ maps to a
75 * logfreq constant of 0, the note-neutral reset default. (1.0f here
76 * parked the osc ~8.8 octaves down, muting PCM drums until reboot.)
77 * Osc 0 only, mirroring the service push: synth-wide would stomp
78 * patch-internal modulator oscs' rates (stopgap - revisit). */
79 case LFO_TARGET_PITCH: e->osc = 0;
80 e->freq_coefs[COEF_CONST] = SEQ_LFO_PITCH_BASE_HZ; break;
81 case LFO_TARGET_PAN: e->pan_coefs[COEF_CONST] = 0.5f; break;
82 /* DIST has no context-free neutral - resting state is the caller's
83 * committed seq_dist_t, so every restore path pushes it directly
84 * via voice_apply_dist (same shape as the FILTER special case). */
85 case LFO_TARGET_DIST_DRIVE:
86 case LFO_TARGET_DIST_MIX:
87 default: break;
88 }
89 amy_helpers_event_send(e);
90}

94void sequencer_core_set_bpm(uint16_t new_bpm)
95{
96 s_bpm = sequencer_clamp_bpm(new_bpm);
97 sequencer_push_tempo(s_bpm);
98 for (int li = 0; li < s_num_layers; li++) {
99#if CONFIG_SEQ_MELODIC_AMY_NATIVE_LFO
100 /* Native-LFO tracks must keep s_lfo_hz == 0: their carrier is retuned
101 * by melodic_lfo_refresh_native_freq() below, and a nonzero value would
102 * start the software stepper double-modulating on top of it. Only
103 * melodic layers can be native - a drum layer's `patch` is just its
104 * stored SYNTH-mode selection and must not shadow the software rate. */
105 if (s_layers[li].type == SEQ_LAYER_MELODIC &&
106 sequencer_core_lfo_native_layout(s_layers[li].patch, NULL, NULL)) continue;
107#endif
108 for (int tr = 0; tr < SEQ_TRACKS; tr++) {
109 const voice_params_t *vp = seq_track_vp((uint8_t)li, (uint8_t)tr);
110 if (vp->lfo_authored && vp->lfo.enabled)
111 s_lfo_hz[li][tr] = seq_lfo_sw_hz(vp->lfo.rate, s_bpm);
112 }
113 }
114 /* Sync every native LFO carrier to the new BPM (each a no-op when the
115 * owner has none active). */
116 arp_core_refresh_lfo_freq();
117 drone_std_core_refresh_lfo_freq();
118 melodic_lfo_refresh_native_freq();
119#if CONFIG_SYNTH_WIRELESS
120 live_play_refresh_lfo_freq();
121#endif
122}
