3/* ════════════════════════════════════════════════════════════════════════
4 * Per-step probability / ratchet / conditional-trig engine
5 * ════════════════════════════════════════════════════════════════════════
6 *
7 * A step is "plain" when step_prob==100 && step_ratchet==1 &&
8 * step_cond_type==NONE: it keeps using sequencer_emit_step()'s original
9 * always-on repeating AMY sequence tag (seq_core_engine.c) — zero added cost,
10 * zero behaviour change from before this file existed.
11 *
12 * Any other combination makes the step "decorated". sequencer_emit_step()
13 * (seq_core_engine.c) leaves such a step's plain ON/OFF tag pair cleared, and
14 * sequencer_core_service_tick() — called once per AMY sequencer tick from the
15 * existing 500us sequencer poll hook (main.c) — takes over: it detects the
16 * moment a layer's playhead crosses into a new step (O(1) integer compare per
17 * layer per tick; the per-track decoration check only runs on that edge, so
18 * the steady-state per-tick cost is a handful of layer-count comparisons, not
19 * a scan of every step), evaluates the step's conditional trig and
20 * probability roll, and — if it should sound — one-shot schedules
21 * step_ratchet sub-hits on a dedicated tag range reserved solely for this
22 * (SEQ_RATCHET_TAG_BASE.., see seq_core_config.h).
23 */

25/* ── State (per MAX_LAYERS[/xSEQ_TRACKS]) ────────────────────────────────

35/* Dedicated xorshift32 PRNG, independent of seq_core_editors.c's s_lfo_rng_state:
36 * that one is mutated from synth_ui_task (Core 0, 20 Hz); this one is mutated
37 * exclusively from sequencer_core_service_tick(), which itself only ever runs
38 * on the esp_timer "sequencer" task callback (also Core 0, but a different
39 * task) — sharing one PRNG across two independently-scheduled tasks would be
40 * a real (if narrow) data race. */
41static uint32_t s_trig_rng_state = 0xA53C9E17u;

43static inline uint8_t trig_rand_pct(void)
44{
45 uint32_t x = s_trig_rng_state;
46 x ^= x << 13;
47 x ^= x >> 17;
48 x ^= x << 5;
49 s_trig_rng_state = x;
50 return (uint8_t)(x % 100u);
51}

53/* Semitone span shared by every non-NONE transform mode: RANDOM offsets by
54 * +/-SEQ_STEP_TRANSFORM_SPAN, RAMP walks 0..SEQ_STEP_TRANSFORM_SPAN across
55 * successive layer loops then wraps. Fixed (no per-step param array) to keep
56 * the OP-Z step-component subset minimal; one octave reads musically once the
57 * result is re-snapped to the scale. */
58#define SEQ_STEP_TRANSFORM_SPAN 12
59
60/* Per-step pitch transform (spec 20 §3.1). Offsets the step's authored note per
61 * fire, then re-snaps the result to the active chord/scale via the same engine
62 * resolve the plain per-track path uses — unless step_quant_bypass is set, in
63 * which case the transformed note is only bounds-clamped and emitted
64 * chromatically (spec 20 §3.2). SEQ_STEP_TRANSFORM_NONE returns `base`
65 * unchanged (this helper is only reached for decorated steps anyway). All
66 * integer math: no float, no lock, safe on the Core-0 service-tick task. */
67static uint8_t trig_transform_note(uint8_t layer_idx, const seq_layer_t *layer,
68 uint8_t track, uint8_t step, uint8_t base)
69{
70 int offset;
71 switch ((seq_step_transform_t)layer->step_transform[track][step]) {
72 case SEQ_STEP_TRANSFORM_RANDOM:
73 offset = (int)(trig_rand_pct() % (2u * SEQ_STEP_TRANSFORM_SPAN + 1u))
74 - SEQ_STEP_TRANSFORM_SPAN;
75 break;
76 case SEQ_STEP_TRANSFORM_RAMP_UP:
77 offset = (int)(s_layer_loop_count[layer_idx]
78 % (SEQ_STEP_TRANSFORM_SPAN + 1u));
79 break;
80 case SEQ_STEP_TRANSFORM_RAMP_DOWN:
81 offset = -(int)(s_layer_loop_count[layer_idx]
82 % (SEQ_STEP_TRANSFORM_SPAN + 1u));
83 break;
84 case SEQ_STEP_TRANSFORM_NONE:
85 default:
86 return base;
87 }
88
89 int transformed = (int)base + offset;
90 if (transformed < 0) transformed = 0;
91 if (transformed > 127) transformed = 127;
92
93 if (layer->step_quant_bypass[track][step]) {
94 return sequencer_clamp_layer_note(layer, (uint8_t)transformed);
95 }
96 return sequencer_resolve_track_note(layer, (uint8_t)transformed);
97}

99/* ── Ratchet tag formula ──────────────────────────────────────────────────
100 * Statically assigned per (layer, track, ratchet-slot) — never pooled/shared
101 * across tracks — so a slow ratchet group can never overwrite another
102 * track's in-flight schedule. Only one step per track is ever "current" at
103 * a time, so SEQ_MAX_RATCHET dedicated slots per track fully cover the worst
104 * case (see seq_core_config.h for the tag-space layout comment). */
105static inline uint32_t ratchet_slot_index(uint8_t layer, uint8_t track, uint8_t slot)
106{
107 return ((uint32_t)layer * SEQ_TRACKS + track) * SEQ_MAX_RATCHET + slot;
108}

110static inline uint32_t ratchet_on_tag(uint8_t layer, uint8_t track, uint8_t slot)
111{
112 return SEQ_RATCHET_TAG_BASE + ratchet_slot_index(layer, track, slot) * 2u;
113}

115static inline uint32_t ratchet_off_tag(uint8_t layer, uint8_t track, uint8_t slot)
116{
117 return ratchet_on_tag(layer, track, slot) + 1u;

157/* ── Conditional trig evaluation ──────────────────────────────────────── */
158static bool trig_eval_condition(uint8_t layer_idx, const seq_layer_t *layer,
159 uint8_t track, uint8_t step)
160{
161 switch ((seq_step_cond_type_t)layer->step_cond_type[track][step]) {
162 case SEQ_STEP_COND_FILL: {
163 uint8_t n = layer->step_cond_param[track][step];
164 if (n < 2) n = 2;
165 return (s_layer_loop_count[layer_idx] % n) == 0;
166 }
167 case SEQ_STEP_COND_PREV:
168 return s_track_last_played[layer_idx][track];
169 case SEQ_STEP_COND_NONE:
170 default:
171 return true;
172 }
173}

182/* One-shot schedule step_ratchet[track][step] sub-hits, evenly spaced across
183 * the step's slot. n==1 reuses the exact plain-path gate math (including the
184 * melodic off-beat shortening) from sequencer_emit_step() (seq_core_engine.c)
185 * so a merely-probabilistic or conditional (ratchet==1) step still feels
186 * identical to a plain one when it does fire; n>1 subdivides the slot. */
187static void trig_schedule_ratchets(uint8_t layer_idx, const seq_layer_t *layer,
188 uint8_t track, uint8_t step, uint32_t now_ticks)
189{
190 uint8_t n = layer->step_ratchet[track][step];
191 if (n < 1) n = 1;
192 if (n > SEQ_MAX_RATCHET) n = SEQ_MAX_RATCHET;
193
194 uint16_t sub_ticks;
195 uint16_t gate;
196 if (n == 1) {
197 sub_ticks = SEQ_TICKS_PER_STEP;
198 gate = seq_step_gate(layer, step);
199 } else {
200 sub_ticks = SEQ_TICKS_PER_STEP / n;
201 if (sub_ticks < 2) sub_ticks = 2;
202 gate = (uint16_t)(sub_ticks - 1);
203 }
204
205 float velocity = sequencer_step_velocity(layer, track, step) * layer->vp[track].amp_trim;
206 /* Per-step velocity offset (patch-06): signed percentage points, default 0.
207 * Applied here as well as the plain path so decorated steps match. */
208 velocity += (float)layer->step_velocity_adj[track][step] * 0.01f;
209 if (velocity < 0.0f) velocity = 0.0f;
210 if (velocity > 1.0f) velocity = 1.0f;
211 /* Single per-fire pitch source for both ratchet sub-hits and the plain
212 * decorated (n==1) path: apply any per-step note transform here so every
213 * sub-hit of one fire shares the same transformed pitch. */
214 uint8_t note = trig_transform_note(layer_idx, layer, track, step,
215 layer->step_note[track][step]);
216 uint8_t synth = layer->synth_id[track];
217
218 /* Ratchet velocity taper (patch-06): sub-hit k is scaled by (1 - taper*k%).
219 * Default 0 = flat (every sub-hit at full velocity — the historical
220 * behaviour). Positive taper decays toward the tail; negative ramps up.
221 * k==0 is always full velocity. */
222 int8_t taper = layer->step_ratchet_taper[track][step];
223 for (uint8_t k = 0; k < n; k++) {
224 float scale = 1.0f - (float)taper * 0.01f * (float)k;
225 if (scale < 0.0f) scale = 0.0f;
226 if (scale > 1.0f) scale = 1.0f;
227 float v = velocity * scale;
228 uint32_t tick_on = now_ticks + 1u + (uint32_t)k * sub_ticks;
229 uint32_t tick_off = tick_on + gate;
230 amy_helpers_note_send(synth, (float)note, v,
231 ratchet_on_tag(layer_idx, track, k), tick_on, 0);
232 amy_helpers_note_send(synth, (float)note, 0.0f,
233 ratchet_off_tag(layer_idx, track, k), tick_off, 0);
234 }

237/* ── The one new per-tick entry point (called from main.c's AMY sequencer
238 * hook, ~2 kHz worst case, esp_timer task, Core 0) ───────────────────── */
239void sequencer_core_service_tick(void)
240{
241 if (!s_playing) return;
242 /* A structural s_layers edit (delete_layer's compaction) is in flight on
243 * another task; skip this tick rather than read a half-shifted table. The
244 * engine tolerates skipped ticks (trig_reset_all runs after the edit), so a
245 * dropped tick here is inaudible. */
246 if (s_layers_mutating) return;
247 uint32_t now_ticks = sequencer_ticks();
248
249 for (uint8_t li = 0; li < s_num_layers; li++) {
250 seq_layer_t *layer = &s_layers[li];
251 /* bar_ticks stays local for the repeat-rate bar-window math below;
252 * the step derivation itself goes through the shared helper. */
253 uint32_t bar_ticks = (uint32_t)layer->num_steps * SEQ_TICKS_PER_STEP;
254 if (bar_ticks == 0) continue;
255 uint8_t cur_step = seq_playhead_step(layer, now_ticks);
256 if (cur_step == s_layer_last_step[li]) continue; /* still mid-step: O(1) exit */
257
258 if (cur_step == 0 && s_layer_last_step[li] != 0xFF) {
259 s_layer_loop_count[li]++;
260 }
261 s_layer_last_step[li] = cur_step;
262
263 for (uint8_t tr = 0; tr < layer->num_tracks; tr++) {
264 bool on = layer->grid[tr][cur_step];
265 bool decorated = sequencer_core_step_is_decorated(layer, tr, cur_step);
266 bool fire;
267
268 if (!on) {
269 fire = false;
270 } else if (!decorated) {
271 fire = true; /* AMY's own periodic tag already fires this one */
272 } else {
273 fire = trig_eval_condition(li, layer, tr, cur_step)
274 && trig_roll_probability(layer->step_prob[tr][cur_step]);
275 /* Per-track repeat_rate: the plain emit path repeats its
276 * note-on tag every bar_ticks*rr (seq_core_engine.c:177-179),
277 * so a plain step only sounds on the 0th bar of each rr-bar
278 * window: (now_ticks/bar_ticks) % rr == 0. Gate the decorated
279 * one-shot's output the same way so SEQ_REPEAT_N behaves
280 * identically for decorated and plain steps. Like the mute/solo
281 * gate below this only silences output — condition/probability
282 * state keeps evaluating every bar so the track's rhythmic
283 * position stays seamless. */
284 uint32_t rr = (layer->repeat_rate[tr] >= SEQ_REPEAT_2)
285 ? (uint32_t)layer->repeat_rate[tr] : 1u;
286 bool rr_bar = ((now_ticks / bar_ticks) % rr) == 0;
287 /* Mute/solo silences output only — condition/probability state
288 * (fire, s_track_last_played below) keeps evaluating normally
289 * so a track resumes its rhythmic position seamlessly when
290 * unmuted rather than freezing while muted. */
291 if (fire && rr_bar && sequencer_track_audible(layer, tr)) {
292 trig_schedule_ratchets(li, layer, tr, cur_step, now_ticks);
293 }
294 }
295 s_track_last_played[li][tr] = fire;
296 }
297 }
298}
