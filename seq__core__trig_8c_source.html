4/* ════════════════════════════════════════════════════════════════════════
5 * Per-step probability / ratchet / conditional-trig engine
6 * ════════════════════════════════════════════════════════════════════════
7 *
8 * A step is "plain" when step_prob==100 && step_ratchet==1 &&
9 * step_every<=1 && !step_prev: it keeps sequencer_emit_step()'s always-on
10 * repeating AMY sequence tag (seq_core_engine.c) at zero added cost.
11 *
12 * Any other combination makes the step "decorated". sequencer_emit_step()
13 * leaves such a step's plain ON/OFF tag pair cleared and
14 * sequencer_core_service_tick() - called once per AMY sequencer tick from the
15 * sequencer hook in main.c - takes over: it detects a layer's playhead
16 * crossing into a new step (O(1) compare per layer per tick, with the per-track
17 * decoration check only on that edge, so steady state costs a few comparisons
18 * rather than a scan of every step), evaluates the conditional trig and
19 * probability roll, and one-shot schedules step_ratchet sub-hits on the
20 * dedicated tag range (SEQ_RATCHET_TAG_BASE.., seq_core_config.h).
21 */
22
23/* ── State (per MAX_LAYERS[/xSEQ_TRACKS]) ────────────────────────────────
24 * Runtime bookkeeping only - never persisted, never pattern data (that lives
25 * in seq_layer_t). Deliberately reset wholesale by
26 * sequencer_core_trig_reset_all() on any layer add/delete rather than shifted
27 * in lockstep with compaction; rationale at the call sites in
28 * seq_core_state.c. */
29uint32_t s_layer_loop_count[MAX_LAYERS];
30uint8_t s_layer_last_step[MAX_LAYERS];
31bool s_track_last_played[MAX_LAYERS][SEQ_TRACKS];

33/* Dedicated xorshift32 PRNG, kept separate from seq_core_editors.c's
34 * s_lfo_rng_state: that one is mutated from synth_ui_task, this one only from
35 * sequencer_core_service_tick() on the AMY sequencer-hook task. Sharing one
36 * PRNG across two independently-scheduled tasks would be a real data race. */
37static uint32_t s_trig_rng_state = 0xA53C9E17u;
38
39static inline uint8_t trig_rand_pct(void)
40{
41 uint32_t x = s_trig_rng_state;
42 x ^= x << 13;
43 x ^= x >> 17;
44 x ^= x << 5;
45 s_trig_rng_state = x;
46 return (uint8_t)(x % 100u);
47}

49/* Semitone span shared by every non-NONE transform mode: RANDOM offsets by
50 * +/-SPAN, RAMP walks 0..SPAN across successive layer loops then wraps. Fixed
51 * rather than a per-step param, to keep the step-component subset minimal; one
52 * octave reads musically once the result is re-snapped to the scale. */
53#define SEQ_STEP_TRANSFORM_SPAN 12
54
55/* Per-step pitch transform, factored as "which offset does this fire get" so
56 * single notes and chord voicings share one PRNG/loop-counter draw per fire.
57 * Returns true and writes the semitone offset when a transform is active, false
58 * (offset 0) for NONE. Integer math only: no float, no lock, safe on the
59 * service-tick task. */
60static bool trig_transform_offset(uint8_t layer_idx, const seq_layer_t *layer,
61 uint8_t track, uint8_t step, int *offset)
62{
63 switch ((seq_step_transform_t)layer->step_transform[track][step]) {
64 case SEQ_STEP_TRANSFORM_RANDOM:
65 *offset = (int)(trig_rand_pct() % (2u * SEQ_STEP_TRANSFORM_SPAN + 1u))
66 - SEQ_STEP_TRANSFORM_SPAN;
67 return true;
68 case SEQ_STEP_TRANSFORM_RAMP_UP:
69 *offset = (int)(s_layer_loop_count[layer_idx]
70 % (SEQ_STEP_TRANSFORM_SPAN + 1u));
71 return true;
72 case SEQ_STEP_TRANSFORM_RAMP_DOWN:
73 *offset = -(int)(s_layer_loop_count[layer_idx]
74 % (SEQ_STEP_TRANSFORM_SPAN + 1u));
75 return true;
76 case SEQ_STEP_TRANSFORM_NONE:
77 default:
78 *offset = 0;
79 return false;
80 }
81}

83/* ── Ratchet tag formula ──────────────────────────────────────────────────
84 * Statically assigned per (layer, track, ratchet-slot), never pooled across
85 * tracks, so a slow ratchet group cannot overwrite another track's in-flight
86 * schedule. Only one step per track is ever current, so SEQ_MAX_RATCHET slots
87 * per track cover the worst case. Tag-space layout: seq_core_config.h. */
88static inline uint32_t ratchet_slot_index(uint8_t layer, uint8_t track, uint8_t slot)
89{
90 return ((uint32_t)layer * SEQ_TRACKS + track) * SEQ_MAX_RATCHET + slot;
91}
92
93static inline uint32_t ratchet_on_tag(uint8_t layer, uint8_t track, uint8_t slot)
94{
95 return SEQ_RATCHET_TAG_BASE + ratchet_slot_index(layer, track, slot) * 2u;
96}
97
98static inline uint32_t ratchet_off_tag(uint8_t layer, uint8_t track, uint8_t slot)
99{
100 return ratchet_on_tag(layer, track, slot) + 1u;
101}

103/* ── Chord-tone tag formula ───────────────────────────────────────────────
104 * Extra chord tones (1..SEQ_CHORD_MAX_NOTES-1; tone 0 rides the ratchet tags
105 * above) get a statically assigned one-shot pair per (layer, track, slot,
106 * tone), same never-pooled discipline as the ratchets. Layout in
107 * seq_core_config.h (SEQ_CHORD_TAG_BASE..MAX). */
108static inline uint32_t chord_on_tag(uint8_t layer, uint8_t track, uint8_t slot,
109 uint8_t tone /* 1.. */)
110{
111 return SEQ_CHORD_TAG_BASE
112 + (ratchet_slot_index(layer, track, slot) * (SEQ_CHORD_MAX_NOTES - 1)
113 + (uint32_t)(tone - 1)) * 2u;
114}
115
116static inline uint32_t chord_off_tag(uint8_t layer, uint8_t track, uint8_t slot,
117 uint8_t tone /* 1.. */)
118{
119 return chord_on_tag(layer, track, slot, tone) + 1u;
120}

122bool sequencer_core_step_is_decorated(const seq_layer_t *layer, uint8_t track, uint8_t step)
123{
124 if (track >= SEQ_TRACKS || step >= SEQ_MAX_STEPS) return false;
125 /* A chord sentinel forces the decorated path too: the plain path emits ONE
126 * periodic tag pair at a fixed pitch, while a chord fires up to
127 * SEQ_CHORD_MAX_NOTES tones whose pitches (progression transpose) and count
128 * (live table edits) resolve per fire. One-shots self-consume, so chords
129 * add no resident periodic events and cannot leave a tag ringing. */
130 return layer->step_prob[track][step] != 100
131 || layer->step_ratchet[track][step] != 1
132 || layer->step_every[track][step] > 1
133 || layer->step_prev[track][step] != 0
134 || layer->step_transform[track][step] != (uint8_t)SEQ_STEP_TRANSFORM_NONE
135 || SEQ_NOTE_IS_CHORD(layer->step_note[track][step]);
136}

181/* ── Conditional trig evaluation ──────────────────────────────────────────
182 * EVERY and PREV are independent conditions; both must hold. every==0 (an
183 * uninitialised or hand-edited value) is treated as 1 (neutral). */
184static bool trig_eval_condition(uint8_t layer_idx, const seq_layer_t *layer,
185 uint8_t track, uint8_t step)
186{
187 uint8_t n = layer->step_every[track][step];
188 if (n > 1 && (s_layer_loop_count[layer_idx] % n) != 0) return false;
189 if (layer->step_prev[track][step] && !s_track_last_played[layer_idx][track])
190 return false;
191 return true;
192}

194static bool trig_roll_probability(uint8_t prob_pct)
195{
196 if (prob_pct >= 100) return true;
197 if (prob_pct == 0) return false;
198 return trig_rand_pct() < prob_pct;
199}

201/* One-shot schedule step_ratchet sub-hits, evenly spaced across the step's
202 * slot. n==1 reuses sequencer_emit_step()'s exact gate math (incl. the off-beat
203 * shortening) so a merely probabilistic or conditional step feels identical to
204 * a plain one when it fires; n>1 subdivides the slot.
205 *
206 * Not static: called only from seq_trig_pump.c's urgent-source drain on the
207 * AMY ingest pump task, never from this file's own render-task tick hook
208 * below (which only enqueues a job) - this function ends in
209 * amy_helpers_note_send(), which asserts if called from the render task and
210 * would block that task even if it didn't. */
211void trig_schedule_ratchets(uint8_t layer_idx, const seq_layer_t *layer,
212 uint8_t track, uint8_t step, uint32_t now_ticks)
213{
214 uint8_t n = layer->step_ratchet[track][step];
215 n = SEQ_CLAMP_U8(n, 1, SEQ_MAX_RATCHET);
216
217 uint16_t sub_ticks;
218 uint16_t gate;
219 if (n == 1) {
220 sub_ticks = SEQ_TICKS_PER_STEP;
221 gate = seq_step_gate(layer, step);
222 } else {
223 sub_ticks = SEQ_TICKS_PER_STEP / n;
224 if (sub_ticks < 2) sub_ticks = 2;
225 gate = (uint16_t)(sub_ticks - 1);
226 }
227
228 float velocity = sequencer_step_velocity(layer, track, step) * layer->vp[track].amp_trim;
229 /* Per-step velocity offset in signed percentage points, applied here as
230 * well as on the plain path so decorated steps match. */
231 velocity += (float)layer->step_velocity_adj[track][step] * 0.01f;
232 velocity = SEQ_CLAMP_F32(velocity, 0.0f, 1.0f);
233
234 /* Single per-fire pitch source for both ratchet sub-hits and the n==1 path.
235 * A chord sentinel expands here with the progression transpose applied, and
236 * a per-step transform draws ONE offset per fire so every sub-hit and every
237 * chord tone shares it. Single notes offset then re-snap unless bypassed;
238 * chord tones take the offset chromatically with a bounds clamp, never
239 * per-tone re-quantization - the intervals are the feature. */
240 uint8_t stored = layer->step_note[track][step];
241 uint8_t tones[SEQ_CHORD_MAX_NOTES];
242 uint8_t ntones = seq_track_fire_notes(layer, stored, tones);
243 if (ntones == 0) return; /* undefined chord slot: fire nothing */
244
245 int toff;
246 if (trig_transform_offset(layer_idx, layer, track, step, &toff)) {
247 if (SEQ_NOTE_IS_CHORD(stored)) {
248 for (uint8_t i = 0; i < ntones; i++) {
249 tones[i] = sequencer_core_clamp_melodic_note((int32_t)tones[i] + toff);
250 }
251 } else {
252 int tn = SEQ_CLAMP_INT((int)tones[0] + toff, 0, 127);
253 tones[0] = layer->step_quant_bypass[track][step]
254 ? sequencer_clamp_layer_note(layer, (uint8_t)tn)
255 : sequencer_resolve_track_note(layer, (uint8_t)tn);
256 }
257 }
258 /* Per-step pitch offset, applied last so the transform's re-snap above
259 * cannot quantize it away: chromatic semitones on single notes and chord
260 * tones alike (block transpose preserves the chord's intervals).
261 * TODO: revisit quantizer interplay - bypassed by design for now. */
262 int8_t pofs = layer->step_pitch_ofs[track][step];
263 if (pofs != 0) {
264 for (uint8_t i = 0; i < ntones; i++)
265 tones[i] = (uint8_t)SEQ_CLAMP_INT((int)tones[i] + (int)pofs, 0, 127);
266 }
267 uint8_t synth = layer->synth_id[track];
268
269 /* PCM drums get no scheduled note-off - same rule and rationale as the
270 * plain path in sequencer_emit_step(): the off is a hard cut for PCM, and
271 * the sample self-terminates under EG0. */
272 bool send_offs = !(layer->type == SEQ_LAYER_DRUM &&
273 s_drum_engine == SEQ_DRUM_PCM);
274
275 /* Ratchet velocity taper: sub-hit k scales by (1 - taper*k%). 0 = flat,
276 * positive decays toward the tail, negative ramps up; k==0 is always
277 * full velocity. */
278 int8_t taper = layer->step_ratchet_taper[track][step];
279 for (uint8_t k = 0; k < n; k++) {
280 float scale = 1.0f - (float)taper * 0.01f * (float)k;
281 scale = SEQ_CLAMP_F32(scale, 0.0f, 1.0f);
282 float v = velocity * scale;
283 uint32_t tick_on = now_ticks + 1u + (uint32_t)k * sub_ticks;
284 uint32_t tick_off = tick_on + gate;
285 amy_helpers_note_send(synth, (float)tones[0], v,
286 ratchet_on_tag(layer_idx, track, k), tick_on, 0);
287 if (send_offs)
288 amy_helpers_note_send(synth, (float)tones[0], 0.0f,
289 ratchet_off_tag(layer_idx, track, k), tick_off, 0);
290 for (uint8_t i = 1; i < ntones; i++) {
291 amy_helpers_note_send(synth, (float)tones[i], v,
292 chord_on_tag(layer_idx, track, k, i), tick_on, 0);
293 if (send_offs)
294 amy_helpers_note_send(synth, (float)tones[i], 0.0f,
295 chord_off_tag(layer_idx, track, k, i), tick_off, 0);
296 }
297 }

300/* ── Per-tick entry point, called from main.c's AMY sequencer hook ──────── */
301void sequencer_core_service_tick(void)
302{
303 if (!s_playing) return;
304 /* A structural s_layers edit (delete_layer's compaction) is in flight on
305 * another task: skip this tick rather than read a half-shifted table. The
306 * engine tolerates skipped ticks - trig_reset_all runs after the edit - so
307 * a dropped tick is inaudible. */
308 if (s_layers_mutating) return;
309 uint32_t now_ticks = sequencer_ticks();
310
311 for (uint8_t li = 0; li < s_num_layers; li++) {
312 seq_layer_t *layer = &s_layers[li];
313 /* bar_ticks is local to the repeat-rate window math below; the step
314 * derivation goes through the shared helper. */
315 uint32_t bar_ticks = (uint32_t)layer->num_steps * SEQ_TICKS_PER_STEP;
316 if (bar_ticks == 0) continue;
317 uint8_t cur_step = seq_playhead_step(layer, now_ticks);
318 if (cur_step == s_layer_last_step[li]) continue; /* still mid-step: O(1) exit */
319
320 if (cur_step == 0 && s_layer_last_step[li] != 0xFF) {
321 s_layer_loop_count[li]++;
322 }
323 s_layer_last_step[li] = cur_step;
324
325 for (uint8_t tr = 0; tr < layer->num_tracks; tr++) {
326 bool on = layer->grid[tr][cur_step];
327 bool decorated = sequencer_core_step_is_decorated(layer, tr, cur_step);
328 bool fire;
329
330 if (!on) {
331 fire = false;
332 } else if (!decorated) {
333 fire = true; /* AMY's own periodic tag already fires this one */
334 } else {
335 fire = trig_eval_condition(li, layer, tr, cur_step)
336 && trig_roll_probability(layer->step_prob[tr][cur_step]);
337 /* Per-track repeat_rate: the plain emit path repeats its
338 * note-on every bar_ticks*rr, so a plain step only sounds on
339 * the 0th bar of each rr-bar window. Gate the decorated
340 * one-shot the same way so SEQ_REPEAT_N behaves identically for
341 * both. Like mute/solo below, this silences output only -
342 * condition/probability state keeps evaluating every bar. */
343 uint32_t rr = (layer->repeat_rate[tr] >= SEQ_REPEAT_2)
344 ? (uint32_t)layer->repeat_rate[tr] : 1u;
345 bool rr_bar = ((now_ticks / bar_ticks) % rr) == 0;
346 /* Mute/solo silences output only: fire and s_track_last_played
347 * keep evaluating, so a track resumes its rhythmic position
348 * seamlessly on unmute instead of freezing. */
349 if (fire && rr_bar && sequencer_track_audible(layer, tr)) {
350 /* trig_schedule_ratchets() ends in amy_helpers_note_send(),
351 * which asserts if called from this (the render) task and
352 * can block for up to 250ms even when it doesn't - both
353 * forbidden on the render path. Hand off a tiny job
354 * descriptor to the ingest pump's urgent source instead
355 * (seq_trig_pump.c); the non-blocking enqueue + doorbell
356 * is the render task's entire touch on this path. */
357 sequencer_core_trig_enqueue(li, tr, cur_step, now_ticks);
358 }
359 }
360 s_track_last_played[li][tr] = fire;
361 }
362 }
363}

473/* ── Per-step note transform + quantize bypass ────────────────────────────
474 * Both re-emit the step so sequencer_emit_step() re-resolves plain-vs-decorated
475 * (a non-NONE transform forces the decorated path). The quantize bypass rides
476 * on the transform and only alters output while one is active: under NONE the
477 * base note is already snapped and uniform per track. */
478void sequencer_core_set_step_transform(uint8_t layer_idx, uint8_t track, uint8_t step,
479 seq_step_transform_t mode, bool quant_bypass)
480{
481 if (layer_idx >= s_num_layers) return;
