46/* ── Per-step conditional trig ──
47 * Two independent, composable conditions (both must hold for the step to
48 * fire, and only then is probability rolled):
49 * EVERY n: fires on one loop pass out of every n of the layer's pattern
50 * (1 = every loop = neutral; the first loop after play-start always
51 * counts, since loop_count 0 satisfies any divisor).
52 * PREV: fires only if this track's previous step-attempt sounded on its
53 * last evaluation, so any miss breaks the chain. */
54#define SEQ_STEP_EVERY_MAX 4

56/* ── Per-step pitch offset ──
57 * Chromatic semitones added to the step's stored pitch at emit time, on both
58 * the plain periodic path and the decorated one-shot path (chord tones
59 * transpose as a block - the intervals are the feature). Offset-relative by
60 * design: track pitch edits and drum bank changes rewrite step_note only, so
61 * an authored offset survives them untouched. Deliberately NOT re-quantized
62 * (TODO: revisit quantizer interplay). */
63#define SEQ_STEP_PITCH_OFS_MAX 24

65/* ── Per-step note transform ──
66 * NONE is the zeroed default: the authored pitch, on the plain periodic-tag
67 * path. Any other mode offsets the pitch per fire and so forces the step onto
68 * the decorated one-shot path (sequencer_core_step_is_decorated) - the plain
69 * path emits one repeating tag at a fixed pitch and AMY has no per-repetition
70 * hook.
71 * RANDOM re-pitches every fire by +/- a fixed semitone span.
72 * RAMP_UP walks the pitch up across successive layer loops, then wraps.
73 * RAMP_DOWN walks it down.
74 * The transformed pitch re-snaps to the active scale/chord unless the step's
75 * step_quant_bypass bit is set, in which case it is emitted chromatically. */
76typedef enum {
77 SEQ_STEP_TRANSFORM_NONE = 0,
78 SEQ_STEP_TRANSFORM_RANDOM = 1,
79 SEQ_STEP_TRANSFORM_RAMP_UP = 2,
80 SEQ_STEP_TRANSFORM_RAMP_DOWN = 3,
81 SEQ_STEP_TRANSFORM_COUNT,
82} seq_step_transform_t;

218/* ── Per-osc distortion stage (AMY DIST_*) ──
219 * A static waveshaper ahead of the filter: soft-clip, wavefolder, or bit
220 * crusher. Stored as concrete AMY units so it survives patch changes, and
221 * scoped per voice like the filter - `type == 0` (OFF) is the disable, there is
222 * no separate enable flag. `bits` tops out at 24 because AMY samples are s8.23
223 * (S_FRAC_BITS, amy_fixedpoint.h): one sign bit plus 23 fraction bits is where
224 * quantization becomes a no-op, whatever the output stage does. */
225typedef struct {
226 uint8_t type; /* AMY stage mask: bit0 CLIP, bit1 FOLD, bit2 CRUSH;
227 0 = OFF. Stages stack in clip->fold->crush order. */
228 uint8_t drive; /* 1..16 pre-gain (fold depth for FOLD) */
229 uint8_t bits; /* 1..24 CRUSH bit depth; 24 = no-op */
230 uint8_t rate; /* 1..64 CRUSH sample-hold length in samples */
231 uint8_t mix; /* 0..100 wet % */
232} seq_dist_t;

234/* Stage-set cycle for the dist Type rows (track editor and bus FX row cycle
235 * identically): raw masks ordered singles-first, labels indexed by mask. */
236#define SEQ_DIST_STAGE_STATES 8u
237static inline uint8_t seq_dist_stage_mask(uint8_t idx) {
238 static const uint8_t order[SEQ_DIST_STAGE_STATES] = {0,1,2,4,3,5,6,7};
239 return order[idx & 7u];
240}
241static inline uint8_t seq_dist_stage_index(uint8_t mask) {
242 for (uint8_t i = 0; i < SEQ_DIST_STAGE_STATES; ++i)
243 if (seq_dist_stage_mask(i) == (mask & 7u)) return i;
244 return 0;
245}
246static inline uint8_t seq_dist_stage_step(uint8_t mask, int step) {
247 uint8_t i = seq_dist_stage_index(mask);
248 return seq_dist_stage_mask((uint8_t)((i + SEQ_DIST_STAGE_STATES + (uint8_t)step)
249 % SEQ_DIST_STAGE_STATES));
250}
251static inline const char *seq_dist_stage_label(uint8_t mask) {
252 static const char *names[SEQ_DIST_STAGE_STATES] =
253 { "OFF", "CLIP", "FOLD", "C+F", "CRSH", "C+H", "F+H", "ALL" };
254 return names[mask & 7u];
255}
