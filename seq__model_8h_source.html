39/* ── Per-step ratchet (sub-trigger count within one step's slot) ──
40 * 1 = plain single trigger (the historical/default behaviour). >1 subdivides
41 * the step evenly into that many evenly-spaced hits, each with its own short
42 * gate — an Elektron-style "ratchet". */
43#define SEQ_MAX_RATCHET 4

45/* ── Per-step conditional trig (Elektron-style) ──
46 * NONE: unconditional — the step fires whenever it is ON, subject only to
47 * probability/ratchet.
48 * FILL: fires only once every step_cond_param loops of the layer's pattern
49 * (param clamped 2..8; the first loop after play-start always counts).
50 * PREV: fires only if the immediately preceding step on the SAME track
51 * actually sounded on its own last evaluation — chains a run of hits,
52 * any miss breaks the chain for the following step. */
53typedef enum {
54 SEQ_STEP_COND_NONE = 0,
55 SEQ_STEP_COND_FILL = 1,
56 SEQ_STEP_COND_PREV = 2,
57 SEQ_STEP_COND_COUNT,
58} seq_step_cond_type_t;

60/* ── Per-step note transform (OP-Z step-component subset, spec 20 §3.1) ──
61 * NONE keeps the step's authored pitch verbatim — the zeroed default, a pure
62 * no-op that leaves the step on the plain periodic-tag path. Any non-NONE mode
63 * offsets the emitted pitch per fire and therefore forces the step onto the
64 * decorated one-shot path (see sequencer_core_step_is_decorated): the plain
65 * path emits one repeating tag with a fixed pitch and AMY has no
66 * per-repetition hook.
67 * RANDOM re-pitches every fire by +/- a fixed semitone span.
68 * RAMP_UP walks the pitch up across successive layer loops, then wraps.
69 * RAMP_DOWN walks it down.
70 * The transformed pitch is re-snapped to the active scale/chord unless the
71 * step's step_quant_bypass bit is set (spec 20 §3.2), in which case it is
72 * emitted chromatically. */
73typedef enum {
74 SEQ_STEP_TRANSFORM_NONE = 0,
75 SEQ_STEP_TRANSFORM_RANDOM = 1,
76 SEQ_STEP_TRANSFORM_RAMP_UP = 2,
77 SEQ_STEP_TRANSFORM_RAMP_DOWN = 3,
78 SEQ_STEP_TRANSFORM_COUNT,
79} seq_step_transform_t;

240 /* ── Per-step probability / ratchet / conditional trig ──
241 * A step with prob==100 && ratchet==1 && cond_type==NONE is "plain" and
242 * keeps using the original always-on repeating AMY sequence tag (zero
243 * extra cost). Any other combination makes the step "decorated": the
244 * periodic tag is left cleared and sequencer_core_service_tick() (called
245 * once per AMY sequencer tick) decides per loop-iteration whether/how it
246 * fires. MUST be initialised to prob=100, ratchet=1 in
247 * sequencer_core_add_layer — memset zeroes them, and 0% probability would
248 * silence every step by default. cond_type=0 (NONE) is the correct zeroed
249 * default and needs no explicit init. */
250 uint8_t step_prob[SEQ_TRACKS][SEQ_MAX_STEPS]; /* 0..100 %, trigger probability */
251 uint8_t step_ratchet[SEQ_TRACKS][SEQ_MAX_STEPS]; /* 1..SEQ_MAX_RATCHET sub-hits */
252 uint8_t step_cond_type[SEQ_TRACKS][SEQ_MAX_STEPS]; /* seq_step_cond_type_t */
253 uint8_t step_cond_param[SEQ_TRACKS][SEQ_MAX_STEPS]; /* FILL: loop divisor 2..8 */
254 /* ── Per-step OP-Z note transform (spec 20 §3.1/§3.2) ──
255 * Both default to 0 = current behaviour: SEQ_STEP_TRANSFORM_NONE keeps the
256 * authored pitch and leaves the step on the plain path; step_quant_bypass=0
257 * re-snaps a transformed pitch to the scale. A zeroed layer needs no init. */
258 uint8_t step_transform[SEQ_TRACKS][SEQ_MAX_STEPS]; /* seq_step_transform_t (0=NONE) */
259 uint8_t step_quant_bypass[SEQ_TRACKS][SEQ_MAX_STEPS]; /* 1 = skip scale snap on transform */
260
261 /* ── Per-step micro-timing / velocity / ratchet taper (patch-06) ──
262 * All three default to 0, which is each field's neutral value: 0 nudge =
263 * on-grid, 0 velocity offset = unchanged, 0 taper = flat ratchet. The
264 * memset(0) in sequencer_core_add_layer therefore yields byte-identical
265 * behaviour with NO explicit init — unlike step_prob/step_ratchet, whose
266 * neutral values are 100/1 and must be initialised there. */
267 int8_t step_nudge[SEQ_TRACKS][SEQ_MAX_STEPS]; /* signed ticks, +-SEQ_STEP_NUDGE_MAX; plain-step only */
268 int8_t step_velocity_adj[SEQ_TRACKS][SEQ_MAX_STEPS]; /* signed percentage points added to velocity (0=none) */
269 int8_t step_ratchet_taper[SEQ_TRACKS][SEQ_MAX_STEPS];/* %-per-sub-hit velocity decay across a ratchet (0=flat) */
270} seq_layer_t;
