6/*
7 * Decorated-step trig jobs (the ingest pump's urgent source).
8 *
9 * sequencer_core_service_tick() always runs on amy_usb_render_task (it's
10 * called from amy_execute_deltas() -> sequencer_check_and_fill(), inline in
11 * amy_update()). trig_schedule_ratchets() ends in amy_helpers_note_send(),
12 * whose amy_helpers_event_begin() asserts if the caller is the render task -
13 * and would be wrong to allow even without the assert, since it can block for
14 * up to AMY_HELPERS_EVENT_TIMEOUT_MS on a mutex/FIFO the render task must
15 * never wait on.
16 *
17 * So the render task's tick hook only enqueues a tiny job descriptor
18 * (non-blocking, never waits) and rings the ingest pump's doorbell. The pump
19 * drains this queue as its registered urgent source, doing the actual
20 * velocity/pitch/tag-emission work on the pump task, where a send applies
21 * inline - ahead of whatever parse backlog the event FIFO holds.
22 *
23 * Deadline: a ratchet's k==0 sub-hit is scheduled for now_ticks+1 - the very
24 * next sequencer tick (single-digit ms at typical tempos) - and AMY silently
25 * drops any event whose tick has already passed. The urgent-before-FIFO drain
26 * order bounds the wait to at most one in-flight amy_add_event() (worst case
27 * a patch-string parse), not the whole backlog.
28 */

53/* Runs on the ingest pump task. Drains one job; returns false when empty. */
54static bool seq_trig_drain_one(void)
55{
56 seq_trig_job_t job;
57 if (xQueueReceive(s_trig_queue, &job, 0) != pdTRUE) return false;
58 /* Closes the race the deferral itself introduces: a delete_layer()
59 * compaction landing between the render task's enqueue and this dequeue
60 * would otherwise read a shifted or reused s_layers[] slot. Re-check both
61 * the in-flight flag and the (possibly now smaller) layer count before
62 * touching s_layers. */
63 if (s_layers_mutating || job.layer_idx >= s_num_layers) {
64 s_trig_drops_mutating++;
65 return true;
66 }
67 trig_schedule_ratchets(job.layer_idx, &s_layers[job.layer_idx],
68 job.track, job.step, job.now_ticks);
69 return true;
70}

85void sequencer_core_trig_enqueue(uint8_t layer_idx, uint8_t track, uint8_t step,
86 uint32_t now_ticks)
87{
88 seq_trig_job_t job = { layer_idx, track, step, now_ticks };
89 /* Zero timeout: the render task must never block here. A full queue means
90 * the consumer has stalled or this tick produced more decorated fires
91 * than the queue was sized for - drop and count, never wait. */
92 if (xQueueSend(s_trig_queue, &job, 0) != pdTRUE) {
93 s_trig_drops_full++;
94 return;
95 }
96 amy_helpers_pump_wake();
97}
