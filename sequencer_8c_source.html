47/* ── LOCAL EDIT (S3-Amysynth): Active-tag index ──────────────────────────
48 * The per-tick scan used to walk 0..highest_tag, where highest_tag is a
49 * high-water mark that only ever grows (e.g. the arp pushes it to ~1119).
50 * That made every 500µs tick O(highest tag ever used), scanning thousands of
51 * mostly-NULL slots on the audio-critical Core-0 timer.
52 *
53 * Instead we keep a compact list of only the tags that currently have deltas,
54 * so the tick is O(active events) regardless of tag magnitude.
55 * s_active_tags is the dense list; s_tag_slot maps tag -> its index (-1 = inactive)
56 * for O(1) swap-removal. Both are guarded by SEQ_LOCK. See AMY-EDITS.md. */
57static int32_t *s_active_tags = NULL;
58static int32_t *s_tag_slot = NULL;
59static int32_t s_num_active = 0;
60
61static inline void seq_active_add(int32_t tag) {
62 if (s_tag_slot == NULL || s_tag_slot[tag] >= 0) return;
63 s_tag_slot[tag] = s_num_active;
64 s_active_tags[s_num_active++] = tag;
65}
66
67static inline void seq_active_remove(int32_t tag) {
68 if (s_tag_slot == NULL) return;
69 int32_t pos = s_tag_slot[tag];
70 if (pos < 0) return;
71 int32_t last = --s_num_active;
72 int32_t moved = s_active_tags[last];
73 s_active_tags[pos] = moved;
74 s_tag_slot[moved] = pos;
75 s_tag_slot[tag] = -1;
76}

150static void sequencer_process_tick(void) {
151 amy_global.sequencer_tick_count++;
152 // LOCAL EDIT (S3-Amysynth): scan only the currently-active tags O(active),
153 // not 0..highest_tag O(highest_tag_ever). Walk backwards so swap-remove
154 // inside seq_active_remove never skips an entry. See AMY-EDITS.md.
155 SEQ_LOCK();
156 for (int32_t i = s_num_active - 1; i >= 0; --i) {
157 int32_t tag = s_active_tags[i];
158 if (sequences[tag].deltas != NULL) {
159 bool hit = false;
160 bool delete = false;
161 if(sequences[tag].period != 0) { // period set
162 uint32_t offset = amy_global.sequencer_tick_count % sequences[tag].period;
163 if (offset == sequences[tag].tick) hit = true;
164 } else {
165 // Test for absolute tick (no period set)
166 if (sequences[tag].tick == amy_global.sequencer_tick_count) { hit = true; delete = true; }
167 }
168 if(hit) {
169 struct delta *d = sequences[tag].deltas;
170 while(d) {
171 // assume the d->time is 0 and that's good.
172 add_delta_to_queue(d, &amy_global.delta_queue);
173 d = d->next;
174 }
175 // Delete absolute tick addressed sequence entry if sent
176 if(delete) {
177 delta_release_list(sequences[tag].deltas);
178 sequences[tag].deltas = NULL;
179 seq_active_remove(tag);
180 }
181 }
182 }
183 }
184 SEQ_UNLOCK();
185 // call the right hook:
186#ifdef __EMSCRIPTEN__
187 EM_ASM({
188 if(typeof amy_sequencer_js_hook === 'function') {
189 amy_sequencer_js_hook($0);
190 }
191 }, amy_global.sequencer_tick_count);
192#endif
193 if(amy_global.config.amy_external_sequencer_hook != NULL) {
194 amy_global.config.amy_external_sequencer_hook(amy_global.sequencer_tick_count);
195 }
196}

223void sequencer_external_clock_disable() {
224 // Leave external-clock mode and hand back to the internal timer. Without
225 // this, sequencer_external_clock latches true on the first F8 tick and is
226 // never cleared, so the internal sequencer stays dead once an external
227 // clock stops -- even after the caller turns external sync back off. Called
228 // from amy_external_midi_sync(0) so disabling sync is a real recovery path.
229 sequencer_external_clock = false;
230 sequencer_running = true;
231 // Re-anchor the tick timer to now so sequencer_check_and_fill doesn't try to
232 // replay every tick that elapsed while we were on external clock.
233 amy_global.next_amy_tick_us = (uint64_t)amy_sysclock() * 1000L;
234}

236uint8_t sequencer_add_event(amy_event *e) {
237 // add this event to the list of sequencer events in the LL.
238 // e->sequence is set up.
239 // if the tag already exists - if there's tick/period, overwrite, if there's no tick / period, we should remove the entry
240 //fprintf(stderr, "sequencer_add_event: e->instrument %d e->note %.0f e->vel %.2f tick %d period %d tag %d\n", e->instrument, e->midi_note, e->velocity, e->sequence[SEQUENCE_TICK], e->sequence[SEQUENCE_PERIOD], e->sequence[SEQUENCE_TAG]);
241 int32_t tag = e->sequence[SEQUENCE_TAG];
242 if (tag > max_sequences) {
243 fprintf(stderr, "sequencer tag %" PRIi32" (with tick %" PRIu32", period %" PRIu32") is greater than or eq max_sequences %" PRIi32"\n",
244 tag, e->sequence[SEQUENCE_TICK], e->sequence[SEQUENCE_PERIOD], max_sequences);
245 // ignore
246 return 0;
247 }
248 SEQ_LOCK();
249 // Release any existing deltas for this tag, even if we're just going to rewrite them.
250 delta_release_list(sequences[tag].deltas);
251 sequences[tag].deltas = NULL;
252
253 // Clearing a tag (no tick/period, or scheduled in the past) makes it
254 // inactive: drop it from the active-tag index so the per-tick scan shrinks.
255 if(e->sequence[SEQUENCE_TICK] == 0 && e->sequence[SEQUENCE_PERIOD] == 0) {
256 seq_active_remove(tag); SEQ_UNLOCK(); return 0;
257 }
258 if(e->sequence[SEQUENCE_TICK] != 0 && e->sequence[SEQUENCE_PERIOD] == 0 && e->sequence[SEQUENCE_TICK] <= amy_global.sequencer_tick_count) {
259 seq_active_remove(tag); SEQ_UNLOCK(); return 0;
260 }
261
262 // Save the tick & period.
263 sequences[tag].tick = e->sequence[SEQUENCE_TICK];
264 sequences[tag].period = e->sequence[SEQUENCE_PERIOD];
265 // Copy all the deltas for this event to the sequences entry.
266 amy_event_to_deltas_queue(e, 0, &sequences[tag].deltas);
267
268 seq_active_add(tag);
269 if (tag > highest_tag) highest_tag = tag; // kept for sequencer_debug() only
270 SEQ_UNLOCK();
271 return 1;
272}

275void sequencer_check_and_fill() {
276 if (!sequencer_running || sequencer_external_clock) return;
277 // If we've fallen behind by more than 1 second (e.g. sequencer was stopped
278 // and restarted, or a long blocking operation occurred), skip ahead instead
279 // of processing hundreds of backed-up ticks at once.
280 uint64_t now_us = (uint64_t)amy_sysclock() * 1000L;
281 if (now_us > amy_global.next_amy_tick_us + 1000000ULL) {
282 amy_global.next_amy_tick_us = now_us;
283 }
284 // The while is in case the timer fires later than a tick; (on esp this would be due to SPI or wifi ops)
285 // LOCAL EDIT (S3-Amysynth): compare in the us domain against the now_us
286 // already computed above. The old ms-domain compare re-read amy_sysclock()
287 // AND paid a 64-bit divide (next_amy_tick_us / 1000) on every check - a
288 // real __udivdi3 libcall per 500 us timer callback on 32-bit targets,
289 // since GCC cannot strength-reduce a 64-bit divide by a constant there.
290 // Behavior delta: a tick's deadline is no longer rounded down to the ms
291 // boundary, so a tick can fire up to 1 ms later (never earlier) within
292 // the same 500 us callback grid; the accumulated tick rate is unchanged.
293 // Upstream-PR candidate.
294 while(now_us >= amy_global.next_amy_tick_us) {
295 sequencer_process_tick();
296 amy_global.next_amy_tick_us = amy_global.next_amy_tick_us + (uint64_t)amy_global.us_per_tick;
297 }
298}
