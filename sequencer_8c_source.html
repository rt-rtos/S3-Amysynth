10// Sequenced ticks events are stored as the raw wire-message string (with its
11// leading 'H' command stripped) plus the scheduling metadata needed to play
12// it back. The string is only parsed when the entry comes due.
13typedef struct sequence_info_t {
14 char *wire; // Stored wire message; NULL means the tag is unused.
15 //uint32_t tag; // tag is implicit, it's its index in the table
16 uint32_t tick; // 0 means not used
17 uint32_t period; // 0 means not used
18 // Next OCCUPIED slot, or -1 for the end. Only meaningful while this
19 // entry has a wire -- `wire != NULL` is what "in the list" means, so
20 // there is one source of truth and not two to keep in step.
21 int32_t next_active;
22} sequence_info_t;

24struct sequence_info_t *sequences = NULL; // An array indexed by tag.
25int32_t max_sequences = 0; // Number of user-addressable tags.
26// Head of the ascending list of occupied slots (user tags and anonymous
27// entries alike); -1 when nothing is scheduled. This replaces `highest_tag`,
28// which was a HIGH-WATER MARK: it only ever grew, so one event at a high tag
29// made every tick scan that far for the rest of the session, long after that
30// sequence was cleared. The anonymous pool made that the common case, not a
31// corner: anonymous entries are allocated round-robin at indices past
32// max_sequences, so a burst of ticks= one-shots pinned the mark at the very
33// end of the table permanently. The cost is proportional to what is
34// scheduled now.
35int32_t first_active = -1;
36// Anonymous (no-tag) entries live past the user-addressable tag range, at
37// indices [max_sequences .. max_sequences+AMY_ANON_SEQUENCE_SLOTS), so a
38// user-supplied tag (bounds-checked against max_sequences) can never reach
39// or clobber one. Allocated round-robin; a new anonymous entry silently
40// evicts the oldest one once the pool wraps around.
41#define AMY_ANON_SEQUENCE_SLOTS 256
42static int32_t anon_cursor = 0;
43static volatile bool sequencer_running = true;
44static volatile bool sequencer_external_clock = false;
45// Firing a stored message parses it, and parsing (e.g. a patch load) can call
46// amy_execute_deltas(), which calls back into the check functions below. This
47// flag makes those nested calls no-ops so a tick is never processed twice.
48static volatile bool wire_firing = false;

50void sequencer_init(int max_sequencer_tags) {
51 // These are statics, so a stop/start of AMY within one process needs them
52 // put back to their boot state (internal clock, running).
53 sequencer_running = true;
54 sequencer_external_clock = false;
55 wire_firing = false;
56 anon_cursor = 0;
57 max_sequences = max_sequencer_tags;
58 int32_t total_slots = max_sequences + AMY_ANON_SEQUENCE_SLOTS;
59 sequences = (struct sequence_info_t *)malloc_caps(total_slots * sizeof(struct sequence_info_t),
60 amy_global.config.ram_caps_synth);
61 // LOCAL EDIT (S3-Amysynth): OOM guard - disable the sequencer wholesale
62 // instead of writing through a NULL table (sequencer_add_wire and
63 // sequencer_check_and_fill already treat sequences == NULL as "not
64 // initialized", so every downstream path degrades to a no-op).
65 if (sequences == NULL) {
66 amy_oom("sequencer_init");
67 max_sequences = 0;
68 first_active = -1;
69 return;
70 }

72 sequences[i].wire = NULL;
73 sequences[i].tick = 0;
74 sequences[i].period = 0;
75 sequences[i].next_active = -1;
76 }
77 first_active = -1;
78 // We are read to go.
79 sequencer_recompute();
80}

118/* The occupied slots, threaded through the table as an ASCENDING list.
119 *
120 * Why threaded rather than a list of its own: the table has to stay
121 * indexable, because add and clear both reach a tag directly and want O(1)
122 * to do it. This gets the tick scan down to the number of sequences
123 * actually scheduled without giving that up, and without allocating
124 * anything the render thread could walk into while it is being freed.
125 *
126 * WHY ASCENDING, and it is not tidiness: two sequences that hit on the same
127 * tick play in the order they are visited, so the order decides which one
128 * wins if they touch the same parameter. That order was slot order when
129 * this was an indexed sweep, and keeping the list sorted keeps it slot
130 * order. An insertion-ordered list would make a pattern sound different
131 * after an edit.
132 *
133 * THREAD SAFETY. Link mutations happen only under the amy lock --
134 * sequencer_add_wire() takes it, the tick loop's delete path takes it, and
135 * sequencer_reset() is called with it already held -- so writers are
136 * serialized. The tick WALK, though, runs without the lock, which is safe
137 * because the links are INDICES INTO A FIXED ARRAY, not pointers:
138 *
139 * - publishing a splice is one aligned 32-bit store, so a walker sees
140 * either the old link or the new one, never half of one;
141 * - every stored link is greater than the slot holding it, so walking
142 * strictly increases the index. A stale link can make a walker skip a
143 * sequence or revisit one for a single tick; it cannot form a cycle,
144 * cannot hang, and cannot leave the array.
145 *
146 * So the worst a race costs is one tick's events being wrong, which is the
147 * same class of hazard the indexed sweep already had. A list of malloc'd
148 * nodes would be a different class entirely -- a torn next pointer walks
149 * the render thread into freed memory.
150 */
151static void active_link(int32_t tag)
152{
153 int32_t *prev = &first_active;
154 while (*prev != -1 && *prev < tag)
155 prev = &sequences[*prev].next_active;
156 if (*prev == tag)
157 return; /* already in */
158 sequences[tag].next_active = *prev; /* point at the tail we found... */
159 *prev = tag; /* ...then publish, in one store */
160}

162static void active_unlink(int32_t tag)
163{
164 int32_t *prev = &first_active;
165 while (*prev != -1 && *prev != tag)
166 prev = &sequences[*prev].next_active;
167 if (*prev == tag)
168 *prev = sequences[tag].next_active; /* one store, again */
169}

171void sequencer_recompute() {
172 // 60000000 us/min / (bpm * ticks per beat); keep it single-precision -
173 // unsuffixed double literals pull in software double emulation on 32-bit.
174 amy_global.us_per_tick = (uint32_t) (60000000.0f / (amy_global.tempo * (float)AMY_SEQUENCER_PPQ));
175 // A wire message can set an absurd tempo; us_per_tick == 0 would make the
176 // catch-up loop in sequencer_check_and_fill spin forever.
177 if (amy_global.us_per_tick < 50) amy_global.us_per_tick = 50;
178 amy_global.next_amy_tick_us = (amy_sysclock64() * 1000ULL) + (uint64_t)amy_global.us_per_tick;

181// Store a wire message in the sequencer. Takes ownership of wire (malloc'd).
182//
183// has_tag false means tag wasn't supplied by the caller (a 1- or 2-value
184// ticks= form): the entry is allocated round-robin from the anonymous pool
185// instead of the given tag value, so it's stored but not addressable or
186// individually cancelable. has_tag true is the normal tag-indexed form: tick
187// and period both zero clears that tag's entry (the only way to cancel one).
188//
189// A one-off whose tick is already due or overdue is not stored at all -- it
190// plays immediately, before returning. See the comment at that branch.
191uint8_t sequencer_add_wire(uint32_t tick, uint32_t period, uint32_t tag, bool has_tag, char *wire) {
192 if (sequences == NULL) { // sequencer_init hasn't run
193 free(wire);
194 return 0;
195 }
196 if (has_tag) {
197 if (tag >= (uint32_t)max_sequences) {
198 fprintf(stderr, "sequencer tag %" PRIu32" (with tick %" PRIu32", period %" PRIu32") is greater than or eq max_sequences %" PRIi32"\n",
199 tag, tick, period, max_sequences);
200 free(wire);
201 return 0;
202 }
203 } else {
204 // Anonymous: tick==0 && period==0 has nothing to cancel (no tag was
205 // given), so just drop it rather than allocating a slot for a no-op.
206 if (tick == 0 && period == 0) {
207 free(wire);
208 return 0;
209 }
210 tag = (uint32_t)(max_sequences + anon_cursor);
211 anon_cursor = (anon_cursor + 1) % AMY_ANON_SEQUENCE_SLOTS;
212 }
213 amy_grab_lock();
214 // Release any existing message for this tag, even if we're just going to rewrite it.
215 if (sequences[tag].wire) free(sequences[tag].wire);
216 sequences[tag].wire = NULL;
217 sequences[tag].tick = 0;
218 sequences[tag].period = 0;
219 active_unlink(tag); // out of the list while it has nothing in it
220 if (tick == 0 && period == 0) { // Non-schedulable event: just clear the tag.
221 amy_release_lock();
222 free(wire);
223 return 0;
224 }
225 if (period == 0 && tick <= amy_global.sequencer_tick_count) {
226 // A one-off that is already due or overdue. Play it NOW rather than
227 // dropping it: a caller that reads the tick clock and schedules
228 // relative to it always runs a little after the tick it read (every
229 // Python path arrives via a deferred callback), and at 48 PPQ any
230 // offset under a tick rounds straight back onto the current count.
231 // Dropping made that lost race silent -- it is what made the Tulip
232 // arpeggiator play nothing at all. Late by a fraction of a tick beats
233 // never, and it restores what the old millisecond time= did with a
234 // past-due event. NB tick==0 is the cancel form, handled above, so it
235 // never reaches here.
236 //
237 // Play outside the lock and free after, exactly as the tick loop does:
238 // amy_queue_lock is a plain non-recursive mutex and amy_play_message()
239 // re-enters the parser, which can land back in this function.
240 amy_release_lock();
241 amy_play_message(wire);
242 free(wire);
243 return 1;
244 }
245 sequences[tag].tick = tick;
246 sequences[tag].period = period;
247 sequences[tag].wire = wire;
248 active_link(tag); // ...and back in, now that it has a message again
249 amy_release_lock();
250 return 1;
251}

253static void sequencer_process_tick(void) {
254 amy_global.sequencer_tick_count++;
255 midi_clock_out_tick(); // no-op unless in AMY_MIDI_SYNC_SEND mode
256 // Guard nested check-and-fire calls (via a fired message's own parse)
257 // while still processing this tick's fires; restore on the way out.
258 bool was_firing = wire_firing;
259 wire_firing = true;
260 // Walk only the slots that have something scheduled. This used to sweep
261 // 0..highest_tag, a mark that never came down.
262 int32_t tag = first_active;
263 while (tag != -1) {
264 // Read the link BEFORE anything below can unlink this entry.
265 int32_t next = sequences[tag].next_active;
266 if (sequences[tag].wire != NULL) {
267 bool hit = false;
268 bool delete = false;
269 if(sequences[tag].period != 0) { // period set
270 uint32_t offset = amy_global.sequencer_tick_count % sequences[tag].period;
271 if (offset == sequences[tag].tick) hit = true;
272 } else {
273 // Test for absolute tick (no period set). <= rather than ==:
274 // the walk above runs without the lock, and a stale link can
275 // make it skip an entry for a single tick (see the thread
276 // safety note). Under ==, a slot skipped on exactly its tick
277 // would sit there forever, holding an anon slot and never
278 // playing. <= lets it fire on the next tick instead, matching
279 // the play-it-late rule sequencer_add_wire() uses for a
280 // one-off that is already due when it arrives.
281 if (sequences[tag].tick <= amy_global.sequencer_tick_count) { hit = true; delete = true; }
282 }
283 if(hit) {
284 // Take the message out (one-shot) or a copy of it (repeating)
285 // under the lock, so an ingest thread rewriting the tag can't
286 // free the string while we parse it.
287 char *wire = NULL;
288 amy_grab_lock();
289 if (sequences[tag].wire != NULL) {
290 if (delete) {
291 wire = sequences[tag].wire;
292 sequences[tag].wire = NULL;
293 sequences[tag].tick = 0;
294 sequences[tag].period = 0;
295 active_unlink(tag);
296 } else {
297 size_t len = strlen(sequences[tag].wire);
298 wire = (char *)malloc_caps(len + 1, amy_global.config.ram_caps_sequencer);
299 // LOCAL EDIT (S3-Amysynth): sequencer caps + events-
300 // pool fallback, matching amy_add_event (api.c) - a
301 // periodic fire must not go silent because osc state
302 // exhausted a pool.
303 if (wire == NULL)
304 wire = (char *)malloc_caps(len + 1, amy_global.config.ram_caps_events);
305 if (wire != NULL) memcpy(wire, sequences[tag].wire, len + 1);
306 else amy_oom("sequencer fire");
307 }
308 }
309 amy_release_lock();
310 if (wire != NULL) {
311 // Parse and play now; the deltas play back within this block.
312 amy_play_message(wire);
313 free(wire);
314 }
315 }
316 }
317 tag = next;
318 }
319 wire_firing = was_firing;
320 if(amy_global.config.amy_external_sequencer_hook != NULL) {
321 amy_global.config.amy_external_sequencer_hook(amy_global.sequencer_tick_count);
322 }
323}

330void sequencer_check_and_call_js_hook() {
331 static uint32_t last_reported_tick = 0;
332 uint32_t tick = amy_global.sequencer_tick_count;
333 if (tick < last_reported_tick) last_reported_tick = tick; // sequencer was reset
334 // If we're more than a second of ticks behind (e.g. the page was
335 // backgrounded and the main loop paused), skip ahead rather than firing a
336 // burst of stale hook calls.
337 if (amy_global.us_per_tick > 0) {
338 uint32_t ticks_per_sec = 1000000 / amy_global.us_per_tick;
339 if (tick - last_reported_tick > ticks_per_sec) last_reported_tick = tick - ticks_per_sec;
340 }
341 while (last_reported_tick < tick) {
342 ++last_reported_tick;
343 EM_ASM({
344 if(typeof amy_sequencer_js_hook === 'function') {
345 amy_sequencer_js_hook($0);
346 }
347 }, last_reported_tick);
348 }

379void sequencer_external_clock_disable() {
380 // Leave external-clock mode and hand back to the internal timer. Without
381 // this, sequencer_external_clock latches true on the first F8 tick and is
382 // never cleared, so the internal sequencer stays dead once an external
383 // clock stops -- even after the caller turns external sync back off. Called
384 // from amy_external_midi_sync(0) so disabling sync is a real recovery path.
385 sequencer_external_clock = false;
386 sequencer_running = true;
387 // Re-anchor the tick timer to now so sequencer_check_and_fill doesn't try to
388 // replay every tick that elapsed while we were on external clock.
389 amy_global.next_amy_tick_us = amy_sysclock64() * 1000ULL;
390}

392// Called once per block from amy_execute_deltas(). Ticks are decided against
393// amy_sysclock(), which counts rendered samples, so the sequencer advances on
394// AMY time in any rendering context (live, offline, tests).
395void sequencer_check_and_fill() {
396 if (sequences == NULL) return; // sequencer_init hasn't run
397 if (sequencer_external_clock) return;
398 if (wire_firing) return; // nested via a fired message's own parse
399 // When we're the MIDI clock master, realtime clock keeps flowing even while
400 // the transport is stopped so slaves stay tempo-locked; otherwise a stopped
401 // sequencer has nothing to do.
402 if (!sequencer_running && !midi_clock_out_enabled()) return;
403 // If we've fallen behind by more than 1 second (e.g. sequencer was stopped
404 // and restarted, or a long blocking operation occurred), skip ahead instead
405 // of processing hundreds of backed-up ticks at once.
406 // next_amy_tick_us is a 64-bit accumulator, so it must be anchored to the
407 // 64-bit clock. Seeding it from the 32-bit amy_sysclock() used to kill the
408 // sequencer permanently at the 49.7-day rollover: now_us collapsed to ~0
409 // while next_amy_tick_us stayed at ~4.3e12, and neither the catch-up guard
410 // (which only handles falling behind) nor the tick loop could ever fire.
411 uint64_t now_us = amy_sysclock64() * 1000ULL;
412 if (now_us > amy_global.next_amy_tick_us + 1000000ULL) {
413 amy_global.next_amy_tick_us = now_us;
414 }
415 // The while is in case the timer fires later than a tick; (on esp this would be due to SPI or wifi ops)
416 while(now_us >= amy_global.next_amy_tick_us) {
417 if (sequencer_running) sequencer_process_tick();
418 else midi_clock_out_tick(); // transport stopped: clock only, no sequence events
419 amy_global.next_amy_tick_us = amy_global.next_amy_tick_us + (uint64_t)amy_global.us_per_tick;
420 }
