15/* ── Virtual wave-patch IDs ─────────────────────────────────────────────────
16 * Patch numbers beyond the 0..256 built-in (Juno/DX7/piano) range.
17 * Intercepted before amy_send_patch() so they never collide with real patches.
18 * Melodic tracks, arp, and drone all use these constants for wave-patch routing.
19 * Melodic and arp route the FULL 0..SEQ_PATCH_FULL_MAX range; the drone opts
20 * out of the ranges its excitation model can't play via drone_patch_excluded()
21 * (drone_core.c). Exclusions are per-consumer predicates on one shared catalog
22 * (patch_cycle.h / ui_patch_cycle.c) — never per-consumer copies of the list. */
23#define SEQ_PATCH_WAVE_BASE 257
24#define SEQ_PATCH_SINE 257 /* AMY SINE */
25#define SEQ_PATCH_SAW_DOWN 258 /* AMY SAW_DOWN */
26#define SEQ_PATCH_SAW_UP 259 /* AMY SAW_UP */
27#define SEQ_PATCH_PULSE 260 /* AMY PULSE */
28#define SEQ_PATCH_TRIANGLE 261 /* AMY TRIANGLE */
29#define SEQ_PATCH_NOISE 262 /* AMY NOISE — arp only; drone excludes */
30#define SEQ_PATCH_KS 263 /* AMY KS — arp only; drone excludes */
31#define SEQ_PATCH_WAVE_MAX 263

33/* ── Multi-osc bass presets (melodic only; oscs_per_voice=2) ────────────
34 * Intercepted before amy_send_patch() — never routed to the Juno/DX7 loader.
35 * Osc 0 is the primary audio carrier; osc 1 is the secondary/sub layer.
36 * Osc 0 envelope is overwritten by sequencer_configure_melodic_envelope()
37 * after configure; characteristic sound comes from osc structure + filter. */
38#define SEQ_PATCH_BASS_BASE 264
39#define SEQ_PATCH_BASS_1 264 /* Classic Sub-Heavy Detune (PULSE + detuned SAW, LPF24) */
40#define SEQ_PATCH_BASS_2 265 /* Solid Sine-Reinforced Acid/Pluck (SINE + SAW, LPF24) */
41#define SEQ_PATCH_BASS_3 266 /* FM DX7-Style (SINE carrier + sub-octave SINE, DX7 env) */
42#define SEQ_PATCH_BASS_MAX 266

44/* ── Wavetable virtual patches (melodic, arp, drone; AMY_WAVETABLE only) ──
45 * One virtual patch per built-in wavetable bank (see pcm_tiny.h /
46 * pcm_wavetable_base). wave=WAVETABLE, preset=pcm_wavetable_base+index.
47 * Intercepted the same way as SEQ_PATCH_WAVE_BASE — never sent to
48 * amy_send_patch(). Range kept separate from SEQ_PATCH_WAVE_BASE so it can be
49 * added without renumbering the bass presets above. */
50/* The numbers exist unconditionally (like every virtual range) so that
51 * compile-aware code — sequencer_core_patch_compiled_out() below — can name
52 * them in any build; only their *routability* is gated on the Kconfig flag. */
53#define SEQ_PATCH_WAVETABLE_BASE 267
54#define SEQ_PATCH_WAVETABLE_0 267 /* 111.WAV */
55#define SEQ_PATCH_WAVETABLE_1 268 /* BRAIDS01.WAV */
56#define SEQ_PATCH_WAVETABLE_2 269 /* PPG_WA00.WAV */
57#define SEQ_PATCH_WAVETABLE_3 270 /* SINE2SAW.WAV */
58#define SEQ_PATCH_WAVETABLE_4 271 /* VIRAL.WAV */
59#define SEQ_PATCH_WAVETABLE_MAX 271
60
61/* True for any raw-waveform virtual patch (SINE..KS, and — when AMY_WAVETABLE
62 * is compiled in — the wavetable bank patches). Shared by the melodic
63 * synth/LFO configurators so both ranges route the same way (direct oscillator
64 * config, no amy_send_patch(), native-LFO eligible). */
65static inline bool sequencer_core_is_wave_patch(uint16_t patch)
66{
67 if (patch >= SEQ_PATCH_WAVE_BASE && patch <= SEQ_PATCH_WAVE_MAX) return true;
68#if CONFIG_AMY_WAVETABLE
69 if (patch >= SEQ_PATCH_WAVETABLE_BASE && patch <= SEQ_PATCH_WAVETABLE_MAX) return true;
70#endif
71 return false;
72}

74/* ── DX7-style 6-operator FM/ALGO voices (melodic only; oscs_per_voice=7) ──
75 * Osc 0 is the AMY ALGO control osc (algorithm + algo_source[0..5] wired to
76 * relative oscs 1..6); oscs 1..6 are SINE operators. Intercepted before
77 * amy_send_patch() exactly like the bass presets above.
78 * FM_BASS/EPIANO/BELL/LEAD are fixed starter presets (see fm_presets.c).
79 * FM_CUSTOM is the single live-editable voice driven by the FM UI screen
80 * (see custompatches/fm_voice.h); its parameters are global, not per-layer —
81 * every melodic row currently on this patch shares the same edited voice.
82 * Numbered to start unconditionally after the wavetable range (267-271, only
83 * ever compiled in under CONFIG_AMY_WAVETABLE) rather than reusing it, so FM
84 * patch numbers never shift under a build-flag change and the two virtual
85 * patch features can never collide. */
86#define SEQ_PATCH_FM_BASE 272
87#define SEQ_PATCH_FM_BASS 272 /* FM Bass (2-op chain, algorithm 0) */
88#define SEQ_PATCH_FM_EPIANO 273 /* FM E.Piano (2 carriers, algorithm 0) */
89#define SEQ_PATCH_FM_BELL 274 /* FM Bell (inharmonic ratios, algorithm 0) */
90#define SEQ_PATCH_FM_LEAD 275 /* FM Lead (brighter 2-op chain, algorithm 0) */
91#define SEQ_PATCH_FM_CUSTOM 276 /* Live-editable voice — opens the FM screen */
92#define SEQ_PATCH_FM_MAX 276

94/* ── Additive/partials voices (melodic + arp; oscs_per_voice = N+1) ────────
95 * Osc 0 is the AMY BYO_PARTIALS control osc (preset = partial count, shared
96 * envelope); oscs 1..N are PARTIAL sines summed additively — no algo_source
97 * routing, AMY derives the child set from preset + osc adjacency. Intercepted
98 * before amy_send_patch() exactly like the FM range. ORGAN/BELL are fixed
99 * presets (see additive_presets.c); ADDITIVE_CUSTOM is the single
100 * live-editable voice (custompatches/additive_voice.h), global not per-layer,
101 * playing its drawbar-organ default until an additive editor screen lands.
102 * Numbered to start unconditionally after the FM range for the same reason FM
103 * sits above the wavetables: patch numbers never shift under a build-flag
104 * change and the virtual ranges can never collide. */
105#define SEQ_PATCH_ADDITIVE_BASE 277
106#define SEQ_PATCH_ADDITIVE_ORGAN 277 /* drawbar organ, 8 harmonics, 1/n */
107#define SEQ_PATCH_ADDITIVE_BELL 278 /* inharmonic free-bar bell ratios */
108#define SEQ_PATCH_ADDITIVE_CUSTOM 279 /* live-editable additive voice */
109#define SEQ_PATCH_ADDITIVE_MAX 279
110#define SEQ_PATCH_ROUTABLE_MAX SEQ_PATCH_ADDITIVE_MAX

112/* True when `patch` falls in a virtual range whose feature is NOT compiled
113 * into this build. The single compile-awareness point for the whole patch
114 * space: browse domains skip these numbers via their `excluded` predicate
115 * (patch_cycle.h) and the patch-kind dispatch snaps them to a safe fallback,
116 * so a new gated range only needs a clause here — no ceiling cascade, no
117 * per-consumer hardcoded holes. Patch numbers are fixed unconditionally
118 * (ranges never renumber under build flags), which is what makes holes
119 * possible in the first place. */
120static inline bool sequencer_core_patch_compiled_out(uint16_t patch)
121{
122#if !CONFIG_AMY_WAVETABLE
123 if (patch >= SEQ_PATCH_WAVETABLE_BASE && patch <= SEQ_PATCH_WAVETABLE_MAX) return true;
124#endif
125#if !CONFIG_SYNTH_CUSTOM_FM
126 if (patch >= SEQ_PATCH_FM_BASE && patch <= SEQ_PATCH_FM_MAX) return true;
127#endif
128#if !CONFIG_SYNTH_ADDITIVE
129 if (patch >= SEQ_PATCH_ADDITIVE_BASE && patch <= SEQ_PATCH_ADDITIVE_MAX) return true;
130#endif
131 (void)patch;
132 return false;
133}

187/* ── Drum sound source (whole-layer Synth vs PCM) ──
