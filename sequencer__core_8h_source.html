79/* Native-LFO layout for a patch: which voice-relative osc carries the LFO pair
80 * (carrier; wobble = carrier+1) and which audible oscs take the COEF_MOD
81 * coupling (bitmask, bit n = osc n). True for every topology reserving a
82 * trailing carrier pair (raw-wave/wavetable 3-osc, custom bass 4-osc); false
83 * means the 20 Hz software stepper serves the LFO. THE single
84 * native-vs-software predicate - every gate must use it, since a site left on
85 * sequencer_core_is_wave_patch would stack the software stepper on a native
86 * carrier (double modulation). Out-params may be NULL. */
87static inline bool sequencer_core_lfo_native_layout(uint16_t patch,
88 uint8_t *carrier_osc,
89 uint8_t *coupled_mask)
90{
91 if (sequencer_core_is_wave_patch(patch)) {
92 if (carrier_osc) *carrier_osc = 1;
93 if (coupled_mask) *coupled_mask = 0x01;
94 return true;
95 }
96 if (patch >= SEQ_PATCH_BASS_BASE && patch <= SEQ_PATCH_BASS_MAX) {
97 if (carrier_osc) *carrier_osc = 2; /* osc0/1 audible pair */
98 if (coupled_mask) *coupled_mask = 0x03; /* couple BOTH audible oscs */
99 return true;
100 }
101 return false;
102}
