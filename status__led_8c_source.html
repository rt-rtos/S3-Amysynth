1// Status LED updater (contract in status_led.h). A low-priority Core-0 task

3// rewrite happen only once per CONFIG_AMYSYNTH_STATUS_LED_PERIOD_MS window,
4// and the WS2812 latches its color, so the steady-state cost is one

29typedef struct {
30 uint8_t r, g, b;
31} rgb_t;

33// Full-scale band colors, scaled by BRIGHTNESS/255 at write time. Each color
34// is pre-normalized to roughly equal photopic luma (weights R 0.30 / G 0.59 /
35// B 0.11, anchored to red = dimmest primary at full scale): the eye - and the
36// WS2812 green die - make an unweighted green read far brighter than red, and
37// two-die yellow brighter still. led_strip offers no gamma/luma facility (its
38// HSV V is just the max channel), so the weighting lives in this table.
39// Orange keeps a low green share so it stays distinct from yellow after the
40// brightness scaling quantizes the channels.
41static const rgb_t s_band_color[] = {
42 {0, 130, 0}, // green: < 50 %
43 {86, 86, 0}, // yellow: >= 50 %
44 {171, 43, 0}, // orange: >= 75 %
45 {255, 0, 0}, // red: >= 90 %
46};
47// Band i is entered upward at load >= s_band_enter[i], left downward at
48// load < s_band_enter[i] - STATUS_LED_HYST.
49static const uint8_t s_band_enter[] = {0, 50, 75, 90};
50#define STATUS_LED_NUM_BANDS (sizeof(s_band_enter) / sizeof(s_band_enter[0]))
51#define STATUS_LED_HYST 5
52
53static const rgb_t s_blink_color = {186, 0, 186}; // magenta (luma-matched): alloc failure
54static const rgb_t s_off_color = {0, 0, 0};

56static led_strip_handle_t s_strip;

59static void write_color(rgb_t c)
60{
61 static rgb_t s_current = {0, 0, 0};
62 static bool s_written;
63 if (s_written && c.r == s_current.r && c.g == s_current.g && c.b == s_current.b) {
64 return;
65 }
66 const uint32_t scale = CONFIG_AMYSYNTH_STATUS_LED_BRIGHTNESS;
67 esp_err_t err = led_strip_set_pixel(s_strip, 0,
68 (c.r * scale) / 255,
69 (c.g * scale) / 255,
70 (c.b * scale) / 255);
71 if (err == ESP_OK) {
72 err = led_strip_refresh(s_strip);
73 }
74 if (err == ESP_OK) {
75 s_current = c;
76 s_written = true;
77 }
78}

80static unsigned band_update(unsigned band, uint8_t load)
81{
82 while (band + 1 < STATUS_LED_NUM_BANDS && load >= s_band_enter[band + 1]) {
83 band++;
84 }
85 while (band > 0 && load + STATUS_LED_HYST < s_band_enter[band]) {
86 band--;
87 }
88 return band;
89}

91static void status_led_task(void *arg)
92{
93 (void)arg;
94 const uint32_t ticks_per_sample =
95 CONFIG_AMYSYNTH_STATUS_LED_PERIOD_MS / STATUS_LED_TICK_MS;
96 unsigned band = 0;
97 unsigned blink_ticks = 0;
98 uint32_t tick = 0;
99
100 TickType_t last_wake = xTaskGetTickCount();
101 for (;;) {
102 vTaskDelayUntil(&last_wake, pdMS_TO_TICKS(STATUS_LED_TICK_MS));
103
104 if (atomic_exchange(&s_alloc_failed, false)) {
105 blink_ticks = STATUS_LED_BLINK_TICKS;
106 }
107
108 if (blink_ticks > 0) {
109 write_color((blink_ticks & 1) ? s_blink_color : s_off_color);
110 blink_ticks--;
111 continue;
112 }
113
114 if (++tick >= ticks_per_sample) {
115 tick = 0;
116 uint8_t busy[CORE_LOAD_NUM_CORES];
117 if (core_load_sample(busy)) {
118 band = band_update(band, busy[STATUS_LED_RENDER_CORE]);
119 }
120 }
121 write_color(s_band_color[band]);
122 }
123}

141void status_led_start(void)
142{
143 led_strip_config_t strip_config = {
144 .strip_gpio_num = CONFIG_AMYSYNTH_STATUS_LED_GPIO,
145 .max_leds = 1,
146 .led_model = LED_MODEL_WS2812,
147 .color_component_format = LED_STRIP_COLOR_COMPONENT_FMT_GRB,
148 };
149 led_strip_rmt_config_t rmt_config = {
150 .clk_src = RMT_CLK_SRC_DEFAULT,
151 .resolution_hz = 10 * 1000 * 1000,
152 .flags.with_dma = false, // 1 LED: a 24-bit frame fits the RMT FIFO
153 };
154 esp_err_t err = led_strip_new_rmt_device(&strip_config, &rmt_config, &s_strip);
155 if (err != ESP_OK) {
156 ESP_LOGW(TAG, "LED driver init failed (%s); status LED disabled",
157 esp_err_to_name(err));
158 s_strip = NULL;
159 return;
160 }
161 led_strip_clear(s_strip);
162
163 BaseType_t ok = xTaskCreatePinnedToCore(status_led_task, "status_led",
164 3072, NULL, 2, NULL, 0);
165 if (ok != pdPASS) {
166 ESP_LOGW(TAG, "task create failed; status LED disabled");
167 led_strip_del(s_strip);
168 s_strip = NULL;
169 return;
170 }
171
172#if CONFIG_AMYSYNTH_STATUS_LED_ALLOC_BLINK
173 heap_caps_register_failed_alloc_callback(alloc_failed_hook);
174#endif
175}
