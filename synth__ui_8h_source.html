223/* ─── Canonical view precedence — the single source of truth ─────────────
224 * The "which screen/overlay is showing" decision used to be re-derived by a
225 * hand-copied precedence cascade in four places (draw switch, hint strip,
226 * and the two main.c input routers). ui_active_view() is now the ONLY place
227 * that precedence lives; every consumer resolves once and dispatches on the
228 * result, so input and draw can never disagree about the active view.
229 *
230 * Order (high → low): FILTER > LFO > STEPEDIT > GRAPH > MENU > mode-tail.
231 * The first five are the input-capturing overlays (UI_VIEW_IS_OVERLAY); the
232 * mode-tail is resolved from seq_state.ui_mode when no overlay is up. */
233typedef enum {
234 UI_VIEW_FILTER = 0,
235 UI_VIEW_LFO,
236 UI_VIEW_STEPEDIT,
237 UI_VIEW_GRAPH,
238 UI_VIEW_MENU,
239 UI_VIEW_ARP,
240 UI_VIEW_DRONE_VIS,
241 UI_VIEW_DRONE,
242 UI_VIEW_DRONE_STD,
243 UI_VIEW_PROG,
244 UI_VIEW_TRACKOPTS,
245 UI_VIEW_FM,
246 UI_VIEW_SEQ,
247 UI_VIEW_COUNT
248} ui_view_id_t;
