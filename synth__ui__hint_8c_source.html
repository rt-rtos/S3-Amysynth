6/* The button-hint strip reads its labels straight from the view descriptor
7 * table, indexed by the single precedence resolver (synth_ui_active_view()).
8 * This used to be three functions (hint_b1/b2/b3) that each re-walked the
9 * overlay precedence ladder by hand; the labels now live one row per view in
10 * ui_view_table[] (ui_view_resolve.c), so the hint strip can never disagree
11 * with the draw switch about which view is active. A NULL static label means
12 * the cell is dynamic — it depends on ui_mode, which the view id does not
13 * carry (only the two scope-capable editors' b1 and the LFO editor's b2) — and
14 * is filled by the row's b*_fn. */
15static const char *hint_cell(const char *label, const char *(*fn)(void))
16{
17 return label ? label : fn();
18}
