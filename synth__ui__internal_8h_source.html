27/* ─── FNV-1a render-on-change (all view signature functions use this) ── */
28#define FNV1A_OFFSET 2166136261u
29#define FNV1A_PRIME 16777619u
30[[gnu::const]] static inline uint32_t fnv1a_bytes(uint32_t h,
31 const void *data, size_t len)
32{
33 const uint8_t *b = (const uint8_t *)data;
34 for (size_t i = 0; i < len; ++i) { h ^= b[i]; h *= FNV1A_PRIME; }
35 return h;
36}
