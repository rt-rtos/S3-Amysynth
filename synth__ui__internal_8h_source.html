38/* ─── FNV-1a render-on-change (all view signature functions use this) ── */
39#define FNV1A_OFFSET 2166136261u
40#define FNV1A_PRIME 16777619u
41[[gnu::const]] static inline uint32_t fnv1a_bytes(uint32_t h,
42 const void *data, size_t len)
43{
44 const uint8_t *b = (const uint8_t *)data;
45 for (size_t i = 0; i < len; ++i) { h ^= b[i]; h *= FNV1A_PRIME; }
46 return h;
47}
