48/* Mirror the UI's grid of active steps into the audio core so the core
49 * schedules notes for every step the user has toggled on. Only "on" steps are
50 * pushed; "off" steps are the core's default after a fresh layer add. */
51void sync_layer_to_core(uint8_t li)
52{
53 seq_layer_t *layer = &seq_state.layers[li];
54 for (int t = 0; t < SEQ_TRACKS; t++) {
55 for (int s = 0; s < layer->num_steps; s++) {
56 if (layer->grid[t][s]) {
57 sequencer_core_set_step(li, t, s, true);
58 }
59 }
60 }
61}

63/* Re-sync the UI mirror from the core after a project load has rewritten layer
64 * topology/content out from under the UI. Also folds in the transport-stop
65 * mirror update, since project_snapshot_load() already stopped the core
66 * transport just before this runs. */
67void synth_ui_reload_mirror_from_core(void)
68{
69 seq_state.num_layers = sequencer_core_get_num_layers();
70 for (uint8_t i = 0; i < seq_state.num_layers; i++) {
71 sequencer_core_export_layer(i, &seq_state.layers[i]);
72 }
73 seq_state.active_layer_idx = 0;
74 seq_state.selected_track = 0;
75 seq_state.selected_step = 0;
76 seq_state.playing = false;
77 /* Match the boot default: edit_mode gates what a bare encoder turn/push
78 * does, and post-load must behave exactly like post-boot. */
79 seq_state.edit_mode = true;
80 s_force_redraw = true;
81 ESP_LOGI(TAG, "UI mirror reloaded from core: %u layer(s)", seq_state.num_layers);
82}

84/* ── Note-name helper (local; mirrors display_seq.c's static one) ─────── */
85void ui_note_name(uint8_t midi_note, char buf[4])
86{
87 /* Chord preset sentinel: the "note" is a chord slot, shown as CHn. */
88 if (SEQ_NOTE_IS_CHORD(midi_note)) {
89 snprintf(buf, 4, "CH%u", (unsigned)(SEQ_CHORD_INDEX(midi_note) + 1u));
90 return;
91 }
92 static const char *const names[] = {
93 "C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"
94 };
95 int octave = (int)midi_note / 12 - 1;
96 snprintf(buf, 4, "%s%d", names[midi_note % 12], octave);
97}
