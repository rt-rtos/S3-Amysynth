24/* Mirror the UI's grid of active steps into the audio core so the core
25 * schedules notes for every step the user has toggled on. Only "on" steps are
26 * pushed; "off" steps are the core's default after a fresh layer add. */
27void sync_layer_to_core(uint8_t li)
28{
29 seq_layer_t *layer = &seq_state.layers[li];
30 for (int t = 0; t < SEQ_TRACKS; t++) {
31 for (int s = 0; s < layer->num_steps; s++) {
32 if (layer->grid[t][s]) {
33 sequencer_core_set_step(li, t, s, true);
34 }
35 }
36 }
37}

39/* Re-sync the UI mirror from the core after a project load has rewritten
40 * layer topology/content out from under the UI (sequencer_core_add_layer/
41 * delete_layer/import_layer). Folds in the transport-stop mirror update too
42 * (seq_state.playing = false) since project_snapshot_load() already stopped
43 * the core transport immediately before this runs — one less thing for the
44 * loader to track. */
45void synth_ui_reload_mirror_from_core(void)
46{
47 seq_state.num_layers = sequencer_core_get_num_layers();
48 for (uint8_t i = 0; i < seq_state.num_layers; i++) {
49 sequencer_core_export_layer(i, &seq_state.layers[i]);
50 }
51 seq_state.active_layer_idx = 0;
52 seq_state.selected_track = 0;
53 seq_state.selected_step = 0;
54 seq_state.playing = false;
55 /* Match the boot default (seq_state initializer): edit_mode gates what a
56 * bare encoder turn/push does on the sequencer screen, and post-load must
57 * behave exactly like post-boot. */
58 seq_state.edit_mode = true;
59 s_force_redraw = true;
60 ESP_LOGI(TAG, "UI mirror reloaded from core: %u layer(s)", seq_state.num_layers);
61}
