90 * serialized against the other seq_state readers on Core 0. Drained

355uint8_t synth_ui_add_layer(seq_layer_type_t type, uint8_t num_steps)
356{
357 uint8_t li = sequencer_core_add_layer(type, num_steps);
358 if (li == 0xFF) return 0xFF;
359
360 seq_layer_t *layer = &seq_state.layers[li];
361 memset(layer, 0, sizeof(seq_layer_t));
362 layer->type = type;
363 layer->num_steps = (num_steps == SEQ_MAX_STEPS) ? SEQ_MAX_STEPS : SEQ_STEPS;
364 layer->num_tracks = SEQ_TRACKS;
365 layer->step_page = 0;
366
367 if (type == SEQ_LAYER_MELODIC) {
368 layer->patch = sequencer_core_get_layer_patch(li);
369 /* Default: Cmaj7 voicing — C4 E4 G4 B4 */
370 static const uint8_t mel_notes[SEQ_TRACKS] = {60, 64, 67, 71};
371 for (int t = 0; t < SEQ_TRACKS; t++) {
372 layer->track_base_note[t] = mel_notes[t];
373 for (int s = 0; s < SEQ_MAX_STEPS; s++) {
374 layer->step_note[t][s] = mel_notes[t];
375 }
376 }
377 } else {
378 /* Drums are per-track patches with role-based default pitches. Read
379 * both from the core (the single source of truth) rather than mirroring
380 * them here, so labels and pitch display cannot drift. */
381 for (int t = 0; t < SEQ_TRACKS; t++) {
382 uint8_t note = sequencer_core_get_track_source_note(li, t);
383 layer->track_base_note[t] = note;
384 layer->track_patch[t] = sequencer_core_get_drum_patch(li, t);
385 for (int s = 0; s < SEQ_MAX_STEPS; s++) {
386 layer->step_note[t][s] = note;
387 }
388 }
389 layer->patch = layer->track_patch[0];
390 }
391
392 seq_state.num_layers = li + 1;
393 ESP_LOGI(TAG_TASK, "UI layer L%d added (type=%d steps=%d)",
394 li + 1, type, layer->num_steps);
395 return li;
396}

438/* Move the step cursor by `delta` in edit mode; outside it a bare turn is
439 * deliberately a no-op (BPM lives in the main menu only). Running off either
440 * end wraps to the adjacent track, so a long turn scans the whole grid. */
441void synth_ui_handle_encoder(long delta)
442{
443 if (delta == 0) return;
444
445 if (seq_state.edit_mode) {
446 uint8_t li = seq_state.active_layer_idx;
447 uint8_t num_steps = seq_state.layers[li].num_steps;
448 int new_step = (int)seq_state.selected_step + (int)delta;
449
450 if (new_step < 0) {
451 /* Off the start: last step of the previous track. */
452 new_step = (int)num_steps - 1;
453 seq_state.selected_track =
454 (uint8_t)((seq_state.selected_track + SEQ_TRACKS - 1) % SEQ_TRACKS);
455 } else if (new_step >= (int)num_steps) {
456 /* Off the end: first step of the next track. */
457 new_step = 0;
458 seq_state.selected_track =
459 (uint8_t)((seq_state.selected_track + 1) % SEQ_TRACKS);
460 }
461 seq_state.selected_step = (uint8_t)new_step;
462
463 /* 32-step layers show 16 steps per page: select the page holding the
464 * new step so the cursor stays visible. */
465 if (num_steps == SEQ_MAX_STEPS) {
466 seq_state.layers[li].step_page = (uint8_t)(new_step / 16);
467 }
468 }
469}

471/* Toggle the grid step under the cursor and mirror it to the core. Gated on

507/* Transpose the selected track by `delta` semitones. Reads/writes the SOURCE
508 * note (the user's raw choice) so repeated nudges accumulate cleanly, then
509 * reads back the resolved note for display since the core may quantize.
510 * Melodic tracks scroll a wrapping virtual list: C1..C7 then the defined chord
511 * presets (seq_chords_selector_step). Drum tracks take a plain clamped nudge. */
512void synth_ui_adjust_track_note(int delta)
513{
514 uint8_t li = seq_state.active_layer_idx;
515 uint8_t track = seq_state.selected_track;
516 uint8_t note = sequencer_core_get_track_source_note(li, track);
517
518 if (li < seq_state.num_layers &&
519 seq_state.layers[li].type == SEQ_LAYER_MELODIC) {
520 int dir = (delta > 0) ? 1 : -1;
521 for (int k = (delta > 0 ? delta : -delta); k > 0; k--) {
522 note = seq_chords_selector_step(note, dir);
523 }
524 sequencer_core_set_track_midi_note(li, track, note);
525 } else {
526 uint8_t new_note = SEQ_CLAMP_U8(note + delta, 0, 127);
