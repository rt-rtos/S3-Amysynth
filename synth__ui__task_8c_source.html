63static void synth_ui_task(void *pvParameters)
64{
65 (void)pvParameters;
66 TickType_t last_wake_time = xTaskGetTickCount();
67 const TickType_t delay = pdMS_TO_TICKS(50); /* 20 Hz */
68 uint32_t last_sig = 0;
69 /* Which top-level view was rendered last frame; a change forces a redraw. */
70 ui_view_id_t last_view = UI_VIEW_SEQ;
71 for (;;) {
72 /* Coalesced arp re-emit: setters mark the arp dirty; we perform at most
73 * one full re-emit per frame here, collapsing fast encoder edits. */
74 arp_core_service();
75 /* Drone: advance the tempo-locked filter sweep + keep the LFO in sync.
76 * Cheap no-op while the drone is disabled. */
77 drone_core_service();
78 /* Normal drone: drain its coalesced rebuild (no tick machinery). */
79 drone_std_core_service();
80 sequencer_core_lfo_service();
81 sequencer_core_progression_service();
82
83 /* Drain deferred layer-delete request (must run in synth_ui_task so that
84 * the array compaction is serialized against all other seq_state readers
85 * on Core 0). */
86 /* Ordering: DELETE is drained before ADD so we never add into a slot
87 * that a concurrent delete has not yet freed and compacted. */
88 if (s_layer_delete_pending) {
89 s_layer_delete_pending = false;
90 uint8_t del_idx = s_layer_delete_idx;
91 if (sequencer_core_delete_layer(del_idx)) {
92 /* Mirror compaction in the UI-side seq_state. */
93 uint8_t tail = (uint8_t)(seq_state.num_layers - del_idx - 1);
94 if (tail > 0) {
95 memmove(&seq_state.layers[del_idx],
96 &seq_state.layers[del_idx + 1],
97 tail * sizeof(seq_state.layers[0]));
98 }
99 seq_state.num_layers--;
100 /* Clamp indices that may now point past the end. */
101 if (seq_state.active_layer_idx >= seq_state.num_layers)
102 seq_state.active_layer_idx = (uint8_t)(seq_state.num_layers - 1);
103 if (s_graph_layer >= seq_state.num_layers)
104 s_graph_layer = (uint8_t)(seq_state.num_layers - 1);
105 if (s_to_layer >= seq_state.num_layers)
106 s_to_layer = (uint8_t)(seq_state.num_layers - 1);
107 /* Drop the Step Trig overlay: after compaction its cached
108 * (layer,track,step) may point at a different layer's steps. */
109 synth_ui_stepedit_close();
110 ESP_LOGI(TAG_TASK, "UI delete layer L%u (%u layers remain)",
111 del_idx + 1u, seq_state.num_layers);
112 }
113 }
114
115 /* Drain deferred layer-add request (runs in synth_ui_task, 8192-byte
116 * stack — sized for the patch-string parse's ~3.4 KB inline frame,
117 * which would overflow the esp_timer task's 3584-byte stack and
118 * overflowed this task's former 4096 too). */
119 if (s_layer_add_pending) {
120 s_layer_add_pending = false;
121 /* Re-check cap; num_layers may have changed since flag was set. */
122 if (seq_state.num_layers < MAX_LAYERS) {
123 synth_ui_add_layer(SEQ_LAYER_MELODIC, SEQ_STEPS);
124 }
125 }
126
127#if CONFIG_SYNTH_PROJECT_STORE
128 /* Execute any project load/save queued by the Projects menu (clicks
129 * run on the button task). Must run here: a load rebuilds layer
130 * topology via core add/delete, which only this task - the single
131 * s_layers applier - may call. After the add/delete drains above so
132 * pending structural edits resolve before a load replaces them. */
133 projects_menu_service();
134#endif
135
136 /* Output-level watchdog: peeks the USB ring on this core/task (see
137 * usb_audio_watchdog.h). Compiles to nothing when the Kconfig gate is
138 * off, as do the badge draw and signature mix below. */
139 output_wd_poll();
140
141 /* Trailing flush for throttled editor live-previews (amp trim), so the
142 * last encoder value lands even after the user stops turning. */
143 synth_ui_editors_live_service();
144
145 seq_state.current_step =
146 sequencer_core_get_current_step(seq_state.active_layer_idx);
147 if (s_u8g2) {
148 /* The whole screen/overlay precedence now lives in one place,
149 * synth_ui_active_view(); this task renders whatever it resolves.
150 * signature() builds the active view into vw and returns the FNV
151 * render-gate hash, and the draw below reuses vw — never built
152 * twice. Adding a view is a one-row change in ui_view_table[]. */
153 ui_view_id_t view = synth_ui_active_view();
154 const ui_view_desc_t *desc = &ui_view_table[view];
155 ui_view_vw_t vw;
156 uint32_t sig = desc->signature(&vw);
157 /* The watchdog badge participates in the render gate so it
158 * appears/clears without needing any other screen change. */
159 sig ^= (uint32_t)output_wd_state() * 0x9E3779B9u;
160 bool force = s_force_redraw || (view != last_view);
161
162 if (force || sig != last_sig) {
163 desc->draw(s_u8g2, &vw);
164 /* Persistent button-hint strip: composited into the buffer on
165 * top of whatever the view above just drew, so no per-screen
166 * renderer needs to know about it. Every screen's draw_frame
167 * now only fills the buffer -- it no longer sends -- so this
168 * is the single physical transfer per redraw. Sending twice
169 * (once without the hint, once with) used to make the hint
170 * strip visibly flicker on every redraw, in every UI state. */
171 if (synth_ui_hint_visible()) {
172 display_hint_draw(s_u8g2, synth_ui_hint_text());
173 }
174 /* Output-level warning badge, top-right, composited last so it
175 * overlays every screen; part of the single physical send. */
176 output_wd_state_t owd = output_wd_state();
177 if (owd != OUTPUT_WD_OK) {
178 const char *owd_txt =
179 (owd == OUTPUT_WD_CLIPPING) ? "CLIP" : "LOUD";
180 u8g2_SetFont(s_u8g2, u8g2_font_4x6_tr);
181 u8g2_SetDrawColor(s_u8g2, 1);
182 u8g2_DrawBox(s_u8g2, 108, 0, 20, 8);
183 u8g2_SetDrawColor(s_u8g2, 0);
184 u8g2_DrawStr(s_u8g2, 110, 7, owd_txt);
185 u8g2_SetDrawColor(s_u8g2, 1);
186 }
187 u8g2_SendBuffer(s_u8g2);
188 last_sig = sig;
189 last_view = view;
190 s_force_redraw = false;
191 }
192 }
193 vTaskDelayUntil(&last_wake_time, delay);
194 }

289uint8_t synth_ui_add_layer(seq_layer_type_t type, uint8_t num_steps)
290{
291 uint8_t li = sequencer_core_add_layer(type, num_steps);
292 if (li == 0xFF) return 0xFF;
293
294 seq_layer_t *layer = &seq_state.layers[li];
295 memset(layer, 0, sizeof(seq_layer_t));
296 layer->type = type;
297 layer->num_steps = (num_steps == SEQ_MAX_STEPS) ? SEQ_MAX_STEPS : SEQ_STEPS;
298 layer->num_tracks = SEQ_TRACKS;
299 layer->step_page = 0;
300
301 if (type == SEQ_LAYER_MELODIC) {
302 layer->patch = sequencer_core_get_layer_patch(li);
303 /* Default: Cmaj7 voicing — C4 E4 G4 B4 */
304 static const uint8_t mel_notes[SEQ_TRACKS] = {60, 64, 67, 71};
305 for (int t = 0; t < SEQ_TRACKS; t++) {
306 layer->track_base_note[t] = mel_notes[t];
307 for (int s = 0; s < SEQ_MAX_STEPS; s++) {
308 layer->step_note[t][s] = mel_notes[t];
309 }
310 }
311 } else {
312 /* Drums are per-track patches with role-based default pitches. Pull the
313 * actual per-track patch + source note from the core (single source of
314 * truth) so labels and pitch display stay correct as those defaults
315 * evolve — no hardcoded mirror to drift out of sync. */
316 for (int t = 0; t < SEQ_TRACKS; t++) {
317 uint8_t note = sequencer_core_get_track_source_note(li, t);
318 layer->track_base_note[t] = note;
319 layer->track_patch[t] = sequencer_core_get_drum_patch(li, t);
320 for (int s = 0; s < SEQ_MAX_STEPS; s++) {
321 layer->step_note[t][s] = note;
322 }
323 }
324 layer->patch = layer->track_patch[0];
325 }
326
327 seq_state.num_layers = li + 1;
328 ESP_LOGI(TAG_TASK, "UI layer L%d added (type=%d steps=%d)",
329 li + 1, type, layer->num_steps);
330 return li;
331}

374/* Moves the step cursor by `delta` while in edit mode; outside edit mode a
375 * bare turn is deliberately a no-op (BPM lives in the main menu only).
376 * The cursor walks the current track's steps; running off either end wraps to
377 * the adjacent track (and wraps track index too), so a long turn scans the
378 * whole grid track-by-track. */
379void synth_ui_handle_encoder(long delta)
380{
381 if (delta == 0) return;
382
383 if (seq_state.edit_mode) {
384 uint8_t li = seq_state.active_layer_idx;
385 uint8_t num_steps = seq_state.layers[li].num_steps;
386 int new_step = (int)seq_state.selected_step + (int)delta;
387
388 if (new_step < 0) {
389 /* Walked off the start: jump to the last step of the previous track. */
390 new_step = (int)num_steps - 1;
391 seq_state.selected_track =
392 (uint8_t)((seq_state.selected_track + SEQ_TRACKS - 1) % SEQ_TRACKS);
393 } else if (new_step >= (int)num_steps) {
394 /* Walked off the end: jump to the first step of the next track. */
395 new_step = 0;
396 seq_state.selected_track =
397 (uint8_t)((seq_state.selected_track + 1) % SEQ_TRACKS);
398 }
399 seq_state.selected_step = (uint8_t)new_step;
400
401 /* 32-step layers display 16 steps per page; keep the cursor visible by
402 * selecting the page (0 or 1) that contains the new step. */
403 if (num_steps == SEQ_MAX_STEPS) {
404 seq_state.layers[li].step_page = (uint8_t)(new_step / 16);
405 }
406 }
407}

409/* Toggle the grid step under the cursor on/off (and mirror it to the core).
410 * Gated on edit_mode like the encoder push, but with no play/pause fallback,
411 * so it is safe to bind to a dedicated toggle button: a press outside edit
412 * mode does nothing. Returns whether a step was toggled. */
413bool synth_ui_toggle_step_at_cursor(void)
414{

420 sequencer_core_set_step(li, t, s, seq_state.layers[li].grid[t][s]);
421 return true;
422}

446/* Transposes the selected track's note by `delta` semitones. We read/write the
447 * *source* note (the user's raw choice) so repeated nudges accumulate cleanly;
448 * the core may quantize it, so we read back the resolved note for display. */
449void synth_ui_adjust_track_note(int delta)
450{
451 uint8_t li = seq_state.active_layer_idx;
452 uint8_t track = seq_state.selected_track;

456 /* Keep display in sync with resolved note after core clamp/quantize. */
457 seq_state.layers[li].track_base_note[track] =
