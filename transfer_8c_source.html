31static FILE *g_files[MAX_OPEN_FILES]; // index 1..MAX_OPEN_FILES-1 used

37#ifndef __EMSCRIPTEN__
38static uint32_t alloc_handle(FILE *f) {
39 for (uint32_t i = 1; i < MAX_OPEN_FILES; i++) {
40 if (g_files[i] == NULL) {
41 g_files[i] = f;
42 return i;
43 }
44 }
45 return HANDLE_INVALID; // table full
46}

48static FILE *lookup_handle(uint32_t h) {
49 if (h == 0 || h >= MAX_OPEN_FILES) return NULL;
50 return g_files[h];

52#endif

54 if (h == 0 || h >= MAX_OPEN_FILES) return;
55 g_files[h] = NULL;
56#ifdef __EMSCRIPTEN__
57 g_em_handle[h] = 0;
58 g_em_pos[h] = 0;

69uint32_t posix_external_fopen_hook(char * filename, const char *mode) {
70#ifdef __EMSCRIPTEN__
71 uint32_t js_handle = EM_ASM_INT({
72 if (typeof amy_shared_open === 'function') {
73 return amy_shared_open(UTF8ToString($0), UTF8ToString($1));
74 }
75 return 0;
76 }, filename, mode);
77 if (js_handle == 0) {
78 return HANDLE_INVALID;
79 }
80 for (uint32_t i = 1; i < MAX_OPEN_FILES; i++) {
81 if (g_em_handle[i] == 0) {
82 g_em_handle[i] = js_handle;
83 g_em_pos[i] = 0;
84 return i;
85 }
86 }
87 EM_ASM({
88 if (typeof amy_shared_close === 'function') {
89 amy_shared_close($0);
90 }
91 }, js_handle);
92 return HANDLE_INVALID;
93#else
94 FILE *f = fopen(filename, mode);
95 if (!f) {
96 return HANDLE_INVALID;
97 }
98 uint32_t h = alloc_handle(f);
99 if (h == HANDLE_INVALID) {
100 fclose(f);
101 return HANDLE_INVALID;
102 }
103 return h;
104#endif
105}

231void start_receiving_file_transfer(uint32_t length, const char *filename) {
232

235 }

256// takes a wire message and adds it to storage after decoding it. stops transfer when it's done
257void parse_transfer_message(char * message, uint16_t len) {
258 size_t decoded = 0;
259 uint8_t * block = b64_decode_ex (message, len, &decbuf, &decoded);
260 if (amy_global.transfer_flag == AMY_TRANSFER_TYPE_FILE) {
261 if (amy_global.config.amy_external_fwrite_hook != NULL && amy_global.transfer_file_handle != HANDLE_INVALID) {
262 uint32_t wrote = amy_global.config.amy_external_fwrite_hook(amy_global.transfer_file_handle, block, (uint32_t)decoded);
263 amy_global.transfer_stored_bytes += wrote;
264 }
265 } else {
266 for (uint16_t i = 0; i < decoded; i++) {
267 amy_global.transfer_storage[amy_global.transfer_stored_bytes++] = block[i];
268 }
269 }
270 if (amy_global.transfer_stored_bytes >= amy_global.transfer_length_bytes) { // we're done
271 if (amy_global.transfer_flag == AMY_TRANSFER_TYPE_FILE) {
272 if (amy_global.config.amy_external_fclose_hook != NULL && amy_global.transfer_file_handle != HANDLE_INVALID) {
273 amy_global.config.amy_external_fclose_hook(amy_global.transfer_file_handle);
274 }
275 // Pair the sequencer_midi_stop() that zT's start handler fired in
276 // amy_parse_transfer_layer_message(). Do it BEFORE the done hook so
277 // the hook can observe a running sequencer if it needs to, and so
278 // any platform that uses zT without a platform-specific restart in
279 // its file_transfer_done_hook doesn't leave playback wedged.
280 sequencer_midi_start();
281 if (amy_global.config.amy_external_file_transfer_done_hook != NULL) {
282 amy_global.config.amy_external_file_transfer_done_hook(amy_global.transfer_filename);
283 }
284 amy_global.transfer_file_handle = 0;
285 amy_global.transfer_filename[0] = '\0';
286 }
287 amy_global.transfer_flag = AMY_TRANSFER_TYPE_NONE;
288 free(decbuf.ptr);
289 }

302// We don't do encoding in AMY, we rely on python for that, but we'll leave it here if you want it in C
303// Just like decode you pass it an allocated encbuf.
304char * b64_encode (const unsigned char *src, b64_buffer_t * encbuf, size_t len) {
305 int i = 0;
306 int j = 0;
307 size_t size = 0;
308 unsigned char buf[4];
309 unsigned char tmp[3];
310
311
312 // parse until end of source
313 while (len--) {
314 // read up to 3 bytes at a time into `tmp'
315 tmp[i++] = *(src++);
316
317 // if 3 bytes read then encode into `buf'
318 if (3 == i) {
319 buf[0] = (tmp[0] & 0xfc) >> 2;
320 buf[1] = ((tmp[0] & 0x03) << 4) + ((tmp[1] & 0xf0) >> 4);
321 buf[2] = ((tmp[1] & 0x0f) << 2) + ((tmp[2] & 0xc0) >> 6);
322 buf[3] = tmp[2] & 0x3f;
323
324 // allocate 4 new byts for `enc` and
325 // then translate each encoded buffer
326 // part by index from the base 64 index table
327 // into `encbuf.ptr' unsigned char array
328
329 for (i = 0; i < 4; ++i) {
330 encbuf->ptr[size++] = b64_table[buf[i]];
331 }
332
333 // reset index
334 i = 0;
335 }
336 }
337
338 // remainder
339 if (i > 0) {
340 // fill `tmp' with `\0' at most 3 times
341 for (j = i; j < 3; ++j) {
342 tmp[j] = '\0';
343 }
344
345 // perform same codec as above
346 buf[0] = (tmp[0] & 0xfc) >> 2;
347 buf[1] = ((tmp[0] & 0x03) << 4) + ((tmp[1] & 0xf0) >> 4);
348 buf[2] = ((tmp[1] & 0x0f) << 2) + ((tmp[2] & 0xc0) >> 6);
349 buf[3] = tmp[2] & 0x3f;
350
351 // perform same write to `encbuf->ptr` with new allocation
352 for (j = 0; (j < i + 1); ++j) {
353 encbuf->ptr[size++] = b64_table[buf[j]];
354 }
355
356 // while there is still a remainder
357 // append `=' to `encbuf.ptr'
358 while ((i++ < 3)) {
359 encbuf->ptr[size++] = '=';
360 }
361 }
362
363 // Make sure we have enough space to add '\0' character at end.
364 encbuf->ptr[size] = '\0';
365
366 return encbuf->ptr;
367}

371unsigned char *
372b64_decode_ex (const char *src, size_t len, b64_buffer_t * decbuf, size_t *decsize) {
373 int i = 0;
374 int j = 0;
375 int l = 0;
376 size_t size = 0;
377 unsigned char buf[3];
378 unsigned char tmp[4];
379
380 // alloc
381 //if (b64_buf_malloc(&decbuf) == -1) { return NULL; }
382
383 // parse until end of source
384 while (len--) {
385 // break if char is `=' or not base64 char
386 if ('=' == src[j]) { break; }
387 if (!(isalnum((unsigned char)src[j]) || '+' == src[j] || '/' == src[j])) { break; }
388
389 // read up to 4 bytes at a time into `tmp'
390 tmp[i++] = src[j++];
391
392 // if 4 bytes read then decode into `buf'
393 if (4 == i) {
394 // translate values in `tmp' from table
395 for (i = 0; i < 4; ++i) {
396 // find translation char in `b64_table'
397 for (l = 0; l < 64; ++l) {
398 if (tmp[i] == b64_table[l]) {
399 tmp[i] = l;
400 break;
401 }
402 }
403 }
404
405 // decode
406 buf[0] = (tmp[0] << 2) + ((tmp[1] & 0x30) >> 4);
407 buf[1] = ((tmp[1] & 0xf) << 4) + ((tmp[2] & 0x3c) >> 2);
408 buf[2] = ((tmp[2] & 0x3) << 6) + tmp[3];
409
410 // write decoded buffer to `decbuf.ptr'
411 //if (b64_buf_realloc(&decbuf, size + 3) == -1) return NULL;
412 for (i = 0; i < 3; ++i) {
413 ((unsigned char*)decbuf->ptr)[size++] = buf[i];
414 }
415
416 // reset
417 i = 0;
418 }
419 }
420
421 // remainder
422 if (i > 0) {
423 // fill `tmp' with `\0' at most 4 times
424 for (j = i; j < 4; ++j) {
425 tmp[j] = '\0';
426 }
427
428 // translate remainder
429 for (j = 0; j < 4; ++j) {
430 // find translation char in `b64_table'
431 for (l = 0; l < 64; ++l) {
432 if (tmp[j] == b64_table[l]) {
433 tmp[j] = l;
434 break;
435 }
436 }
437 }
438
439 // decode remainder
440 buf[0] = (tmp[0] << 2) + ((tmp[1] & 0x30) >> 4);
441 buf[1] = ((tmp[1] & 0xf) << 4) + ((tmp[2] & 0x3c) >> 2);
442 buf[2] = ((tmp[2] & 0x3) << 6) + tmp[3];
443
444 // write remainer decoded buffer to `decbuf.ptr'
445 //if (b64_buf_realloc(&decbuf, size + (i - 1)) == -1) return NULL;
446 for (j = 0; (j < i - 1); ++j) {
447 ((unsigned char*)decbuf->ptr)[size++] = buf[j];
448 }
449 }
450
451 // Make sure we have enough space to add '\0' character at end.
452 //if (b64_buf_realloc(&decbuf, size + 1) == -1) return NULL;
453 ((unsigned char*)decbuf->ptr)[size] = '\0';
454
455 // Return back the size of decoded string if demanded.
456 if (decsize != NULL) {
457 *decsize = size;
458 }
459
460 return (unsigned char*) decbuf->ptr;
461}

466// Walk all active instruments and emit one wire-command line per callback.
467static void amy_emit_state_lines(void (*cb)(const char *line, int len, void *ctx), void *ctx) {
468 char buf[1024];
469 char line[1100];
470 bool include_fx = true;
471 for (int inst = 0; inst < instruments_max_instruments(); inst++) {
472 if (instrument_number_exists(inst, NULL)) {
473 uint16_t voices[MAX_VOICES_PER_INSTRUMENT];
474 int num_voices = instrument_get_num_voices(inst, voices);
475 if (num_voices < 1) continue; // should never happen.
476 int oscs_per_voice = instrument_get_oscs_per_voice(inst);
477 int n = snprintf(line, sizeof(line), "i%dic255Z", inst);
478 cb(line, n, ctx);
479 n = snprintf(line, sizeof(line), "i%div%din%dZ", inst, num_voices, oscs_per_voice);
480 cb(line, n, ctx);
481 void *state = NULL;
482 do {
483 state = yield_synth_commands((uint8_t)inst, buf, sizeof(buf), include_fx, state);
484 if (buf[0] == '\0') continue;
485 n = snprintf(line, sizeof(line), "i%d%s", inst, buf); // Prepends to FX as well, that's OK.
486 cb(line, n, ctx);
487 } while (state);
488 // We include FX only in the first osc.
489 include_fx = false;
490 }
491 }
492}

514// Max base64 chars per sysex chunk. Empirically Chrome's Web MIDI has trouble
515// delivering sysex messages above a few KB reliably, so we split large dumps
516// into multiple smaller sysex frames. 1024 base64 chars → ~1030-byte sysex
517// frame including header/footer, well within the "reliably delivered" band.
518#define ZDUMP_CHUNK_B64_MAX 1024
519
520// Raw bytes per streamed chunk. 768 = 3 * 256 → exactly 1024 base64 chars with
521// no padding, so non-final chunks can be concatenated on the receiver and
522// decoded as one base64 string. Only the final chunk may emit '=' padding
523// (which is fine: it's appended last).
524#define ZDUMP_STREAM_RAW_CHUNK 768
525
526// Streaming sysex dump state. Lets amy_dump_state_to_sysex and
527// amy_dump_file_to_sysex produce arbitrarily-large dumps without holding the
528// whole payload in RAM. Each ZDUMP_STREAM_RAW_CHUNK bytes of raw data is
529// base64-encoded and emitted as one sysex frame (marker 'C' if more follow,
530// 'E' or '0' on the last frame). The effective upper bound is now the
531// web-side reassembler's SYSEX_REASM_MAX rather than MAX_SYSEX_BYTES.
532typedef struct {
533 uint8_t *raw_buf; // ZDUMP_STREAM_RAW_CHUNK-byte raw accumulator
534 int raw_pos; // current fill of raw_buf
535 uint8_t *enc; // base64 scratch: b64(raw_buf)
536 uint8_t *frame; // sysex frame scratch: F0 00 03 45 <marker> <b64...> F7
537 int frames_sent;
538 int bytes_sent; // raw bytes drained so far, for logging
539 bool ok;
540} _zdump_stream;
541
542static int _zdump_stream_init(_zdump_stream *c) {
543 c->raw_buf = NULL;
544 c->enc = NULL;
545 c->frame = NULL;
546 c->raw_pos = 0;
547 c->frames_sent = 0;
548 c->bytes_sent = 0;
549 c->ok = false;
550 c->raw_buf = (uint8_t *)malloc_caps(ZDUMP_STREAM_RAW_CHUNK, amy_global.config.ram_caps_sysex);
551 c->enc = (uint8_t *)malloc_caps(ZDUMP_CHUNK_B64_MAX + 4, amy_global.config.ram_caps_sysex);
552 c->frame = (uint8_t *)malloc_caps(ZDUMP_CHUNK_B64_MAX + 6, amy_global.config.ram_caps_sysex);
553 if (!c->raw_buf || !c->enc || !c->frame) {
554 fprintf(stderr, "zD: stream init malloc failed (raw=%p enc=%p frame=%p)\n",
555 (void *)c->raw_buf, (void *)c->enc, (void *)c->frame);
556 return -1;
557 }
558 c->ok = true;
559 return 0;
560}

562static void _zdump_stream_destroy(_zdump_stream *c) {
563 if (c->raw_buf) { free(c->raw_buf); c->raw_buf = NULL; }
564 if (c->enc) { free(c->enc); c->enc = NULL; }
565 if (c->frame) { free(c->frame); c->frame = NULL; }
566 c->ok = false;
567}
568
569// Emit one sysex frame carrying b64(raw_buf[0..raw_pos]). If is_last is false
570// the marker is 'C' (more frames follow); otherwise it's 'E' when we've
571// already sent at least one frame, or '0' for a single-frame dump.
572static void _zdump_stream_drain(_zdump_stream *c, bool is_last) {
573 if (!c->ok) return;
574 uint8_t marker;
575 if (is_last) {
576 marker = (c->frames_sent == 0) ? '0' : 'E';
577 } else {
578 marker = 'C';
579 }
580 b64_buffer_t bb = { (char *)c->enc, 1 };
581 b64_encode(c->raw_buf, &bb, (size_t)c->raw_pos);
582 int enc_len = (int)strlen((char *)c->enc);
583 c->frame[0] = 0xF0;
584 c->frame[1] = 0x00;
585 c->frame[2] = 0x03;
586 c->frame[3] = 0x45;
587 c->frame[4] = marker;
588 memcpy(c->frame + 5, c->enc, enc_len);
589 c->frame[5 + enc_len] = 0xF7;
590 midi_out(c->frame, 6 + enc_len);
591 c->bytes_sent += c->raw_pos;
592 c->frames_sent++;
593 c->raw_pos = 0;
594}
595
596// Append `len` bytes from `data` to the raw accumulator, draining as 'C'
597// frames whenever the accumulator is full and more data arrives. The last
598// (possibly partial) chunk is held until _zdump_stream_finish is called.
599static void _zdump_stream_write(_zdump_stream *c, const uint8_t *data, int len) {
600 if (!c->ok) return;
601 while (len > 0) {
602 if (c->raw_pos == ZDUMP_STREAM_RAW_CHUNK) {
603 _zdump_stream_drain(c, false);
604 }
605 int space = ZDUMP_STREAM_RAW_CHUNK - c->raw_pos;
606 int take = len < space ? len : space;
607 memcpy(c->raw_buf + c->raw_pos, data, take);
608 c->raw_pos += take;
609 data += take;
610 len -= take;
611 }
612}

614// Emit any buffered data as the final frame. Always emits at least one frame
615// so the receiver sees a well-formed '0' (empty-payload) dump response for
616// empty inputs.
617static void _zdump_stream_finish(_zdump_stream *c) {
618 if (!c->ok) return;

645void amy_dump_file_to_sysex(const char *filename) {
646 // Only meaningful on platforms with MIDI sysex output (hardware, emscripten).
647 // On desktop, midi_out can overflow its stack buffer with large sysex.
648 if (!amy_global.config.midi) return;
649 sequencer_midi_stop();

651 !amy_global.config.amy_external_fread_hook ||
652 !amy_global.config.amy_external_fclose_hook) {
653 fprintf(stderr, "zD: file I/O hooks unavailable\n");
654 sequencer_midi_start();
655 return;
656 }
657 uint32_t fh = amy_global.config.amy_external_fopen_hook((char *)filename, "r");
658 if (!fh) {
659 fprintf(stderr, "zD: could not open '%s'\n", filename);
660 sequencer_midi_start();
661 return;
662 }
663 _zdump_stream stream;
664 if (_zdump_stream_init(&stream) != 0) {
665 amy_global.config.amy_external_fclose_hook(fh);
666 sequencer_midi_start();
667 return;
668 }
669 uint8_t *read_buf = (uint8_t *)malloc_caps(ZDUMP_STREAM_RAW_CHUNK, amy_global.config.ram_caps_sysex);
670 if (!read_buf) {
671 fprintf(stderr, "zD: malloc read_buf(%d) FAILED\n", ZDUMP_STREAM_RAW_CHUNK);
672 _zdump_stream_destroy(&stream);
673 amy_global.config.amy_external_fclose_hook(fh);
674 sequencer_midi_start();
675 return;
676 }
677 while (1) {
678 uint32_t n = amy_global.config.amy_external_fread_hook(fh, read_buf, ZDUMP_STREAM_RAW_CHUNK);
679 if (n == 0) break;
680 _zdump_stream_write(&stream, read_buf, (int)n);
681 }
682 free(read_buf);
683 _zdump_stream_finish(&stream);
684 _zdump_stream_destroy(&stream);
685 amy_global.config.amy_external_fclose_hook(fh);
686 sequencer_midi_start();
687}

729static int skip_bytes(uint32_t handle, uint32_t len) {
730 uint8_t scratch[64];
731 while (len > 0) {

734 return 0;

736 len -= chunk;
737 }

741int wave_parse_header(uint32_t handle, wave_info_t *info, uint32_t *data_bytes) {
742 if (info == NULL || data_bytes == NULL) {
743 return 0;
744 }
745 uint8_t riff_header[12];
746 if (!read_exact(handle, riff_header, sizeof(riff_header))) {
747 return 0;
748 }
749 if (memcmp(riff_header, "RIFF", 4) != 0 || memcmp(riff_header + 8, "WAVE", 4) != 0) {
750 return 0;
751 }
752 uint8_t fmt_found = 0;
753 wave_info_t tmp_info = {0};
754 while (1) {
755 uint8_t chunk_header[8];
756 if (!read_exact(handle, chunk_header, sizeof(chunk_header))) {
757 return 0;
758 }
759 uint32_t chunk_size = read_u32_le(chunk_header + 4);
760 if (memcmp(chunk_header, "fmt ", 4) == 0) {
761 if (chunk_size < 16) {
762 return 0;
763 }
764 uint8_t fmt_chunk[16];
765 if (!read_exact(handle, fmt_chunk, sizeof(fmt_chunk))) {
766 return 0;
767 }
768 uint16_t audio_format = read_u16_le(fmt_chunk);
769 tmp_info.channels = read_u16_le(fmt_chunk + 2);
770 tmp_info.sample_rate = read_u32_le(fmt_chunk + 4);
771 if (audio_format != 1 || tmp_info.channels == 0 || tmp_info.channels > 2) {
772 return 0;
773 }
774 if (!skip_bytes(handle, chunk_size - sizeof(fmt_chunk))) {
775 return 0;
776 }
777 if (chunk_size & 1) {
778 if (!skip_bytes(handle, 1)) {
779 return 0;
780 }
781 }
782 fmt_found = 1;
783 continue;
784 }
785 if (memcmp(chunk_header, "data", 4) == 0) {
786 if (!fmt_found) {
787 return 0;
788 }
789 *info = tmp_info;
790 *data_bytes = chunk_size;
791 return 1;
792 }
793 if (!skip_bytes(handle, chunk_size)) {
794 return 0;
795 }
796 if (chunk_size & 1) {
797 if (!skip_bytes(handle, 1)) {
798 return 0;
799 }
800 }
801 }
802 return 0;
803}
804uint32_t wave_read_pcm_frames_s16(uint32_t handle, uint16_t channels,
805 uint32_t *bytes_remaining, int16_t *dest, uint32_t max_frames) {
806 if (dest == NULL || bytes_remaining == 0 || channels == 0 || channels > 2) {
807 return 0;
808 }
809 uint32_t bytes_per_frame = channels * 2;
810 if (*bytes_remaining < bytes_per_frame) {
811 return 0;
812 }
813 uint32_t max_bytes = max_frames * bytes_per_frame;
814 if (max_bytes > *bytes_remaining) {
815 max_bytes = *bytes_remaining - (*bytes_remaining % bytes_per_frame);
816 }
817 uint32_t frames_to_read = max_bytes / bytes_per_frame;
818 if (frames_to_read == 0) {
819 return 0;
820 }
821 uint8_t raw_buf[(AMY_BLOCK_SIZE * PCM_FILE_BUFFER_MULT + 1) * 4];
822 if (max_bytes > sizeof(raw_buf)) {
823 max_bytes = sizeof(raw_buf) - (sizeof(raw_buf) % bytes_per_frame);
824 frames_to_read = max_bytes / bytes_per_frame;
825 }
826 if (amy_global.config.amy_external_fread_hook == NULL) {
827 return 0;
828 }
829 uint32_t got = amy_global.config.amy_external_fread_hook(handle, raw_buf, max_bytes);
830 got -= got % bytes_per_frame;
831 if (got == 0) {
832 return 0;
833 }
834 frames_to_read = got / bytes_per_frame;
835 *bytes_remaining -= got;
836 if (channels == 1) {
837 for (uint32_t i = 0; i < frames_to_read; i++) {
838 dest[i] = (int16_t)read_u16_le(raw_buf + i * 2);
839 }
840 } else {
841 for (uint32_t i = 0; i < frames_to_read; i++) {
842 dest[i * 2] = (int16_t)read_u16_le(raw_buf + i * 4);
843 dest[i * 2 + 1] = (int16_t)read_u16_le(raw_buf + i * 4 + 2);
844 }
845 }
846 return frames_to_read;
847}
