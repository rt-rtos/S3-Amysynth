95/* ── Curve-type preview shaping ──────────────────────────────────────────────
96 * Per-segment shape callback for the ADSR plot: a float mirror of AMY's
97 * compute_breakpoint_scale() segment math (envelope.c), evaluated in the
98 * widget's normalised 0..1 level space, so the drawn curve matches what the
99 * selected eg_type will actually sound like. Runs only while the editor is
100 * open, on the 50 ms UI tick — nowhere near the render path. */
101#define GRAPH_ENV_EPS 0.0002f /* BREAKPOINT_EPS: floor for the log-domain types */
102
103static float graph_env_shape(float v0, float v1, float t)
104{
105 uint8_t eg_type = (uint8_t)(s_graph_eg_type_disp & 3u);
106
107 if (eg_type == 1) { /* ENVELOPE_LINEAR */
108 return v0 + (v1 - v0) * t;
109 }
110
111 if (eg_type == 2 || eg_type == 3) { /* DX7 / TRUE_EXPONENTIAL */
112 float a = (v0 > GRAPH_ENV_EPS) ? v0 : GRAPH_ENV_EPS;
113 float b = (v1 > GRAPH_ENV_EPS) ? v1 : GRAPH_ENV_EPS;
114 if (eg_type == 2 && b > a) {
115 /* DX7 attack law. Levels map linear->DX7 (log2 + 12.375), then
116 * through the attack-range curve; segment time is normalised so
117 * only the ratio matters. Degenerate spans fall through to the
118 * true-exp branch below. */
119 float l0 = log2f(a) + 12.375f;
120 float l1 = log2f(b) + 12.375f;
121 float m0 = 1.0f - ((l0 > 4.25f) ? (l0 - 4.25f) : 0.0f) / 9.375f;
122 float m1 = 1.0f - ((l1 > 4.25f) ? (l1 - 4.25f) : 0.0f) / 9.375f;
123 float dl = log2f(m0) - log2f(m1);
124 if (dl > 1e-6f) {
125 float t_const = 1.0f / dl;
126 float my_t0 = -t_const * log2f(m0);
127 float level = 4.25f
128 + 9.375f * (1.0f - exp2f(-(my_t0 + t) / t_const));
129 return exp2f(level - 12.375f);
130 }
131 }
132 /* TRUE_EXPONENTIAL, and DX7 decay/release (also plain true-exp). */
133 float la = log2f(a), lb = log2f(b);
134 return exp2f(la + (lb - la) * t);
135 }
136
137 /* ENVELOPE_NORMAL: overshoot-compensated "false exponential". */
138 const float rate = -4.328085f; /* EXP_RATE_VAL */
139 const float overshoot = 1.0f / (1.0f - exp2f(rate));
140 float y = v0 + (v1 - v0) * overshoot * (1.0f - exp2f(rate * t));
141 if (y < 0.0f) y = 0.0f;
142 return y;
143}

190/* log(1 + GRAPH_LONG_SQUASH), the constant normaliser of the long-range
191 * squash curve — hoisted so the mapping helpers below don't recompute it on
192 * every call. */
193static float graph_log1p_squash(void)
194{
195 static float s_val = 0.0f;
196 if (s_val == 0.0f) s_val = logf(1.0f + GRAPH_LONG_SQUASH);

200/* Normalised X (0..1) -> milliseconds, range/curve aware. */
201static uint32_t graph_x_to_ms(float x)
202{
203 if (x < 0.0f) x = 0.0f;
204 if (x > 1.0f) x = 1.0f;
205 if (!s_graph_long_range) {
206 return (uint32_t)(x * (float)GRAPH_RANGE_SHORT_MS + 0.5f);
207 }
208 /* Long view: expand the squashed display X back to a linear time fraction.
209 * Display compresses with log1p(k*t)/log1p(k); invert it here. */
210 float k = GRAPH_LONG_SQUASH;
211 float t = (expf(x * graph_log1p_squash()) - 1.0f) / k; /* 0..1 linear time */
212 return (uint32_t)(t * (float)GRAPH_RANGE_LONG_MS + 0.5f);
213}

215/* Milliseconds -> normalised X (0..1), range/curve aware (inverse of above). */
216static float graph_ms_to_x(uint32_t ms)
217{
218 if (!s_graph_long_range) {
219 float x = (float)ms / (float)GRAPH_RANGE_SHORT_MS;
220 return x > 1.0f ? 1.0f : x;
221 }
222 float k = GRAPH_LONG_SQUASH;
223 float t = (float)ms / (float)GRAPH_RANGE_LONG_MS; /* 0..1 linear time */
224 if (t < 0.0f) t = 0.0f;
225 if (t > 1.0f) t = 1.0f;
226 /* Compress: more pixels to small t, fewer to the long tail. */
227 return logf(1.0f + k * t) / graph_log1p_squash();
228}

230/* Audio-taper encoder stride for the envelope points' X axis (installed as the
231 * popup's xstep hook). Envelope time knobs are conventionally exponential:
232 * each detent scales the SEGMENT duration (time since the previous point, i.e.
233 * the A/D/R value itself — not the point's cumulative axis position) by ~8%,
234 * with a minimum stride so the bottom of the range is reachable at fine
235 * resolution. A fixed normalised step (2% of a linear 2 s axis = 40 ms) made
236 * the smallest attack move 2→40 ms, which under the log-domain curve types
237 * (DX7/EXP, back-loaded attacks) silences short notes outright.
238 * Per-role feel: the attack is where a few ms decide pluck vs pad, so it gets
239 * a finer ratio and stride than decay, and release coarser still — sweeping a
240 * long tail shouldn't take a hundred detents. */
241static const float grx_ratio[4] = { 1.08f, 1.04f, 1.08f, 1.10f }; /* -,A,D,R */
242static const float grx_stride[4] = { 1.0f, 1.0f, 1.0f, 2.0f }; /* min ms */
243
244static float graph_xstep(uint8_t idx, float x, long delta)
245{
246 uint32_t prev_ms = 0;
247 if (idx > 0 && idx <= s_graph_popup.num_points) {
248 prev_ms = graph_x_to_ms(s_graph_popup.points[idx - 1].x);
249 }
250 uint32_t cum_ms = graph_x_to_ms(x);
251 float seg = (cum_ms > prev_ms) ? (float)(cum_ms - prev_ms) : 0.0f;
252
253 float ratio = grx_ratio[idx & 3u];
254 float stride = grx_stride[idx & 3u];
255 for (long n = delta; n > 0; --n) {
256 float g = seg * ratio;
257 seg = (g > seg + stride) ? g : seg + stride;
258 }
259 for (long n = delta; n < 0; ++n) {
260 float g = seg / ratio;
261 seg = (g < seg - stride) ? g : seg - stride;
262 }
263 if (seg < 0.0f) seg = 0.0f;
264
265 float range = s_graph_long_range ? (float)GRAPH_RANGE_LONG_MS
266 : (float)GRAPH_RANGE_SHORT_MS;
267 float cum = (float)prev_ms + seg;
268 if (cum > range) cum = range;
269 return graph_ms_to_x((uint32_t)(cum + 0.5f));
270}

276/* ── Auto-decay rule (derived-decay mode only) ───────────────────────────────
277 * When CONFIG_SEQ_ADSR_EXPLICIT_DECAY is OFF the decay TIME is derived, not
278 * user-dragged: the sustain point's X is locked (Y-only) in the widget and
279 * recomputed here from attack time + sustain level. Lower sustain -> longer,
280 * more audible fall; decay also scales gently with attack.
281 * When the Kconfig toggle is ON (default, industry norm) the sustain point's X
282 * is user-owned and graph_recompute_decay() is a no-op, so these constants and
283 * the whole rule are inert. */
284#if !CONFIG_SEQ_ADSR_EXPLICIT_DECAY
285#define DECAY_BASE_MS 120u //note 06-20 testing some params moving up from 40
286#define DECAY_ATTACK_K 0.5f
287#define DECAY_SUSTAIN_SPAN_MS 400.0f
288#define DECAY_MIN_MS 20u
289#define DECAY_MAX_MS 2000u
290
291static uint32_t graph_decay_ms(uint32_t attack_ms, float sustain_frac)
292{
293 if (sustain_frac < 0.0f) sustain_frac = 0.0f;
294 if (sustain_frac > 1.0f) sustain_frac = 1.0f;
295 float d = (float)DECAY_BASE_MS
296 + (float)attack_ms * DECAY_ATTACK_K
297 + (1.0f - sustain_frac) * DECAY_SUSTAIN_SPAN_MS;
298 if (d < (float)DECAY_MIN_MS) d = (float)DECAY_MIN_MS;
299 if (d > (float)DECAY_MAX_MS) d = (float)DECAY_MAX_MS;
300 return (uint32_t)(d + 0.5f);
301}
302#endif /* !CONFIG_SEQ_ADSR_EXPLICIT_DECAY */

328/* Recompute the sustain point's X (decay time) from the current attack time and
329 * sustain level, then write it back. Keeps all ms math host-side so the widget
330 * stays AMY-agnostic. Expects the standard 4-point ADSR layout. */
331static void graph_recompute_decay(void)
332{
333#if CONFIG_SEQ_ADSR_EXPLICIT_DECAY
334 /* Explicit decay: the user owns the sustain point's X (decay time), so never
335 * re-derive it. Keeps this a single guard covering every call site (seed +
336 * every encoder move). */
337 return;
338#else
339 gpopup_point_t pts[GPOPUP_MAX_POINTS];
340 uint8_t n = graph_popup_get_points(&s_graph_popup, pts, GPOPUP_MAX_POINTS);
341 if (n < 4) return;
342
343 uint32_t attack_ms = graph_x_to_ms(pts[1].x);
344 float sustain_frac = pts[2].y;
345 uint32_t decay_ms = graph_decay_ms(attack_ms, sustain_frac);
346
347 pts[2].x = graph_ms_to_x(attack_ms + decay_ms);
348 /* Preserve edit state across the points rewrite. */
349 uint8_t saved_cursor = s_graph_popup.cursor;
350 bool saved_editing = s_graph_popup.editing_value;
351 bool saved_axis_y = s_graph_popup.adjust_axis_y;
352 graph_popup_set_points(&s_graph_popup, pts, n);
353 s_graph_popup.cursor = saved_cursor;
354 s_graph_popup.editing_value = saved_editing;
355 s_graph_popup.adjust_axis_y = saved_axis_y;
356#endif /* !CONFIG_SEQ_ADSR_EXPLICIT_DECAY */
357}

382/* Seed the editor's 3 points from the stored envelope, applying the current
383 * time-range mapping to the X (time) axis. Cumulative time is squashed segment
384 * by segment so the curve shape matches what write-back will reconstruct. */
385static void graph_seed_from_env(const seq_env_t *env)
386{
387 uint32_t cum_a = env->attack_ms;
388 uint32_t cum_d = env->attack_ms + env->decay_ms;
389 uint32_t cum_r = env->attack_ms + env->decay_ms + env->release_ms;
390
391 gpopup_point_t pts[4];
392 pts[0].x = 0.0f; pts[0].y = 0.0f; /* origin */
393 pts[1].x = graph_ms_to_x(cum_a); pts[1].y = 1.0f; /* attack peak*/
394 pts[2].x = graph_ms_to_x(cum_d); pts[2].y = (float)env->sustain_pct / 100.0f; /* sustain */
395 pts[3].x = graph_ms_to_x(cum_r); pts[3].y = 0.0f; /* release end*/
396 graph_popup_set_points(&s_graph_popup, pts, 4);
397 /* Derived-decay mode: snap the sustain point's X to the auto-decay rule so
398 * the opening curve already obeys it. Explicit-decay mode: no-op, so the
399 * stored decay_ms (already mapped into pts[2].x above) is shown as-is. */
400 graph_recompute_decay();
401}

579/* Convert the popup's current points to a seq_env_t and write it to the given
580 * eg_index's store on the bound target. Shared by graph_commit_to_env() (the
581 * currently-shown eg_index) and graph_toggle_eg_index() (writes the
582 * DEPARTING eg_index through before switching the view). */
583/* Convert the current popup points into `env`'s ADSR fields, plus the shown
584 * curve type from s_graph_eg_type_disp (the editor's scratch — the type is
585 * previewed live and persisted on commit, like the points). `env` should be
586 * pre-seeded from the target so any non-ADSR fields carry through. Returns
587 * false when the popup has no full ADSR point set. */
588static bool graph_points_to_env(seq_env_t *env)
589{
590 gpopup_point_t pts[GPOPUP_MAX_POINTS];
591 uint8_t n = graph_popup_get_points(&s_graph_popup, pts, GPOPUP_MAX_POINTS);
592 if (n < 4) return false; /* expect origin + A + D + R */
593
594 uint32_t cum_a = graph_x_to_ms(pts[1].x);
595 uint32_t cum_d = graph_x_to_ms(pts[2].x);
596 uint32_t cum_r = graph_x_to_ms(pts[3].x);
597
598 /* Convert cumulative times back to per-segment durations (clamp monotonic). */
599 uint32_t a = cum_a;
600 if (a < 2) a = 2; /* minimum 2 ms attack prevents DAC pop on trigger */
601 env->attack_ms = a;
602 env->decay_ms = (cum_d > cum_a) ? (cum_d - cum_a) : 0;
603 env->release_ms = (cum_r > cum_d) ? (cum_r - cum_d) : 0;
604 env->sustain_pct = (uint8_t)(pts[2].y * 100.0f + 0.5f);
605 env->eg_type = s_graph_eg_type_disp;
606 return true;
607}

832/* Set the time-range mode (long_range=true → LONG 15s, false → SHORT 2s) and
833 * remap current on-screen points so in-progress edits survive the switch.
834 * No-op when already in the target range. This is the single place that flips
835 * s_graph_long_range — all callers (manual toggle, auto-range) go through here. */
836static void graph_set_range(bool long_range)
837{
838 if (s_graph_long_range == long_range) return;
839
840 /* Snapshot each point's time in ms under the OLD range, then flip the range
841 * and remap ms → x under the NEW range. Y is range-independent. */
842 gpopup_point_t pts[GPOPUP_MAX_POINTS];
843 uint8_t n = graph_popup_get_points(&s_graph_popup, pts, GPOPUP_MAX_POINTS);
844 uint32_t ms[GPOPUP_MAX_POINTS];
845 for (uint8_t i = 0; i < n; ++i) ms[i] = graph_x_to_ms(pts[i].x);
846
847 s_graph_long_range = long_range;
848
849 for (uint8_t i = 0; i < n; ++i) pts[i].x = graph_ms_to_x(ms[i]);
850 uint8_t saved_cursor = s_graph_popup.cursor;
851 bool saved_editing = s_graph_popup.editing_value;
852 bool saved_axis_y = s_graph_popup.adjust_axis_y;
853 graph_popup_set_points(&s_graph_popup, pts, n);
854 s_graph_popup.cursor = saved_cursor;
855 s_graph_popup.editing_value = saved_editing;
856 s_graph_popup.adjust_axis_y = saved_axis_y;
857
858 graph_update_ticks();
859 ESP_LOGI(TAG, "graph range -> %s", long_range ? "LONG(15s)" : "SHORT(2s)");
860}

862/* Auto-switch the range based on the total envelope time (pts[3].x = cum_r).
863 * Metric: rightmost point because that is the literal x-extent of the curve —
864 * a long attack also overflows the SHORT axis, and cum_r = A+D+R captures all.
865 * Hysteresis: SHORT→LONG at ≥2000ms, LONG→SHORT only at ≤1700ms so the range
866 * does not flicker at the boundary. Called after every encoder edit and once at
867 * editor open. No-op when the popup is not active. */
868static void graph_auto_range_check(void)
869{
870 if (!graph_popup_is_active(&s_graph_popup)) return;
871 gpopup_point_t pts[GPOPUP_MAX_POINTS];
872 uint8_t n = graph_popup_get_points(&s_graph_popup, pts, GPOPUP_MAX_POINTS);
873 if (n < 4) return;
874 uint32_t total_ms = graph_x_to_ms(pts[3].x);
875 if (!s_graph_long_range && total_ms >= GRAPH_RANGE_SHORT_MS) {
876 graph_set_range(true); /* grow to LONG at the SHORT axis ceiling */
877 } else if (s_graph_long_range && total_ms <= 1700u) {
878 graph_set_range(false); /* shrink back to SHORT below hysteresis floor */
879 }
880}

935/* ── Perceived-duration equivalence across curve types ───────────────────────
936 * Fraction of a segment's nominal time at which each type has covered its
937 * first/last 40 dB — where a falling segment becomes effectively silent, or a
938 * rising one becomes audible. Derived from envelope.c: EXP/DX7 move linearly
939 * in dB down to the -74 dB BREAKPOINT_EPS floor, so -40 dB lands at ~40/74 of
940 * T; NORMAL's RC shape crosses it at ~0.94 T; LINEAR at 0.99 T; the DX7
941 * attack law is RC-like by design. Cycling types rescales each segment by
942 * g_old/g_new so the AUDIBLE length is preserved even though the stored ms
943 * (and the drawn point positions) change — without this, LIN→EXP at fixed ms
944 * turns a long ringing release into a fast pluck. */
945static const float graph_g_fall[4] = { 0.94f, 0.99f, 0.54f, 0.54f }; /* NRM LIN DX7 EXP */
946static const float graph_g_rise[4] = { 0.94f, 0.99f, 0.94f, 0.54f };
947
948static void graph_rescale_points_for_type(uint8_t old_type, uint8_t new_type)
949{
950 gpopup_point_t pts[GPOPUP_MAX_POINTS];
951 uint8_t n = graph_popup_get_points(&s_graph_popup, pts, GPOPUP_MAX_POINTS);
952 if (n < 4) return;
953
954 float cum_a = (float)graph_x_to_ms(pts[1].x);
955 float cum_d = (float)graph_x_to_ms(pts[2].x);
956 float cum_r = (float)graph_x_to_ms(pts[3].x);
957 float a = cum_a;
958 float d = (cum_d > cum_a) ? (cum_d - cum_a) : 0.0f;
959 float r = (cum_r > cum_d) ? (cum_r - cum_d) : 0.0f;
960
961 a *= graph_g_rise[old_type & 3u] / graph_g_rise[new_type & 3u];
962 float kf = graph_g_fall[old_type & 3u] / graph_g_fall[new_type & 3u];
963 d *= kf;
964 r *= kf;
965 if (a < 2.0f) a = 2.0f;
966
967 /* Grow into the LONG range up front so a lengthening rescale isn't clipped
968 * by the SHORT axis; the auto-range check below shrinks back when a
969 * shortening rescale allows it. */
970 if (!s_graph_long_range && (a + d + r) >= (float)GRAPH_RANGE_SHORT_MS) {
971 graph_set_range(true);
972 }
973 float range = s_graph_long_range ? (float)GRAPH_RANGE_LONG_MS
974 : (float)GRAPH_RANGE_SHORT_MS;
975 float ca = (a > range) ? range : a;
976 float cd = (ca + d > range) ? range : (ca + d);
977 float cr = (cd + r > range) ? range : (cd + r);
978 pts[1].x = graph_ms_to_x((uint32_t)(ca + 0.5f));
979 pts[2].x = graph_ms_to_x((uint32_t)(cd + 0.5f));
980 pts[3].x = graph_ms_to_x((uint32_t)(cr + 0.5f));
981
982 uint8_t saved_cursor = s_graph_popup.cursor;
983 bool saved_editing = s_graph_popup.editing_value;
984 bool saved_axis_y = s_graph_popup.adjust_axis_y;
985 graph_popup_set_points(&s_graph_popup, pts, n);
986 s_graph_popup.cursor = saved_cursor;
987 s_graph_popup.editing_value = saved_editing;
988 s_graph_popup.adjust_axis_y = saved_axis_y;
989
990 graph_recompute_decay();
991 graph_update_ticks();
992 graph_auto_range_check();
993}

1043/* Route an encoder delta to the pop-up. Returns true if the pop-up consumed it
1044 * (host should then skip its normal sequencer routing). */

1048
1049 if (s_graph_amp_mode) {
1050 if (s_graph_eg_index == 1 && s_graph_target == GRAPH_TGT_MELODIC) {
1051 /* EG1 page: encoder adjusts EG1->cutoff depth, 0.25 oct/detent.
1052 * Bipolar -8..+8; negative = inverted/downward sweep (same range
1053 * as the filter editor's EG cursor — one shared field). */
1054 float v = s_graph_fenv_edit + (float)delta * 0.25f;
1055 if (v < -8.0f) v = -8.0f;
1056 if (v > 8.0f) v = 8.0f;

1058 s_graph_fenv_dirty = true;

1061 return true;
1062 }

1064 * Applied live but throttled (melodic applies re-emit the track). */
1065 float v = s_graph_amp_edit + (float)delta * 0.05f;
1066 if (v < 0.0f) v = 0.0f;
1067 if (v > 1.0f) v = 1.0f;
1068 s_graph_amp_edit = v;
1069 s_amp_live_pending = true;
1070 graph_amp_live_flush(false);
1071 s_force_redraw = true;
1072 return true;
1073 }

1075 s_graph_env_dirty = true; /* user moved an ADSR control point */
1076 bool adjusting = s_graph_popup.editing_value;
1077 graph_popup_handle_encoder(&s_graph_popup, delta);
1078 /* Derived-decay mode: moving A (time) or the S level changes the derived
1079 * decay, so re-snap S.x. No-op in explicit-decay mode (user owns S.x). */
1080 graph_recompute_decay();
1081 /* Auto-switch range if total envelope time crosses the threshold. */
1082 graph_auto_range_check();
1083 /* Point actually moved (not just cursor selection): audition it. */
1084 if (adjusting) graph_live_push_env();

1086}

1136/* Normalisation helpers shared between open and commit. */
1137static float filter_hz_to_norm(float hz)
1138{
1139 if (hz < FGRAPH_CUTOFF_HZ_MIN) hz = FGRAPH_CUTOFF_HZ_MIN;
1140 if (hz > FGRAPH_CUTOFF_HZ_MAX) hz = FGRAPH_CUTOFF_HZ_MAX;
1141 return log2f(hz / FGRAPH_CUTOFF_HZ_MIN) /
1142 log2f(FGRAPH_CUTOFF_HZ_MAX / FGRAPH_CUTOFF_HZ_MIN);
1143}
1144
1145static float filter_norm_to_hz(float norm)
1146{
1147 return FGRAPH_CUTOFF_HZ_MIN *
1148 powf(FGRAPH_CUTOFF_HZ_MAX / FGRAPH_CUTOFF_HZ_MIN,
1149 SEQ_CLAMP_F32(norm, 0.0f, 1.0f));
1150}
1151
1152static float filter_q_to_norm(float q)
1153{
1154 float n = (q - FGRAPH_RES_MIN) / (FGRAPH_RES_MAX - FGRAPH_RES_MIN);
1155 return SEQ_CLAMP_F32(n, 0.0f, 1.0f);
1156}
1157
1158static float filter_norm_to_q(float n)
1159{
1160 return FGRAPH_RES_MIN + SEQ_CLAMP_F32(n, 0.0f, 1.0f) *
1161 (FGRAPH_RES_MAX - FGRAPH_RES_MIN);
1162}

1174/* Read the current filter state from the active target into s_filter_edit. */
1175static void filter_load_from_target(void)
1176{
1177 seq_filter_t f = {0};
1178 if (seq_state.ui_mode == UI_MODE_DRONE) {
1179 /* Drone: always LPF24, cutoff = sweep midpoint, Q from drone_get_resonance. */
1180 float lo = drone_get_sweep_lo();
1181 float hi = drone_get_sweep_hi();
1182 f.filter_type = 4; /* SEQ_FILTER_LPF24 */
1183 f.cutoff_hz = SEQ_CLAMP_F32((lo + hi) * 0.5f, FGRAPH_CUTOFF_HZ_MIN, FGRAPH_CUTOFF_HZ_MAX);
1184 f.resonance = drone_get_resonance();
1185 f.enabled = true;
1186
1187 snprintf(s_fgraph.label, sizeof(s_fgraph.label), "STUTR");
1188 } else if (seq_state.ui_mode == UI_MODE_DRONE_STD) {
1189 drone_std_get_filter(&f);
1190 /* Same never-authored sentinel handling as the arp: seed sensible
1191 * starting values but honestly read "OFF" until the user enables. */
1192 if (f.cutoff_hz <= 0.0f) {
1193 f.filter_type = SEQ_FILTER_LPF;
1194 f.cutoff_hz = 1200.0f;
1195 f.resonance = 1.0f;
1196 f.enabled = false;
1197 }
1198 snprintf(s_fgraph.label, sizeof(s_fgraph.label), "DRONE");
1199 } else if (seq_state.ui_mode == UI_MODE_ARP) {
1200 arp_get_filter(&f);
1201 /* Only apply display defaults when the filter was never authored
1202 * (cutoff_hz == 0 from zero-init); preserve authored values even
1203 * when disabled so enabled=false round-trips correctly. Default to
1204 * disabled (enabled=false) so the graph honestly reads "OFF" for a
1205 * never-authored track — the type/cutoff/resonance below are only a
1206 * sensible starting point for when the user toggles the filter on. */
1207 if (f.cutoff_hz <= 0.0f) {
1208 f.filter_type = SEQ_FILTER_LPF;
1209 f.cutoff_hz = 800.0f;
1210 f.resonance = 1.0f;
1211 f.enabled = false;
1212 }
1213 snprintf(s_fgraph.label, sizeof(s_fgraph.label), "ARP");
1214 } else {
1215 uint8_t li = seq_state.active_layer_idx;
1216 uint8_t tr = seq_state.selected_track;
1217 sequencer_core_get_melodic_filter(li, tr, &f);
1218 /* Same sentinel: zero cutoff means never-authored; disabled-but-authored
1219 * keeps its authored values for correct round-trip display. Default to
1220 * disabled so the graph reads "OFF" when the track has no filter — the
1221 * type/cutoff/resonance are just the starting point for enabling one. */
1222 if (f.cutoff_hz <= 0.0f) {
1223 f.filter_type = SEQ_FILTER_LPF;
1224 f.cutoff_hz = 800.0f;
1225 f.resonance = 1.0f;
1226 f.enabled = false;
1227 }
1228 snprintf(s_fgraph.label, sizeof(s_fgraph.label), "L%u T%u%s",
1229 (unsigned)(li + 1), (unsigned)(tr + 1),
1230 s_editor_apply_all ? ">L" : ">T");
1231 }
1232 /* Drone filter is a fixed LPF24 that is always on (cursor tops out at 1), so
1233 * it hides the type/enable header controls; melodic + arp expose both. */
1234 s_fgraph.show_toggles = (seq_state.ui_mode != UI_MODE_DRONE);
1235 s_filter_edit = f;
1236}

1238/* ── Filter live preview ─────────────────────────────────────────────────────
1239 * Same model as the envelope editor: each edit pushes the scratch filter to
1240 * AMY only; cancel re-pushes the store (or reloads the layer when a touched
1241 * melodic row was never authored). The drone has no preview path — its sweep
1242 * window/resonance are live drone state — so its values are snapshotted at
1243 * open and re-set on cancel. */
1244static bool s_filter_live = false; /* any live push this session */
1245static float s_fdrone_open_lo, s_fdrone_open_hi, s_fdrone_open_res;
1246
1247static void filter_live_push(void)
1248{
1249 if (seq_state.ui_mode == UI_MODE_DRONE) {
1250 /* Same midpoint math as commit: move the sweep window, keep width. */
1251 float lo = drone_get_sweep_lo();
1252 float hi = drone_get_sweep_hi();
1253 float half = (hi - lo) * 0.5f;
1254 float mid = filter_norm_to_hz(s_fgraph.cutoff_norm);
1255 drone_set_sweep_lo(SEQ_CLAMP_F32(mid - half, 65.0f, 7900.0f));
1256 drone_set_sweep_hi(SEQ_CLAMP_F32(mid + half, 100.0f, 8000.0f));
1257 drone_set_resonance(s_filter_edit.resonance);
1258 } else if (seq_state.ui_mode == UI_MODE_DRONE_STD) {
1259 drone_std_preview_filter(&s_filter_edit);
1260 } else if (seq_state.ui_mode == UI_MODE_ARP) {
1261 arp_preview_filter(&s_filter_edit);
1262 } else {
1263 uint8_t li = seq_state.active_layer_idx;
1264 if (s_editor_apply_all) {
1265 for (uint8_t t = 0; t < SEQ_TRACKS; ++t)
1266 sequencer_core_preview_melodic_filter(li, t, &s_filter_edit);
1267 } else {
1268 sequencer_core_preview_melodic_filter(li, seq_state.selected_track,
1269 &s_filter_edit);
1270 }
1271 }

1275static void filter_live_cancel_restore(void)
1276{
1277 if (!s_filter_live) return;
1278 s_filter_live = false;
1279
1280 if (seq_state.ui_mode == UI_MODE_DRONE) {
1281 drone_set_sweep_lo(s_fdrone_open_lo);
1282 drone_set_sweep_hi(s_fdrone_open_hi);
1283 drone_set_resonance(s_fdrone_open_res);
1284 return;
1285 }
1286 if (seq_state.ui_mode == UI_MODE_DRONE_STD) {
1287 /* Re-push the stored filter (bypass when it was never enabled). */
1288 seq_filter_t f;
1289 drone_std_get_filter(&f);
1290 drone_std_preview_filter(&f);
1291 return;
1292 }
1293 if (seq_state.ui_mode == UI_MODE_ARP) {
1294 /* Best effort: re-push the stored filter. A never-authored PATCH-mode
1295 * arp keeps the previewed bypass until its next patch reload. */
1296 seq_filter_t f;
1297 arp_get_filter(&f);
1298 arp_preview_filter(&f);
1299 return;
1300 }
1301 uint8_t li = seq_state.active_layer_idx;
1302 uint8_t t0 = s_editor_apply_all ? 0 : seq_state.selected_track;
1303 uint8_t t1 = s_editor_apply_all ? (uint8_t)(SEQ_TRACKS - 1)
1304 : seq_state.selected_track;
1305 bool need_reload = false;
1306 for (uint8_t t = t0; t <= t1; ++t) {
1307 if (!sequencer_core_melodic_filter_authored(li, t)) need_reload = true;
1308 }
1309 if (need_reload) {
1310 sequencer_core_reload_layer_synth(li);
1311 return;
1312 }
1313 for (uint8_t t = t0; t <= t1; ++t) {
1314 seq_filter_t f;
1315 if (sequencer_core_get_melodic_filter(li, t, &f))
1316 sequencer_core_preview_melodic_filter(li, t, &f);
1317 }
1318}

1346bool synth_ui_filter_handle_encoder(long delta)
1347{
1348 if (!s_filter_active) return false;
1349 bool drone = (seq_state.ui_mode == UI_MODE_DRONE);
1350
1351 bool arp = (seq_state.ui_mode == UI_MODE_ARP);
1352
1353 if (!s_fgraph.editing) {
1354 /* Not editing: scroll cursor position.
1355 * Drone: 0=cutoff 1=resonance (type fixed, no EN cursor).
1356 * Arp/melodic: 0=cutoff 1=resonance 2=type 3=enable. The melodic
1357 * EG1 sweep depth/polarity live on the envelope editor's EG1 page
1358 * (arp uses the fixed ARP_FILTER_EG1_DEPTH_OCT). */
1359 uint8_t max_cursor = drone ? 1 : 3;
1360 if (delta > 0) {
1361 s_fgraph.cursor = (uint8_t)((s_fgraph.cursor + 1) % (max_cursor + 1));
1362 } else if (delta < 0) {
1363 s_fgraph.cursor = (s_fgraph.cursor == 0) ? max_cursor : (s_fgraph.cursor - 1);
1364 }
1365 s_force_redraw = true;
1366 return true;
1367 }
1368
1369 /* Editing: adjust the selected parameter. */
1370 switch (s_fgraph.cursor) {
1371 case 0: { /* cutoff */
1372 float step = 0.015f * (float)delta;
1373 s_fgraph.cutoff_norm = SEQ_CLAMP_F32(s_fgraph.cutoff_norm + step, 0.0f, 1.0f);
1374 s_filter_edit.cutoff_hz = filter_norm_to_hz(s_fgraph.cutoff_norm);
1375 break;
1376 }
1377 case 1: { /* resonance */
1378 float step = 0.02f * (float)delta;
1379 s_fgraph.resonance_norm = SEQ_CLAMP_F32(s_fgraph.resonance_norm + step, 0.0f, 1.0f);
1380 s_filter_edit.resonance = filter_norm_to_q(s_fgraph.resonance_norm);
1381 break;
1382 }
1383 case 2: { /* type (melodic/arp only) */
1384 if (!drone) {
1385 int t = (int)s_filter_edit.filter_type + (int)delta;
1386 /* Wrap within 1..4 (NONE is toggled via MY_BUTTON_1, not the type cursor). */
1387 if (t < 1) t = 4;
1388 if (t > 4) t = 1;
1389 s_filter_edit.filter_type = (uint8_t)t;
1390 s_fgraph.filter_type = (uint8_t)t;
1391 }
1392 break;
1393 }
1394 case 3: { /* enable toggle (melodic/arp only; drone filter is always on) */
1395 if (!drone) {
1396 s_filter_edit.enabled = !s_filter_edit.enabled;
1397 s_fgraph.enabled = s_filter_edit.enabled;
1398 ESP_LOGI(TAG, "filter enabled -> %d", (int)s_filter_edit.enabled);
1399 }
1400 break;
1401 }
1402 default: break;
1403 }
1404 filter_sync_fgraph();
1405 filter_live_push();
1406 s_force_redraw = true;
1407 return true;
1408}

1488uint32_t lfo_view_signature(void)
1489{
1490 const seq_lfo_t *l = &s_lfo_view.lfo;
1491 return (uint32_t)l->enabled
1492 | ((uint32_t)l->wave << 1) /* 3 bits */
1493 | ((uint32_t)l->rate << 4) /* 3 bits */
1494 | ((uint32_t)l->depth << 7) /* 7 bits (0..100) */

1536bool synth_ui_lfo_handle_encoder(long delta)
1537{
1538 if (!s_lfo_active) return false;
1539 /* Fields: 5 target checkboxes (0..LFO_TARGET_COUNT-1), then WAVE/RATE/DEPTH/EN. */
1540 const uint8_t N = LFO_FLD_COUNT;
1541 if (!s_lfo_view.editing) {
1542 if (delta > 0) s_lfo_view.cursor = (s_lfo_view.cursor + 1) % N;

1544 s_force_redraw = true;
1545 return true;
1546 }
1547 /* Only the multi-value fields (WAVE/RATE/DEPTH) use adjust mode; checkbox
1548 * and EN fields toggle on press and never set `editing`. */
1549 seq_lfo_t *l = &s_lfo_view.lfo;
1550 int d = (delta > 0) ? 1 : -1;
1551 switch (s_lfo_view.cursor) {
1552 case LFO_FLD_WAVE:
1553 l->wave = (lfo_wave_t)((l->wave + LFO_WAVE_COUNT + d) % LFO_WAVE_COUNT);
1554 break;
1555 case LFO_FLD_RATE:
1556 l->rate = (lfo_rate_t)((l->rate + LFO_RATE_COUNT + d) % LFO_RATE_COUNT);
1557 break;
1558 case LFO_FLD_DEPTH: /* ±5%, clamped 0..100 */
1559 if (d > 0) l->depth = (l->depth < 95) ? l->depth + 5 : 100;
1560 else l->depth = (l->depth > 5) ? l->depth - 5 : 0;
1561 break;
1562 default: break;
1563 }
1564 s_force_redraw = true;
1565 return true;
1566}

1733 * which is now set automatically and no longer meaningful to the user).
1734 * The melodic EG1 page shows the signed sweep depth instead whenever the
1735 * middle readout is idle, so the shoulder-button polarity flip has a
1736 * visible readout. */

1751 snprintf(amp_buf, sizeof(amp_buf), "ENV%+.2f", (double)s_graph_fenv_edit);
