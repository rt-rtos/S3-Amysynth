86void synth_ui_cycle_melodic_patch(int delta)
87{
88 if (delta == 0) return;
89
90 uint8_t li = seq_state.active_layer_idx;
91 if (li >= seq_state.num_layers) return;
92 if (seq_state.layers[li].type != SEQ_LAYER_MELODIC) return;
93
94 int dir = (delta > 0) ? 1 : -1;
95 uint16_t next = patch_domain_step(&s_melodic_domain, sequencer_core_get_layer_patch(li), dir);
96
97 sequencer_core_set_layer_patch(li, next);
98
99 uint16_t applied = sequencer_core_get_layer_patch(li);
100 seq_state.layers[li].patch = applied;
101
102 const char *name = patch_name_for(applied);
103 if (name) {
104 ESP_LOGI(TAG, "L%u melodic patch -> %u (%s)", (unsigned)li + 1u, (unsigned)applied, name);
105 } else {
106 ESP_LOGI(TAG, "L%u melodic patch -> %u", (unsigned)li + 1u, (unsigned)applied);
107 }
108}

110/* Cycle the SELECTED drum track's patch through the curated drum list: the drum
111 * layer is per-track, so this targets seq_state.selected_track on the active
112 * layer. Mirrors the applied patch into the UI copy for the on-screen label. */
113void synth_ui_cycle_drum_patch(int delta)
114{
115 if (delta == 0) return;
116
117 uint8_t li = seq_state.active_layer_idx;
118 if (li >= seq_state.num_layers) return;
119 if (seq_state.layers[li].type != SEQ_LAYER_DRUM) return;
120
121 uint8_t track = seq_state.selected_track;
122 if (track >= SEQ_TRACKS) return;
123
124 int dir = (delta > 0) ? 1 : -1;
125
126 /* PCM engine: browse the drum samples instead of the curated SYNTH list,
127 * which stays stored for the next engine switch. */
128 if (sequencer_core_get_drum_engine() == SEQ_DRUM_PCM) {
129 uint16_t preset = sequencer_core_cycle_drum_pcm_preset(li, track, dir);
130 seq_state.layers[li].track_pcm_preset[track] = preset;
131
132 const char *pcm_name = pcm_preset_name_for(preset);
133 if (pcm_name) {
134 ESP_LOGI(TAG, "drum PCM preset cycle L%u T%u -> %u (%s)",
135 li + 1u, track + 1u, (unsigned)preset, pcm_name);
136 } else {
137 ESP_LOGI(TAG, "drum PCM preset cycle L%u T%u -> %u",
138 li + 1u, track + 1u, (unsigned)preset);
139 }
140 return;
141 }
142
143 uint16_t applied = sequencer_core_cycle_drum_patch(li, track, dir);
144
145 /* Keep the UI mirror in sync (track_patch[] drives the label). */
146 seq_state.layers[li].track_patch[track] = applied;
147 if (track == 0) seq_state.layers[li].patch = applied;
148
149 const char *name = patch_name_for(applied);
150 if (name) {
151 ESP_LOGI(TAG, "drum patch cycle L%u T%u -> %u (%s)",
152 li + 1u, track + 1u, (unsigned)applied, name);
153 } else {
154 ESP_LOGI(TAG, "drum patch cycle L%u T%u -> %u",
155 li + 1u, track + 1u, (unsigned)applied);
156 }
157}

159/* Shift+Turn on the SEQ screen: step the active melodic layer's FM algorithm
160 * live (audible mid-note - the whole point is auditioning the 30+ variations
161 * by ear). Non-FM patches get a "NOT FM" banner instead of a dead detent;
162 * drum layers ignore the gesture. */
163void synth_ui_cycle_fm_algo(int delta)
164{
165 if (delta == 0) return;
166
167 uint8_t li = seq_state.active_layer_idx;
168 if (li >= seq_state.num_layers) return;
169 if (seq_state.layers[li].type != SEQ_LAYER_MELODIC) return;
170
171 int a = sequencer_core_cycle_layer_fm_algo(li, (delta > 0) ? 1 : -1);
172 seq_state.algo_banner_value =
173 (a >= 0) ? (uint8_t)a : DISPLAY_ALGO_BANNER_NOFM;
174 seq_state.algo_banner_ticks = 24; /* ~1.2 s at the 20 Hz UI rate */
175 s_force_redraw = true;
176}
