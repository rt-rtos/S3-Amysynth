88void synth_ui_cycle_melodic_patch(int delta)
89{
90 if (delta == 0) return;
91
92 uint8_t li = seq_state.active_layer_idx;
93 if (li >= seq_state.num_layers) return;
94 if (seq_state.layers[li].type != SEQ_LAYER_MELODIC) return;
95
96 int dir = (delta > 0) ? 1 : -1;
97 uint16_t next = patch_domain_step(&s_melodic_domain, sequencer_core_get_layer_patch(li), dir);
98
99 sequencer_core_set_layer_patch(li, next);
100
101 uint16_t applied = sequencer_core_get_layer_patch(li);
102 seq_state.layers[li].patch = applied;
103
104 const char *name = patch_name_for(applied);
105 if (name) {
106 ESP_LOGI(TAG, "L%u melodic patch -> %u (%s)", (unsigned)li + 1u, (unsigned)applied, name);
107 } else {
108 ESP_LOGI(TAG, "L%u melodic patch -> %u", (unsigned)li + 1u, (unsigned)applied);
109 }
110}

112/* Cycle the SELECTED drum track's patch through the curated drum list. The drum
113 * layer is per-track, so this targets seq_state.selected_track on the active
114 * layer (must be a drum layer). Mirrors the applied patch into the UI copy so
115 * the on-screen patch number updates. */
116void synth_ui_cycle_drum_patch(int delta)
117{
118 if (delta == 0) return;
119
120 uint8_t li = seq_state.active_layer_idx;
121 if (li >= seq_state.num_layers) return;
122 if (seq_state.layers[li].type != SEQ_LAYER_DRUM) return;
123
124 uint8_t track = seq_state.selected_track;
125 if (track >= SEQ_TRACKS) return;
126
127 int dir = (delta > 0) ? 1 : -1;
128
129 /* PCM engine: the control browses the ROM drum samples instead of the
130 * curated SYNTH patch list (which stays stored for the next engine
131 * switch). Mirror + name lookup use the PCM preset table. */
132 if (sequencer_core_get_drum_engine() == SEQ_DRUM_PCM) {
133 uint16_t preset = sequencer_core_cycle_drum_pcm_preset(li, track, dir);
134 seq_state.layers[li].track_pcm_preset[track] = preset;
135
136 const char *pcm_name = pcm_preset_name_for(preset);
137 if (pcm_name) {
138 ESP_LOGI(TAG, "drum PCM preset cycle L%u T%u -> %u (%s)",
139 li + 1u, track + 1u, (unsigned)preset, pcm_name);
140 } else {
141 ESP_LOGI(TAG, "drum PCM preset cycle L%u T%u -> %u",
142 li + 1u, track + 1u, (unsigned)preset);
143 }
144 return;
145 }
146
147 uint16_t applied = sequencer_core_cycle_drum_patch(li, track, dir);
148
149 /* Keep the UI mirror in sync (track_patch[] drives the label). */
150 seq_state.layers[li].track_patch[track] = applied;
151 if (track == 0) seq_state.layers[li].patch = applied;
152
153 const char *name = patch_name_for(applied);
154 if (name) {
155 ESP_LOGI(TAG, "drum patch cycle L%u T%u -> %u (%s)",
156 li + 1u, track + 1u, (unsigned)applied, name);
157 } else {
158 ESP_LOGI(TAG, "drum patch cycle L%u T%u -> %u",
159 li + 1u, track + 1u, (unsigned)applied);
160 }
161}
