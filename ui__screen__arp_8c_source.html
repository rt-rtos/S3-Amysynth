25/* Build the flat arp view from arp_core for the renderer. */
26void arp_build_view(arp_view_t *out)
27{
28 static const char *s_dir_names[ARP_DIR_COUNT] = { "UP", "DOWN", "SLOT" };
29 out->enabled = arp_get_enabled();
30 out->mode_str = s_dir_names[arp_get_direction()];
31 out->octaves = arp_get_octaves();
32 out->rate_str = arp_rate_name(arp_get_rate());
33 out->gate_pct = arp_get_gate_pct();
34 for (uint8_t i = 0; i < ARP_VIEW_SLOTS; i++) {
35 int16_t raw = arp_get_slot(i);
36 int16_t snapped = arp_get_slot_snapped(i);
37 out->slot_rest[i] = (raw == ARP_REST);
38 if (snapped >= 0) {
39 out->slot_active[i] = true;
40 ui_note_name((uint8_t)snapped, out->slot_name[i]);
41 } else {
42 out->slot_active[i] = false;
43 out->slot_name[i][0] = '\0';
44 }
45 }
46 out->cursor = s_arp_cursor;
47 out->editing = s_arp_editing;
48 out->portamento_ms = arp_get_portamento_ms();
49
50 /* Patch indicator mirrors the sequencer view: number always, name banner
51 * only while the patch hold+turn gesture is active. */
52 out->patch = arp_get_patch();
53 out->patch_select = seq_state.patch_select_mode;
54 out->patch_name = patch_name_for(out->patch);
55}

57/* Builds the view into *out so the caller can draw from it without rebuilding. */
58uint32_t arp_view_signature(arp_view_t *out)
59{
60 uint32_t h = FNV1A_OFFSET;
61 arp_build_view(out);
62 h = fnv1a_bytes(h, &out->enabled, sizeof(out->enabled));
63 h = fnv1a_bytes(h, &out->octaves, sizeof(out->octaves));
64 h = fnv1a_bytes(h, &out->gate_pct, sizeof(out->gate_pct));
65 h = fnv1a_bytes(h, &out->cursor, sizeof(out->cursor));
66 h = fnv1a_bytes(h, &out->editing, sizeof(out->editing));
67 h = fnv1a_bytes(h, &out->patch, sizeof(out->patch));
68 h = fnv1a_bytes(h, &out->patch_select, sizeof(out->patch_select));
69 h = fnv1a_bytes(h, &out->portamento_ms, sizeof(out->portamento_ms));
70 h = fnv1a_bytes(h, out->rate_str, 4);
71 h = fnv1a_bytes(h, out->mode_str, 4);
72 for (uint8_t i = 0; i < ARP_VIEW_SLOTS; i++) {
73 h = fnv1a_bytes(h, &out->slot_active[i], sizeof(out->slot_active[i]));
74 h = fnv1a_bytes(h, &out->slot_rest[i], sizeof(out->slot_rest[i]));
75 h = fnv1a_bytes(h, out->slot_name[i], sizeof(out->slot_name[i]));
76 }
77 return h;
78}

80static void arp_edit_value(uint8_t cursor, int delta)
81{
82 int dir = (delta > 0) ? 1 : (delta < 0 ? -1 : 0);
83 switch (cursor) {
84 case ARP_CUR_ENABLE:
85 if (dir != 0) arp_set_enabled(!arp_get_enabled());
86 break;
87 case ARP_CUR_MODE:
88 if (dir != 0) {
89 int nd = ((int)arp_get_direction() + dir + ARP_DIR_COUNT) % ARP_DIR_COUNT;
90 arp_set_direction((arp_dir_t)nd);
91 }
92 break;
93 case ARP_CUR_OCT:
94 arp_set_octaves((uint8_t)SEQ_CLAMP_INT(
95 (int)arp_get_octaves() + dir, 1, ARP_OCT_MAX));
96 break;
97 case ARP_CUR_RATE: {
98 int r = (int)arp_get_rate() + dir;
99 r = SEQ_CLAMP_INT(r, 0, ARP_RATE_COUNT - 1);
100 arp_set_rate((arp_rate_t)r);
101 break;
102 }
103 case ARP_CUR_GATE:
104 arp_set_gate_pct((uint8_t)SEQ_CLAMP_INT(
105 (int)arp_get_gate_pct() + dir * 5, 10, 100));
106 break;
107 case ARP_CUR_PORTA:
108 /* 1ms/detent: fine control for short, snappy glides. */
109 arp_set_portamento_ms((uint16_t)SEQ_CLAMP_INT(
110 (int)arp_get_portamento_ms() + dir * 1, 0, ARP_PORTAMENTO_MAX_MS));
111 break;
112 default: {
113 /* Slot edit: chromatic note, or clear below the floor. */
114 uint8_t slot = (uint8_t)(cursor - ARP_CUR_SLOT0);
115 if (slot >= ARP_VIEW_SLOTS) break;
116 int16_t cur = arp_get_slot(slot);
117 if (cur == -1) {
118 /* Unused: up seeds a note, down sets REST (one step above floor). */
119 if (dir > 0) arp_set_slot(slot, (int16_t)arp_get_root_note());
120 else arp_set_slot(slot, ARP_REST);
121 } else if (cur == ARP_REST) {
122 /* REST is the floor: up goes to empty, not root note, so recovery
123 * is two turns rather than scrolling through octaves. */
124 if (dir > 0) arp_set_slot(slot, -1);
125 /* dir < 0: stay at REST, no bounce back to empty */
126 } else {
127 int nv = (int)cur + dir;
128 if (nv < 24) {
129 arp_set_slot(slot, -1); /* below floor → clear to empty */
130 } else if (nv > 127) {
131 arp_set_slot(slot, 127); /* ceiling clamp, no wrap */
132 } else {
133 arp_set_slot(slot, (int16_t)nv);
134 }
135 }
136 break;
137 }
138 }

141void synth_ui_arp_handle_encoder(long delta)
142{
143 if (delta == 0) return;
144 if (s_arp_editing) {
145 arp_edit_value(s_arp_cursor, (int)delta);
146 } else {
147 int c = (int)s_arp_cursor + (int)delta;
148 c = SEQ_CLAMP_INT(c, 0, ARP_CUR_COUNT - 1);
149 s_arp_cursor = (uint8_t)c;
150 }
151 s_force_redraw = true;
152}
