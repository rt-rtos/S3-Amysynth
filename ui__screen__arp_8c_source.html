26/* Build the flat arp view from arp_core for the renderer. */
27void arp_build_view(arp_view_t *out)
28{
29 static const char *s_dir_names[ARP_DIR_COUNT] = { "UP", "DOWN", "SLOT" };
30 out->enabled = arp_get_enabled();
31 out->mode_str = s_dir_names[arp_get_direction()];
32 out->octaves = arp_get_octaves();
33 out->rate_str = arp_rate_name(arp_get_rate());
34 out->gate_pct = arp_get_gate_pct();
35 for (uint8_t i = 0; i < ARP_VIEW_SLOTS; i++) {
36 int16_t raw = arp_get_slot(i);
37 int16_t snapped = arp_get_slot_snapped(i);
38 out->slot_rest[i] = (raw == ARP_REST);
39 if (snapped >= 0) {
40 out->slot_active[i] = true;
41 ui_note_name((uint8_t)snapped, out->slot_name[i]);
42 } else {
43 out->slot_active[i] = false;
44 out->slot_name[i][0] = '\0';
45 }
46 }
47 out->cursor = s_arp_cursor;
48 out->editing = s_arp_editing;
49
50 /* Source / wave (F-UI). */
51 bool wave_mode = (arp_get_source() == ARP_SRC_WAVE);
52 out->wave_mode = wave_mode;
53 out->source_str = wave_mode ? "WAVE" : "PTCH";
54 out->wave_str = drone_wave_name(arp_get_wave());
55 out->portamento_ms = arp_get_portamento_ms();
56
57 /* Patch indicator: mirror the sequencer view. Number is always available;
58 * the name banner shows only while the patch hold+turn gesture is active. */
59 out->patch = arp_get_patch();
60 out->patch_select = seq_state.patch_select_mode;
61 out->patch_name = patch_name_for(out->patch);
62}

64/* Signature of the arp screen. Builds the view into *out so the caller can
65 * draw from it without a second build. */
66uint32_t arp_view_signature(arp_view_t *out)
67{
68 uint32_t h = FNV1A_OFFSET;

73 h = fnv1a_bytes(h, &out->cursor, sizeof(out->cursor));
74 h = fnv1a_bytes(h, &out->editing, sizeof(out->editing));
75 h = fnv1a_bytes(h, &out->patch, sizeof(out->patch));
76 h = fnv1a_bytes(h, &out->patch_select, sizeof(out->patch_select));
77 h = fnv1a_bytes(h, &out->wave_mode, sizeof(out->wave_mode));
78 h = fnv1a_bytes(h, &out->portamento_ms, sizeof(out->portamento_ms));
79 h = fnv1a_bytes(h, out->rate_str, 4);
80 h = fnv1a_bytes(h, out->mode_str, 4);
81 if (out->source_str) h = fnv1a_bytes(h, out->source_str, 4);
82 if (out->wave_str) h = fnv1a_bytes(h, out->wave_str, 4);
83 for (uint8_t i = 0; i < ARP_VIEW_SLOTS; i++) {
84 h = fnv1a_bytes(h, &out->slot_active[i], sizeof(out->slot_active[i]));
85 h = fnv1a_bytes(h, &out->slot_rest[i], sizeof(out->slot_rest[i]));
86 h = fnv1a_bytes(h, out->slot_name[i], sizeof(out->slot_name[i]));
87 }
88 return h;
89}

91static void arp_edit_value(uint8_t cursor, int delta)
92{
93 int dir = (delta > 0) ? 1 : (delta < 0 ? -1 : 0);
94 switch (cursor) {
95 case ARP_CUR_ENABLE:
96 if (dir != 0) arp_set_enabled(!arp_get_enabled());
97 break;
98 case ARP_CUR_MODE:
99 if (dir != 0) {
100 int nd = ((int)arp_get_direction() + dir + ARP_DIR_COUNT) % ARP_DIR_COUNT;
101 arp_set_direction((arp_dir_t)nd);
102 }
103 break;
104 case ARP_CUR_OCT:
105 arp_set_octaves((uint8_t)SEQ_CLAMP_INT(
106 (int)arp_get_octaves() + dir, 1, ARP_OCT_MAX));
107 break;
108 case ARP_CUR_RATE: {
109 int r = (int)arp_get_rate() + dir;
110 r = SEQ_CLAMP_INT(r, 0, ARP_RATE_COUNT - 1);
111 arp_set_rate((arp_rate_t)r);
112 break;
113 }
114 case ARP_CUR_GATE:
115 arp_set_gate_pct((uint8_t)SEQ_CLAMP_INT(
116 (int)arp_get_gate_pct() + dir * 5, 10, 100));
117 break;
118 case ARP_CUR_SOURCE:
119 if (dir != 0)
120 arp_set_source(arp_get_source() == ARP_SRC_WAVE
121 ? ARP_SRC_PATCH : ARP_SRC_WAVE);
122 break;
123 case ARP_CUR_WAVE: {
124 /* Arp keeps NOISE and KS; drone excludes them (see DROW_WAVE). */
125 static const uint16_t s_arp_waves[] = {
126 SAW_DOWN, SAW_UP, PULSE, TRIANGLE, SINE, NOISE, KS
127 };
128 const int wn = (int)(sizeof(s_arp_waves) / sizeof(s_arp_waves[0]));
129 int idx = 0;
130 uint16_t cur_wave = arp_get_wave();
131 for (int i = 0; i < wn; i++) {
132 if (s_arp_waves[i] == cur_wave) { idx = i; break; }
133 }
134 idx = (idx + dir + wn) % wn;
135 arp_set_wave(s_arp_waves[idx]);
136 break;
137 }
138 case ARP_CUR_PORTA:
139 /* 1ms/detent over a 0..100ms range: fine control for short, snappy
140 * glides (100 turns edge-to-edge) without a drag UI for one scalar. */
141 arp_set_portamento_ms((uint16_t)SEQ_CLAMP_INT(
142 (int)arp_get_portamento_ms() + dir * 1, 0, ARP_PORTAMENTO_MAX_MS));
143 break;
144 default: {
145 /* Slot edit: chromatic note, or clear below the floor. */
146 uint8_t slot = (uint8_t)(cursor - ARP_CUR_SLOT0);
147 if (slot >= ARP_VIEW_SLOTS) break;
148 int16_t cur = arp_get_slot(slot);
149 if (cur == -1) {
150 /* Unused: up seeds a note, down sets REST (one step above floor). */
151 if (dir > 0) arp_set_slot(slot, (int16_t)arp_get_root_note());
152 else arp_set_slot(slot, ARP_REST);
153 } else if (cur == ARP_REST) {
154 /* REST is the floor: up goes to empty (not root note, so recovery
155 * is two turns up rather than scrolling through octaves); down clamps. */
156 if (dir > 0) arp_set_slot(slot, -1);
157 /* dir < 0: stay at REST — no bounce back to empty */
158 } else {
159 int nv = (int)cur + dir;
160 if (nv < 24) {
161 arp_set_slot(slot, -1); /* below floor → clear to empty */
162 } else if (nv > 127) {
163 arp_set_slot(slot, 127); /* ceiling clamp, no wrap */
164 } else {
165 arp_set_slot(slot, (int16_t)nv);
166 }
167 }
168 break;
169 }
170 }

173void synth_ui_arp_handle_encoder(long delta)
174{
175 if (delta == 0) return;
176 if (s_arp_editing) {
177 arp_edit_value(s_arp_cursor, (int)delta);
178 } else {
179 int c = (int)s_arp_cursor + (int)delta;
180 c = SEQ_CLAMP_INT(c, 0, ARP_CUR_COUNT - 1);
181 /* Skip the WAVE cursor when source is PATCH — it has no effect there. */
182 if (c == ARP_CUR_WAVE && arp_get_source() == ARP_SRC_PATCH) {
183 c = (delta > 0) ? ARP_CUR_PORTA : ARP_CUR_SOURCE;
184 }
185 s_arp_cursor = (uint8_t)c;
186 }
187 s_force_redraw = true;
188}
