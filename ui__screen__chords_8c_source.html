10/* ════════════════════════════════════════════════════════════════════════
11 * CHORDS PAGE (Main Menu -> "Chords")
12 * ════════════════════════════════════════════════════════════════════════
13 * Item model for the chord-preset editor; page state and input routing live
14 * in ui_screen_menu.c (same split as the FX/NoteFX/Projects pages).
15 *
16 * Two levels inside one page: the slot list (CH1..CH8 + Back) and, after
17 * clicking a slot, the edit view (Root + Type + Clear + Back). Every edit
18 * commits immediately through seq_chords_set(), so the engine sweep (re-emit,
19 * voice reconfig, delete fallback) rides each commit and closing the menu
20 * mid-edit can never lose a voicing.
21 *
22 * Authoring is root + chord type, the same model the drone and progression
23 * screens use, so one mental model covers every place chords are chosen.
24 * STORAGE IS UNCHANGED: seq_chord_t still holds absolute MIDI pitches, which
25 * the engine expands and transposes exactly as before - root/type are an
26 * authoring surface, generated into pitches on every edit. That is why this
27 * needed no snapshot version change.
28 *
29 * The previous editor let each note position be set freely. If that is ever
30 * wanted back, it is in git before this commit; nothing it relied on
31 * (normalization, the engine sweep, audition) lives in this file, so it is a
32 * self-contained restore rather than a rewrite.
33 *
34 * Changes audition through the selected melodic track's actual patch; falls
35 * back to the first melodic track on a drum layer, silent with none. */

55/* ── Chord type helpers ──────────────────────────────────────────────────
56 * The picker offers only types within SEQ_CHORD_MAX_SELECT tones (the 9ths sit
57 * above it - see seq_chords.h for the voice-budget rationale). The fit check
58 * reads the shared interval table, so the ceiling drops the wide chords from
59 * the list instead of silently storing a truncated voicing under their name. */
60
61static uint8_t chord_type_tones(chord_type_t t)
62{
63 const int8_t *row = quantizer_chord_intervals(t);
64 if (!row) return 0;
65 uint8_t n = 0;
66 while (n < 6 && row[n] >= 0) n++;
67 return n;
68}

70static bool chord_type_fits(chord_type_t t)
71{
72 uint8_t n = chord_type_tones(t);
73 uint8_t cap = (SEQ_CHORD_MAX_SELECT < SEQ_CHORD_MAX_NOTES)
74 ? SEQ_CHORD_MAX_SELECT : SEQ_CHORD_MAX_NOTES;
75 return n >= 2 && n <= cap;
76}

78/* Highest interval of a type, so the root can be bounded to keep the whole
79 * chord inside the melodic range instead of letting storage clamp the top
80 * tones together into a collapsed voicing. */
81static uint8_t chord_type_span(chord_type_t t)
82{
83 const int8_t *row = quantizer_chord_intervals(t);
84 if (!row) return 0;
85 uint8_t span = 0;
86 for (uint8_t i = 0; i < 6 && row[i] >= 0; i++) span = (uint8_t)row[i];
87 return span;
88}

90static uint8_t chord_root_max(chord_type_t t)
91{
92 int hi = (int)SEQ_MEL_NOTE_MAX - (int)chord_type_span(t);
93 return (hi < SEQ_MEL_NOTE_MIN) ? SEQ_MEL_NOTE_MIN : (uint8_t)hi;
94}

96/* Walk to the next fitting type, wrapping within the fitting set. */
97static chord_type_t chord_type_step(chord_type_t cur, int dir)
98{
99 int n = (int)cur;
100 for (int guard = 0; guard < CHORD_REAL_COUNT; guard++) {
101 n = (n + dir + CHORD_REAL_COUNT) % CHORD_REAL_COUNT;
102 if (chord_type_fits((chord_type_t)n)) return (chord_type_t)n;
103 }
104 return cur;
105}

107/* Generate the stored voicing for root + type. */
108static void chord_build(uint8_t root, chord_type_t type, seq_chord_t *out)
109{
110 memset(out, 0, sizeof(*out));
111 const int8_t *row = quantizer_chord_intervals(type);
112 if (!row) return;
113 for (uint8_t i = 0; i < SEQ_CHORD_MAX_NOTES && row[i] >= 0; i++) {
114 int n = (int)root + (int)row[i];
115 if (n < SEQ_MEL_NOTE_MIN) n = SEQ_MEL_NOTE_MIN;
116 if (n > SEQ_MEL_NOTE_MAX) n = SEQ_MEL_NOTE_MAX;
117 out->notes[out->count++] = (uint8_t)n;
118 }
119}

121/* Recover root + type from a stored voicing, so reopening a slot shows what was
122 * authored. Exact match against the selectable types only: a voicing wider than
123 * the picker's ceiling (e.g. a 9th saved while 5 tones were admitted) misses on
124 * purpose and gets rewritten to a fitting chord on its first edit. */
125static bool chord_match(const seq_chord_t *c, uint8_t *root, chord_type_t *type)
126{
127 if (!c || c->count < 2) return false;
128 for (uint8_t t = 0; t < CHORD_REAL_COUNT; t++) {
129 if (!chord_type_fits((chord_type_t)t)) continue;
130 const int8_t *row = quantizer_chord_intervals((chord_type_t)t);
131 if (chord_type_tones((chord_type_t)t) != c->count) continue;
132 bool ok = true;
133 for (uint8_t i = 0; i < c->count; i++) {
134 if ((int)c->notes[i] - (int)c->notes[0] != (int)row[i]) { ok = false; break; }
135 }
136 if (ok) { *root = c->notes[0]; *type = (chord_type_t)t; return true; }
137 }
138 return false;
139}

141/* Audition target: the selected track when its layer is melodic, else the
142 * first melodic layer's first track. Returns false with no melodic layer. */
143static bool chords_audition_target(uint8_t *li, uint8_t *track)
144{
145 uint8_t a = seq_state.active_layer_idx;
146 if (a < seq_state.num_layers &&
147 seq_state.layers[a].type == SEQ_LAYER_MELODIC) {
148 *li = a;
149 *track = seq_state.selected_track;
150 return true;
151 }
152 for (uint8_t i = 0; i < seq_state.num_layers; i++) {
153 if (seq_state.layers[i].type == SEQ_LAYER_MELODIC) {
154 *li = i;
155 *track = 0;
156 return true;
157 }
158 }
159 return false;
160}

250/* Returns the new menu_editing state (mirrors projects_menu_handle_click). */
251bool chords_menu_handle_click(uint8_t idx)
252{
253 if (s_mode == CHORDS_MODE_LIST) {
254 if (idx < SEQ_CHORD_SLOTS) {
255 s_slot = idx;
256 /* Zeroed up front: the audition below reads c.count on every path,
257 * including the one where the getter never filled it. */
258 seq_chord_t c = { 0 };
259 if (seq_chords_get(s_slot, &c) && c.count > 0) {
260 /* Generated voicings always match; a miss can only come from a
261 * malformed slot, so fall back to its lowest tone as root. */
262 if (!chord_match(&c, &s_root, &s_type)) s_root = c.notes[0];
263 } else {
264 /* Fresh slot: a middle-register major, audible immediately. */
265 s_root = 60;
266 s_type = CHORD_MAJ;
267 }
268 if (!chord_type_fits(s_type)) s_type = chord_type_step(s_type, 1);
269 s_mode = CHORDS_MODE_EDIT;
270 seq_state.menu_cursor = 0;
271 if (c.count > 0) chords_audition(&c); /* hear it on open */
272 }
273 return false;
274 }
275
276 if (idx == CHORDS_EDIT_ROOT || idx == CHORDS_EDIT_TYPE) {
277 return !seq_state.menu_editing; /* toggle value editing */
278 }
279 if (idx == CHORDS_EDIT_CLEAR) {
280 seq_chords_clear(s_slot);
281 return false;
282 }
283 /* Back: voicing is already committed per edit; just pop to the list. */
284 s_mode = CHORDS_MODE_LIST;
285 seq_state.menu_cursor = s_slot;
286 return false;
287}

289void chords_menu_edit_value(uint8_t idx, int delta)
290{
291 if (s_mode != CHORDS_MODE_EDIT || delta == 0) return;
292
293 int dir = (delta > 0) ? 1 : -1;
294 int steps = (delta > 0) ? delta : -delta;
295
296 if (idx == CHORDS_EDIT_ROOT) {
297 int n = (int)s_root + dir * steps;
298 int hi = (int)chord_root_max(s_type);
299 if (n < SEQ_MEL_NOTE_MIN) n = SEQ_MEL_NOTE_MIN;
300 if (n > hi) n = hi;
301 s_root = (uint8_t)n;
302 } else if (idx == CHORDS_EDIT_TYPE) {
303 for (int k = 0; k < steps; k++) s_type = chord_type_step(s_type, dir);
304 /* A wider chord can push the top tone out of range - pull the root
305 * down rather than let the voicing collapse against the ceiling. */
306 uint8_t hi = chord_root_max(s_type);
307 if (s_root > hi) s_root = hi;
308 } else {
309 return;
310 }
311
312 chords_commit();
313}
