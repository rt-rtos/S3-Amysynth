15/* ════════════════════════════════════════════════════════════════════════
16 * DEV menu — quarantine zone for temporary controls, diagnostics, readouts
17 * ════════════════════════════════════════════════════════════════════════
18 * Deliberately free of the UX constraints real screens carry: the ONLY design
19 * goals are cheap authoring and trivial navigation (turn = move, click = act,
20 * "< back" row = up). Controls graduate to real screens if they prove out.
21 *
22 * Adding a control = one table row + (usually) a fmt/adjust pair:
23 * readout { .label="X", .fmt=... } click: nothing
24 * toggle { .label="X", .fmt=..., .fire=... } click: fire
25 * value { .label="X", .fmt=..., .adjust=... } click: edit mode,
26 * turn adjusts
27 * action { .label="X", .fire=... } click: fire
28 * submenu { .label="X", .sub=&page } click: enter
29 * (+ .fmt previews state on the row: "VALUE >")
30 * `arg` is passed to every callback so one handler serves N rows.
31 * Adding a submenu = one dev_item_t[] + one dev_page_t + one submenu row.
32 *
33 * DEV state is VOLATILE by design - no snapshot fields, no persistence.
34 * (Controls whose backing state persists, like the PCM mode, own that in
35 * their real module.) Everything here runs on the UI task. */
36
37typedef struct dev_page dev_page_t;
38
39typedef struct {
40 const char *label;
41 void (*fmt)(char *buf, size_t n, int arg); /* value text; NULL = none */
42 void (*adjust)(int delta, int arg); /* encoder while editing */
43 void (*fire)(int arg); /* one-shot on click */
44 const dev_page_t *sub; /* submenu target */
45 int arg;
46} dev_item_t;
47
48struct dev_page {
49 const char *title;
50 const dev_item_t *items;
51 uint8_t count;
52};

76/* Melodic unison prototype (sequencer_core_set_unison): pick a melodic
77 * layer (L2-L4), then edit its spec. The field id rides `arg`; LAYER is
78 * dev-local navigation state, the spec itself lives in the core. COUNT
79 * changes rebuild the layer's voices (notes stop); the rest push live. */
80enum { UNI_LAYER, UNI_COUNT, UNI_DETUNE, UNI_SPREAD, UNI_BLEND };
81
82static uint8_t s_uni_layer = 1; /* layer INDEX 1..3 = UI L2..L4 */
83
84static void uni_fmt(char *buf, size_t n, int arg)
85{
86 voice_unison_t u = sequencer_core_get_unison(s_uni_layer);
87 switch (arg) {
88 case UNI_LAYER: snprintf(buf, n, "L%u", (unsigned)s_uni_layer + 1u); break;
89 case UNI_COUNT: snprintf(buf, n, "%u", (unsigned)u.count); break;
90 case UNI_DETUNE: snprintf(buf, n, "%uc", (unsigned)u.detune_cents); break;
91 case UNI_SPREAD: snprintf(buf, n, "%u%%", (unsigned)u.spread_pct); break;
92 case UNI_BLEND: snprintf(buf, n, "%u%%", (unsigned)u.blend_pct); break;
93 default: buf[0] = '\0'; break;
94 }
95}
96
97static void uni_adjust(int delta, int arg)
98{
99 if (arg == UNI_LAYER) {
100 int l = (int)s_uni_layer - 1 + delta; /* wrap over indices 1..3 */
101 l %= 3;
102 if (l < 0) l += 3;
103 s_uni_layer = (uint8_t)(l + 1);
104 return;
105 }
106 voice_unison_t u = sequencer_core_get_unison(s_uni_layer);
107 switch (arg) {
108 case UNI_COUNT:
109 u.count = (uint8_t)SEQ_CLAMP_INT((int)u.count + delta,
110 1, (int)VOICE_UNISON_MAX_COPIES);
111 break;
112 case UNI_DETUNE:
113 u.detune_cents = (uint8_t)SEQ_CLAMP_INT((int)u.detune_cents + delta,
114 0, (int)VOICE_UNISON_MAX_DETUNE);
115 break;
116 case UNI_SPREAD:
117 u.spread_pct = (uint8_t)SEQ_CLAMP_INT((int)u.spread_pct + delta, 0, 100);
118 break;
119 case UNI_BLEND:
120 u.blend_pct = (uint8_t)SEQ_CLAMP_INT((int)u.blend_pct + delta, 0, 100);
121 break;
122 default:
123 return;
124 }
125 sequencer_core_set_unison(s_uni_layer, &u);
126}

161/* Heap+CPU status bar: while enabled, replaces the bottom hint strip on EVERY
162 * screen (including ones that normally hide the hint) so internal-heap and
163 * CPU headroom can be watched live under real load. Sampled from the UI
164 * task's pre-gate service pass at HEAPBAR_PERIOD frames (20 Hz loop ->
165 * 500 ms); the sampled text feeds the render gate so the bar refreshes on
166 * change. Dev-facing abbreviations (deliberately terse, not end-user text):
167 * TF = total free internal heap (MALLOC_CAP_INTERNAL|8BIT), K
168 * LB = largest free block of the same heap, K
169 * CPU = core0/core1 busy %, from the status-LED sampler via
170 * core_load_last() (1 s cadence; omitted until first publish) */
171#define HEAPBAR_PERIOD 10

197void synth_ui_dev_heapbar_poll(void)
198{
199 if (!s_heapbar_on) return;
200 if (s_heapbar_tick == 0) {
201 size_t f = heap_caps_get_free_size(MALLOC_CAP_INTERNAL | MALLOC_CAP_8BIT);
202 size_t b = heap_caps_get_largest_free_block(MALLOC_CAP_INTERNAL |
203 MALLOC_CAP_8BIT);
204 uint8_t busy[CORE_LOAD_NUM_CORES];
205 char cpu[12];
206 if (core_load_last(busy))
207 snprintf(cpu, sizeof cpu, "%u/%u",
208 (unsigned)busy[0], (unsigned)busy[1]);
209 else
210 snprintf(cpu, sizeof cpu, "--");
211 snprintf(s_heapbar_text, sizeof s_heapbar_text,
212 "TF:%u.%uK LB:%u.%uK CPU:%s",
213 (unsigned)(f / 1024u), (unsigned)((f % 1024u) * 10u / 1024u),
214 (unsigned)(b / 1024u), (unsigned)((b % 1024u) * 10u / 1024u),
215 cpu);
216 /* Hash the text, not the raw sizes: sub-0.1K jitter then never forces
217 * a physically identical redraw through the gate. */
218 s_heapbar_sig = fnv1a_bytes(FNV1A_OFFSET, s_heapbar_text,
219 strlen(s_heapbar_text));
220 }
221 s_heapbar_tick = (uint8_t)((s_heapbar_tick + 1) % HEAPBAR_PERIOD);
222}

352bool synth_ui_dev_handle_encoder(int delta)
353{
354 if (!synth_ui_dev_is_active()) return false;
355 dev_lvl_t *lv = &s_stack[s_depth];
356 if (s_editing && lv->cur > 0) {
357 const dev_item_t *it = &lv->pg->items[lv->cur - 1];
358 if (it->adjust) it->adjust(delta, it->arg);
359 } else {
360 int n = (int)dev_row_count(lv->pg);
361 int c = (int)lv->cur + delta;
362 c %= n;
363 if (c < 0) c += n;
364 lv->cur = (uint8_t)c;
365 }
366 s_force_redraw = true;
367 return true;
368}
369
370bool synth_ui_dev_handle_button(void)
371{
372 if (!synth_ui_dev_is_active()) return false;
373 dev_lvl_t *lv = &s_stack[s_depth];
374 if (s_editing) {
375 s_editing = false;
376 } else if (lv->cur == 0) {
377 /* "< back": up one page, or from the root back to the main menu. */
378 if (s_depth > 0) s_depth--;
379 else seq_state.menu_open = true;
380 } else {
381 const dev_item_t *it = &lv->pg->items[lv->cur - 1];
382 if (it->sub && s_depth + 1 < DEV_STACK_MAX) {
383 s_depth++;
384 s_stack[s_depth] = (dev_lvl_t){ it->sub, 0 };
385 } else if (it->fire) {
386 it->fire(it->arg);
387 } else if (it->adjust) {
388 s_editing = true;
389 }
390 /* plain readout: click is a no-op */
391 }
392 s_force_redraw = true;
393 return true;
394}

396/* ── View build + render-gate signature ────────────────────────────────── */
397
398uint32_t dev_view_signature(dev_view_t *out)
399{
400 const dev_lvl_t *lv = &s_stack[s_depth];
401 const dev_page_t *pg = lv->pg;
402
403 /* Zero first: rows are hashed full-width below, so bytes past each NUL
404 * must be deterministic or the render gate never settles. */
405 memset(out, 0, sizeof *out);
406 out->title = pg->title;
407 out->cursor = lv->cur;
408 out->editing = s_editing;
409
410 uint8_t n = dev_row_count(pg);
411 if (n > DEV_VIEW_MAX_ROWS) n = DEV_VIEW_MAX_ROWS;
412 out->count = n;
413
414 snprintf(out->rows[0].label, DEV_LABEL_LEN, "%s",
415 s_depth > 0 ? "< back" : "< menu");
416 out->rows[0].value[0] = '\0';
417
418 for (uint8_t i = 1; i < n; i++) {
419 const dev_item_t *it = &pg->items[i - 1];
420 snprintf(out->rows[i].label, DEV_LABEL_LEN, "%s", it->label);
421 if (it->sub && it->fmt) {
422 /* Submenu that also previews state: "<value> >". */
423 char v[DEV_VALUE_LEN];
424 it->fmt(v, sizeof v, it->arg);
425 snprintf(out->rows[i].value, DEV_VALUE_LEN, "%.*s >",
426 DEV_VALUE_LEN - 3, v);
427 }
428 else if (it->sub) snprintf(out->rows[i].value, DEV_VALUE_LEN, ">");
429 else if (it->fmt) it->fmt(out->rows[i].value, DEV_VALUE_LEN, it->arg);
430 else out->rows[i].value[0] = '\0';
431 }
432
433 uint32_t h = FNV1A_OFFSET;
434 h = fnv1a_bytes(h, &s_depth, sizeof s_depth);
435 h = fnv1a_bytes(h, &out->cursor, sizeof out->cursor);
436 h = fnv1a_bytes(h, &out->editing, sizeof out->editing);
437 h = fnv1a_bytes(h, &out->count, sizeof out->count);
438 for (uint8_t i = 0; i < n; i++) {
439 h = fnv1a_bytes(h, out->rows[i].label, sizeof out->rows[i].label);
440 h = fnv1a_bytes(h, out->rows[i].value, sizeof out->rows[i].value);
441 }
442 return h;
443}
