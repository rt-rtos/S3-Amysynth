46/* Fill `out` with the logical rows visible for the current source; return count.
47 * out[] must hold at least DROW_ALL_COUNT entries. */
48static uint8_t drone_visible_rows(drone_logical_row_t out[DROW_ALL_COUNT])
49{
50 bool wave = (drone_get_source() == DRONE_SRC_WAVE);
51 uint8_t n = 0;
52 for (drone_logical_row_t r = 0; r < DROW_ALL_COUNT; r++) {
53 if (wave && r == DROW_PATCH) continue;
54 if (!wave && (r == DROW_WAVE || r == DROW_CONST || r == DROW_MOD
55 || r == DROW_RATE || r == DROW_GATE_LEN)) continue;
56 out[n++] = r;
57 }
58 return n;
59}

159void drone_build_view(drone_view_t *out)
160{
161 drone_logical_row_t vis[DROW_ALL_COUNT];
162 uint8_t n = drone_visible_rows(vis);
163 for (uint8_t i = 0; i < n; i++) {
164 drone_row_label_value(vis[i], s_drone_rows[i].label, s_drone_rows[i].value);
165 }
166 if (s_drone_cursor >= n) s_drone_cursor = (n ? (uint8_t)(n - 1) : 0);
167 out->rows = s_drone_rows;
168 out->count = n;
169 out->cursor = s_drone_cursor;
170 out->editing = s_drone_editing;
171}

173static void drone_edit_row(drone_logical_row_t r, int delta)
174{
175 int dir = (delta > 0) ? 1 : (delta < 0 ? -1 : 0);
176 switch (r) {
177 case DROW_ENABLE:
178 if (dir != 0) drone_set_enabled(!drone_get_enabled());
179 break;
180 case DROW_SOURCE:
181 if (dir != 0)
182 drone_set_source(drone_get_source() == DRONE_SRC_WAVE
183 ? DRONE_SRC_PATCH : DRONE_SRC_WAVE);
184 break;
185 case DROW_WAVE: {
186 /* NOISE and KS are excluded: their excitation model misbehaves
187 * with the drone's one-shot trigger style. */
188 static const uint16_t waves[] = { SAW_DOWN, SAW_UP, PULSE, TRIANGLE, SINE };
189 const int wn = (int)(sizeof(waves) / sizeof(waves[0]));
190 int idx = 0;
191 for (int i = 0; i < wn; i++) if (waves[i] == drone_get_wave()) { idx = i; break; }
192 idx = (idx + dir + wn) % wn;
193 drone_set_wave(waves[idx]);
194 break;
195 }
196 case DROW_ROOT: {
197 int rv = SEQ_CLAMP_INT((int)drone_get_root_note() + dir, 24, 72);
198 drone_set_root_note((uint8_t)rv);
199 break;
200 }
201 case DROW_CHORD: {
202 /* Wraps, unlike the clamped rows around it: CHORD_OFF is the
203 * default and sits last, so wrapping keeps it one detent from the
204 * first chord instead of stranding it at the end. */
205 int c = ((int)drone_get_chord() + dir + CHORD_TYPE_COUNT) % CHORD_TYPE_COUNT;
206 drone_set_chord((chord_type_t)c);
207 break;
208 }
209 case DROW_RES: {
210 /* Geometric step: Q is perceptually multiplicative, so a linear
211 * step is too coarse low and too fine high. ~15% per detent feels
212 * even across the whole span. */
213 float rv = drone_get_resonance();
214 rv *= (dir > 0) ? 1.15f : (1.0f / 1.15f);
215 drone_set_resonance(rv);
216 break;
217 }
218 case DROW_CONST:
219 drone_set_amp_peak(drone_get_amp_peak() + (float)dir * 0.1f);
220 break;
221 case DROW_MOD:
222 drone_set_amp_duck(drone_get_amp_duck() + (float)dir * 0.1f);
223 break;
224 case DROW_VIS_POPUP:
225 break; /* encoder turns consumed; popup opened/closed via button */
226 case DROW_RATE: {
227 int v = SEQ_CLAMP((int)drone_get_rate() + dir, 0, DRONE_RATE_COUNT - 1);
228 drone_set_rate((drone_rate_t)v);
229 break;
230 }
231 case DROW_GATE_LEN:
232 drone_set_gate_len(drone_get_gate_len() + (float)dir * 0.05f);
233 break;
234 case DROW_SWING:
235 drone_set_swing((uint8_t)SEQ_CLAMP(
236 (int)drone_get_swing() + dir * 2, 0, 66));
237 break;
238 case DROW_PATTERN: {
239 int v = SEQ_CLAMP((int)drone_get_pattern() + dir, 0, DRONE_PAT_COUNT - 1);
240 drone_set_pattern((drone_pattern_t)v);
241 break;
242 }
243 case DROW_BLIP:
244 drone_set_blip(drone_get_blip() + (float)dir * 0.05f);
245 break;
246 case DROW_SWEEP_LO:
247 drone_set_sweep_lo(drone_get_sweep_lo() + (float)dir * 25.0f);
248 break;
249 case DROW_SWEEP_HI:
250 drone_set_sweep_hi(drone_get_sweep_hi() + (float)dir * 25.0f);
251 break;
252 case DROW_SWEEP_BARS:
253 drone_set_sweep_bars((uint8_t)SEQ_CLAMP(
254 (int)drone_get_sweep_bars() + dir, 1, 16));
255 break;
256 case DROW_SUB:
257 if (dir != 0) drone_set_sub_enabled(!drone_get_sub_enabled());
258 break;
259 case DROW_SUB_INTVL:
260 drone_set_sub_interval((int8_t)SEQ_CLAMP(
261 (int)drone_get_sub_interval() + dir, -36, 0));
262 break;
263 case DROW_PATCH:
264 synth_ui_drone_cycle_patch(dir);
265 break;
266 default:
267 break;
268 }
269}

279void synth_ui_drone_handle_encoder(long delta)
280{
281 if (delta == 0) return;
282 if (s_drone_vis_open) {
283 s_force_redraw = true; /* consume turn; popup has nothing to scroll */
284 return;
285 }
286 drone_logical_row_t vis[DROW_ALL_COUNT];
287 uint8_t n = drone_visible_rows(vis);
288 if (n == 0) return;
289 if (s_drone_cursor >= n) s_drone_cursor = (uint8_t)(n - 1);
290
291 if (s_drone_editing) {
292 drone_edit_row(vis[s_drone_cursor], (int)delta);
293 } else {
294 int c = (int)s_drone_cursor + (int)delta;
295 c = SEQ_CLAMP_INT(c, 0, (int)n - 1);
296 s_drone_cursor = (uint8_t)c;
297 }
298 s_force_redraw = true;
299}
300
301void synth_ui_drone_handle_button(void)
302{
303 if (s_drone_vis_open) {
304 s_drone_vis_open = false;
305 s_force_redraw = true;
306 return;
307 }
308 drone_logical_row_t vis[DROW_ALL_COUNT];
309 uint8_t n = drone_visible_rows(vis);
310 if (n > 0 && s_drone_cursor < n && vis[s_drone_cursor] == DROW_VIS_POPUP) {
311 s_drone_vis_open = true;
312 s_force_redraw = true;
313 return;
314 }
315 s_drone_editing = !s_drone_editing;
316 s_force_redraw = true;
317}

319/* Signature of the drone screen. Builds the view into *out so the caller draws
320 * from it without a second build. */
321uint32_t drone_view_signature(drone_view_t *out)
322{
323 uint32_t h = FNV1A_OFFSET;
324 drone_build_view(out);
325 h = fnv1a_bytes(h, &out->cursor, sizeof(out->cursor));
326 h = fnv1a_bytes(h, &out->editing, sizeof(out->editing));
327 h = fnv1a_bytes(h, &out->count, sizeof(out->count));
328 for (uint8_t i = 0; i < out->count; i++) {
329 h = fnv1a_bytes(h, out->rows[i].label, sizeof(out->rows[i].label));
330 h = fnv1a_bytes(h, out->rows[i].value, sizeof(out->rows[i].value));
331 }
332 return h;
