36static uint8_t drone_std_visible_rows(drone_std_row_t out[DSROW_ALL_COUNT])
37{
38 bool wave = (drone_std_get_source() == DRONE_SRC_WAVE);
39 uint8_t n = 0;
40 for (drone_std_row_t r = 0; r < DSROW_ALL_COUNT; r++) {
41 if (wave && r == DSROW_PATCH) continue;
42 if (!wave && r == DSROW_WAVE) continue;
43 out[n++] = r;
44 }
45 return n;

114void drone_std_build_view(drone_view_t *out)
115{
116 drone_std_row_t vis[DSROW_ALL_COUNT];
117 uint8_t n = drone_std_visible_rows(vis);
118 for (uint8_t i = 0; i < n; i++) {
119 drone_std_row_label_value(vis[i], s_dstd_rows[i].label, s_dstd_rows[i].value);
120 }
121 if (s_dstd_cursor >= n) s_dstd_cursor = (n ? (uint8_t)(n - 1) : 0);
122 out->rows = s_dstd_rows;
123 out->count = n;
124 out->cursor = s_dstd_cursor;
125 out->editing = s_dstd_editing;
126}

128static void drone_std_edit_row(drone_std_row_t r, int delta)
129{
130 int dir = (delta > 0) ? 1 : (delta < 0 ? -1 : 0);
131 switch (r) {
132 case DSROW_ENABLE:
133 if (dir != 0) drone_std_set_enabled(!drone_std_get_enabled());
134 break;
135 case DSROW_SOURCE:
136 if (dir != 0)
137 drone_std_set_source(drone_std_get_source() == DRONE_SRC_WAVE
138 ? DRONE_SRC_PATCH : DRONE_SRC_WAVE);
139 break;
140 case DSROW_WAVE: {
141 /* Same wave cycle as the stutter drone (NOISE/KS excluded). */
142 static const uint16_t waves[] = { SAW_DOWN, SAW_UP, PULSE, TRIANGLE, SINE };
143 const int wn = (int)(sizeof(waves) / sizeof(waves[0]));
144 int idx = 0;
145 for (int i = 0; i < wn; i++) {
146 if (waves[i] == drone_std_get_wave()) { idx = i; break; }
147 }
148 idx = (idx + dir + wn) % wn;
149 drone_std_set_wave(waves[idx]);
150 break;
151 }
152 case DSROW_ROOT: {
153 int rv = SEQ_CLAMP_INT((int)drone_std_get_root_note() + dir, 24, 72);
154 drone_std_set_root_note((uint8_t)rv);
155 break;
156 }
157 case DSROW_CHORD: {
158 int c = SEQ_CLAMP_INT((int)drone_std_get_chord() + dir,
159 0, CHORD_TYPE_COUNT - 1);
160 drone_std_set_chord((chord_type_t)c);
161 break;
162 }
163 case DSROW_LEVEL:
164 drone_std_set_level(drone_std_get_level() + (float)dir * 0.1f);
165 break;
166 case DSROW_SUB:
167 if (dir != 0) drone_std_set_sub_enabled(!drone_std_get_sub_enabled());
168 break;
169 case DSROW_SUB_INTVL:
170 drone_std_set_sub_interval((int8_t)SEQ_CLAMP(
171 (int)drone_std_get_sub_interval() + dir, -36, 0));
172 break;
173 case DSROW_PATCH:
174 synth_ui_drone_std_cycle_patch(dir);
175 break;
176 case DSROW_STUTTER:
177 default:
178 break; /* dive row: handled on button press, not encoder turn */
179 }
180}

189void synth_ui_drone_std_handle_encoder(long delta)
190{
191 if (delta == 0) return;
192 drone_std_row_t vis[DSROW_ALL_COUNT];
193 uint8_t n = drone_std_visible_rows(vis);
194 if (n == 0) return;
195 if (s_dstd_cursor >= n) s_dstd_cursor = (uint8_t)(n - 1);
196
197 if (s_dstd_editing) {
198 drone_std_edit_row(vis[s_dstd_cursor], (int)delta);
199 } else {
200 int c = (int)s_dstd_cursor + (int)delta;
201 c = SEQ_CLAMP_INT(c, 0, (int)n - 1);
202 s_dstd_cursor = (uint8_t)c;
203 }
204 s_force_redraw = true;
205}

207void synth_ui_drone_std_handle_button(void)
208{
209 drone_std_row_t vis[DSROW_ALL_COUNT];
210 uint8_t n = drone_std_visible_rows(vis);
211 if (n > 0 && s_dstd_cursor < n && vis[s_dstd_cursor] == DSROW_STUTTER) {

213 seq_state.ui_mode = UI_MODE_DRONE;
214 s_dstd_editing = false;
215 s_force_redraw = true;
216 return;

218 s_dstd_editing = !s_dstd_editing;

222/* Signature of the normal drone screen (everything the renderer reads). */
223uint32_t drone_std_view_signature(drone_view_t *out)
224{
225 uint32_t h = FNV1A_OFFSET;
226 drone_std_build_view(out);
227 h = fnv1a_bytes(h, &out->cursor, sizeof(out->cursor));
228 h = fnv1a_bytes(h, &out->editing, sizeof(out->editing));
229 h = fnv1a_bytes(h, &out->count, sizeof(out->count));
230 for (uint8_t i = 0; i < out->count; i++) {
231 h = fnv1a_bytes(h, out->rows[i].label, sizeof(out->rows[i].label));
232 h = fnv1a_bytes(h, out->rows[i].value, sizeof(out->rows[i].value));
233 }
234 return h;
