20/* Curated DX7-style harmonic ratios plus a few inharmonic ones, kept short
21 * enough to encoder through. Editing snaps to the nearest step, then walks. */
22static const float s_fm_ratio_steps[] = {
23 0.5f, 1.0f, 1.5f, 2.0f, 2.5f, 3.0f, 3.5f, 4.0f, 5.0f,
24 6.0f, 7.0f, 8.0f, 9.0f, 10.0f, 11.0f, 12.0f, 14.0f, 16.0f,
25};
26#define FM_RATIO_STEP_COUNT ((int)(sizeof(s_fm_ratio_steps) / sizeof(s_fm_ratio_steps[0])))

32static int fm_ratio_nearest_index(float ratio)
33{
34 int best = 0;
35 float best_d = 1e9f;
36 for (int i = 0; i < FM_RATIO_STEP_COUNT; i++) {
37 float d = ratio - s_fm_ratio_steps[i];
38 if (d < 0) d = -d;
39 if (d < best_d) { best_d = d; best = i; }
40 }
41 return best;
42}

44static float fm_ratio_step(float current, int delta)
45{
46 int idx = fm_ratio_nearest_index(current) + delta;
47 idx = SEQ_CLAMP_INT(idx, 0, FM_RATIO_STEP_COUNT - 1);
48 return s_fm_ratio_steps[idx];
49}

102/* Walk the TO row's candidates (OUT, then the other operators) from the
103 * current target in `delta`'s direction until one the compiler accepts. */
104static void fm_edit_target(uint8_t op, int delta)
105{
106 fm_graph_view_t g;
107 fm_voice_graph(&s_fm_voice, &g);
108 /* Candidate ring: index 0 = OUT, 1..6 = operator 0..5 (self skipped). */
109 int cur = 0;
110 for (uint8_t i = 0; i < FM_NUM_OPS; i++) { if (g.out_mask[op] & (1u << i)) { cur = i + 1; break; } }
111 int step = (delta > 0) ? 1 : -1;
112 int n = FM_NUM_OPS + 1;
113 int c = cur;
114 for (int tries = 0; tries < n; tries++) {
115 c = ((c + step) % n + n) % n;
116 uint8_t target = (c == 0) ? FM_TO_OUT : (uint8_t)(c - 1);
117 if (target == op) continue;
118 if (fm_voice_set_op_target(&s_fm_voice, op, target)) {
119 sequencer_core_fm_voice_changed(FM_PUSH_ROUTING);
120 return;
121 }
122 }
123}

125bool synth_ui_fm_handle_encoder(int delta)
126{
127 if (!synth_ui_fm_is_active()) return false;
128 if (delta == 0) return true;
129
130 if (s_fm_editing) {
131 uint8_t op = s_fm_selected;
132 switch (s_fm_cursor) {
133 case FM_CUR_RATIO:
134 s_fm_voice.op_ratio[op] = fm_ratio_step(s_fm_voice.op_ratio[op], delta);
135 sequencer_core_fm_voice_changed(op);
136 break;
137 case FM_CUR_LEVEL: {
138 float lvl = s_fm_voice.op_level[op] + (float)delta * 0.05f;
139 s_fm_voice.op_level[op] = SEQ_CLAMP_F32(lvl, 0.0f, 1.0f);
140 sequencer_core_fm_voice_changed(op);
141 break;
142 }
143 case FM_CUR_TO:
144 fm_edit_target(op, delta);
145 break;
146 case FM_CUR_FB: {
147 float fb = s_fm_voice.feedback + (float)delta * 0.05f;
148 s_fm_voice.feedback = SEQ_CLAMP_F32(fb, 0.0f, 1.2f);
149 sequencer_core_fm_voice_changed(FM_PUSH_ROUTING);
150 break;
151 }
152 case FM_CUR_ALGO:
153 fm_voice_step_algorithm(&s_fm_voice, delta);
154 sequencer_core_fm_voice_changed(FM_PUSH_ROUTING);
155 break;
156 default:
157 break;
158 }
159 } else {
160 int c = (int)s_fm_cursor + delta;
161 c = SEQ_CLAMP_INT(c, 0, (int)FM_CUR_COUNT - 1);
162 s_fm_cursor = (uint8_t)c;
163 if (s_fm_cursor < FM_CUR_RATIO) s_fm_selected = (uint8_t)(s_fm_cursor - FM_CUR_OP_BASE);
164 }
165 s_force_redraw = true;
166 return true;
167}
