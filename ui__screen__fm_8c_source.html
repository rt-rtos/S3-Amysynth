8/* ════════════════════════════════════════════════════════════════════════
9 * FM/ALGO voice editor — algorithm + per-operator ratio/level
10 * ════════════════════════════════════════════════════════════════════════
11 * Edits the single live SEQ_PATCH_FM_CUSTOM voice (s_fm_voice, owned by
12 * custompatches/fm_voice.c). Reuses the generic scrollable label/value list
13 * renderer from display_menu.c (display_menu_draw_frame_titled) rather than
14 * a bespoke layout — this screen's shape (a scrollable list of rows, each
15 * either navigated or entered for editing) is identical to the menu overlay.
16 *
17 * Rows: ALGO, FEEDBACK, then (RATIO, LEVEL) for each of the 6 operators.
18 * Operators are labelled by their array index (OP 0..5), which is also the
19 * AMY algo_source[] slot index — see fm_voice.h for how that maps onto a
20 * real DX7 algorithm chart's OP1..OP6 numbering (reversed order). Full
21 * per-algorithm operator-role labelling (carrier vs modulator) is deferred —
22 * see docs/handoff/a2-fm-dx7.md. */

38/* Curated ratio steps: common DX7-style integer/harmonic ratios plus a few
39 * inharmonic values, kept small enough to step through quickly with an
40 * encoder. Editing lands on the nearest step, then walks the table. */
41static const float s_fm_ratio_steps[] = {
42 0.5f, 1.0f, 1.5f, 2.0f, 2.5f, 3.0f, 3.5f, 4.0f, 5.0f,
43 6.0f, 7.0f, 8.0f, 9.0f, 10.0f, 11.0f, 12.0f, 14.0f, 16.0f,
44};
45#define FM_RATIO_STEP_COUNT ((int)(sizeof(s_fm_ratio_steps) / sizeof(s_fm_ratio_steps[0])))

50static int fm_ratio_nearest_index(float ratio)
51{
52 int best = 0;
53 float best_d = 1e9f;
54 for (int i = 0; i < FM_RATIO_STEP_COUNT; i++) {
55 float d = ratio - s_fm_ratio_steps[i];
56 if (d < 0) d = -d;
57 if (d < best_d) { best_d = d; best = i; }
58 }
59 return best;
60}

62static float fm_ratio_step(float current, int delta)
63{
64 int idx = fm_ratio_nearest_index(current) + delta;
65 idx = SEQ_CLAMP_INT(idx, 0, FM_RATIO_STEP_COUNT - 1);
66 return s_fm_ratio_steps[idx];
67}

124bool synth_ui_fm_handle_encoder(int delta)
125{
126 if (!synth_ui_fm_is_active()) return false;
127 if (delta == 0) return true;
128
129 if (s_fm_editing) {
130 fm_row_t row = (fm_row_t)s_fm_cursor;
131 switch (row) {
132 case FM_ROW_ALGO: {
133 int n = FM_ALGO_COUNT;
134 int a = (int)s_fm_voice.algorithm + delta;
135 a = ((a % n) + n) % n;
136 s_fm_voice.algorithm = (uint8_t)a;
137 break;
138 }
139 case FM_ROW_FEEDBACK: {
140 float fb = s_fm_voice.feedback + (float)delta * 0.05f;
141 s_fm_voice.feedback = SEQ_CLAMP_F32(fb, 0.0f, 1.2f);
142 break;
143 }
144 default: {
145 uint8_t op = fm_row_op_index(row);
146 if (fm_row_is_ratio(row)) {
147 s_fm_voice.op_ratio[op] = fm_ratio_step(s_fm_voice.op_ratio[op], delta);
148 } else {
149 float lvl = s_fm_voice.op_level[op] + (float)delta * 0.05f;
150 s_fm_voice.op_level[op] = SEQ_CLAMP_F32(lvl, 0.0f, 1.0f);
151 }
152 break;
153 }
154 }
155 sequencer_core_fm_voice_changed();
156 } else {
157 int c = (int)s_fm_cursor + delta;
158 c = SEQ_CLAMP_INT(c, 0, (int)FM_ROW_COUNT - 1);
159 s_fm_cursor = (uint8_t)c;
160 }
161 s_force_redraw = true;
162 return true;
163}
