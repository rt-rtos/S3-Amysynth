6/* ════════════════════════════════════════════════════════════════════════
7 * GLOBAL FX SUBMENU PAGE
8 * ════════════════════════════════════════════════════════════════════════
9 * Item model for the FX page of the menu overlay (page state and input
10 * routing live in ui_screen_menu.c; this file only builds the rows and
11 * applies encoder edits). Covers the three-band EQ, the echo/chorus/reverb
12 * levels, and the extended per-effect params behind them (fx_state_t).
13 *
14 * Extended params use the FX_PARAM_UNSET sentinel: while unset, fx_push_*
15 * never writes the matching amy_event field, so AMY keeps its factory value.
16 * The rows therefore DISPLAY the AMY default while unset, and the first
17 * encoder step seeds the field from that same default before stepping — the
18 * shown value and the audible value never disagree. */
19

141/* Step a sentinel-gated param: the first edit lifts the field from the AMY
142 * default it was displaying, then steps and pushes like any other value. */
143static void fx_step(int16_t *field, int def, int step, int lo, int hi,
144 int dir, void (*push)(void))
145{
146 int cur = (*field == FX_PARAM_UNSET) ? def : (int)*field;
147 *field = (int16_t)SEQ_CLAMP_INT(cur + dir * step, lo, hi);
148 push();
149}

151void fx_menu_edit_value(uint8_t idx, int delta)
152{
153 int dir = (delta > 0) ? 1 : (delta < 0 ? -1 : 0);
154 if (dir == 0) return;
155
156 switch ((fx_menu_item_id_t)idx) {
157 case FXI_EQ_LOW: {
158 int v = SEQ_CLAMP_INT((int)s_fx.eq_low_db + dir, -15, 15);
159 s_fx.eq_low_db = (int8_t)v; fx_push_eq();
160 break;
161 }
162 case FXI_EQ_MID: {
163 int v = SEQ_CLAMP_INT((int)s_fx.eq_mid_db + dir, -15, 15);
164 s_fx.eq_mid_db = (int8_t)v; fx_push_eq();
165 break;
166 }
167 case FXI_EQ_HIGH: {
168 int v = SEQ_CLAMP_INT((int)s_fx.eq_high_db + dir, -15, 15);
169 s_fx.eq_high_db = (int8_t)v; fx_push_eq();
170 break;
171 }
172 case FXI_ECHO_LEVEL: {
173 int v = SEQ_CLAMP_INT((int)s_fx.echo_level + dir * 5, 0, 100);
174 s_fx.echo_level = (uint8_t)v; fx_push_echo();
175 break;
176 }
177 case FXI_ECHO_FEEDBACK:
178 fx_step(&s_fx.echo_feedback, 0, 5, 0, 99, dir, fx_push_echo);
179 break;
180 case FXI_ECHO_TIME:
181 /* 743 ms = AMY's default-allocated echo delay line; longer values
182 * would be clamped inside config_echo anyway. */
183 fx_step(&s_fx.echo_delay_ms, 500, 10, 0, 743, dir, fx_push_echo);
184 break;
185 case FXI_ECHO_TONE:
186 fx_step(&s_fx.echo_tone, 0, 5, -99, 99, dir, fx_push_echo);
187 break;
188 case FXI_CHORUS_LEVEL: {
189 int v = SEQ_CLAMP_INT((int)s_fx.chorus_level + dir * 5, 0, 100);
190 s_fx.chorus_level = (uint8_t)v; fx_push_chorus();
191 break;
192 }
193 case FXI_CHORUS_RATE:
194 /* centi-Hz: 5..1000 = 0.05..10 Hz in 0.05 Hz steps. */
195 fx_step(&s_fx.chorus_rate, 50, 5, 5, 1000, dir, fx_push_chorus);
196 break;
197 case FXI_CHORUS_DEPTH:
198 fx_step(&s_fx.chorus_depth, 50, 5, 0, 100, dir, fx_push_chorus);
199 break;
200 case FXI_REVERB_LEVEL: {
201 int v = SEQ_CLAMP_INT((int)s_fx.reverb_level + dir * 5, 0, 100);
202 s_fx.reverb_level = (uint8_t)v; fx_push_reverb();
203 break;
204 }
205 case FXI_REVERB_LIVENESS:
206 fx_step(&s_fx.reverb_liveness, 85, 5, 0, 100, dir, fx_push_reverb);
207 break;
208 case FXI_REVERB_DAMPING:
209 fx_step(&s_fx.reverb_damping, 50, 5, 0, 100, dir, fx_push_reverb);
210 break;
211 case FXI_REVERB_XOVER:
212 fx_step(&s_fx.reverb_xover_hz, 3000, 250, 500, 8000, dir, fx_push_reverb);
213 break;
214 case FXI_PRESET_GLOBAL_FX:
215 s_fx.presets_alter_global = !s_fx.presets_alter_global;
216 /* Turning the guard back ON re-imposes the user's cached FX
217 * immediately, undoing whatever the last preset left behind. */
218 if (!s_fx.presets_alter_global) synth_ui_fx_reassert_global();
219 break;

221 break;
