21/* ════════════════════════════════════════════════════════════════════════
22 * MENU OVERLAY
23 * ════════════════════════════════════════════════════════════════════════
24 * A small modal list. Items are either ACTIONS (run on click, no value) or
25 * VALUE items (click to enter editing, encoder changes the value, click to
26 * exit). The model is a static table; values are read/written live from the
27 * sequencer_core quantizer + arp_core + global FX cache.
28 *
29 * The overlay has two pages: the main list below, and the global-FX page
30 * (item model in ui_screen_fxmenu.c) reached through MI_FX_MENU. This file
31 * owns the page state and routes encoder/button input to the active page;
32 * both pages share seq_state.menu_cursor/menu_editing. */

236/* Signature of the menu overlay. Builds the view into *out so the caller
237 * can draw from it without a second full snprintf pass. */
238uint32_t menu_view_signature(menu_view_t *out)
239{
240 uint32_t h = FNV1A_OFFSET;
241 menu_build_view(out);
242 h = fnv1a_bytes(h, &out->cursor, sizeof(out->cursor));
243 h = fnv1a_bytes(h, &out->editing, sizeof(out->editing));
244 for (uint8_t i = 0; i < out->count; i++) {
245 h = fnv1a_bytes(h, out->items[i].label, sizeof(out->items[i].label));
246 h = fnv1a_bytes(h, out->items[i].value, sizeof(out->items[i].value));
247 }
248 return h;
249}

270/* Apply an encoder delta to the currently-entered menu value. */
271static void menu_edit_value(menu_item_id_t id, int delta)
272{
273 int dir = (delta > 0) ? 1 : (delta < 0 ? -1 : 0);
274 switch (id) {
275 case MI_BPM:
276 synth_ui_set_bpm((uint16_t)((int)seq_get_bpm() + delta));
277 break;
278 case MI_QUANT_ENABLED:
279 if (dir != 0)
280 sequencer_core_set_quantizer_enabled(
281 !sequencer_core_get_quantizer_enabled());
282 break;
283 case MI_QUANT_SCALE: {
284 int n = (int)quantizer_scale_count();
285 int cur = (int)sequencer_core_get_quantizer_scale();
286 int ni = (cur + dir + n) % n;
287 sequencer_core_set_quantizer_scale((uint8_t)ni);
288 break;
289 }
290 case MI_QUANT_ROOT: {
291 int r = (int)sequencer_core_get_quantizer_root_note() + delta;
292 r = SEQ_CLAMP_INT(r, 0, 127);
293 sequencer_core_set_quantizer_root_note((uint8_t)r);
294 break;
295 }
296 case MI_ARP_ENABLED:
297 if (dir != 0) arp_set_enabled(!arp_get_enabled());
298 break;
299 case MI_DRONE_ENABLED:
300 if (dir != 0) drone_std_set_enabled(!drone_std_get_enabled());
301 break;
302 case MI_STUTTER_ENABLED:
303 if (dir != 0) drone_set_enabled(!drone_get_enabled());
304 break;
305 case MI_DRUM_ENGINE:
306 if (dir != 0) {
307 int n = (int)sequencer_core_drum_source_count();
308 int ni = ((int)sequencer_core_get_drum_source() + dir + n) % n;
309 sequencer_core_set_drum_source((uint8_t)ni);
310 /* A bank seed rewrote the core's per-track presets; refresh
311 * the UI mirror so the drum screen's labels follow. */
312 for (uint8_t i = 0; i < seq_state.num_layers; i++) {
313 if (seq_state.layers[i].type != SEQ_LAYER_DRUM) continue;
314 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
315 seq_state.layers[i].track_pcm_preset[t] =
316 sequencer_core_get_drum_pcm_preset(i, t);
317 }
318 }
319 }
320 break;
321 case MI_VOLUME: {
322 /* 5% steps, range 0..200% (0.0..2.0 linear). Clamping and the
323 * write to amy_global.volume[] are handled by the setter. */
324 amy_fx_set_master_volume(amy_fx_get_master_volume() + (float)dir * 0.05f);
325 break;
326 }
327 default:
328 break;
329 }

332void synth_ui_menu_toggle(void)
333{
334 /* The graph editor is the top overlay; don't let the menu fight it. */
335 if (synth_ui_graph_is_active()) return;
336 seq_state.menu_open = !seq_state.menu_open;
337 seq_state.menu_editing = false;
338 if (seq_state.menu_open) {
339 /* Always reopen on the main page so the menu lands somewhere known. */
340 s_fx_page = false;
341 s_notefx_page = false;
342#if CONFIG_SYNTH_PROJECT_STORE
343 s_projects_page = false;
344#endif
345 if (seq_state.menu_cursor >= MI_COUNT) seq_state.menu_cursor = 0;
346 }
347 s_force_redraw = true;
348 ESP_LOGI(TAG, "menu %s", seq_state.menu_open ? "open" : "closed");
349}

385bool synth_ui_menu_handle_encoder(long delta)
386{
387 if (!seq_state.menu_open) return false;
388 if (delta == 0) return true;
389
390 if (seq_state.menu_editing) {
391 if (s_notefx_page) {
392 notefx_menu_edit_value(seq_state.menu_cursor, (int)delta);
393 } else if (s_fx_page) {
394 fx_menu_edit_value(seq_state.menu_cursor, (int)delta);
395#if CONFIG_SYNTH_PROJECT_STORE
396 } else if (s_projects_page) {
397 projects_menu_edit_value(seq_state.menu_cursor, (int)delta);
398#endif
399 } else {
400 menu_edit_value((menu_item_id_t)seq_state.menu_cursor, (int)delta);
401 }
402 } else {
403 int n = s_notefx_page ? (int)notefx_menu_item_count() :
404 s_fx_page ? (int)fx_menu_item_count() :
405#if CONFIG_SYNTH_PROJECT_STORE
406 s_projects_page ? (int)projects_menu_item_count() :
407#endif
408 (int)MI_COUNT;
409 int c = (int)seq_state.menu_cursor + (int)delta;
410 /* clamp (no wrap) so the list feels bounded */
411 c = SEQ_CLAMP_INT(c, 0, n - 1);
412 seq_state.menu_cursor = (uint8_t)c;
413 }
414 s_force_redraw = true;
415 return true;
416}

418bool synth_ui_menu_handle_button(void)
419{
420 if (!seq_state.menu_open) return false;
421
422 if (s_notefx_page) {
423 uint8_t idx = seq_state.menu_cursor;
424 if (notefx_menu_item_is_value(idx)) {
425 seq_state.menu_editing = !seq_state.menu_editing;
426 } else if (notefx_menu_item_is_back(idx)) {
427 /* Back returns to the FX page it was dived from. */
428 s_notefx_page = false;
429 s_fx_page = true;
430 seq_state.menu_cursor = s_fx_cursor;
431 seq_state.menu_editing = false;
432 }
433 s_force_redraw = true;
434 return true;
435 }
436
437 if (s_fx_page) {
438 uint8_t idx = seq_state.menu_cursor;
439 if (fx_menu_item_is_notefx(idx)) {
440 /* Dive into the per-layer NoteFX page; park the FX cursor. */
441 s_fx_cursor = seq_state.menu_cursor;
442 s_fx_page = false;
443 s_notefx_page = true;
444 seq_state.menu_cursor = 0;
445 seq_state.menu_editing = false;
446 } else if (fx_menu_item_is_value(idx)) {
447 seq_state.menu_editing = !seq_state.menu_editing;
448 } else if (fx_menu_item_is_back(idx)) {
449 s_fx_page = false;
450 seq_state.menu_cursor = s_main_cursor;
451 seq_state.menu_editing = false;
452 }
453 s_force_redraw = true;
454 return true;
455 }
456
457#if CONFIG_SYNTH_PROJECT_STORE
458 if (s_projects_page) {
459 uint8_t idx = seq_state.menu_cursor;
460 if (projects_menu_item_is_back(idx)) {
461 s_projects_page = false;
462 seq_state.menu_cursor = s_main_cursor;
463 seq_state.menu_editing = false;
464 } else if (projects_menu_item_is_value(idx)) {
465 seq_state.menu_editing = projects_menu_handle_click(idx);
466 }
467 s_force_redraw = true;
468 return true;
469 }
470#endif
471
472 menu_item_id_t id = (menu_item_id_t)seq_state.menu_cursor;
473
474 if (menu_item_is_value(id)) {
475 /* Toggle in/out of editing this value. */
476 seq_state.menu_editing = !seq_state.menu_editing;
477 } else {
478 /* Action item: run it and close the menu. */
479 switch (id) {
480 case MI_SCREEN_SEQ:
481 seq_state.ui_mode = UI_MODE_SEQUENCER;
482 seq_state.menu_open = false;
483 break;
484 case MI_SCREEN_ARP:
485 seq_state.ui_mode = UI_MODE_ARP;
486 seq_state.menu_open = false;
487 break;
488 case MI_SCREEN_DRONE:
489 /* Opens the normal drone; its STUTTER row dives to the
490 * stutter screen (UI_MODE_DRONE). */
491 seq_state.ui_mode = UI_MODE_DRONE_STD;
492 seq_state.menu_open = false;
493 break;
494 case MI_SCREEN_PROG:
495 seq_state.ui_mode = UI_MODE_PROG;
496 seq_state.menu_open = false;
497 break;
498 case MI_SCREEN_TRACKOPTS:
499 seq_state.ui_mode = UI_MODE_TRACKOPTS;
500 seq_state.menu_open = false;
501 s_to_layer = seq_state.active_layer_idx;
502 s_to_track = seq_state.selected_track;
503 break;
504#if CONFIG_SYNTH_CUSTOM_FM
505 case MI_SCREEN_FM:
506 seq_state.ui_mode = UI_MODE_FM;
507 seq_state.menu_open = false;
508 break;
509#endif
510 case MI_ADD_LAYER:
511 if (seq_state.num_layers < MAX_LAYERS)
512 synth_ui_request_add_layer();
513 seq_state.menu_open = false;
514 break;
515 case MI_REMOVE_LAYER:
516 s_to_layer = seq_state.active_layer_idx;
517 synth_ui_request_delete_to_layer();
518 seq_state.menu_open = false;
519 break;
520 case MI_FX_MENU:
521 /* Dive into the global-FX page; the menu stays open. */
522 s_main_cursor = seq_state.menu_cursor;
523 s_fx_page = true;
524 seq_state.menu_cursor = 0;
525 break;
526#if CONFIG_SYNTH_PROJECT_STORE
527 case MI_PROJECTS:
528 /* Dive into the projects page; the menu stays open. */
529 s_main_cursor = seq_state.menu_cursor;
530 s_projects_page = true;
531 seq_state.menu_cursor = 0;
532 projects_menu_reset();
533 break;
534#endif
535 case MI_SAMPLE:
536 switch (sample_rec_get_state()) {
537 case SAMPLE_REC_IDLE:
538 if (seq_state.layers[seq_state.active_layer_idx].type == SEQ_LAYER_DRUM) {
539 sample_rec_arm(seq_state.active_layer_idx, seq_state.selected_track);
540 } else {
541 ESP_LOGW(TAG, "sample_rec: select a drum track first");
542 }
543 break;
544 case SAMPLE_REC_ARMED:
545 sample_rec_start();
546 break;
547 case SAMPLE_REC_READY:
548 sample_rec_assign();
549 break;
550 case SAMPLE_REC_RECORDING:
551 default:
552 break; /* capture runs in the background regardless of the menu */
553 }
554 seq_state.menu_open = false;
555 break;
556 case MI_SAMPLE_CANCEL:
557 sample_rec_cancel();
558 seq_state.menu_open = false;
559 break;
560 default:
561 break;
562 }
563 seq_state.menu_editing = false;
564 }
565 s_force_redraw = true;
566 return true;
567}
