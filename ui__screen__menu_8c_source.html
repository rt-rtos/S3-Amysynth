25 * MENU OVERLAY
26 * ════════════════════════════════════════════════════════════════════════
27 * A small modal list. Items are either ACTIONS (run on click) or VALUE items
28 * (click to enter editing, encoder changes the value, click to exit). The model
29 * is a static table; values are read/written live from sequencer_core, arp_core
30 * and the global FX cache.
31 *
32 * This file owns the page state (main list plus the sub-pages dived into from
33 * it) and routes encoder/button input to the active page; every page shares
34 * seq_state.menu_cursor/menu_editing. */

306/* Signature of the menu overlay. Builds the view into *out so the caller draws
307 * from it without a second full snprintf pass. */
308uint32_t menu_view_signature(menu_view_t *out)
309{
310 uint32_t h = FNV1A_OFFSET;
311 menu_build_view(out);
312 h = fnv1a_bytes(h, &out->cursor, sizeof(out->cursor));
313 h = fnv1a_bytes(h, &out->editing, sizeof(out->editing));
314 for (uint8_t i = 0; i < out->count; i++) {
315 h = fnv1a_bytes(h, out->items[i].label, sizeof(out->items[i].label));
316 h = fnv1a_bytes(h, out->items[i].value, sizeof(out->items[i].value));
317 }
318 return h;
319}

341/* Apply an encoder delta to the currently-entered menu value. */
342static void menu_edit_value(menu_item_id_t id, int delta)
343{
344 int dir = (delta > 0) ? 1 : (delta < 0 ? -1 : 0);
345 switch (id) {
346 case MI_BPM:
347 synth_ui_set_bpm((uint16_t)((int)seq_get_bpm() + delta));
348 break;
349 case MI_QUANT_ENABLED:
350 if (dir != 0)
351 sequencer_core_set_quantizer_enabled(
352 !sequencer_core_get_quantizer_enabled());
353 break;
354 case MI_QUANT_SCALE: {
355 int n = (int)quantizer_scale_count();
356 int cur = (int)sequencer_core_get_quantizer_scale();
357 int ni = (cur + dir + n) % n;
358 sequencer_core_set_quantizer_scale((uint8_t)ni);
359 break;
360 }
361 case MI_QUANT_ROOT: {
362 /* Cycle the 12 pitch classes - only the class matters to the
363 * snapper - but store as a MIDI note in octave 4 (60..71) so
364 * snapshots and the Kconfig default keep their meaning. */
365 int pc = ((int)sequencer_core_get_quantizer_root_note() % 12 + delta) % 12;
366 if (pc < 0) pc += 12;
367 sequencer_core_set_quantizer_root_note((uint8_t)(60 + pc));
368 break;
369 }
370 case MI_ARP_QUANT:
371 if (dir != 0) arp_set_follow_quant(!arp_get_follow_quant());
372 break;
373 case MI_ARP_ENABLED:
374 if (dir != 0) arp_set_enabled(!arp_get_enabled());
375 break;
376 case MI_DRONE_ENABLED:
377 if (dir != 0) drone_std_set_enabled(!drone_std_get_enabled());
378 break;
379 case MI_STUTTER_ENABLED:
380 if (dir != 0) drone_set_enabled(!drone_get_enabled());
381 break;
382 case MI_DRUM_ENGINE:
383 if (dir != 0) {
384 int n = (int)sequencer_core_drum_source_count();
385 int ni = ((int)sequencer_core_get_drum_source() + dir + n) % n;
386 sequencer_core_set_drum_source((uint8_t)ni);
387 /* The bank seed rewrote the core's per-track presets; refresh
388 * the UI mirror so the drum screen's labels follow. */
389 for (uint8_t i = 0; i < seq_state.num_layers; i++) {
390 if (seq_state.layers[i].type != SEQ_LAYER_DRUM) continue;
391 for (uint8_t t = 0; t < SEQ_TRACKS; t++) {
392 seq_state.layers[i].track_pcm_preset[t] =
393 sequencer_core_get_drum_pcm_preset(i, t);
394 }
395 }
396 }
397 break;
398 case MI_VOLUME: {
399 /* 5% steps; the setter clamps and writes amy_global.volume[]. */
400 amy_fx_set_master_volume(amy_fx_get_master_volume() + (float)dir * 0.05f);
401 break;
402 }
403 default:
404 break;
405 }

408void synth_ui_menu_toggle(void)
409{
410 /* The graph editor is the top overlay; don't let the menu fight it. */
411 if (synth_ui_graph_is_active()) return;
412 seq_state.menu_open = !seq_state.menu_open;
413 seq_state.menu_editing = false;
414 if (seq_state.menu_open) {
415 /* Always reopen on the main page so the menu lands somewhere known. */
416 s_fx_page = false;
417 s_notefx_page = false;
418#if CONFIG_SYNTH_PROJECT_STORE
419 s_projects_page = false;
420#endif
421 s_chords_page = false;
422#if CONFIG_SYNTH_WIRELESS
423 s_wireless_page = false;
424#endif
425 if (seq_state.menu_cursor >= MI_COUNT) seq_state.menu_cursor = 0;
426 }
427 s_force_redraw = true;
428 ESP_LOGI(TAG, "menu %s", seq_state.menu_open ? "open" : "closed");
429}

464bool synth_ui_menu_handle_encoder(long delta)
465{
466 if (!seq_state.menu_open) return false;
467 if (delta == 0) return true;
468
469 if (seq_state.menu_editing) {
470 if (s_notefx_page) {
471 notefx_menu_edit_value(seq_state.menu_cursor, (int)delta);
472 } else if (s_fx_page) {
473 fx_menu_edit_value(seq_state.menu_cursor, (int)delta);
474#if CONFIG_SYNTH_PROJECT_STORE
475 } else if (s_projects_page) {
476 projects_menu_edit_value(seq_state.menu_cursor, (int)delta);
477#endif
478 } else if (s_chords_page) {
479 chords_menu_edit_value(seq_state.menu_cursor, (int)delta);
480#if CONFIG_SYNTH_WIRELESS
481 } else if (s_wireless_page) {
482 wireless_menu_edit_value(seq_state.menu_cursor, (int)delta);
483#endif
484 } else {
485 menu_edit_value((menu_item_id_t)seq_state.menu_cursor, (int)delta);
486 }
487 } else {
488 int n = s_notefx_page ? (int)notefx_menu_item_count() :
489 s_fx_page ? (int)fx_menu_item_count() :
490#if CONFIG_SYNTH_PROJECT_STORE
491 s_projects_page ? (int)projects_menu_item_count() :
492#endif
493 s_chords_page ? (int)chords_menu_item_count() :
494#if CONFIG_SYNTH_WIRELESS
495 s_wireless_page ? (int)wireless_menu_item_count() :
496#endif
497 (int)MI_COUNT;
498 int c = (int)seq_state.menu_cursor + (int)delta;
499 /* wrap: the lists are long enough that top-to-bottom in one click
500 * beats scrolling the whole way */
501 c %= n;
502 if (c < 0) c += n;
503 seq_state.menu_cursor = (uint8_t)c;
504 }
505 s_force_redraw = true;
506 return true;
507}

509bool synth_ui_menu_handle_button(void)
510{
511 if (!seq_state.menu_open) return false;
512
513 if (s_notefx_page) {
514 uint8_t idx = seq_state.menu_cursor;
515 if (notefx_menu_item_is_value(idx)) {
516 seq_state.menu_editing = !seq_state.menu_editing;
517 } else if (notefx_menu_item_is_back(idx)) {
518 /* Back returns to the FX page it was dived from. */
519 s_notefx_page = false;
520 s_fx_page = true;
521 seq_state.menu_cursor = s_fx_cursor;
522 seq_state.menu_editing = false;
523 }
524 s_force_redraw = true;
525 return true;
526 }
527
528 if (s_fx_page) {
529 uint8_t idx = seq_state.menu_cursor;
530 if (fx_menu_item_is_notefx(idx)) {
531 /* Dive into the per-layer NoteFX page; park the FX cursor. */
532 s_fx_cursor = seq_state.menu_cursor;
533 s_fx_page = false;
534 s_notefx_page = true;
535 seq_state.menu_cursor = 0;
536 seq_state.menu_editing = false;
537 } else if (fx_menu_item_is_value(idx)) {
538 seq_state.menu_editing = !seq_state.menu_editing;
539 } else if (fx_menu_item_is_back(idx)) {
540 s_fx_page = false;
541 seq_state.menu_cursor = s_main_cursor;
542 seq_state.menu_editing = false;
543 }
544 s_force_redraw = true;
545 return true;
546 }
547
548#if CONFIG_SYNTH_PROJECT_STORE
549 if (s_projects_page) {
550 uint8_t idx = seq_state.menu_cursor;
551 if (projects_menu_item_is_back(idx)) {
552 s_projects_page = false;
553 seq_state.menu_cursor = s_main_cursor;
554 seq_state.menu_editing = false;
555 } else if (projects_menu_item_is_value(idx)) {
556 seq_state.menu_editing = projects_menu_handle_click(idx);
557 }
558 s_force_redraw = true;
559 return true;
560 }
561#endif
562
563 if (s_chords_page) {
564 uint8_t idx = seq_state.menu_cursor;
565 if (chords_menu_item_is_back(idx)) {
566 s_chords_page = false;
567 seq_state.menu_cursor = s_main_cursor;
568 seq_state.menu_editing = false;
569 } else {
570 seq_state.menu_editing = chords_menu_handle_click(idx);
571 }
572 s_force_redraw = true;
573 return true;
574 }
575
576#if CONFIG_SYNTH_WIRELESS
577 if (s_wireless_page) {
578 uint8_t idx = seq_state.menu_cursor;
579 if (wireless_menu_item_is_back(idx)) {
580 s_wireless_page = false;
581 seq_state.menu_cursor = s_main_cursor;
582 seq_state.menu_editing = false;
583 } else {
584 seq_state.menu_editing = wireless_menu_handle_click(idx);
585 }
586 s_force_redraw = true;
587 return true;
588 }
589#endif
590
591 menu_item_id_t id = (menu_item_id_t)seq_state.menu_cursor;
592
593 if (menu_item_is_value(id)) {
594 /* Toggle in/out of editing this value. */
595 seq_state.menu_editing = !seq_state.menu_editing;
596 } else {
597 /* Action item: run it and close the menu. */
598 switch (id) {
599 case MI_SCREEN_SEQ:
600 seq_state.ui_mode = UI_MODE_SEQUENCER;
601 seq_state.menu_open = false;
602 break;
603 case MI_SCREEN_ARP:
604 seq_state.ui_mode = UI_MODE_ARP;
605 seq_state.menu_open = false;
606 break;
607 case MI_SCREEN_DRONE:
608 /* Opens the normal drone; its STUTTER row dives to the
609 * stutter screen (UI_MODE_DRONE). */
610 seq_state.ui_mode = UI_MODE_DRONE_STD;
611 seq_state.menu_open = false;
612 break;
613 case MI_SCREEN_PROG:
614 seq_state.ui_mode = UI_MODE_PROG;
615 seq_state.menu_open = false;
616 break;
617 case MI_SCREEN_TRACKOPTS:
618 seq_state.ui_mode = UI_MODE_TRACKOPTS;
619 seq_state.menu_open = false;
620 s_to_layer = seq_state.active_layer_idx;
621 s_to_track = seq_state.selected_track;
622 break;
623#if CONFIG_SYNTH_DEV_MENU
624 case MI_DEV:
625 seq_state.ui_mode = UI_MODE_DEV;
626 seq_state.menu_open = false;
627 break;
628#endif
629#if CONFIG_SYNTH_CUSTOM_FM
630 case MI_SCREEN_FM:
631 seq_state.ui_mode = UI_MODE_FM;
632 seq_state.menu_open = false;
633 break;
634#endif
635 case MI_ADD_LAYER:
636 if (seq_state.num_layers < MAX_LAYERS)
637 synth_ui_request_add_layer();
638 seq_state.menu_open = false;
639 break;
640 case MI_REMOVE_LAYER:
641 s_to_layer = seq_state.active_layer_idx;
642 synth_ui_request_delete_to_layer();
643 seq_state.menu_open = false;
644 break;
645 case MI_CHORDS:
646 /* Dive into the chord-preset page; the menu stays open. */
647 s_main_cursor = seq_state.menu_cursor;
648 s_chords_page = true;
649 seq_state.menu_cursor = 0;
650 chords_menu_reset();
651 break;
652 case MI_FX_MENU:
653 /* Dive into the global-FX page; the menu stays open. */
654 s_main_cursor = seq_state.menu_cursor;
655 s_fx_page = true;
656 seq_state.menu_cursor = 0;
657 break;
658#if CONFIG_SYNTH_PROJECT_STORE
659 case MI_PROJECTS:
660 /* Dive into the projects page; the menu stays open. */
661 s_main_cursor = seq_state.menu_cursor;
662 s_projects_page = true;
663 seq_state.menu_cursor = 0;
664 projects_menu_reset();
665 break;
666#endif
667#if CONFIG_SYNTH_WIRELESS
668 case MI_WIRELESS:
669 /* Dive into the wireless page; the menu stays open. */
670 s_main_cursor = seq_state.menu_cursor;
671 s_wireless_page = true;
672 seq_state.menu_cursor = 0;
673 wireless_menu_reset();
674 break;
675#endif
676 case MI_SAMPLE:
677 switch (sample_rec_get_state()) {
678 case SAMPLE_REC_IDLE:
679 if (seq_state.layers[seq_state.active_layer_idx].type == SEQ_LAYER_DRUM) {
680 sample_rec_arm(seq_state.active_layer_idx, seq_state.selected_track);
681 } else {
682 ESP_LOGW(TAG, "sample_rec: select a drum track first");
683 }
684 break;
685 case SAMPLE_REC_ARMED:
686 sample_rec_start();
687 break;
688 case SAMPLE_REC_READY:
689 sample_rec_assign();
690 break;
691 case SAMPLE_REC_RECORDING:
692 default:
693 break; /* capture runs regardless of the menu */
694 }
695 seq_state.menu_open = false;
696 break;
697 case MI_SAMPLE_CANCEL:
698 sample_rec_cancel();
699 seq_state.menu_open = false;
700 break;
701 default:
702 break;
703 }
704 seq_state.menu_editing = false;
705 }
706 s_force_redraw = true;
707 return true;
708}
