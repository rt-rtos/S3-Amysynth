7/* ════════════════════════════════════════════════════════════════════════
8 * NOTE FX SUBMENU PAGE
9 * ════════════════════════════════════════════════════════════════════════
10 * Per-layer melodic note controls, reached from a dive row on the global-FX
11 * page (page state and input routing live in ui_screen_menu.c; this file only
12 * builds the rows and applies encoder edits). Two controls, both acting on the
13 * currently ACTIVE melodic layer (seq_state.active_layer_idx):
14 * GATE — note-hold as % of the step (10..100; 100 = full-step legato)
15 * GLIDE — AMY-native portamento between step pitches (0..cap ms)
16 * Kept off the crowded global-FX list because these are per-layer, not global.
17 *
18 * When the active layer is a drum layer (or otherwise not melodic) there is
19 * nothing to edit: the rows render "--" and edits no-op, so the page is always
20 * safe to open regardless of which layer is selected. */
21

80void notefx_menu_edit_value(uint8_t idx, int delta)
81{

83 if (dir == 0) return;

85 uint8_t li = notefx_active_melodic_layer();
86 if (li == 0xFF) return; /* non-melodic layer: nothing to edit */
87
88 switch ((notefx_item_id_t)idx) {

95 }

101 sequencer_core_set_melodic_portamento_ms(li, (uint16_t)v);
