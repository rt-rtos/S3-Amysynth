72bool synth_ui_prog_handle_encoder(int delta)
73{
74 if (!synth_ui_prog_is_active()) return false;
75 uint8_t count = sequencer_core_progression_get_count();
76 /* Cursor range: 0=enable toggle, 1..count=entries, count+1=apply mode. */
77 uint8_t max_cursor = (uint8_t)(count + 1);
78
79 if (s_prog_editing && s_prog_cursor >= 1 && s_prog_cursor <= count) {
80 /* Editing the focused field of the selected entry. */
81 uint8_t ei = (uint8_t)(s_prog_cursor - 1);
82 uint8_t root; chord_type_t ct; uint8_t dur;
83 sequencer_core_progression_get_entry(ei, &root, &ct, &dur);
84 switch (s_prog_field) {
85 case 0: { /* root: chromatic 0–11 */
86 int nr = ((int)root + delta) % 12;
87 if (nr < 0) nr += 12;
88 root = (uint8_t)nr;
89 break;
90 }
91 case 1: { /* chord type */
92 int nct = (int)ct + delta;
93 /* Real chords only - CHORD_OFF is a drone affordance. */
94 while (nct < 0) nct += CHORD_REAL_COUNT;
95 nct %= CHORD_REAL_COUNT;
96 ct = (chord_type_t)nct;
97 break;
98 }
99 default: { /* duration: cycle the allowed set */
100 int di = prog_duration_index(dur) + delta;
101 while (di < 0) di += PROG_NUM_DURATIONS;
102 di %= PROG_NUM_DURATIONS;
103 dur = s_prog_durations[di];
104 break;
105 }
106 }
107 sequencer_core_progression_set_entry(ei, root, ct, dur);
108 } else {
109 int nc = (int)s_prog_cursor + delta;
110 if (nc < 0) nc = (int)max_cursor;
111 if (nc > (int)max_cursor) nc = 0;
112 s_prog_cursor = (uint8_t)nc;
113 }
114 s_force_redraw = true;
115 return true;
116}
