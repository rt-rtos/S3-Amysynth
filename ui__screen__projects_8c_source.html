15/* ════════════════════════════════════════════════════════════════════════
16 * PROJECTS PAGE (persistent project storage)
17 * ════════════════════════════════════════════════════════════════════════
18 * Item model for the Projects page of the menu overlay (page state and
19 * input routing live in ui_screen_menu.c, mirroring the global-FX page).
20 *
21 * Layout: item 0 = Back, item 1 = a read-only storage-usage line, items
22 * 2..2+N-1 = one row per slot. Clicking a slot enters a per-slot ACTION
23 * sub-state (Load/Save/Ren/Del/Exit, cycled with the encoder); clicking
24 * again executes the selected action. Save on a used slot arms on the
25 * first click and executes on the second ("Save!"); Rename swaps the value
26 * field for a compact character editor (encoder cycles the alphabet,
27 * click advances position, the '#' sentinel commits early).
28 *
29 * There is no toast/transient-message overlay anywhere in this UI (verified
30 * against synth_ui_hint.c and ui_view_resolve.c: the "hint" strip is a
31 * static per-view button-hint label, not a message queue). Action results
32 * ("SAVED", "LOAD FAIL", ...) are therefore shown inline in the acted-on
33 * row's own value field until the cursor moves off it — no new draw path,
34 * still fill-only, still generic menu_item_view_t rendering. */
35

131/* Compact rename-editor value: the char under the cursor, bracketed, plus a
132 * 1-based position/length pair. A full 15-char window doesn't fit
133 * MENU_VALUE_LEN (14 incl. NUL); this stays legible and simple to reason
134 * about instead of a scrolling text window. */
135static void fmt_rename_value(char *out, size_t outsz)
136{
137 char c = s_name_buf[s_name_pos];
138 if (c == '\0') c = '_';
139 snprintf(out, outsz, "[%c]%u/%u", c,
140 (unsigned)(s_name_pos + 1), (unsigned)(PROJECT_NAME_LEN - 1));
141}

145 if (c == '\0') c = ' ';

223const menu_item_view_t *projects_menu_build_items(void)
224{
225 if (s_dirty) {
226 for (uint8_t i = 0; i < CONFIG_SYNTH_PROJECT_MAX_SLOTS; i++) {
227 if (!project_store_slot_info(i, &s_info[i])) {
228 s_info[i].used = false;
229 }
230 }
231 s_dirty = false;
232 }
233
234 /* A status message only ever belongs to the row the action ran on;
235 * moving off that row (the only way to reach a different item once
236 * editing has ended) clears it so it can't linger on the wrong slot. */
237 if (s_status_active && seq_state.menu_cursor != s_status_idx) {
238 s_status_active = false;
239 }
240
241 snprintf(s_items[0].label, MENU_LABEL_LEN, "< Back");
242 s_items[0].value[0] = '\0';
243
244 snprintf(s_items[1].label, MENU_LABEL_LEN, "Storage");
245 {
246 size_t total = 0, used = 0;
247 if (project_fs_ok() && project_fs_stats(&total, &used)) {
248 /* 6 chars each so "u/t" (13 + NUL) provably fits the value
249 * field - keeps -Werror=format-truncation satisfied. */
250 char u[7], t[7];
251 fmt_bytes(u, sizeof(u), used);
252 fmt_bytes(t, sizeof(t), total);
253 snprintf(s_items[1].value, MENU_VALUE_LEN, "%s/%s", u, t);
254 } else {
255 snprintf(s_items[1].value, MENU_VALUE_LEN, "ERR");
256 }
257 }
258
259 for (uint8_t slot = 0; slot < CONFIG_SYNTH_PROJECT_MAX_SLOTS; slot++) {
260 uint8_t idx = (uint8_t)(2 + slot);
261 const project_slot_info_t *info = &s_info[slot];
262
263 fmt_label(s_items[idx].label, MENU_LABEL_LEN, slot, info);
264
265 bool editing_this = seq_state.menu_editing && seq_state.menu_cursor == idx;
266 if (editing_this && s_renaming) {
267 fmt_rename_value(s_items[idx].value, MENU_VALUE_LEN);
268 } else if (editing_this) {
269 snprintf(s_items[idx].value, MENU_VALUE_LEN, "%s",
270 action_name(s_action, s_armed));
271 } else if (s_status_active && s_status_idx == idx) {
272 snprintf(s_items[idx].value, MENU_VALUE_LEN, "%s", s_status_msg);
273 } else {
274 fmt_value(s_items[idx].value, MENU_VALUE_LEN, info);
275 }
276 }
277
278 return s_items;
279}

281void projects_menu_edit_value(uint8_t idx, int delta)
282{
283 if (!projects_menu_item_is_value(idx)) return;
284 int dir = (delta > 0) ? 1 : (delta < 0 ? -1 : 0);
285 if (dir == 0) return;
286
287 if (s_renaming) {
288 int n = (int)(sizeof(PROJ_NAME_CHARS) - 1);
289 int cur = char_index(s_name_buf[s_name_pos]);
290 cur = ((cur + dir) % n + n) % n;
291 s_name_buf[s_name_pos] = PROJ_NAME_CHARS[cur];
292 return;
293 }
294
295 int a = ((int)s_action + dir) % (int)PA_COUNT;
296 if (a < 0) a += PA_COUNT;
297 s_action = (proj_action_t)a;
298 s_armed = false; /* action changed: any pending confirm is void */
299}

301/* Click on item idx. Returns the desired seq_state.menu_editing value —
302 * true keeps the row's action/rename sub-state open (needs another click),
303 * false executes/aborts and returns to browsing the slot list. */
304bool projects_menu_handle_click(uint8_t idx)
305{
306 if (!projects_menu_item_is_value(idx)) return false;
307 /* A queued load/save is still in flight (one UI frame): ignore clicks so
308 * a second request can't clobber the pending one's fields. */
309 if (s_req != PREQ_NONE) return seq_state.menu_editing;
310 uint8_t slot = (uint8_t)(idx - 2);
311 const project_slot_info_t *info = &s_info[slot];
312 bool corrupt = slot_corrupt(info);
313
314 if (!seq_state.menu_editing) {
315 /* First click: enter the action sub-menu fresh. */
316 s_action = PA_LOAD;
317 s_armed = false;
318 s_renaming = false;
319 return true;
320 }
321
322 if (s_renaming) {
323 char c = s_name_buf[s_name_pos];
324 if (c == '#') {
325 s_name_buf[s_name_pos] = '\0';
326 commit_rename(idx, slot);
327 return false;
328 }
329 s_name_pos++;
330 if (s_name_pos >= PROJECT_NAME_LEN - 1) {
331 s_name_buf[PROJECT_NAME_LEN - 1] = '\0';
332 commit_rename(idx, slot);
333 return false;
334 }
335 return true;
336 }
337
338 switch (s_action) {
339 case PA_LOAD:
340 if (!info->used || corrupt) {
341 set_status(idx, "NO PROJ");
342 s_armed = false;
343 return false;
344 }
345 s_req_slot = slot;
346 s_req_idx = idx;
347 s_req = PREQ_LOAD;
348 set_status(idx, "LOAD..");
349 s_armed = false;
350 return false;
351
352 case PA_SAVE:
353 if (info->used && !s_armed) {
354 s_armed = true;
355 return true; /* second click required to overwrite */
356 }
357 s_req_slot = slot;
358 s_req_idx = idx;
359 s_req_has_name = info->used;
360 if (info->used) {
361 snprintf(s_req_name, sizeof(s_req_name), "%s", info->name);
362 }
363 s_req = PREQ_SAVE;
364 set_status(idx, "SAVE..");
365 s_armed = false;
366 return false;
367
368 case PA_REN:
369 if (!info->used || corrupt) {
370 set_status(idx, "NO PROJ");
371 s_armed = false;
372 return false;
373 }
374 enter_rename(slot);
375 return true;
376
377 case PA_DEL:
378 if (!info->used) {
379 set_status(idx, "NO PROJ");
380 s_armed = false;
381 return false;
382 }
383 set_status(idx, project_store_delete(slot) ? "DELETED" : "DEL FAIL");
384 s_armed = false;
385 s_dirty = true;
386 return false;
387
388 case PA_EXIT:
389 default:
390 s_armed = false;
391 return false;
392 }
393}

395/* Runs once per synth_ui_task frame: executes the queued load/save on the
396 * layers-applier task, then rewrites the acted-on row's inline status. */
397void projects_menu_service(void)
398{
399 if (s_req == PREQ_NONE) return;
400
401 if (s_req == PREQ_LOAD) {
402 bool ok = project_snapshot_load(s_req_slot);
403 set_status(s_req_idx, ok ? "LOADED" : "LOAD FAIL");
404 } else {
405 bool ok = project_snapshot_save(s_req_slot,
406 s_req_has_name ? s_req_name : NULL);
407 set_status(s_req_idx, ok ? "SAVED" : "SAVE FAIL");
408 s_dirty = true;
409 }
410 s_req = PREQ_NONE;
411}
