23bool synth_ui_stepedit_is_active(void)
24{
25 /* The overlay decorates the grid's own cursor, so it is active only while
26 * that grid is the live screen. Gating here (as every sibling
27 * *_is_active() does) means draw cascade, input routers and hint strip all
28 * drop it the instant the user leaves the sequencer or opens the menu - no
29 * per-consumer teardown. */
30 return s_se_active
31 && seq_state.ui_mode == UI_MODE_SEQUENCER
32 && !seq_state.menu_open;

73bool synth_ui_stepedit_handle_encoder(long delta)
74{
75 if (!s_se_active) return false;
76 if (!s_se_editing) {
77 /* Navigate: one field per event, direction only. */
78 if (delta > 0) s_se_field = (uint8_t)((s_se_field + 1) % SE_FIELD_COUNT);
79 else if (delta < 0) s_se_field = (uint8_t)((s_se_field + SE_FIELD_COUNT - 1) % SE_FIELD_COUNT);
80 s_force_redraw = true;
81 return true;
82 }
83 uint8_t li = seq_state.active_layer_idx;
84 uint8_t t = seq_state.selected_track;
85 uint8_t s = seq_state.selected_step;
86 int d = (int)delta;
87
88 switch (s_se_field) {
89 case SE_FIELD_PITCH: {
90 int v = (int)sequencer_core_get_step_pitch_ofs(li, t, s) + d;
91 v = SEQ_CLAMP_INT(v, -SEQ_STEP_PITCH_OFS_MAX, SEQ_STEP_PITCH_OFS_MAX);
92 sequencer_core_set_step_pitch_ofs(li, t, s, (int8_t)v);
93 break;
94 }
95 case SE_FIELD_PROB: {
96 int v = (int)sequencer_core_get_step_prob(li, t, s) + d * 5;
97 sequencer_core_set_step_prob(li, t, s, SEQ_CLAMP_U8(v, 0, 100));
98 break;
99 }
100 case SE_FIELD_RATCHET: {
101 int v = (int)sequencer_core_get_step_ratchet(li, t, s) + d;
102 sequencer_core_set_step_ratchet(li, t, s, SEQ_CLAMP_U8(v, 1, SEQ_MAX_RATCHET));
103 break;
104 }
105 case SE_FIELD_EVERY: {
106 int v = (int)sequencer_core_get_step_every(li, t, s) + d;
107 sequencer_core_set_step_every(li, t, s, SEQ_CLAMP_U8(v, 1, SEQ_STEP_EVERY_MAX));
108 break;
109 }
110 default:
111 /* SE_FIELD_PREV never enters adjust mode (click-toggle). */
112 break;
113 }
114 s_force_redraw = true;
115 return true;
116}
