27bool synth_ui_stepedit_is_active(void)
28{
29 /* The overlay decorates the sequencer grid's own cursor, so it is only
30 * ever "active" while that grid is the live screen. Gating here (matching
31 * every sibling *_is_active()) means the draw cascade, encoder/button
32 * routers, and hint strip all stop treating a stray editor as active the
33 * instant the user leaves the sequencer or opens the menu — no separate
34 * teardown needed in each consumer. */
35 return s_se_active
36 && seq_state.ui_mode == UI_MODE_SEQUENCER
37 && !seq_state.menu_open;
38}

71bool synth_ui_stepedit_handle_encoder(long delta)
72{
73 if (!s_se_active) return false;
74 uint8_t li = seq_state.active_layer_idx;
75 uint8_t t = seq_state.selected_track;
76 uint8_t s = seq_state.selected_step;
77 int d = (int)delta;
78
79 /* Cond type may have changed since the field cursor was last set (e.g.
80 * an earlier edit left PARAM selected, then the user cycled COND away
81 * from FILL) — clamp defensively before dispatching. */
82 uint8_t max_field = se_max_field(li, t, s);
83 if (s_se_field > max_field) s_se_field = max_field;
84
85 switch (s_se_field) {
86 case SE_FIELD_PROB: {
87 int v = (int)sequencer_core_get_step_prob(li, t, s) + d * 5;
88 sequencer_core_set_step_prob(li, t, s, SEQ_CLAMP_U8(v, 0, 100));
89 break;
90 }
91 case SE_FIELD_RATCHET: {
92 int v = (int)sequencer_core_get_step_ratchet(li, t, s) + d;
93 sequencer_core_set_step_ratchet(li, t, s, SEQ_CLAMP_U8(v, 1, SEQ_MAX_RATCHET));
94 break;
95 }
96 case SE_FIELD_COND: {
97 seq_step_cond_type_t type = SEQ_STEP_COND_NONE;
98 uint8_t param = 0;
99 sequencer_core_get_step_cond(li, t, s, &type, &param);
100 int v = (int)type + d;
101 v %= (int)SEQ_STEP_COND_COUNT;
102 if (v < 0) v += (int)SEQ_STEP_COND_COUNT;
103 sequencer_core_set_step_cond(li, t, s, (seq_step_cond_type_t)v, param);
104 break;
105 }
106 case SE_FIELD_PARAM: {
107 seq_step_cond_type_t type = SEQ_STEP_COND_NONE;
108 uint8_t param = 0;
109 sequencer_core_get_step_cond(li, t, s, &type, &param);
110 int v = (int)param + d;
111 sequencer_core_set_step_cond(li, t, s, type, SEQ_CLAMP_U8(v, 2, 8));
112 break;
113 }
114 default:
115 break;
116 }
117 s_force_redraw = true;
118 return true;
119}
