15/* Next repeat rate in the 1→2→4→8→1 cycle. */
16static uint8_t to_next_repeat_rate(uint8_t rr, int delta)
17{
18 static const uint8_t rates[] = { 1, 2, 4, 8 };
19 int n = (int)(sizeof(rates) / sizeof(rates[0]));
20 int idx = 0;
21 for (int i = 0; i < n; i++) if (rates[i] == rr) idx = i;
22 idx = (idx + delta) % n;
23 if (idx < 0) idx += n;
24 return rates[idx];
25}

27/* The rows the cursor can actually land on, in navigation order. Two of them
28 * come and go: the chord rows are melodic-only, and CLR exists only while
29 * something is soloed. Building the list is what keeps navigation correct as
30 * they appear and disappear - a plain cursor range cannot express a gap.
31 * Returns the count; `rows` must hold TO_ROW_COUNT entries. */
32static uint8_t to_row_list(uint8_t li, uint8_t *rows)
33{
34 uint8_t n = 0;
35 rows[n++] = TO_ROW_LAYER;
36 rows[n++] = TO_ROW_TRACK;
37 rows[n++] = TO_ROW_REPEAT;
38 rows[n++] = TO_ROW_MUTE;
39 rows[n++] = TO_ROW_SOLO;
40 if (sequencer_core_get_layer_type(li) == SEQ_LAYER_MELODIC) {
41 rows[n++] = TO_ROW_CHORD;
42 rows[n++] = TO_ROW_ROOT;
43 rows[n++] = TO_ROW_TYPE;
44 }
45 if (sequencer_core_any_solo()) rows[n++] = TO_ROW_CLRSOLO;
46 return n;
47}

49/* Pull the cursor back onto a row that still exists - after a layer switch onto
50 * a drum layer, or after the last solo is cleared out from under CLR. Snapping
51 * to the last row keeps the cursor near where it was rather than jumping home. */
52static void to_clamp_cursor(void)
53{
54 uint8_t rows[TO_ROW_COUNT];
55 uint8_t n = to_row_list(s_to_layer, rows);
56 for (uint8_t i = 0; i < n; i++) {
57 if (rows[i] == s_to_cursor) return;
58 }
59 s_to_cursor = rows[n - 1];
60 s_to_editing = false;
61}

119bool synth_ui_trackopts_handle_encoder(int delta)
120{
121 if (!synth_ui_trackopts_is_active()) return false;
122 uint8_t li = s_to_layer;
123 uint8_t tr = s_to_track;
124
125 if (s_to_editing) {
126 switch (s_to_cursor) {
127 case TO_ROW_LAYER: {
128 int nl = (int)s_to_layer + delta;
129 int nc = (int)seq_state.num_layers;
130 if (nl < 0) nl += nc; else if (nl >= nc) nl -= nc;
131 s_to_layer = (uint8_t)nl;
132 to_clamp_cursor(); /* the new layer may not have chord rows */
133 break;
134 }
135 case TO_ROW_TRACK: {
136 int nt = (int)s_to_track + delta;
137 if (nt < 0) nt += SEQ_TRACKS; else if (nt >= SEQ_TRACKS) nt -= SEQ_TRACKS;
138 s_to_track = (uint8_t)nt;
139 break;
140 }
141 case TO_ROW_REPEAT: {
142 uint8_t rr = (uint8_t)sequencer_core_get_track_repeat_rate(li, tr);
143 rr = to_next_repeat_rate(rr, delta);
144 sequencer_core_set_track_repeat_rate(li, tr, (seq_repeat_rate_t)rr);
145 break;
146 }
147 case TO_ROW_MUTE: {
148 sequencer_core_set_track_mute(li, tr, !sequencer_core_get_track_mute(li, tr));
149 break;
150 }
151 case TO_ROW_SOLO: {
152 sequencer_core_set_track_solo(li, tr, !sequencer_core_get_track_solo(li, tr));
153 break;
154 }
155 case TO_ROW_CHORD: {
156 if (sequencer_core_progression_get_enabled()) break;
157 bool mode = false; uint8_t root = 0; chord_type_t ct = CHORD_MAJ;
158 sequencer_core_get_layer_chord(li, &mode, &root, &ct);
159 if (mode) sequencer_core_progression_clear_layer_chord(li);
160 else sequencer_core_progression_set_layer_chord(li, root, ct);
161 break;
162 }
163 case TO_ROW_ROOT:
164 case TO_ROW_TYPE: {
165 if (sequencer_core_progression_get_enabled()) break;
166 bool mode = false; uint8_t root = 0; chord_type_t ct = CHORD_MAJ;
167 sequencer_core_get_layer_chord(li, &mode, &root, &ct);
168 if (!mode) break;
169 if (s_to_cursor == TO_ROW_ROOT) {
170 int nr = ((int)root + delta) % 12;
171 if (nr < 0) nr += 12;
172 root = (uint8_t)nr;
173 } else {
174 int nct = (int)ct + delta;
175 /* Real chords only - CHORD_OFF is a drone affordance. */
176 while (nct < 0) nct += CHORD_REAL_COUNT;
177 nct %= CHORD_REAL_COUNT;
178 ct = (chord_type_t)nct;
179 }
180 sequencer_core_progression_set_layer_chord(li, root, ct);
181 break;
182 }
183 default: break;
184 }
185 } else {
186 /* Walk the live row list rather than a numeric range: it wraps over the
187 * rows that are currently hidden instead of stalling on them. */
188 uint8_t rows[TO_ROW_COUNT];
189 uint8_t n = to_row_list(s_to_layer, rows);
190 int idx = 0;
191 for (uint8_t i = 0; i < n; i++) if (rows[i] == s_to_cursor) idx = (int)i;
192 idx = (idx + delta) % (int)n;
193 if (idx < 0) idx += (int)n;
194 s_to_cursor = rows[idx];
195 }
196 s_force_redraw = true;
197 return true;
198}
