115/* Click on item idx. Returns the desired seq_state.menu_editing value (the
116 * menu router feeds it back), matching the Chords/Projects click contract. */
117bool wireless_menu_handle_click(uint8_t idx)
118{
119 if (idx == WI_TOGGLE) {
120 radio_state_t st = radio_manager_state();
121 if (st == RADIO_ACTIVE) {
122 radio_manager_request_stop();
123 } else {
124 radio_manager_request_start();
125 }
126 return false;
127 }
128 if (idx == WI_SOURCE) {
129 /* Immediate toggle, no edit mode - the two-state row pattern. */
130 live_play_set_wave_mode(!live_play_get_wave_mode());
131 return false;
132 }
133 if (idx == WI_PATCH || idx == WI_GLIDE) {
134 return !seq_state.menu_editing; /* click in, click out */
135 }
136 return false;
137}

139void wireless_menu_edit_value(uint8_t idx, int delta)
140{
141 if (idx == WI_PATCH) {
142 synth_ui_cycle_live_patch(delta);
143 } else if (idx == WI_GLIDE) {
144 int g = (int)live_play_get_glide_ms() + delta; /* 1 ms per detent */
145 if (g < 0) g = 0;
146 if (g > (int)LIVE_PLAY_GLIDE_MAX_MS) g = (int)LIVE_PLAY_GLIDE_MAX_MS;
147 live_play_set_glide_ms((uint16_t)g);
148 }
149}
