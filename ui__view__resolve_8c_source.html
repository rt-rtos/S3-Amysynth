6/* ════════════════════════════════════════════════════════════════════════
7 * Single source of view precedence + the descriptor table over it
8 * ════════════════════════════════════════════════════════════════════════
9 * synth_ui_active_view() is the ONE place the screen/overlay precedence lives
10 * (it used to be hand-copied into the draw switch, the hint strip, and both
11 * main.c input routers). Every consumer calls it, indexes ui_view_table[], and
12 * dispatches — so input and draw always resolve the same view. This mirrors the
13 * former draw cascade (synth_ui_task.c) branch-for-branch, and depends on the
14 * Phase-1 gate inside synth_ui_stepedit_is_active(): the STEPEDIT branch is only
15 * correct because that accessor now self-gates to the sequencer screen. */

17ui_view_id_t synth_ui_active_view(void)
18{
19 if (s_filter_active) return UI_VIEW_FILTER;
20 if (s_lfo_active) return UI_VIEW_LFO;
21 if (synth_ui_stepedit_is_active()) return UI_VIEW_STEPEDIT; /* gated, Phase 1 */
22 if (synth_ui_graph_is_active()) return UI_VIEW_GRAPH;
23 if (seq_state.menu_open) return UI_VIEW_MENU;
24 switch (seq_state.ui_mode) {
25 case UI_MODE_ARP: return UI_VIEW_ARP;
26 case UI_MODE_DRONE: return s_drone_vis_open ? UI_VIEW_DRONE_VIS : UI_VIEW_DRONE;
27 case UI_MODE_DRONE_STD: return UI_VIEW_DRONE_STD;
28 case UI_MODE_PROG: return UI_VIEW_PROG;
29 case UI_MODE_TRACKOPTS: return UI_VIEW_TRACKOPTS;
30#if CONFIG_SYNTH_CUSTOM_FM
31 case UI_MODE_FM: return UI_VIEW_FM;
32#endif
33 default: return UI_VIEW_SEQ;
34 }
35}

63/* DRONE_VIS composites the sweep/amp visualiser from the live drone getters
64 * rather than a prebuilt view struct (the signature still uses drone_view for
65 * render-gating). Lifted verbatim from the former draw switch. */
66static void draw_drone_vis(u8g2_t *g, ui_view_vw_t *vw)
67{
68 (void)vw;
69 const float SWEEP_MIN = 100.0f, SWEEP_MAX = 8000.0f;
70 float span = SWEEP_MAX - SWEEP_MIN;
71 float c = drone_get_amp_peak();
72 float m = drone_get_amp_duck();
73 float fl, cl;
74 drone_get_amp_levels_norm(&fl, &cl);
75 drone_vis_t dvis = {
76 .sweep_lo_norm = (drone_get_sweep_lo() - SWEEP_MIN) / span,
77 .sweep_hi_norm = (drone_get_sweep_hi() - SWEEP_MIN) / span,
78 .amp_const = c,
79 .amp_mod = m,
80 .amp_floor_norm = fl,
81 .amp_ceil_norm = cl,
82 .resonance = drone_get_resonance(),
83 .rate_idx = (uint8_t)drone_get_rate(),
84 .sweep_bars = drone_get_sweep_bars(),
85 .pattern_mask = (uint8_t)0xFF, /* filled below */
86 .gate_len = drone_get_gate_len(),
87 .wave_mode = (drone_get_source() == DRONE_SRC_WAVE),
88 };
89 static const uint8_t pat_masks[] = { 0xFF, 0x55, 0xAA, 0x5B, 0x51 };
90 drone_pattern_t pat = drone_get_pattern();
91 if ((size_t)pat < sizeof(pat_masks))
92 dvis.pattern_mask = pat_masks[pat];
93 display_drone_vis_draw(g, &dvis);
94}
