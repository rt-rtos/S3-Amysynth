6/* ════════════════════════════════════════════════════════════════════════
7 * Single source of view precedence + the descriptor table over it
8 * ════════════════════════════════════════════════════════════════════════
9 * synth_ui_active_view() is the ONE place the screen/overlay precedence lives.
10 * Every consumer calls it, indexes ui_view_table[] and dispatches, so input and
11 * draw always resolve the same view. The STEPEDIT branch is only correct
12 * because synth_ui_stepedit_is_active() self-gates to the sequencer screen. */
13
14ui_view_id_t synth_ui_active_view(void)
15{
16 if (s_filter_active) return UI_VIEW_FILTER;
17 if (s_lfo_active) return UI_VIEW_LFO;
18 if (s_dist_active) return UI_VIEW_DIST;
19 if (synth_ui_stepedit_is_active()) return UI_VIEW_STEPEDIT; /* self-gated */
20 if (synth_ui_graph_is_active()) return UI_VIEW_GRAPH;
21 if (seq_state.menu_open) return UI_VIEW_MENU;
22 switch (seq_state.ui_mode) {
23 case UI_MODE_ARP: return UI_VIEW_ARP;
24 case UI_MODE_DRONE: return s_drone_vis_open ? UI_VIEW_DRONE_VIS : UI_VIEW_DRONE;
25 case UI_MODE_DRONE_STD: return UI_VIEW_DRONE_STD;
26 case UI_MODE_PROG: return UI_VIEW_PROG;
27 case UI_MODE_TRACKOPTS: return UI_VIEW_TRACKOPTS;
28#if CONFIG_SYNTH_CUSTOM_FM
29 case UI_MODE_FM: return UI_VIEW_FM;
30#endif
31#if CONFIG_SYNTH_DEV_MENU
32 case UI_MODE_DEV: return UI_VIEW_DEV;
33#endif
34 default: return UI_VIEW_SEQ;
35 }
36}

66/* DRONE_VIS composites the sweep/amp visualiser from the live drone getters
67 * rather than a prebuilt view struct; the signature still uses drone_view for
68 * render-gating. */
69static void draw_drone_vis(u8g2_t *g, ui_view_vw_t *vw)
70{
71 (void)vw;
72 const float SWEEP_MIN = 100.0f, SWEEP_MAX = 8000.0f;
73 float span = SWEEP_MAX - SWEEP_MIN;
74 float c = drone_get_amp_peak();
75 float m = drone_get_amp_duck();
76 float fl, cl;
77 drone_get_amp_levels_norm(&fl, &cl);
78 drone_vis_t dvis = {
79 .sweep_lo_norm = (drone_get_sweep_lo() - SWEEP_MIN) / span,
80 .sweep_hi_norm = (drone_get_sweep_hi() - SWEEP_MIN) / span,
81 .amp_const = c,
82 .amp_mod = m,
83 .amp_floor_norm = fl,
84 .amp_ceil_norm = cl,
85 .resonance = drone_get_resonance(),
86 .rate_idx = (uint8_t)drone_get_rate(),
87 .sweep_bars = drone_get_sweep_bars(),
88 .pattern_mask = (uint8_t)0xFF, /* filled below */
89 .gate_len = drone_get_gate_len(),
90 .wave_mode = (drone_get_source() == DRONE_SRC_WAVE),
91 };
92 static const uint8_t pat_masks[] = { 0xFF, 0x55, 0xAA, 0x5B, 0x51 };
93 drone_pattern_t pat = drone_get_pattern();
94 if ((size_t)pat < sizeof(pat_masks))
95 dvis.pattern_mask = pat_masks[pat];
96 display_drone_vis_draw(g, &dvis);
97}
