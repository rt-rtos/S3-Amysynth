99const uint16_t *tud_descriptor_string_cb(uint8_t index, uint16_t langid) {
100 char serial_buf[MICROPY_HW_USB_DESC_STR_MAX + 1]; // Includes terminating NUL byte
101 static uint16_t desc_wstr[MICROPY_HW_USB_DESC_STR_MAX + 1]; // Includes prefix uint16_t
102 const char *desc_str = NULL;
103 uint16_t desc_len;
104
105 #if MICROPY_HW_ENABLE_USB_RUNTIME_DEVICE
106 desc_str = amy_usbd_runtime_string_cb(index);
107 #endif
108
109 if (index == 0) {
110 // String descriptor 0 is special, see USB 2.0 section 9.6.7 String
111 //
112 // Expect any runtime value in desc_str to be a fully formed descriptor
113 if (desc_str == NULL) {
114 desc_str = "\x04\x03\x09\x04"; // Descriptor for "English"
115 }
116 if (desc_str[0] < sizeof(desc_wstr)) {
117 memcpy(desc_wstr, desc_str, desc_str[0]);
118 return desc_wstr;
119 }
120 return NULL; // Descriptor length too long (or malformed), stall endpoint
121 }
122
123 // Otherwise, generate a "UNICODE" string descriptor from the C string
124
125 if (desc_str == NULL) {
126 // Fall back to the "static" string
127 switch (index) {
128 case USBD_STR_SERIAL:
129 amy_usbd_port_get_serial_number(serial_buf);
130 desc_str = serial_buf;
131 break;
132 case USBD_STR_MANUF:
133 desc_str = MICROPY_HW_USB_MANUFACTURER_STRING;
134 break;
135 case USBD_STR_PRODUCT:
136 desc_str = MICROPY_HW_USB_PRODUCT_FS_STRING;
137 break;
138 #if CFG_TUD_CDC
139 case USBD_STR_CDC:
140 desc_str = MICROPY_HW_USB_CDC_INTERFACE_STRING;
141 break;
142 #endif
143 #if CFG_TUD_MSC
144 case USBD_STR_MSC:
145 desc_str = MICROPY_HW_USB_MSC_INTERFACE_STRING;
146 break;
147 #endif
148 default:
149 break;
150 }
151 }
152
153 if (desc_str == NULL) {
154 return NULL; // No string, STALL the endpoint
155 }
156
157 // Convert from narrow string to wide string
158 desc_len = 2;
159 for (int i = 0; i < MICROPY_HW_USB_DESC_STR_MAX && desc_str[i] != 0; i++) {
160 desc_wstr[1 + i] = desc_str[i];
161 desc_len += 2;
162 }
163
164 // first byte is length (including header), second byte is string type
165 desc_wstr[0] = (TUSB_DESC_STRING << 8) | desc_len;
166
167 return desc_wstr;
168}

193void amy_usbd_hex_str(char *out_str, const uint8_t *bytes, size_t bytes_len) {
194 for (size_t i = 0; i < bytes_len; i++) {
195 uint8_t b = bytes[i];
196 uint8_t hi = b >> 4;
197 uint8_t lo = b & 0x0F;
198 out_str[i * 2] = hi < 10 ? '0' + hi : 'A' + hi - 10;
199 out_str[i * 2 + 1] = lo < 10 ? '0' + lo : 'A' + lo - 10;
200 }
201 out_str[bytes_len * 2] = '\0';
202}
