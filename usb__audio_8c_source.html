23// ~170 ms buffer @ 48 kHz stereo 16-bit → very safe for drum prototyping
24#define RING_BUFFER_SIZE (32768) // must be power-of-2 for fast modulo
25// Allocated from PSRAM in usb_audio_init() (64 KB). It used to be a static
26// int16_t[RING_BUFFER_SIZE] in internal DRAM, which alone consumed 64 KB of the
27// scarce internal SRAM (~54 KB free at the time). The buffer is touched at
28// USB-frame / render-block granularity, not in the per-sample DSP inner loop, so
29// PSRAM latency is irrelevant here while the internal-RAM relief is significant.
30//
31// LOCK-FREE SPSC: there is exactly ONE producer (the AMY render task, Core 1,
32// via usb_audio_write_stereo) and exactly ONE consumer (usb_mic_task, Core 0,
33// via uac_input_cb). The old design guarded the indices with a mutex taken at
34// portMAX_DELAY, which let the lower-priority consumer block the high-priority
35// render task across PSRAM copies (priority inversion on a strict-1:1 5333 us
36// budget). With single-producer/single-consumer we drop the mutex entirely:
37// - s_write_idx is WRITER-owned: only the producer stores it; the consumer
38// only loads it.
39// - s_read_idx is READER-owned: only the consumer stores it; the producer
40// only loads it.
41// Each side publishes its own index with release ordering AFTER its data copy
42// and loads the other side's index with acquire ordering BEFORE its copy, so a
43// reader never observes an advanced s_write_idx before the samples it points to
44// are committed to PSRAM (and vice-versa for free-slot accounting).
45static int16_t *s_ring_buffer = NULL;
46static _Atomic size_t s_write_idx = 0; // writer-owned (producer)
47static _Atomic size_t s_read_idx = 0; // reader-owned (consumer)
48
49static bool s_initialized = false;

52// Written only by the consumer; read relaxed cross-core for advisory gating.
53#define USB_AUDIO_CONSUMER_TIMEOUT_MS 200
54static _Atomic uint32_t s_last_consumed_tick = 0;

56bool usb_audio_consumer_active(void)
57{
58 if (!s_initialized) return false;
59 uint32_t last = atomic_load_explicit(&s_last_consumed_tick, memory_order_relaxed);
60 if (last == 0) return false; // host never pulled yet
61 return (xTaskGetTickCount() - last) < pdMS_TO_TICKS(USB_AUDIO_CONSUMER_TIMEOUT_MS);
62}

74#if CONFIG_OUTPUT_WATCHDOG
75// Third-party PEEK reader (output-level watchdog, UI task on Core 0). Does not
76// participate in the SPSC index protocol: it never stores either index, only
77// acquire-loads s_write_idx so every sample behind it is committed. The region
78// [w - n, w) is the most recently produced audio; the producer cannot rewrite
79// it until the write index wraps the entire ring (~170 ms), so a short scan at
80// UI cadence cannot observe torn data.
81bool usb_audio_peek_levels(size_t n_samples, int32_t *peak_abs,
82 int32_t *mean_abs, size_t *write_idx)
83{
84 *peak_abs = 0;
85 *mean_abs = 0;
86 *write_idx = 0;
87 if (!s_initialized || n_samples == 0) return false;
88 // Cap the window so the |sample| accumulator cannot overflow int32 and the
89 // scan stays deep inside the not-yet-rewritable region behind the index.
90 if (n_samples > RING_BUFFER_SIZE / 4) n_samples = RING_BUFFER_SIZE / 4;
91
92 size_t w = atomic_load_explicit(&s_write_idx, memory_order_acquire);
93 size_t start = (w + RING_BUFFER_SIZE - n_samples) % RING_BUFFER_SIZE;
94 int32_t peak = 0, acc = 0;
95 for (size_t i = 0; i < n_samples; ++i) {
96 int32_t v = s_ring_buffer[(start + i) & (RING_BUFFER_SIZE - 1)];
97 if (v < 0) v = -v;
98 acc += v;
99 if (v > peak) peak = v;
100 }
101 *peak_abs = peak;
102 *mean_abs = acc / (int32_t)n_samples;
103 *write_idx = w;
104 return true;
105}
106#endif

108// Advisory fill level for diagnostics. Loads both indices relaxed; the result
109// is a momentary estimate, never used for correctness decisions.
110static inline size_t usb_audio_fill_samples_relaxed(void)
111{
112 size_t w = atomic_load_explicit(&s_write_idx, memory_order_relaxed);
113 size_t r = atomic_load_explicit(&s_read_idx, memory_order_relaxed);
114 return (w - r + RING_BUFFER_SIZE) % RING_BUFFER_SIZE;
115}

117static esp_err_t uac_input_cb(uint8_t *buf, size_t len, size_t *bytes_read, void *cb_ctx)
118{
119 (void)cb_ctx;
120
121 if (!s_initialized) {
122 memset(buf, 0, len);
123 *bytes_read = len;
124 return ESP_OK;
125 }
126
127 // Consumer liveness + flush-on-resume. The consumer owns s_read_idx, so it
128 // may legally advance it to s_write_idx to discard stale buffered audio when
129 // the host resumes after a gap (so playback starts fresh, not 200 ms late).
130 uint32_t now = xTaskGetTickCount();
131 uint32_t last = atomic_load_explicit(&s_last_consumed_tick, memory_order_relaxed);
132 if (last != 0 && (now - last) > pdMS_TO_TICKS(USB_AUDIO_CONSUMER_TIMEOUT_MS)) {
133 size_t w = atomic_load_explicit(&s_write_idx, memory_order_acquire);
134 atomic_store_explicit(&s_read_idx, w, memory_order_release);
135 }
136 atomic_store_explicit(&s_last_consumed_tick, now, memory_order_relaxed);
137
138 // CONSUMER side (single reader). s_read_idx is ours to advance; we acquire
139 // s_write_idx so every sample below [read, write) is guaranteed committed.
140 size_t read_idx = atomic_load_explicit(&s_read_idx, memory_order_relaxed);
141 size_t write_idx = atomic_load_explicit(&s_write_idx, memory_order_acquire);
142
143 size_t available_samples = (write_idx - read_idx + RING_BUFFER_SIZE) % RING_BUFFER_SIZE;
144 size_t samples_to_copy = (len / 2 < available_samples) ? len / 2 : available_samples;
145
146#if CONFIG_USB_AUDIO_DIAGNOSTICS
147 if (unlikely(samples_to_copy < (len / 2))) {
148 s_underrun_events++; // consumer-owned counter
149 }
150#endif
151
152 // Copy with wrap-around
153 size_t first = RING_BUFFER_SIZE - read_idx;
154 if (first > samples_to_copy) first = samples_to_copy;
155
156 memcpy(buf, &s_ring_buffer[read_idx], first * sizeof(int16_t));
157 if (samples_to_copy > first) {
158 memcpy(buf + first * sizeof(int16_t),
159 s_ring_buffer,
160 (samples_to_copy - first) * sizeof(int16_t));
161 }
162
163 // Publish the advanced read index with release ordering so the producer
164 // only sees the slots freed after the copy above has completed.
165 atomic_store_explicit(&s_read_idx,
166 (read_idx + samples_to_copy) % RING_BUFFER_SIZE,
167 memory_order_release);
168 *bytes_read = samples_to_copy * 2; // bytes
169
170 // Zero-pad on underrun (prevents pops)
171 if (*bytes_read < len) {
172 memset(buf + *bytes_read, 0, len - *bytes_read);
173 *bytes_read = len;
174 }
175
176 return ESP_OK;
177}

189esp_err_t usb_audio_init(void)
190{
191 if (s_initialized) return ESP_OK;
192
193 // 64 KB ring buffer in PSRAM (was a static array in internal DRAM).
194 s_ring_buffer = heap_caps_malloc(RING_BUFFER_SIZE * sizeof(int16_t),
195 MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
196 if (s_ring_buffer == NULL) {
197 ESP_LOGE(TAG, "ring buffer alloc failed (%u bytes PSRAM)",
198 (unsigned)(RING_BUFFER_SIZE * sizeof(int16_t)));
199 return ESP_ERR_NO_MEM;
200 }
201 // Zero the ring so neither a level peek nor the host's first read can ever
202 // observe uninitialized PSRAM as (loud) audio.
203 memset(s_ring_buffer, 0, RING_BUFFER_SIZE * sizeof(int16_t));
204
205 atomic_store_explicit(&s_write_idx, 0, memory_order_relaxed);
206 atomic_store_explicit(&s_read_idx, 0, memory_order_relaxed);
207
208 uac_device_config_t cfg = {
209 .skip_tinyusb_init = false,
210 .output_cb = NULL, // we only need mic direction
211 .input_cb = uac_input_cb,
212 .set_mute_cb = uac_set_mute_cb,
213 .set_volume_cb = uac_set_volume_cb,
214 .cb_ctx = NULL
215 };
216
217 esp_err_t ret = uac_device_init(&cfg);
218 if (ret != ESP_OK) {
219 ESP_LOGE(TAG, "uac_device_init failed: %s", esp_err_to_name(ret));
220 heap_caps_free(s_ring_buffer);
221 s_ring_buffer = NULL;
222 return ret;
223 }
224
225 s_initialized = true;
226 ESP_LOGI(TAG, "USB Audio ready – 48 kHz 16-bit stereo mic");
227 ESP_LOGI(TAG, "Plug into PC and select the new audio input device");
228 return ESP_OK;
229}

231esp_err_t usb_audio_write_stereo(const int16_t *data, size_t num_frames)
232{
233 if (!s_initialized || !data || num_frames == 0) return ESP_ERR_INVALID_STATE;
234
235 // No host consuming → do not churn the ring. Keep it empty so the host gets
236 // fresh audio on open. Not a drop (nobody is listening); report success.
237 if (!usb_audio_consumer_active()) {
238 return ESP_OK;
239 }
240
241 const size_t samples = num_frames * 2; // interleaved L,R
242
243#if CONFIG_USB_AUDIO_DIAGNOSTICS
244 s_write_calls++; // producer-owned counter
245#endif
246
247 // PRODUCER side (single writer). s_write_idx is ours to advance; we acquire
248 // s_read_idx so the free-slot count reflects everything the consumer has
249 // already drained.
250 size_t write_idx = atomic_load_explicit(&s_write_idx, memory_order_relaxed);
251 size_t read_idx = atomic_load_explicit(&s_read_idx, memory_order_acquire);
252
253 size_t free_slots = (read_idx - write_idx - 1 + RING_BUFFER_SIZE) % RING_BUFFER_SIZE;
254
255 // ALL-OR-NOTHING: if the whole block does not fit we write nothing and
256 // report the drop. A partial write would splice a half-block against the
257 // next full block and mangle the waveform; a clean whole-block gap is
258 // absorbed by the ~170 ms buffer / host resync instead.
259 if (unlikely(samples > free_slots)) {
260#if CONFIG_USB_AUDIO_DIAGNOSTICS
261 s_write_drop_events++;
262#endif
263 return ESP_ERR_NO_MEM;
264 }
265
266#if CONFIG_USB_AUDIO_DIAGNOSTICS
267 int16_t local_peak = 0;
268#endif
269
270 // Copy the block into the ring with wrap-around (two memcpys, no per-sample
271 // index churn). The producer is the sole writer of these slots, so this is
272 // safe before publishing the new write index.
273 size_t first = RING_BUFFER_SIZE - write_idx;
274 if (first > samples) first = samples;
275 memcpy(&s_ring_buffer[write_idx], data, first * sizeof(int16_t));
276 if (samples > first) {
277 memcpy(s_ring_buffer, data + first, (samples - first) * sizeof(int16_t));
278 }
279
280#if CONFIG_USB_AUDIO_DIAGNOSTICS
281 for (size_t i = 0; i < samples; i++) {
282 int16_t sample = data[i];
283 int16_t abs_sample = (sample < 0) ? (int16_t)(-sample) : sample;
284 if (abs_sample > local_peak) {
285 local_peak = abs_sample;
286 }
287 }
288 if (local_peak > s_peak_abs_sample) {
289 s_peak_abs_sample = local_peak;
290 }
291#endif
292
293 // Publish the advanced write index with release ordering so the consumer
294 // only observes it after all sample stores above are committed to PSRAM.
295 atomic_store_explicit(&s_write_idx,
296 (write_idx + samples) % RING_BUFFER_SIZE,
297 memory_order_release);
298
299#if CONFIG_USB_AUDIO_DIAGNOSTICS
300 size_t fill_samples = usb_audio_fill_samples_relaxed();
301 if (fill_samples > s_peak_fill_samples) {
302 s_peak_fill_samples = fill_samples;
303 }
304#endif
305
306 return ESP_OK;
307}

309esp_err_t usb_audio_write_mono(const int16_t *data, size_t num_samples)
310{
311 if (!s_initialized || !data || num_samples == 0) return ESP_ERR_INVALID_STATE;
312
313 // Expand to interleaved stereo on the stack and hand off in one all-or-
314 // nothing, lock-free write (the old version called write_stereo once PER
315 // sample). Chunked so the stack frame stays small for arbitrary lengths.
316 int16_t frames[128 * 2];
317 const size_t chunk = 128;
318 for (size_t off = 0; off < num_samples; off += chunk) {
319 size_t n = (num_samples - off < chunk) ? (num_samples - off) : chunk;
320 for (size_t i = 0; i < n; i++) {
321 frames[i * 2] = data[off + i];
322 frames[i * 2 + 1] = data[off + i];
323 }
324 esp_err_t err = usb_audio_write_stereo(frames, n);
325 if (err != ESP_OK) return err; // drop the rest of this call on full buffer
326 }
327 return ESP_OK;
328}
