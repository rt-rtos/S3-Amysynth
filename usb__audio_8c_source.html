25// ~170 ms buffer @ 48 kHz stereo 16-bit → very safe for drum prototyping
26#define RING_BUFFER_SIZE (32768) // must be power-of-2 for fast modulo
27// 64 KB, allocated from PSRAM in usb_audio_init(): touched at USB-frame /
28// render-block granularity, never in the per-sample DSP loop, so PSRAM latency
29// is irrelevant while the internal-RAM relief is significant.
30//
31// LOCK-FREE SPSC, no mutex (a mutex here let the lower-priority consumer block
32// the render task across PSRAM copies on a strict-1:1 5333 us budget). Exactly
33// ONE producer (AMY render task, Core 1, usb_audio_write_stereo) and ONE
34// consumer (usb_mic_task, Core 0, uac_input_cb):
35// - s_write_idx is WRITER-owned: only the producer stores it.
36// - s_read_idx is READER-owned: only the consumer stores it.
37// Each side publishes its own index release-ordered AFTER its data copy and
38// acquire-loads the other's BEFORE its copy, so an advanced index is never
39// observed before the samples it points to are committed.
40static int16_t *s_ring_buffer = NULL;
41static _Atomic size_t s_write_idx = 0; // writer-owned (producer)
42static _Atomic size_t s_read_idx = 0; // reader-owned (consumer)

44static bool s_initialized = false;

47// Written only by the consumer; read relaxed cross-core for advisory gating.
48#define USB_AUDIO_CONSUMER_TIMEOUT_MS 200
49static _Atomic uint32_t s_last_consumed_tick = 0;

51bool usb_audio_consumer_active(void)
52{
53 if (!s_initialized) return false;
54 uint32_t last = atomic_load_explicit(&s_last_consumed_tick, memory_order_relaxed);
55 if (last == 0) return false; // host never pulled yet
56 return (xTaskGetTickCount() - last) < pdMS_TO_TICKS(USB_AUDIO_CONSUMER_TIMEOUT_MS);
57}

67#if CONFIG_OUTPUT_WATCHDOG
68// Third-party PEEK reader (output-level watchdog, UI task on Core 0). Outside
69// the SPSC protocol: never stores an index, only acquire-loads s_write_idx so
70// everything behind it is committed. The producer cannot rewrite [w - n, w)
71// until the index wraps the whole ring (~170 ms), so a short scan at UI cadence
72// cannot see torn data.
73bool usb_audio_peek_levels(size_t n_samples, int32_t *peak_abs,
74 int32_t *mean_abs, size_t *write_idx)
75{
76 *peak_abs = 0;
77 *mean_abs = 0;
78 *write_idx = 0;
79 if (!s_initialized || n_samples == 0) return false;
80 // Cap the window so the |sample| accumulator cannot overflow int32 and the
81 // scan stays deep inside the not-yet-rewritable region behind the index.
82 if (n_samples > RING_BUFFER_SIZE / 4) n_samples = RING_BUFFER_SIZE / 4;
83
84 size_t w = atomic_load_explicit(&s_write_idx, memory_order_acquire);
85 size_t start = (w + RING_BUFFER_SIZE - n_samples) % RING_BUFFER_SIZE;
86 int32_t peak = 0, acc = 0;
87 for (size_t i = 0; i < n_samples; ++i) {
88 int32_t v = s_ring_buffer[(start + i) & (RING_BUFFER_SIZE - 1)];
89 if (v < 0) v = -v;
90 acc += v;
91 if (v > peak) peak = v;
92 }
93 *peak_abs = peak;
94 *mean_abs = acc / (int32_t)n_samples;
95 *write_idx = w;
96 return true;
97}
98#endif

100// Advisory fill level for diagnostics. Loads both indices relaxed; the result
101// is a momentary estimate, never used for correctness decisions.
102static inline size_t usb_audio_fill_samples_relaxed(void)
103{
104 size_t w = atomic_load_explicit(&s_write_idx, memory_order_relaxed);
105 size_t r = atomic_load_explicit(&s_read_idx, memory_order_relaxed);
106 return (w - r + RING_BUFFER_SIZE) % RING_BUFFER_SIZE;
107}

109static esp_err_t uac_input_cb(uint8_t *buf, size_t len, size_t *bytes_read, void *cb_ctx)
110{
111 (void)cb_ctx;
112
113 if (!s_initialized) {
114 *bytes_read = 0;
115 return ESP_OK;
116 }
117
118 // Flush-on-resume: the consumer owns s_read_idx, so it may advance it to
119 // s_write_idx to discard audio buffered during a host gap - playback starts
120 // fresh instead of 200 ms late.
121 uint32_t now = xTaskGetTickCount();
122 uint32_t last = atomic_load_explicit(&s_last_consumed_tick, memory_order_relaxed);
123 if (last != 0 && (now - last) > pdMS_TO_TICKS(USB_AUDIO_CONSUMER_TIMEOUT_MS)) {
124 size_t w = atomic_load_explicit(&s_write_idx, memory_order_acquire);
125 atomic_store_explicit(&s_read_idx, w, memory_order_release);
126 }
127 atomic_store_explicit(&s_last_consumed_tick, now, memory_order_relaxed);
128
129 // CONSUMER side. s_read_idx is ours to advance; acquire s_write_idx so all
130 // of [read, write) is guaranteed committed.
131 size_t read_idx = atomic_load_explicit(&s_read_idx, memory_order_relaxed);
132 size_t write_idx = atomic_load_explicit(&s_write_idx, memory_order_acquire);
133
134 size_t available_samples = (write_idx - read_idx + RING_BUFFER_SIZE) % RING_BUFFER_SIZE;
135 size_t samples_to_copy = (len / 2 < available_samples) ? len / 2 : available_samples;
136
137 // Copy with wrap-around
138 size_t first = RING_BUFFER_SIZE - read_idx;
139 if (first > samples_to_copy) first = samples_to_copy;
140
141 memcpy(buf, &s_ring_buffer[read_idx], first * sizeof(int16_t));
142 if (samples_to_copy > first) {
143 memcpy(buf + first * sizeof(int16_t),
144 s_ring_buffer,
145 (samples_to_copy - first) * sizeof(int16_t));
146 }
147
148 // Release-store: the producer sees the slots freed only after the copy.
149 atomic_store_explicit(&s_read_idx,
150 (read_idx + samples_to_copy) % RING_BUFFER_SIZE,
151 memory_order_release);
152 // Short returns are the contract, not a fault: the SOF-context pull
153 // sizes requests by EP FIFO room, and handing back only real audio is
154 // what lets the flow controller's packet sizing track the render rate
155 // (async source). Zero-padding here would pin the EP FIFO full and
156 // insert silence at the correction rate.
157 *bytes_read = samples_to_copy * 2; // bytes
158
159 return ESP_OK;
160}

196esp_err_t usb_audio_init(void)
197{
198 if (s_initialized) return ESP_OK;
199
200 s_ring_buffer = heap_caps_malloc(RING_BUFFER_SIZE * sizeof(int16_t),
201 MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
202 if (s_ring_buffer == NULL) {
203 ESP_LOGE(TAG, "ring buffer alloc failed (%u bytes PSRAM)",
204 (unsigned)(RING_BUFFER_SIZE * sizeof(int16_t)));
205 return ESP_ERR_NO_MEM;
206 }
207 // Zero the ring: a level peek or the host's first read must never see
208 // uninitialized PSRAM as (loud) audio.
209 memset(s_ring_buffer, 0, RING_BUFFER_SIZE * sizeof(int16_t));
210
211 // A SPIRAM cap request can be served from internal RAM under pressure, and
212 // this buffer is large enough that the fallback would matter. Record where
213 // it actually landed.
214 diag_mem_track("usb-ring", s_ring_buffer,
215 RING_BUFFER_SIZE * sizeof(int16_t));
216
217 atomic_store_explicit(&s_write_idx, 0, memory_order_relaxed);
218 atomic_store_explicit(&s_read_idx, 0, memory_order_relaxed);
219
220 uac_device_config_t cfg = {
221 .skip_tinyusb_init = false,
222 .output_cb = NULL, // we only need mic direction
223 .input_cb = uac_input_cb,
224 .set_mute_cb = uac_set_mute_cb,
225 .set_volume_cb = uac_set_volume_cb,
226 .cb_ctx = NULL
227 };
228
229 esp_err_t ret = uac_device_init(&cfg);
230 if (ret != ESP_OK) {
231 ESP_LOGE(TAG, "uac_device_init failed: %s", esp_err_to_name(ret));
232 heap_caps_free(s_ring_buffer);
233 s_ring_buffer = NULL;
234 return ret;
235 }
236
237 s_initialized = true;
238 ESP_LOGI(TAG, "USB Audio ready – 48 kHz 16-bit stereo mic");
239 ESP_LOGI(TAG, "Plug into PC and select the new audio input device");
240 return ESP_OK;
241}

243esp_err_t usb_audio_write_stereo(const int16_t *data, size_t num_frames)
244{
245 if (!s_initialized || !data || num_frames == 0) return ESP_ERR_INVALID_STATE;
246
247 // No host consuming → do not churn the ring. Keep it empty so the host gets
248 // fresh audio on open. Not a drop (nobody is listening); report success.
249 if (!usb_audio_consumer_active()) {
250 return ESP_OK;
251 }
252
253 const size_t samples = num_frames * 2; // interleaved L,R
254
255#if CONFIG_USB_AUDIO_DIAGNOSTICS
256 s_write_calls++; // producer-owned counter
257#endif
258
259 // PRODUCER side. s_write_idx is ours to advance; acquire s_read_idx so the
260 // free-slot count reflects everything the consumer already drained.
261 size_t write_idx = atomic_load_explicit(&s_write_idx, memory_order_relaxed);
262 size_t read_idx = atomic_load_explicit(&s_read_idx, memory_order_acquire);
263
264 size_t free_slots = (read_idx - write_idx - 1 + RING_BUFFER_SIZE) % RING_BUFFER_SIZE;
265
266 // ALL-OR-NOTHING: a partial write would splice a half-block against the
267 // next full block and mangle the waveform. A whole-block gap is absorbed
268 // by the ~170 ms buffer / host resync instead.
269 if (unlikely(samples > free_slots)) {
270#if CONFIG_USB_AUDIO_DIAGNOSTICS
271 s_write_drop_events++;
272#endif
273 return ESP_ERR_NO_MEM;
274 }
275
276#if CONFIG_USB_AUDIO_DIAGNOSTICS
277 int16_t local_peak = 0;
278#endif
279
280 // Copy with wrap-around. The producer is the sole writer of these slots, so
281 // this is safe before publishing the new write index.
282 size_t first = RING_BUFFER_SIZE - write_idx;
283 if (first > samples) first = samples;
284 memcpy(&s_ring_buffer[write_idx], data, first * sizeof(int16_t));
285 if (samples > first) {
286 memcpy(s_ring_buffer, data + first, (samples - first) * sizeof(int16_t));
287 }
288
289#if CONFIG_USB_AUDIO_DIAGNOSTICS
290 for (size_t i = 0; i < samples; i++) {
291 int16_t sample = data[i];
292 int16_t abs_sample = (sample < 0) ? (int16_t)(-sample) : sample;
293 if (abs_sample > local_peak) {
294 local_peak = abs_sample;
295 }
296 }
297 if (local_peak > s_peak_abs_sample) {
298 s_peak_abs_sample = local_peak;
299 }
300#endif
301
302 // Release-store: the consumer observes it only after the sample stores are
303 // committed to PSRAM.
304 atomic_store_explicit(&s_write_idx,
305 (write_idx + samples) % RING_BUFFER_SIZE,
306 memory_order_release);
307
308#if CONFIG_USB_AUDIO_DIAGNOSTICS
309 size_t fill_samples = usb_audio_fill_samples_relaxed();
310 if (fill_samples > s_peak_fill_samples) {
311 s_peak_fill_samples = fill_samples;
312 }
313#endif
314
315 return ESP_OK;
316}

318esp_err_t usb_audio_write_mono(const int16_t *data, size_t num_samples)
319{
320 if (!s_initialized || !data || num_samples == 0) return ESP_ERR_INVALID_STATE;
321
322 // Expand to interleaved stereo on the stack, chunked so the frame stays
323 // small for arbitrary lengths.
324 int16_t frames[128 * 2];
325 const size_t chunk = 128;
326 for (size_t off = 0; off < num_samples; off += chunk) {
327 size_t n = (num_samples - off < chunk) ? (num_samples - off) : chunk;
328 for (size_t i = 0; i < n; i++) {
329 frames[i * 2] = data[off + i];
330 frames[i * 2 + 1] = data[off + i];
331 }
332 esp_err_t err = usb_audio_write_stereo(frames, n);
333 if (err != ESP_OK) return err; // drop the rest of this call on full buffer
334 }
335 return ESP_OK;
336}
