19/* ── Lazy LFO-sibling materialization state ──────────────────────────────
20 * Per synth: has anything ever sent an event to the LFO sibling oscs
21 * (osc1/osc2) since the pool was last provably reset? AMY allocates an osc's
22 * ~532 B struct the first time an event addresses it (ensure_osc_allocd), so
23 * the disabled-path "park" events below would themselves BE the allocation.
24 * Skipping them is safe exactly when this bit is clear: the oscs are then NULL
25 * or AMY-reset, both silent and free of mod coupling.
26 *
27 * Clear the bit ONLY on provable freshness (a known pool-shape change in
28 * voice_build_wave); anything ambiguous keeps it SET. A wrongly-set bit costs
29 * only memory; a wrongly-clear bit skips parking a live carrier (the
30 * stale-COEF_MOD DC-rail class). The shape cache exists solely for that proof:
31 * 0 = unknown (never built here, or foreign config since, see
32 * voice_lfo_mark_foreign), and unknown never clears the bit.
33 *
34 * Core-0 / UI-task only, like every entry point in this file. */
35#define VOICE_LFO_SYNTH_MAX 128u

37static uint16_t s_pool_shape[VOICE_LFO_SYNTH_MAX]; /* (voices<<8)|oscs; 0 = unknown */
38static uint8_t s_lfo_materialized[VOICE_LFO_SYNTH_MAX / 8u];

40static inline bool lfo_materialized(uint8_t synth)
41{
42 if (synth >= VOICE_LFO_SYNTH_MAX) return true; /* out of range: park always */
43 return (s_lfo_materialized[synth >> 3] >> (synth & 7u)) & 1u;
44}

65void voice_lfo_note_pool_shape(uint8_t synth, uint8_t num_voices,
66 uint8_t oscs_per_voice)
67{
68 if (synth >= VOICE_LFO_SYNTH_MAX) return;
69 /* A *known different* shape forces AMY to reallocate the voice oscs
70 * (patches.c no-ops only on an identical shape) and RESET_OSC each one, so
71 * the carrier pair is provably fresh. An unknown cache proves nothing. */
72 uint16_t shape = (uint16_t)(((uint16_t)num_voices << 8) | oscs_per_voice);
73 if (s_pool_shape[synth] != 0 && s_pool_shape[synth] != shape)
74 lfo_set_materialized(synth, false);
75 s_pool_shape[synth] = shape;
76}

103/* ââ Unison copy math (spec: voice_unison_t, seq_model.h) âââââââââââââââ
104 * Copy i of n sits at position s in -1..+1 (0 alone for n = 1; the center
105 * copy of an odd n at 0). Detune fans log-linearly: freq CONST is Hz through
106 * the event API and logfreq_of_freq() anchors it at 440 Hz, so a
107 * note-tracking offset of c cents is SEQ_LFO_PITCH_BASE_HZ * 2^(c/1200).
108 * Blend tapers the outer copies (weight 1 - (1-blend)*|s|), normalized so
109 * summed power stays at the single-copy level (detuned copies decorrelate,
110 * so power - not amplitude - is what adds). */
111static float unison_pos(uint8_t i, uint8_t n)
112{
113 return (n > 1u) ? (2.0f * (float)i / (float)(n - 1u) - 1.0f) : 0.0f;
114}
115
116static float unison_weight(uint8_t i, uint8_t n, float blend)
117{
118 return 1.0f - (1.0f - blend) * fabsf(unison_pos(i, n));
119}
120
121static float unison_amp_norm(uint8_t n, float blend)
122{
123 float sum = 0.0f;
124 for (uint8_t i = 0; i < n; i++) {
125 float w = unison_weight(i, n, blend);
126 sum += w * w;
127 }
128 /* blend 0 with no center copy (even n) zeroes every weight; fall back to
129 * the equal-weight norm instead of dividing by zero. */
130 if (sum < 1e-6f) return 1.0f / sqrtf((float)n);
131 return 1.0f / sqrtf(sum);
132}

134/* Per-copy CONST terms, shared by the build and the live push so the two
135 * paths cannot drift. Overwrites amp CONST with the blend law. */
136static void unison_copy_coefs(amy_event *e, uint8_t i, const voice_unison_t *u,
137 float base_amp)
138{
139 uint8_t n = u->count;
140 float s = unison_pos(i, n);
141 float blend = (float)u->blend_pct / 100.0f;
142 e->freq_coefs[COEF_CONST] = SEQ_LFO_PITCH_BASE_HZ *
143 powf(2.0f, (float)u->detune_cents * s / 1200.0f);
144 e->pan_coefs[COEF_CONST] = 0.5f +
145 0.5f * ((float)u->spread_pct / 100.0f) * s;
146 e->amp_coefs[COEF_CONST] = base_amp *
147 unison_weight(i, n, blend) * unison_amp_norm(n, blend);
148 /* Deterministic, decorrelated start phase per copy: left phase-aligned,
149 * the copies comb through the whole attack. */
150 e->trigger_phase = (float)i / (float)n;
151}

153void voice_build_wave(const voice_wave_cfg_t *cfg)
154{
155 if (!cfg) return;
156
157 voice_lfo_note_pool_shape(cfg->synth, cfg->num_voices, cfg->oscs_per_voice);
158
159 /* Pool definition. Re-sending an unchanged shape is a no-op in AMY, so
160 * callers may rebuild freely without resetting live voices. */
161 amy_event *e = amy_helpers_event_begin();
162 e->synth = cfg->synth;
163 e->num_voices = cfg->num_voices;
164 e->oscs_per_voice = cfg->oscs_per_voice;
165 amy_helpers_event_send(e);
166
167 /* Audible copies: osc 0 alone, or the unison fan 0..n-1 (any reserved
168 * LFO pair sits above it - the caller sized oscs_per_voice). n == 1
169 * emits exactly the single-osc skeleton, with no pan/phase/freq CONST
170 * terms, so unison off is bit-for-bit the old build. Each copy is a
171 * note-following carrier (COEF_NOTE=1) with EG0-gated amplitude. */
172 uint8_t n = 1;
173 if (cfg->unison && cfg->unison->count > 1u && cfg->wave != KS)
174 n = cfg->unison->count;
175 if (n > VOICE_UNISON_MAX_COPIES) n = VOICE_UNISON_MAX_COPIES;
176
177 for (uint8_t i = 0; i < n; i++) {
178 e = amy_helpers_event_begin();
179 e->synth = cfg->synth;
180 e->osc = i;
181 e->wave = cfg->wave;
182 if (cfg->wt_preset >= 0) e->preset = cfg->wt_preset;
183 if (cfg->wave == KS) {
184 /* Authored feedback drives KS string decay; the 0 "never set"
185 * sentinel falls back to the fixed default. */
186 float fb = cfg->ks_feedback;
187 if (fb > 1.0f) fb = 1.0f; /* > 1 would make the KS buffer diverge */
188 e->feedback = (cfg->ks_feedback_authored && fb > 0.0f) ? fb : 0.9f;
189 }
190 e->freq_coefs[COEF_NOTE] = 1.0f;
191 e->amp_coefs[COEF_CONST] = cfg->osc0_amp_const;
192 e->amp_coefs[COEF_VEL] = cfg->osc0_amp_vel;
193 e->amp_coefs[COEF_EG0] = 1.0f;
194 if (n > 1u) unison_copy_coefs(e, i, cfg->unison, cfg->osc0_amp_const);
195 amy_helpers_event_send(e);
196 }
197
198 /* A rebuild must not drop the stage, so the owner's block rides every
199 * build. cfg->dist == NULL means the caller has none to assert yet. */
200 if (cfg->dist) voice_apply_dist(cfg->synth, cfg->dist);
201}

228void voice_apply_dist(uint8_t synth, const seq_dist_t *d)
229{
230 if (!d) return;
231 seq_dist_t v = *d;
232 voice_dist_clamp(&v);
233
234 /* osc 0 addresses the BASE osc of every voice in the synth (AMY's
235 * patches_event_has_voices). For wave voices and for ALGO/FM voices - where
236 * osc 0 already carries the summed operator output - that is the whole
237 * voice. Multi-osc patch strings distort their base osc only; distorting
238 * each osc and summing afterwards is a different, harsher effect, so the
239 * reach stops here deliberately. */
240 amy_event *e = amy_helpers_event_begin();
241 e->synth = synth;
242 e->osc = 0;
243 /* v.type is a stage mask; author all three enables explicitly so a
244 * mask change turns dropped stages off. */
245 e->dist_clip = !!(v.type & 1u);
246 e->dist_fold = !!(v.type & 2u);
247 e->dist_crush = !!(v.type & 4u);
248 e->dist_bits = (uint8_t)v.bits;
249 e->dist_rate = (uint16_t)v.rate;
250 /* Drive and mix ride AMY's control-coef rails now: author the CONST term
251 * only, leaving the MOD rail for a native LFO. Drive's CONST is linear
252 * (AMY maps it onto its log2 drive rail on the way in); mix is linear 0..1. */
253 e->dist_drive_coefs[COEF_CONST] = (float)v.drive;
254 e->dist_mix_coefs[COEF_CONST] = (float)v.mix / 100.0f;
255#if CONFIG_SEQ_DIST_DRIVE_AMP_EG
256 /* WIP PROTOTYPE (Kconfig-gated, default off): route the amp envelope (EG0)
257 * into drive on the log2 rail - the unipolar 0..1 env times a signed octave
258 * depth, exactly the shape filter_env_amount uses to sweep cutoff. The
259 * authored drive above is the release-tail floor; this adds up to DEPTH_Q/4
260 * octaves at the envelope peak (COEF0_SPECIAL only remaps CONST, so the EG
261 * slot is already in octaves). Shares EG0 with amplitude, so drive tracks
262 * loudness. TODO: promote to a per-track amount + EG0/EG1 source choice if
263 * it proves itself (see the dist coef rail in AMY-EDITS.md). */
264 e->dist_drive_coefs[COEF_EG0] =
265 (float)CONFIG_SEQ_DIST_DRIVE_AMP_EG_DEPTH_Q / 4.0f;
266#endif
267 amy_helpers_event_send(e);
268}

270/* PROTOTYPE LFO->distortion tick: contract in voice_config.h. Partial event -
271 * EVENT_TO_DELTA emits deltas only for the fields set here, so the committed
272 * type/bits/rate are never re-sent, and a stepper tick costs one event. */
273void voice_push_dist_lfo(uint8_t synth, const seq_dist_t *base,
274 const seq_lfo_t *lfo, float val)
275{
276 if (!base || !lfo || base->type == 0u) return;
277 float d = (float)lfo->depth / 100.0f;
278
279 amy_event *e = amy_helpers_event_begin();
280 e->synth = synth;
281 e->osc = 0; /* base osc of every voice - same reach as voice_apply_dist */
282 if (LFO_HAS_TGT(lfo, LFO_TARGET_DIST_DRIVE)) {
283 /* Octave-denominated pre-gain swing around the committed drive,
284 * clamped to the documented wire range. */
285 float drv = (float)base->drive *
286 powf(2.0f, d * VOICE_LFO_DEPTH_DIST_OCT * val);
287 e->dist_drive_coefs[COEF_CONST] = SEQ_CLAMP_F32(drv, 1.0f, 16.0f);
288 }
289 if (LFO_HAS_TGT(lfo, LFO_TARGET_DIST_MIX)) {
290 float mix = (float)base->mix / 100.0f +
291 d * VOICE_LFO_DEPTH_DIST_MIX * val;
292 e->dist_mix_coefs[COEF_CONST] = SEQ_CLAMP_F32(mix, 0.0f, 1.0f);
293 }
294 amy_helpers_event_send(e);
295}

297void voice_apply_native_lfo_topo(uint8_t synth, const seq_lfo_t *lfo,
298 uint16_t bpm, uint8_t carrier_osc,
299 uint8_t coupled_mask)
300{
301 amy_event *e;
302 uint8_t wobble_osc = (uint8_t)(carrier_osc + 1u);
303
304 if (lfo && lfo->enabled && lfo->targets) {
305 /* Coupled (audible) oscs: wire mod_source to the carrier (voice-local -
306 * AMY adds the base_osc offset) and set the COEF_MOD depth for every
307 * checked target, clearing every sibling rail first. One carrier feeds
308 * all targets on all coupled oscs, each scaled by its own normalizing
309 * constant. Bass couples both audible oscs so vibrato/tremolo hits the
310 * sub; the wave build couples osc0 only. */
311 float d = (float)lfo->depth / 100.0f;
312 for (uint8_t o = 0; (uint8_t)(coupled_mask >> o) != 0u; o++) {
313 if (!(coupled_mask & (uint8_t)(1u << o))) continue;
314 e = amy_helpers_event_begin();
315 e->synth = synth;
316 e->osc = o;
317 e->mod_source[0] = carrier_osc;
318 e->filter_freq_coefs[COEF_MOD] = 0.0f;
319 e->amp_coefs[COEF_MOD] = 0.0f;
320 e->freq_coefs[COEF_MOD] = 0.0f;
321 e->duty_coefs[COEF_MOD] = 0.0f;
322 e->pan_coefs[COEF_MOD] = 0.0f;
323 /* Distortion lives on the base osc (osc0) only - the same reach as
324 * voice_apply_dist - so its COEF_MOD rides osc0 alone; clear it on
325 * every coupled osc so a stale rail cannot keep modulating. */
326 e->dist_drive_coefs[COEF_MOD] = 0.0f;
327 e->dist_mix_coefs[COEF_MOD] = 0.0f;
328 if (LFO_HAS_TGT(lfo, LFO_TARGET_FILTER)) e->filter_freq_coefs[COEF_MOD] = voice_lfo_filter_octaves(lfo);
329 if (LFO_HAS_TGT(lfo, LFO_TARGET_AMP)) e->amp_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_AMP;
330 if (LFO_HAS_TGT(lfo, LFO_TARGET_PITCH)) e->freq_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_PITCH;
331 if (LFO_HAS_TGT(lfo, LFO_TARGET_SCAN)) e->duty_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_SCAN;
332 if (o == 0) {
333 /* Drive rides AMY's log2 rail, so the drive COEF_MOD is in
334 * octaves: the carrier's +/-1 swing is +/-(d*OCT) octaves of
335 * pre-gain, matching the old software stepper's law. Mix is a
336 * linear rail. Both stay inert until a dist stage is enabled. */
337 if (LFO_HAS_TGT(lfo, LFO_TARGET_DIST_DRIVE))
338 e->dist_drive_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_DIST_OCT;
339 if (LFO_HAS_TGT(lfo, LFO_TARGET_DIST_MIX))
340 e->dist_mix_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_DIST_MIX;
341 }
342 if (LFO_HAS_TGT(lfo, LFO_TARGET_PAN)) {
343 /* Pan is [0,1], not bipolar: set the center baseline and swing
344 * COEF_MOD around it in the same event. */
345 e->pan_coefs[COEF_CONST] = 0.5f;
346 e->pan_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_PAN;
347 }
348 amy_helpers_event_send(e);
349 }
350
351 /* Carrier: BPM-synced, no pitch tracking / velocity / envelope. WOBBLE
352 * (second-order LFO) chains the next osc as the carrier's own
353 * mod_source per AMY's chained-modulator semantics
354 * (mod_osc_would_cause_loop / compute_mod_scale); the reach picks
355 * which carrier rails it drives. Depth breathing is DOWNWARD-ONLY:
356 * amp CONST is pre-dropped by the swing so the breath peaks exactly
357 * at the authored LFO depth (an unclamped +swing would multiply the
358 * effective depth through the exponential MOD rail - voice_config.h).
359 * All coupled coefs are ALWAYS written so a stale wobble cannot keep
360 * modulating after it is turned off or its reach changes (the
361 * clear-siblings contract). */
362 float w = (float)lfo->wob_depth / 100.0f;
363 float wcoef = w * VOICE_WOB_DEPTH_AMP;
364 bool wob_depth_live = lfo->wob_depth > 0 &&
365 lfo->wob_reach != WOB_REACH_RATE;
366 bool wob_rate_live = lfo->wob_depth > 0 &&
367 lfo->wob_reach != WOB_REACH_DEPTH;
368 e = amy_helpers_event_begin();
369 e->synth = synth;
370 e->osc = carrier_osc;
371 e->wave = voice_lfo_wave_to_amy(lfo->wave);
372 e->freq_coefs[COEF_CONST] = lfo_rate_to_hz(lfo->rate, bpm);
373 e->freq_coefs[COEF_NOTE] = 0.0f;
374 e->freq_coefs[COEF_BEND] = 0.0f;
375 e->amp_coefs[COEF_CONST] = wob_depth_live
376 ? powf(10.0f, -3.0f * wcoef) : 1.0f;
377 e->amp_coefs[COEF_VEL] = 0.0f;
378 e->amp_coefs[COEF_EG0] = 0.0f;
379 e->mod_source[0] = wobble_osc;
380 e->amp_coefs[COEF_MOD] = wob_depth_live ? wcoef : 0.0f;
381 e->freq_coefs[COEF_MOD] = wob_rate_live
382 ? w * VOICE_WOB_DEPTH_RATE : 0.0f;
383 amy_helpers_event_send(e);
384
385 /* The wobble modulator itself: fixed TRIANGLE (smooth, no steps on the
386 * depth/rate rails), BPM-synced, dormant at 0 %. */
387 e = amy_helpers_event_begin();
388 e->synth = synth;
389 e->osc = wobble_osc;
390 e->wave = TRIANGLE;
391 e->freq_coefs[COEF_CONST] = lfo_rate_to_hz((lfo_rate_t)lfo->wob_rate, bpm);
392 e->freq_coefs[COEF_NOTE] = 0.0f;
393 e->freq_coefs[COEF_BEND] = 0.0f;
394 e->amp_coefs[COEF_CONST] = (lfo->wob_depth > 0) ? 1.0f : 0.0f;
395 e->amp_coefs[COEF_VEL] = 0.0f;
396 e->amp_coefs[COEF_EG0] = 0.0f;
397 amy_helpers_event_send(e);
398
399 lfo_set_materialized(synth, true);
400 } else {
401 /* Disabled: clear the mod coupling on every coupled osc, silence the
402 * carrier. The coupled-osc events always go out (those oscs exist as
403 * part of the patch and cost nothing). */
404 for (uint8_t o = 0; (uint8_t)(coupled_mask >> o) != 0u; o++) {
405 if (!(coupled_mask & (uint8_t)(1u << o))) continue;
406 e = amy_helpers_event_begin();
407 e->synth = synth;
408 e->osc = o;
409 e->filter_freq_coefs[COEF_MOD] = 0.0f;
410 e->amp_coefs[COEF_MOD] = 0.0f;
411 e->freq_coefs[COEF_MOD] = 0.0f;
412 e->duty_coefs[COEF_MOD] = 0.0f;
413 e->pan_coefs[COEF_MOD] = 0.0f;
414 e->dist_drive_coefs[COEF_MOD] = 0.0f;
415 e->dist_mix_coefs[COEF_MOD] = 0.0f;
416 amy_helpers_event_send(e);
417 }
418
419 /* A never-materialized carrier pair is NULL or AMY-reset: silent, no
420 * coupling, nothing to park. Sending the events would allocate the
421 * oscs and forfeit the lazy reservation. */
422 if (!lfo_materialized(synth))
423 return;
424
425 e = amy_helpers_event_begin();
426 e->synth = synth;
427 e->osc = carrier_osc;
428 e->amp_coefs[COEF_CONST] = 0.0f; /* dormant */
429 e->amp_coefs[COEF_MOD] = 0.0f; /* clear wobble depth coupling */
430 e->freq_coefs[COEF_MOD] = 0.0f; /* clear wobble rate coupling */
431 amy_helpers_event_send(e);
432
433 e = amy_helpers_event_begin();
434 e->synth = synth;
435 e->osc = wobble_osc;
436 e->amp_coefs[COEF_CONST] = 0.0f; /* wobble modulator dormant */
437 amy_helpers_event_send(e);
438 }
439}
