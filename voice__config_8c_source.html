38void voice_build_wave(const voice_wave_cfg_t *cfg)
39{
40 if (!cfg) return;
41
42 /* Pool definition. Re-sending an unchanged shape is a no-op in AMY, so
43 * callers may rebuild freely without resetting live voices. */
44 amy_event *e = amy_helpers_event_begin();
45 e->synth = cfg->synth;
46 e->num_voices = cfg->num_voices;
47 e->oscs_per_voice = cfg->oscs_per_voice;
48 amy_helpers_event_send(e);
49
50 /* osc 0: note-following carrier (COEF_NOTE=1), EG0-gated amplitude. */
51 e = amy_helpers_event_begin();
52 e->synth = cfg->synth;
53 e->osc = 0;
54 e->wave = cfg->wave;
55 if (cfg->wt_preset >= 0) e->preset = cfg->wt_preset;
56 if (cfg->wave == KS) {
57 /* Authored Q drives KS string decay once the user has dialed it;
58 * otherwise keep the fixed default until they touch Q. */
59 e->feedback = cfg->ks_feedback_authored
60 ? sequencer_core_ks_feedback_from_q(cfg->ks_feedback_q)
61 : 0.9f;
62 }
63 e->freq_coefs[COEF_NOTE] = 1.0f;
64 e->amp_coefs[COEF_CONST] = cfg->osc0_amp_const;
65 e->amp_coefs[COEF_VEL] = cfg->osc0_amp_vel;
66 e->amp_coefs[COEF_EG0] = 1.0f;
67 amy_helpers_event_send(e);

70void voice_apply_native_lfo(uint8_t synth, const seq_lfo_t *lfo, uint16_t bpm)
71{
72 amy_event *e;
73
74 if (lfo && lfo->enabled && lfo->targets) {
75 /* osc 0: wire mod_source to osc 1 (voice-local — AMY adds the base_osc
76 * offset, so it resolves within each voice) and set the COEF_MOD depth
77 * for every checked target, clearing every sibling first. One carrier
78 * feeds all targets; each gets the shared depth scaled by its own
79 * normalizing constant. */
80 float d = (float)lfo->depth / 100.0f;
81 e = amy_helpers_event_begin();
82 e->synth = synth;
83 e->osc = 0;
84 e->mod_source = 1;
85 e->filter_freq_coefs[COEF_MOD] = 0.0f;
86 e->amp_coefs[COEF_MOD] = 0.0f;
87 e->freq_coefs[COEF_MOD] = 0.0f;
88 e->duty_coefs[COEF_MOD] = 0.0f;
89 e->pan_coefs[COEF_MOD] = 0.0f;
90 if (LFO_HAS_TGT(lfo, LFO_TARGET_FILTER)) e->filter_freq_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_FILTER;
91 if (LFO_HAS_TGT(lfo, LFO_TARGET_AMP)) e->amp_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_AMP;
92 if (LFO_HAS_TGT(lfo, LFO_TARGET_PITCH)) e->freq_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_PITCH;
93 if (LFO_HAS_TGT(lfo, LFO_TARGET_SCAN)) e->duty_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_SCAN;
94 if (LFO_HAS_TGT(lfo, LFO_TARGET_PAN)) {
95 /* Pan is [0,1], not bipolar: set the center baseline and swing
96 * COEF_MOD around it in the same event. */
97 e->pan_coefs[COEF_CONST] = 0.5f;
98 e->pan_coefs[COEF_MOD] = d * VOICE_LFO_DEPTH_PAN;
99 }
100 amy_helpers_event_send(e);
101
102 /* osc 1: BPM-synced carrier — no pitch tracking, no velocity, no
103 * envelope; amp CONST=1 so AMY computes a mod value every block. */
104 e = amy_helpers_event_begin();
105 e->synth = synth;
106 e->osc = 1;
107 e->wave = voice_lfo_wave_to_amy(lfo->wave);
108 e->freq_coefs[COEF_CONST] = lfo_rate_to_hz(lfo->rate, bpm);
109 e->freq_coefs[COEF_NOTE] = 0.0f;
110 e->freq_coefs[COEF_BEND] = 0.0f;
111 e->amp_coefs[COEF_CONST] = 1.0f;
112 e->amp_coefs[COEF_VEL] = 0.0f;
113 e->amp_coefs[COEF_EG0] = 0.0f;
114 amy_helpers_event_send(e);
115 } else {
116 /* Disabled: clear the mod coupling, silence the carrier. */
117 e = amy_helpers_event_begin();
118 e->synth = synth;
119 e->osc = 0;
120 e->filter_freq_coefs[COEF_MOD] = 0.0f;
121 e->amp_coefs[COEF_MOD] = 0.0f;
122 e->freq_coefs[COEF_MOD] = 0.0f;
123 e->duty_coefs[COEF_MOD] = 0.0f;
124 e->pan_coefs[COEF_MOD] = 0.0f;
125 amy_helpers_event_send(e);
126
127 e = amy_helpers_event_begin();
128 e->synth = synth;
129 e->osc = 1;
130 e->amp_coefs[COEF_CONST] = 0.0f; /* dormant */
131 amy_helpers_event_send(e);
132 }
