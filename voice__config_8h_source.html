10/* Canonical LFO-target depth scalars - single source of truth. Each is the
11 * per-domain exchange rate for the SHARED depth%: anchored so equal depth
12 * reads as equal musical intensity across targets. Applies to the native
13 * (COEF_MOD) path AND the software-stepper pushes (sequencer/arp/live),
14 * which must multiply the same scalar inside their powf(2, ...) term. */
15#define VOICE_LFO_DEPTH_FILTER 3.0f
16#define VOICE_LFO_DEPTH_AMP 0.5f
17/* Pitch is log2: 1.0 = a full octave at 100% depth, which made even 5%
18 * (+-60 cents) heavy vibrato. One semitone at 100% puts the whole musical
19 * vibrato range on the knob - depth% ~= 1.2 cents/%. Octave-scale sweeps
20 * are deliberately out of this LFO's reach (per-target field if ever
21 * wanted, like the filter's octave-denominated swing). */
22#define VOICE_LFO_DEPTH_PITCH (1.0f / 12.0f)
23#define VOICE_LFO_DEPTH_SCAN 0.5f
24#define VOICE_LFO_DEPTH_PAN 0.5f /* swing around the 0.5 center baseline */
25/* DIST target (software stepper only - see voice_push_dist_lfo). Drive is
26 * perceived roughly log like cutoff, so the swing is denominated in octaves
27 * of pre-gain around the committed value: 100% depth = +/-2 oct (x1/4..x4).
28 * Mix is bounded [0,1] and linear like pan: 100% depth = +/-0.5 swing. */
29#define VOICE_LFO_DEPTH_DIST_OCT 2.0f
30#define VOICE_LFO_DEPTH_DIST_MIX 0.5f
31
32/* Anchor for software-LFO pitch pushes (every software stepper: sequencer
33 * layers, arp, live voice). AMY's freq COEF_CONST is an absolute frequency in
34 * Hz, converted via logfreq_of_freq(x) = log2(x / 440) (amy.c,
35 * ZERO_LOGFREQ_IN_HZ). Multiplying the desired ratio by this base makes the
36 * resulting logfreq constant the pure octave offset - 440 Hz itself is the
37 * neutral push, identical to the reset default of 0. */
38#define SEQ_LFO_PITCH_BASE_HZ 440.0f
39
40/* FILTER-target swing ceiling, in quarter-octave steps (16 = 4.0 octaves). */
41#define VOICE_LFO_FLT_OCT_Q_MAX 16u
42
43/* Effective FILTER swing in octaves. AMY's filter_freq COEF_MOD rail is
44 * log2-frequency, so this value IS the native coefficient; the software
45 * stepper uses it as the 2^x exponent scale. Resolves the flt_oct_q legacy
46 * sentinel (0 = depth% x VOICE_LFO_DEPTH_FILTER - the pre-octave law). */
47static inline float voice_lfo_filter_octaves(const seq_lfo_t *l)
48{
49 uint8_t q = l->flt_oct_q;
50 if (q == 0) return ((float)l->depth / 100.0f) * VOICE_LFO_DEPTH_FILTER;
51 if (q > VOICE_LFO_FLT_OCT_Q_MAX) q = VOICE_LFO_FLT_OCT_Q_MAX;
52 return 0.25f * (float)q;
53}

55/* WOBBLE (second-order LFO): osc2 chained as the osc1 carrier's mod_source.
56 * AMP rides the dB combine (contribution is linear in the exponent: amp
57 * multiplier = 10^(3 * coef * wob)) and is applied DOWNWARD-ONLY: the
58 * carrier's amp CONST is pre-dropped by the swing (10^(-3*coef)) so the
59 * breathing peaks exactly at the authored LFO depth and dips below it, never
60 * above. Untamed, the +half of the swing multiplies the effective LFO depth
61 * through the unclamped exponential MOD rail (2.8x at the former 0.15 coef)
62 * and overdrives every target - the "too loud" wobble finding, 2026-08-03.
63 * RATE is linear in log2-frequency: 1.0 => +/-1 octave rate swing. */
64#define VOICE_WOB_DEPTH_AMP 0.075f
65#define VOICE_WOB_DEPTH_RATE 1.0f
66
67/* ── WOBBLE authoring unit ───────────────────────────────────────────────
68 * Stored wob_depth is a 0..100 percentage of VOICE_WOB_DEPTH_AMP - meaningless
69 * to the user, so authoring and display speak whole dB of TOTAL DIP below the
70 * authored depth: the LFO breathes from full authored depth down to -N dB and
71 * back (downward-only, see above).
72 *
73 * The AMP coefficient enters AMY's dB combine linearly (amp = 10^(3*coef*mod),
74 * amp_combine_controls) and the carrier CONST is dropped by the same swing, so
75 * total dip = 120 * coef dB with coef = wob/100 * VOICE_WOB_DEPTH_AMP: full
76 * scale is 120 * VOICE_WOB_DEPTH_AMP = 9 dB in 1 dB steps.
77 *
78 * 0 dB is OFF, not a distinct setting: at zero swing the modulator is parked
79 * (osc2 silenced), so nothing exists between "off" and "1 dB". The stored byte
80 * keeps its 0..100 meaning, so old snapshots load unchanged.
81 *
82 * The same control also swings carrier RATE by up to +/-1 octave
83 * (VOICE_WOB_DEPTH_RATE) when the reach includes rate; rate-only reach has no
84 * depth breathing to measure, so its readout speaks octaves instead of dB. */
85#define VOICE_WOB_DB_MAX 9u /* = 120 * VOICE_WOB_DEPTH_AMP; keep in step */

134/* ── Shared WAVE-voice skeleton ──────────────────────────────────────────
135 * The canonical "N voices, osc0 = note-following carrier" build used by the
136 * arp, drone and melodic sequencer. Sends two events: pool definition and osc0
137 * skeleton. Deliberately does NOT touch osc1, envelopes, filters or mod
138 * routing - each engine layers its own specialization on top as deltas.
139 *
140 * AMY does not reset the osc pool when the same num_voices/oscs_per_voice is
141 * re-sent, so rebuilding never glitches held voices; that same property is why
142 * per-target COEF_MOD state must be cleared explicitly on reconfigure
143 * (voice_apply_native_lfo). */
144typedef struct {
145 uint8_t synth;
146 uint8_t num_voices;
147 uint8_t oscs_per_voice; /* 3 when oscs 1-2 are reserved as the LFO pair */
148 uint16_t wave; /* AMY wave constant for osc0 */
149 float osc0_amp_const; /* arp/melodic 1.0; drone const_sent */
150 float osc0_amp_vel; /* arp/melodic 1.0; drone 0.0 */
151 bool ks_feedback_authored; /* authored feedback drives KS decay */
152 float ks_feedback; /* seq_filter_t.feedback (0..1) when authored;
153 <= 0 or unauthored -> fixed 0.9 default */
154 int16_t wt_preset; /* >=0 => e->preset (WAVETABLE); else -1 */
155 const seq_dist_t *dist; /* re-pushed on every build so a rebuild
156 never drops the stage; NULL = leave the
157 synth's distortion untouched */
158 const voice_unison_t *unison; /* copy-fan spec (seq_model.h); NULL or
159 count <= 1 = the single-osc build. The
160 caller passes the EFFECTIVE count: it
161 has sized oscs_per_voice for count
162 copies plus any reserved pair, excluded
163 KS and clamped against the osc budget */
164} voice_wave_cfg_t;

179/* ── Per-voice distortion (AMY DIST_*) ───────────────────────────────────
180 * The distortion block is owned by the caller's voice_params_t (seq_model.h),
181 * exactly like the filter and the LFO: melodic layers hold one per track, the
182 * arp / both drones / live-play hold one each. There is no global or
183 * per-domain set - a synth's distortion is whatever its owner last pushed.

185 * Reach: the event addresses osc 0, which AMY resolves to the BASE osc of every

193 * inspects the target's topology. */

216/* ── Lazy LFO-sibling materialization ────────────────────────────────────
217 * voice_build_wave() reserves the osc1 (LFO carrier) and osc2 (wobble) INDEX
218 * slots in the pool shape, but AMY allocates an osc's ~532 B struct only when
219 * an event first addresses it. voice_config tracks per synth whether the
220 * siblings were ever materialized, so the disabled-path park events - which
221 * would themselves BE the allocation - are skipped while the oscs are provably
222 * fresh (NULL or AMY-reset: silent, no mod coupling).
223 *
224 * Failure modes are asymmetric by design: a wrongly-SET state costs only the
225 * savings, while a wrongly-CLEAR state skips parking a live carrier and
226 * resurrects the stale-COEF_MOD DC rail. So the state clears ONLY on a
227 * provable pool reset inside voice_build_wave, and every path configuring a
228 * wave-built synth with a foreign topology (patch string, bass/FM/additive
229 * preset, PCM pool) must call voice_lfo_mark_foreign() first. */
