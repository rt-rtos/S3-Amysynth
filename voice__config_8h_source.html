35/* ── Shared WAVE-voice skeleton ──────────────────────────────────────────
36 * The canonical "N voices, osc0 = note-following carrier" build used by the
37 * arp, the drone, and the melodic sequencer. voice_build_wave() sends two
38 * events: the pool definition and the osc0 skeleton. It deliberately does
39 * NOT touch osc1, envelopes, filters, or mod routing — each engine layers
40 * its own specialization (native LFO carrier, PULSE stutter gate, sweep
41 * filter) on top as follow-up deltas.
42 *
43 * AMY does not reset the osc pool when the same num_voices/oscs_per_voice
44 * is re-sent, so rebuilding through this function never glitches held
45 * voices; that same property is why per-target COEF_MOD state must be
46 * cleared explicitly on reconfigure (see voice_apply_native_lfo). */
47typedef struct {
48 uint8_t synth;
49 uint8_t num_voices;
50 uint8_t oscs_per_voice; /* 2 when osc1 is reserved as a carrier */
51 uint16_t wave; /* AMY wave constant for osc0 */
52 float osc0_amp_const; /* arp/melodic 1.0; drone const_sent */
53 float osc0_amp_vel; /* arp/melodic 1.0; drone 0.0 */
54 bool ks_feedback_authored; /* authored Q drives KS string decay */
55 float ks_feedback_q; /* seq_filter_t resonance when authored */
56 int16_t wt_preset; /* >=0 => e->preset (WAVETABLE); else -1 */
57} voice_wave_cfg_t;

62/* Apply the AMY-native mod-source LFO routing for one synth built by
63 * voice_build_wave() with oscs_per_voice=2: osc0 gets mod_source=1 plus the
64 * selected target's COEF_MOD depth, osc1 becomes the LFO carrier at the
65 * BPM-synced rate. Pass lfo=NULL (or a disabled lfo) to deactivate: the mod
66 * coupling is cleared and the carrier silenced.
67 *
68 * Idempotent and state-clean: every target's COEF_MOD is cleared before the
69 * selected one is set, because re-sending the same voice count does not reset
70 * the osc pool — a prior target's coef would otherwise persist and keep
71 * modulating (the stale-AMP case rides AMY's convex dB combine path and ramps
72 * the output to a DC rail).
73 *
74 * Core-0 / UI-task only; pushes through amy_helpers (never amy_queue_lock).
75 * This is the function a future per-step p-lock caller uses from the tick
76 * engine. */
77void voice_apply_native_lfo(uint8_t synth, const seq_lfo_t *lfo, uint16_t bpm);
